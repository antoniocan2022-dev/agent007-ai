   183→   183→async function safeJson(res: Response): Promise<any> {
   184→   184→  const text = await res.text()
   185→   185→  try {
   186→   186→    return JSON.parse(text)
   187→   187→  } catch {
   188→   188→    return { error: text || 'parse error' }
   189→   189→  }

   233→   233→      let convs = data.conversations ?? []
   234→   234→      // UPGRADE #70 — DB is the ONLY source of truth (was: localStorage priority)

   238→   238→        try { localStorage.setItem('agent007_conversations', JSON.stringify(convs)) } catch {}
   239→   239→      }
   240→   240→    } catch (e) {
   241→   241→      console.error('loadConversations', e)
   242→   242→      // Fallback: localStorage ONLY if API is unreachable (offline mode)
   243→   243→      if (typeof window !== 'undefined') {
   244→   244→        try {
   245→   245→          const saved = localStorage.getItem('agent007_conversations')

   295→   295→  loadMessages: async (conversationId) => {
   296→   296→    try {
   297→   297→      const res = await fetch(`/api/conversations/${conversationId}`)
   298→   298→      if (!res.ok) {
   299→   299→        console.warn('[loadMessages] API returned', res.status)
   300→   300→        set({ messages: [] })
   301→   301→        return
   302→   302→      }
   303→   303→      const data = await safeJson(res)
   304→   304→      const conv = data.conversation
   305→   305→      if (!conv || !conv.Message || conv.Message.length === 0) {
   306→   306→        // UPGRADE #70 — DB is the ONLY source of truth. No localStorage fallback for messages.
   307→   307→        // Every device sees the SAME messages from Postgres.
   308→   308→        set({ messages: [] })
   309→   309→        return
   310→   310→      }
   311→   311→      // Reconstruct messages from DB rows: collapse tool/thought rows under the *next* assistant message
   312→   312→      const rows: any[] = conv.Message ?? []
   313→   313→      const messages: ChatMessage[] = []
   314→   314→      let pendingSteps: ToolStep[] = []
   315→   315→      // Track open dispatches so subsequent subagent rows can be linked.
   316→   316→      // Map dispatchId → { subagentId, subagentName, subagentColor, subagentIcon, task, stepId }
   317→   317→      const dispatchMap = new Map<string, any>()
   318→   318→      let stepCounter = 0
   319→   319→      for (const r of rows) {
   320→   320→        if (r.role === 'user') {
   321→   321→          const atts = r.attachments ? safeParseAttachments(r.attachments) : undefined
   322→   322→          messages.push({
   323→   323→            id: r.id,
   324→   324→            role: 'user',
   325→   325→            content: r.content,
   326→   326→            attachments: atts,
   327→   327→            createdAt: new Date(r.createdAt).getTime(),
   328→   328→          })
   329→   329→        } else if (r.role === 'thought') {
   330→   330→          // Check if this is a sub-agent thought (prefixed with [subagent:id])
   331→   331→          const subMatch = r.content?.match(/^\[subagent:([^\]]+)\]\s*([\s\S]*)$/)
   332→   332→          if (subMatch) {
   333→   333→            const subId = subMatch[1]
   334→   334→            // Find the most recent dispatch for this sub-agent
   335→   335→            let parentDispatch: any = null
   336→   336→            for (const [did, info] of dispatchMap.entries()) {
   337→   337→              if (info.subagentId === subId) parentDispatch = { did, ...info }
   338→   338→            }
   339→   339→            stepCounter++
   340→   340→            pendingSteps.push({
   341→   341→              id: r.id,
   342→   342→              stepNumber: stepCounter,
   343→   343→              thought: subMatch[2],
   344→   344→              status: 'done',
   345→   345→              startedAt: new Date(r.createdAt).getTime(),
   346→   346→              finishedAt: new Date(r.createdAt).getTime(),
   347→   347→              kind: 'subagent_thought',
   348→   348→              dispatchId: parentDispatch?.did,
   349→   349→              subagentId: subId,
   350→   350→              subagentName: parentDispatch?.subagentName,
   351→   351→              subagentColor: parentDispatch?.subagentColor,
   352→   352→              subagentIcon: parentDispatch?.subagentIcon,
   353→   353→            })
   354→   354→          } else {
   355→   355→            stepCounter++
   356→   356→            pendingSteps.push({
   357→   357→              id: r.id,
   358→   358→              stepNumber: stepCounter,
   359→   359→              thought: r.content,
   360→   360→              status: 'done',
   361→   361→              startedAt: new Date(r.createdAt).getTime(),
   362→   362→              finishedAt: new Date(r.createdAt).getTime(),
   363→   363→              kind: 'super_thought',
   364→   364→            })
   365→   365→          }
   366→   366→        } else if (r.role === 'tool') {
   367→   367→          // Sub-agent dispatch?
   368→   368→          if (r.toolName === 'subagent_dispatch') {
   369→   369→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   370→   370→            stepCounter++
   371→   371→            const stepId = `dispatch_${meta.dispatchId ?? r.id}`
   372→   372→            const stepInfo: any = {
   373→   373→              subagentId: meta.agentId,
   374→   374→              subagentName: meta.agentName,
   375→   375→              subagentColor: meta.color,
   376→   376→              subagentIcon: meta.icon,
   377→   377→              task: meta.task,
   378→   378→            }
   379→   379→            if (meta.dispatchId) dispatchMap.set(meta.dispatchId, stepInfo)
   380→   380→            pendingSteps.push({
   381→   381→              id: stepId,
   382→   382→              stepNumber: stepCounter,
   383→   383→              subagentId: meta.agentId,
   384→   384→              subagentName: meta.agentName,
   385→   385→              subagentColor: meta.color,
   386→   386→              subagentIcon: meta.icon,
   387→   387→              subagentTask: meta.task,
   388→   388→              dispatchId: meta.dispatchId,
   389→   389→              status: 'done',
   390→   390→              startedAt: new Date(r.createdAt).getTime(),
   391→   391→              finishedAt: new Date(r.createdAt).getTime(),
   392→   392→              kind: 'subagent_dispatch',
   393→   393→              // If a subagent_complete row exists for the same dispatchId, the
   394→   394→              // subagentAnswer will be filled below when we encounter that row.
   395→   395→            })
   396→   396→          } else if (r.toolName === 'subagent_complete') {
   397→   397→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   398→   398→            // Find the original dispatch step and set the answer
   399→   399→            const dispatchStep = pendingSteps.find(
   400→   400→              (st) => st.dispatchId === meta.dispatchId && st.kind === 'subagent_dispatch'
   401→   401→            )
   402→   402→            if (dispatchStep) {
   403→   403→              dispatchStep.subagentAnswer = r.toolResult ?? ''
   404→   404→              dispatchStep.status = 'done'
   405→   405→              dispatchStep.finishedAt = new Date(r.createdAt).getTime()
   406→   406→            }
   407→   407→          } else if (r.toolName === 'subagent_tool') {
   408→   408→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   409→   409→            const parentDispatch = meta.dispatchId ? dispatchMap.get(meta.dispatchId) : null
   410→   410→            stepCounter++
   411→   411→            pendingSteps.push({
   412→   412→              id: meta.stepId ?? r.id,
   413→   413→              stepNumber: stepCounter,
   414→   414→              toolName: meta.tool,
   415→   415→              toolArgs: meta.args,
   416→   416→              toolResult: r.toolResult ?? '',
   417→   417→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   418→   418→              toolOk: true,
   419→   419→              status: 'done',
   420→   420→              startedAt: new Date(r.createdAt).getTime(),
   421→   421→              finishedAt: new Date(r.createdAt).getTime(),
   422→   422→              kind: 'subagent_tool',
   423→   423→              dispatchId: meta.dispatchId,
   424→   424→              subagentId: parentDispatch?.subagentId ?? meta.agentId,
   425→   425→              subagentName: parentDispatch?.subagentName,
   426→   426→              subagentColor: parentDispatch?.subagentColor,
   427→   427→              subagentIcon: parentDispatch?.subagentIcon,
   428→   428→            })
   429→   429→          } else if (r.toolName === 'manage_action') {
   430→   430→            // Reconstruct manage_action step from persisted row
   431→   431→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   432→   432→            stepCounter++
   433→   433→            pendingSteps.push({
   434→   434→              id: r.id,
   435→   435→              stepNumber: stepCounter,
   436→   436→              status: 'done',
   437→   437→              startedAt: new Date(r.createdAt).getTime(),
   438→   438→              finishedAt: new Date(r.createdAt).getTime(),
   439→   439→              kind: 'manage_action',
   440→   440→              manageAction: meta.action,
   441→   441→              manageAttrs: meta.attrs,
   442→   442→              manageResult: { ok: !/failed|error/i.test(r.toolResult ?? ''), message: r.toolResult ?? '' },
   443→   443→            })
   444→   444→          } else {
   445→   445→            // Regular super-agent tool
   446→   446→            stepCounter++
   447→   447→            pendingSteps.push({
   448→   448→              id: r.id,
   449→   449→              stepNumber: stepCounter,
   450→   450→              toolName: r.toolName,
   451→   451→              toolArgs: r.toolArgs ? safeParseJson(r.toolArgs) : undefined,
   452→   452→              toolResult: r.toolResult ?? '',
   453→   453→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   454→   454→              toolOk: true,
   455→   455→              status: 'done',
   456→   456→              startedAt: new Date(r.createdAt).getTime(),
   457→   457→              finishedAt: new Date(r.createdAt).getTime(),
   458→   458→              kind: 'super_tool',
   459→   459→            })
   460→   460→          }
   461→   461→        } else if (r.role === 'assistant') {
   462→   462→          messages.push({
   463→   463→            id: r.id,
   464→   464→            role: 'assistant',
   465→   465→            content: r.content,
   466→   466→            steps: pendingSteps,
   467→   467→            createdAt: new Date(r.createdAt).getTime(),
   468→   468→          })
   469→   469→          pendingSteps = []
   470→   470→          stepCounter = 0
   471→   471→          dispatchMap.clear()
   472→   472→        }

   837→   837→function parseSse(raw: string): { event: string; data: any } | null {
   838→   838→  const lines = raw.split('\n')
   839→   839→  let event = 'message'
   840→   840→  let data = ''
   841→   841→  for (const ln of lines) {
   842→   842→    if (ln.startsWith('event:')) event = ln.slice(6).trim()
   843→   843→    else if (ln.startsWith('data:')) data += ln.slice(5).trim()
   844→   844→  }
   845→   845→  try {
   846→   846→    return { event, data: data ? JSON.parse(data) : {} }
   847→   847→  } catch {
   848→   848→    return { event, data: { raw: data } }
   849→   849→  }
   850→   850→}

   852→   852→function applyEvent(
   853→   853→  event: string,
   854→   854→  data: any,

   861→   861→  // Store the reasoning on the assistant message so the UI can display it
   862→   862→  // in a collapsible "Show reasoning" section.
   863→   863→  if (event === 'reasoning') {
   864→   864→    set((s) => ({
   865→   865→      status: 'thinking',
   866→   866→      messages: s.messages.map((m) => {
   867→   867→        if (m.id !== assistantId) return m
   868→   868→        // Append reasoning (there may be multiple reasoning chunks if the
   869→   869→        // agent does multiple iterations — concatenate them)
   870→   870→        const existingReasoning = m.reasoning || ''
   871→   871→        const newReasoning = existingReasoning
   872→   872→          ? `${existingReasoning}\n\n${data.content}`
   873→   873→          : data.content
   874→   874→        return { ...m, reasoning: newReasoning }
   875→   875→      }),
   876→   876→    }))
   877→   877→    return
   878→   878→  }
   879→   879→  if (event === 'thought') {
   880→   880→    set((s) => ({
   881→   881→      status: 'thinking',
   882→   882→      messages: s.messages.map((m) => {
   883→   883→        if (m.id !== assistantId) return m
   884→   884→        const steps = [...(m.steps ?? [])]
   885→   885→        // create a "thinking-only" step so users see the reasoning

   887→   887→        steps.push({
   888→   888→          id: stepId,
   889→   889→          stepNumber: steps.length + 1,
   890→   890→          thought: data.content,
   891→   891→          status: 'done',
   892→   892→          startedAt: Date.now(),
   893→   893→          finishedAt: Date.now(),
   894→   894→          kind: 'super_thought',
   895→   895→        })
   896→   896→        return { ...m, steps }
   897→   897→      }),
   898→   898→    }))
   899→   899→  } else if (event === 'tool_call') {
   900→   900→    const stepId = data.stepId
   901→   901→    set((s) => ({
   902→   902→      status: 'tool_running',
   903→   903→      currentTool: data.name,
   904→   904→      messages: s.messages.map((m) => {
   905→   905→        if (m.id !== assistantId) return m
   906→   906→        const steps = [...(m.steps ?? [])]
   907→   907→        steps.push({
   908→   908→          id: stepId,
   909→   909→          stepNumber: data.stepNumber ?? steps.length + 1,
   910→   910→          thought: data.thought,
   911→   911→          toolName: data.name,
   912→   912→          toolArgs: data.args,
   913→   913→          status: 'running',
   914→   914→          startedAt: Date.now(),
   915→   915→          kind: 'super_tool',
   916→   916→        })
   917→   917→        return { ...m, steps }
   918→   918→      }),
   919→   919→    }))
   920→   920→  } else if (event === 'tool_result') {
   921→   921→    const stepId = data.stepId
   922→   922→    // UPGRADE #124 — Verify tool action (check for real artifact)
   923→   923→    const verification = data.verified !== undefined ? {
   924→   924→      verified: data.verified,
   925→   925→      verificationWarning: data.verificationWarning,
   926→   926→    } : undefined
   927→   927→    set((s) => ({
   928→   928→      status: 'thinking',
   929→   929→      currentTool: null,
   930→   930→      messages: s.messages.map((m) => {
