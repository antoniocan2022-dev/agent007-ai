   121→async function safeJson(res: Response): Promise<any> {
   122→  const text = await res.text()

   124→    return JSON.parse(text)
   125→  } catch {

   130→export const useChatStore = create<ChatState>((set, get) => ({
   131→  conversations: [],
   132→  currentConversationId: null,
   133→  messages: [],
   134→  language: 'en',
   135→  attachments: [],
   136→  status: 'idle',
   137→  currentTool: null,
   138→  memories: [],
   139→  activeSubagents: [],
   140→  subagentActivity: {},
   141→  leftOpen: true,
   142→  rightOpen: true,
   143→  abortFlag: { current: false },
   144→
   145→  loadConversations: async () => {
   146→    try {
   147→      const res = await fetch('/api/conversations')
   148→      const data = await safeJson(res)
   149→      set({ conversations: data.conversations ?? [] })
   150→    } catch (e) {
   151→      console.error('loadConversations', e)
   152→    }
   153→  },
   154→
   155→  createConversation: async () => {
   156→    const res = await fetch('/api/conversations', {
   157→      method: 'POST',
   158→      headers: { 'Content-Type': 'application/json' },
   159→      body: JSON.stringify({ title: 'New Conversation' }),
   160→    })
   161→    const data = await safeJson(res)
   162→    const conv: ConversationMeta = data.conversation
   163→    set((s) => ({
   164→      conversations: [conv, ...s.conversations],
   165→      currentConversationId: conv.id,
   166→      messages: [],
   167→    }))
   168→    return conv.id
   169→  },
   170→
   171→  selectConversation: async (id) => {
   172→    set({ currentConversationId: id, messages: [] })
   173→    await get().loadMessages(id)
   174→  },
   175→
   176→  deleteConversation: async (id) => {
   177→    await fetch(`/api/conversations/${id}`, { method: 'DELETE' })
   178→    set((s) => ({
   179→      conversations: s.conversations.filter((c) => c.id !== id),
   180→      currentConversationId: s.currentConversationId === id ? null : s.currentConversationId,
   181→      messages: s.currentConversationId === id ? [] : s.messages,
   182→    }))
   183→  },
   184→
   185→  loadMessages: async (conversationId) => {
   186→    try {
   187→      const res = await fetch(`/api/conversations/${conversationId}`)
   188→      const data = await safeJson(res)
   189→      const conv = data.conversation
   190→      if (!conv) {
   191→        set({ messages: [] })
   192→        return
   193→      }
   194→      // Reconstruct messages from DB rows: collapse tool/thought rows under the *next* assistant message
   195→      const rows: any[] = conv.messages ?? []
   196→      const messages: ChatMessage[] = []
   197→      let pendingSteps: ToolStep[] = []
   198→      // Track open dispatches so subsequent subagent rows can be linked.
   199→      // Map dispatchId → { subagentId, subagentName, subagentColor, subagentIcon, task, stepId }
   200→      const dispatchMap = new Map<string, any>()
   201→      let stepCounter = 0
   202→      for (const r of rows) {
   203→        if (r.role === 'user') {
   204→          const atts = r.attachments ? safeParseAttachments(r.attachments) : undefined
   205→          messages.push({
   206→            id: r.id,
   207→            role: 'user',
   208→            content: r.content,
   209→            attachments: atts,
   210→            createdAt: new Date(r.createdAt).getTime(),
   211→          })
   212→        } else if (r.role === 'thought') {
   213→          // Check if this is a sub-agent thought (prefixed with [subagent:id])
   214→          const subMatch = r.content?.match(/^\[subagent:([^\]]+)\]\s*(.*)$/s)
   215→          if (subMatch) {
   216→            const subId = subMatch[1]
   217→            // Find the most recent dispatch for this sub-agent
   218→            let parentDispatch: any = null
   219→            for (const [did, info] of dispatchMap.entries()) {
   220→              if (info.subagentId === subId) parentDispatch = { did, ...info }
   221→            }
   222→            stepCounter++
   223→            pendingSteps.push({
   224→              id: r.id,
   225→              stepNumber: stepCounter,
   226→              thought: subMatch[2],
   227→              status: 'done',
   228→              startedAt: new Date(r.createdAt).getTime(),
   229→              finishedAt: new Date(r.createdAt).getTime(),
   230→              kind: 'subagent_thought',
   231→              dispatchId: parentDispatch?.did,
   232→              subagentId: subId,
   233→              subagentName: parentDispatch?.subagentName,
   234→              subagentColor: parentDispatch?.subagentColor,
   235→              subagentIcon: parentDispatch?.subagentIcon,
   236→            })
   237→          } else {
   238→            stepCounter++
   239→            pendingSteps.push({
   240→              id: r.id,
   241→              stepNumber: stepCounter,
   242→              thought: r.content,
   243→              status: 'done',
   244→              startedAt: new Date(r.createdAt).getTime(),
   245→              finishedAt: new Date(r.createdAt).getTime(),
   246→              kind: 'super_thought',
   247→            })
   248→          }
   249→        } else if (r.role === 'tool') {
   250→          // Sub-agent dispatch?
   251→          if (r.toolName === 'subagent_dispatch') {
   252→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   253→            stepCounter++
   254→            const stepId = `dispatch_${meta.dispatchId ?? r.id}`
   255→            const stepInfo: any = {
   256→              subagentId: meta.agentId,
   257→              subagentName: meta.agentName,
   258→              subagentColor: meta.color,
   259→              subagentIcon: meta.icon,
   260→              task: meta.task,
   261→            }
   262→            if (meta.dispatchId) dispatchMap.set(meta.dispatchId, stepInfo)
   263→            pendingSteps.push({
   264→              id: stepId,
   265→              stepNumber: stepCounter,
   266→              subagentId: meta.agentId,
   267→              subagentName: meta.agentName,
   268→              subagentColor: meta.color,
   269→              subagentIcon: meta.icon,
   270→              subagentTask: meta.task,
   271→              dispatchId: meta.dispatchId,
   272→              status: 'done',
   273→              startedAt: new Date(r.createdAt).getTime(),
   274→              finishedAt: new Date(r.createdAt).getTime(),
   275→              kind: 'subagent_dispatch',
   276→              // If a subagent_complete row exists for the same dispatchId, the
   277→              // subagentAnswer will be filled below when we encounter that row.
   278→            })
   279→          } else if (r.toolName === 'subagent_complete') {
   280→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   281→            // Find the original dispatch step and set the answer
   282→            const dispatchStep = pendingSteps.find(
   283→              (st) => st.dispatchId === meta.dispatchId && st.kind === 'subagent_dispatch'
   284→            )
   285→            if (dispatchStep) {
   286→              dispatchStep.subagentAnswer = r.toolResult ?? ''
   287→              dispatchStep.status = 'done'
   288→              dispatchStep.finishedAt = new Date(r.createdAt).getTime()
   289→            }
   290→          } else if (r.toolName === 'subagent_tool') {
   291→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   292→            const parentDispatch = meta.dispatchId ? dispatchMap.get(meta.dispatchId) : null
   293→            stepCounter++
   294→            pendingSteps.push({
   295→              id: meta.stepId ?? r.id,
   296→              stepNumber: stepCounter,
   297→              toolName: meta.tool,
   298→              toolArgs: meta.args,
   299→              toolResult: r.toolResult ?? '',
   300→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   301→              toolOk: true,
   302→              status: 'done',
   303→              startedAt: new Date(r.createdAt).getTime(),
   304→              finishedAt: new Date(r.createdAt).getTime(),
   305→              kind: 'subagent_tool',
   306→              dispatchId: meta.dispatchId,
   307→              subagentId: parentDispatch?.subagentId ?? meta.agentId,
   308→              subagentName: parentDispatch?.subagentName,
   309→              subagentColor: parentDispatch?.subagentColor,
   310→              subagentIcon: parentDispatch?.subagentIcon,
   311→            })
   312→          } else {
   313→            // Regular super-agent tool
   314→            stepCounter++
   315→            pendingSteps.push({
   316→              id: r.id,
   317→              stepNumber: stepCounter,
   318→              toolName: r.toolName,
   319→              toolArgs: r.toolArgs ? safeParseJson(r.toolArgs) : undefined,
   320→              toolResult: r.toolResult ?? '',
   321→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   322→              toolOk: true,
   323→              status: 'done',
   324→              startedAt: new Date(r.createdAt).getTime(),
   325→              finishedAt: new Date(r.createdAt).getTime(),
   326→              kind: 'super_tool',
   327→            })
   328→          }
   329→        } else if (r.role === 'assistant') {
   330→          messages.push({
   331→            id: r.id,
   332→            role: 'assistant',
   333→            content: r.content,
   334→            steps: pendingSteps,
   335→            createdAt: new Date(r.createdAt).getTime(),
   336→          })
   337→          pendingSteps = []
   338→          stepCounter = 0
   339→          dispatchMap.clear()
   340→        }
   341→      }
   342→      set({ messages })
   343→    } catch (e) {
   344→      console.error('loadMessages', e)
   345→      set({ messages: [] })
   346→    }
   347→  },
   348→
   349→  clearMessages: () => set({ messages: [] }),
   350→
   351→  setLanguage: (l) => set({ language: l }),
   352→
   353→  addAttachment: (a) => set((s) => ({ attachments: [...s.attachments, a] })),
   354→  removeAttachment: (filename) =>
   355→    set((s) => ({ attachments: s.attachments.filter((a) => a.filename !== filename) })),
   356→  clearAttachments: () => set({ attachments: [] }),
   357→
   358→  loadMemories: async () => {
   359→    try {
   360→      const res = await fetch('/api/memory')
   361→      const data = await safeJson(res)
   362→      set({ memories: data.memories ?? [] })
   363→    } catch (e) {
   364→      console.error('loadMemories', e)
   365→    }
   366→  },
   367→
   368→  resetSubagentActivity: () => set({ activeSubagents: [], subagentActivity: {} }),
   369→
   370→  sendMessage: async (text) => {
   371→    const trimmed = text.trim()
   372→    if (!trimmed && get().attachments.length === 0) return
   373→    const state = get()
   374→    if (state.status !== 'idle') return
   375→
   376→    let conversationId = state.currentConversationId
   377→    if (!conversationId) {
   378→      conversationId = await get().createConversation()
   379→    }
   380→
   381→    const userMsg: ChatMessage = {
   382→      id: makeId('u'),
   383→      role: 'user',
   384→      content: trimmed,
   385→      attachments: state.attachments.length ? [...state.attachments] : undefined,
   386→      createdAt: Date.now(),
   387→    }
   388→    const assistantId = makeId('a')
   389→    const assistantMsg: ChatMessage = {
   390→      id: assistantId,
   391→      role: 'assistant',
   392→      content: '',
   393→      steps: [],
   394→      isStreaming: true,
   395→      createdAt: Date.now(),
   396→    }
   397→    set((s) => ({
   398→      messages: [...s.messages, userMsg, assistantMsg],
   399→      status: 'thinking',
   400→      currentTool: null,
   401→      attachments: [],
   402→      abortFlag: { current: false },
   403→      activeSubagents: [],
   404→      subagentActivity: {},
   405→    }))
   406→
   407→    const abortFlag = get().abortFlag
   408→    try {
   409→      const res = await fetch('/api/agent', {
   410→        method: 'POST',
   411→        headers: { 'Content-Type': 'application/json' },
   412→        body: JSON.stringify({
   413→          message: trimmed,
   414→          conversationId,
   415→          attachments: userMsg.attachments ?? [],
   416→          language: state.language,
   417→        }),
   418→      })
   419→      if (!res.ok || !res.body) {
   420→        const errText = await res.text()
   421→        throw new Error(errText || `HTTP ${res.status}`)
   422→      }
   423→
   424→      const reader = res.body.getReader()
   425→      const decoder = new TextDecoder()
   426→      let buffer = ''
   427→      // Track which step is "in progress" so we can mark it done
   428→      let currentStepId: string | null = null
   429→
   430→      while (true) {
   431→        if (abortFlag.current) break
   432→        const { done, value } = await reader.read()
   433→        if (done) break
   434→        buffer += decoder.decode(value, { stream: true })
   435→
   436→        // SSE messages separated by \n\n
   437→        let idx: number
   438→        while ((idx = buffer.indexOf('\n\n')) >= 0) {
   439→          const rawEvent = buffer.slice(0, idx)
   440→          buffer = buffer.slice(idx + 2)
   441→          const evt = parseSse(rawEvent)
   442→          if (!evt) continue
   443→          applyEvent(evt.event, evt.data, assistantId, set, get, () => {
   444→            currentStepId = evt.data?.stepId ?? currentStepId
   445→          })
   446→          if (evt.event === 'done' || evt.event === 'error') {
   447→            // finish
   448→          }
   449→        }
   450→      }
   451→
   452→      set((s) => ({
   453→        messages: s.messages.map((m) =>
   454→          m.id === assistantId ? { ...m, isStreaming: false } : m
   455→        ),
   456→        status: 'idle',
   457→        currentTool: null,
   458→        activeSubagents: [],
   459→      }))
   460→      // Refresh conversation list (title may have changed) + memories
   461→      get().loadConversations()
   462→      get().loadMemories()
   463→    } catch (e: any) {
   464→      console.error('sendMessage error', e)
   465→      set((s) => ({
   466→        messages: s.messages.map((m) =>
   467→          m.id === assistantId
   468→            ? {
   469→                ...m,
   470→                isStreaming: false,
   471→                content:
   472→                  m.content +
   473→                  (m.content ? '\n\n' : '') +
   474→                  `⚠️ **Error:** ${e?.message ?? String(e)}`,
   475→              }
   476→            : m
   477→        ),
   478→        status: 'idle',
   479→        currentTool: null,
   480→        activeSubagents: [],
   481→      }))
   482→    }
   483→  },
   484→
   485→  stopStreaming: () => {
   486→    get().abortFlag.current = true
   487→    set({ status: 'idle', currentTool: null })
   488→  },
   489→
   490→  toggleLeft: () => set((s) => ({ leftOpen: !s.leftOpen })),
   491→  toggleRight: () => set((s) => ({ rightOpen: !s.rightOpen })),
   492→  setLeft: (v) => set({ leftOpen: v }),
   493→  setRight: (v) => set({ rightOpen: v }),
   494→}))

   496→function safeParseJson(s: string): any {
   497→  try {
   498→    return JSON.parse(s)
   499→  } catch {
   500→    return s
   501→  }
   502→}
   503→function safeParseAttachments(s: string): AttachmentMeta[] | undefined {
   504→  try {
   505→    const a = JSON.parse(s)
   506→    if (Array.isArray(a)) return a
   507→  } catch {
   508→    /* ignore */
   509→  }
   510→  return undefined
   511→}
   512→
   513→function parseSse(raw: string): { event: string; data: any } | null {
   514→  const lines = raw.split('\n')
   515→  let event = 'message'
   516→  let data = ''
   517→  for (const ln of lines) {
   518→    if (ln.startsWith('event:')) event = ln.slice(6).trim()
   519→    else if (ln.startsWith('data:')) data += ln.slice(5).trim()
   520→  }
   521→  try {
   522→    return { event, data: data ? JSON.parse(data) : {} }
   523→  } catch {
   524→    return { event, data: { raw: data } }
   525→  }
   526→}

   528→function applyEvent(
   529→  event: string,
   530→  data: any,
   531→  assistantId: string,
   532→  set: (fn: (s: ChatState) => Partial<ChatState>) => void,
   533→  get: () => ChatState,
   534→  _track: () => void
   535→) {
   536→  if (event === 'thought') {
   537→    set((s) => ({
   538→      status: 'thinking',
   539→      messages: s.messages.map((m) => {
   540→        if (m.id !== assistantId) return m
   541→        const steps = [...(m.steps ?? [])]
   542→        // create a "thinking-only" step so users see the reasoning
   543→        const stepId = `thought_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`
   544→        steps.push({
   545→          id: stepId,
   546→          stepNumber: steps.length + 1,
   547→          thought: data.content,
   548→          status: 'done',
   549→          startedAt: Date.now(),
   550→          finishedAt: Date.now(),
   551→          kind: 'super_thought',
   552→        })
   553→        return { ...m, steps }
   554→      }),
   555→    }))
   556→  } else if (event === 'tool_call') {
   557→    const stepId = data.stepId
   558→    set((s) => ({
   559→      status: 'tool_running',
   560→      currentTool: data.name,
   561→      messages: s.messages.map((m) => {
   562→        if (m.id !== assistantId) return m
   563→        const steps = [...(m.steps ?? [])]
   564→        steps.push({
   565→          id: stepId,
   566→          stepNumber: data.stepNumber ?? steps.length + 1,
   567→          thought: data.thought,
   568→          toolName: data.name,
   569→          toolArgs: data.args,
   570→          status: 'running',
   571→          startedAt: Date.now(),
   572→          kind: 'super_tool',
   573→        })
   574→        return { ...m, steps }
   575→      }),
   576→    }))
   577→  } else if (event === 'tool_result') {
   578→    const stepId = data.stepId
   579→    set((s) => ({
   580→      status: 'thinking',
   581→      currentTool: null,
   582→      messages: s.messages.map((m) => {
   583→        if (m.id !== assistantId) return m
   584→        const steps = (m.steps ?? []).map((st) =>
   585→          st.id === stepId
   586→            ? {
   587→                ...st,
   588→                status: data.ok === false ? 'error' : 'done',
   589→                toolResult: data.result,
   590→                toolPreview: data.preview,
   591→                toolOk: data.ok,
   592→                artifacts: data.artifacts,
   593→                finishedAt: Date.now(),
   594→              }
   595→            : st
   596→        )
   597→        return { ...m, steps }
   598→      }),
   599→    }))
   600→  } else if (event === 'subagent_dispatch') {
   601→    const dispatchId = data.dispatchId
   602→    const agentId = data.agentId
   603→    set((s) => {
   604→      const activeSubagents = s.activeSubagents.includes(dispatchId)
   605→        ? s.activeSubagents
   606→        : [...s.activeSubagents, dispatchId]
   607→      const subagentActivity = { ...s.subagentActivity, [agentId]: 'working' as const }
   608→      return {
   609→        status: 'thinking',
   610→        activeSubagents,
   611→        subagentActivity,
   612→        messages: s.messages.map((m) => {
   613→          if (m.id !== assistantId) return m
   614→          const steps = [...(m.steps ?? [])]
   615→          const stepId = `dispatch_${dispatchId}`
   616→          steps.push({
   617→            id: stepId,
   618→            stepNumber: data.stepNumber ?? steps.length + 1,
   619→            subagentId: agentId,
   620→            subagentName: data.agentName,
   621→            subagentColor: data.color,
   622→            subagentIcon: data.icon,
   623→            subagentTask: data.task,
   624→            dispatchId,
   625→            status: 'running',
   626→            startedAt: Date.now(),
   627→            kind: 'subagent_dispatch',
   628→          })
   629→          return { ...m, steps }
   630→        }),
   631→      }
   632→    })
   633→  } else if (event === 'subagent_thought') {
   634→    const dispatchId = data.dispatchId
   635→    set((s) => ({
   636→      messages: s.messages.map((m) => {
   637→        if (m.id !== assistantId) return m
   638→        const steps = [...(m.steps ?? [])]
   639→        // Find the dispatch step for this dispatchId to get sub-agent info
   640→        const dispatchStep = steps.find(
   641→          (st) => st.dispatchId === dispatchId && st.kind === 'subagent_dispatch'
   642→        )
   643→        const stepId = `subthought_${dispatchId}_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`
   644→        steps.push({
   645→          id: stepId,
   646→          stepNumber: steps.length + 1,
   647→          thought: data.content,
   648→          status: 'done',
   649→          startedAt: Date.now(),
   650→          finishedAt: Date.now(),
   651→          kind: 'subagent_thought',
   652→          dispatchId,
   653→          subagentId: dispatchStep?.subagentId,
   654→          subagentName: dispatchStep?.subagentName,
   655→          subagentColor: dispatchStep?.subagentColor,
   656→          subagentIcon: dispatchStep?.subagentIcon,
   657→        })
   658→        return { ...m, steps }
   659→      }),
   660→    }))
   661→  } else if (event === 'subagent_tool_call') {
   662→    const dispatchId = data.dispatchId
   663→    const stepId = data.stepId
   664→    set((s) => ({
   665→      status: 'tool_running',
   666→      currentTool: data.name,
   667→      messages: s.messages.map((m) => {
   668→        if (m.id !== assistantId) return m
   669→        const steps = [...(m.steps ?? [])]
   670→        const dispatchStep = steps.find(
   671→          (st) => st.dispatchId === dispatchId && st.kind === 'subagent_dispatch'
   672→        )
   673→        steps.push({
   674→          id: stepId,
   675→          stepNumber: data.stepNumber ?? steps.length + 1,
   676→          thought: data.thought,
   677→          toolName: data.name,
   678→          toolArgs: data.args,
   679→          status: 'running',
   680→          startedAt: Date.now(),
   681→          kind: 'subagent_tool',
   682→          dispatchId,
   683→          subagentId: dispatchStep?.subagentId,
   684→          subagentName: dispatchStep?.subagentName,
   685→          subagentColor: dispatchStep?.subagentColor,
   686→          subagentIcon: dispatchStep?.subagentIcon,
   687→        })
   688→        return { ...m, steps }
   689→      }),
   690→    }))
   691→  } else if (event === 'subagent_tool_result') {
   692→    const stepId = data.stepId
   693→    set((s) => ({
   694→      status: 'thinking',
   695→      currentTool: null,
   696→      messages: s.messages.map((m) => {
   697→        if (m.id !== assistantId) return m
   698→        const steps = (m.steps ?? []).map((st) =>
   699→          st.id === stepId
   700→            ? {
   701→                ...st,
   702→                status: data.ok === false ? 'error' : 'done',
   703→                toolResult: data.result,
   704→                toolPreview: data.preview,
   705→                toolOk: data.ok,
   706→                artifacts: data.artifacts,
   707→                finishedAt: Date.now(),
   708→              }
   709→            : st
   710→        )
   711→        return { ...m, steps }
   712→      }),
   713→    }))
   714→  } else if (event === 'subagent_complete') {
   715→    const dispatchId = data.dispatchId
   716→    set((s) => {
   717→      const activeSubagents = s.activeSubagents.filter((id) => id !== dispatchId)
   718→      // Mark the sub-agent activity as done (find which agent this dispatchId was for)
   719→      const messages = s.messages.map((m) => {
   720→        if (m.id !== assistantId) return m
   721→        const steps = (m.steps ?? []).map((st) => {
   722→          if (st.dispatchId === dispatchId && st.kind === 'subagent_dispatch') {
   723→            const subagentActivity = { ...s.subagentActivity }
   724→            if (st.subagentId) subagentActivity[st.subagentId] = 'done'
   725→            // Update via the outer set below (this closure can't return both)
   726→            return {
   727→              ...st,
   728→              status: 'done' as const,
   729→              subagentAnswer: data.answer,
   730→              finishedAt: Date.now(),
   731→            }
   732→          }
   733→          return st
   734→        })
   735→        return { ...m, steps }
   736→      })
   737→      // Determine subagent id for activity update
   738→      let agentId: string | undefined
   739→      for (const m of s.messages) {
   740→        if (m.id !== assistantId) continue
   741→        const ds = (m.steps ?? []).find(
   742→          (st) => st.dispatchId === dispatchId && st.kind === 'subagent_dispatch'
   743→        )
   744→        if (ds?.subagentId) {
   745→          agentId = ds.subagentId
   746→          break
   747→        }
   748→      }
   749→      const subagentActivity = { ...s.subagentActivity }
   750→      if (agentId) subagentActivity[agentId] = 'done'
   751→      return { activeSubagents, subagentActivity, messages }
   752→    })
   753→  } else if (event === 'synthesis') {
   754→    set((s) => ({
   755→      status: 'streaming',
   756→      messages: s.messages.map((m) => {
   757→        if (m.id !== assistantId) return m
   758→        const steps = [...(m.steps ?? [])]
   759→        // Push a "synthesizing" marker step (only if not already present)
   760→        const hasSynth = steps.some((st) => st.kind === ('super_thought' as any) && st.thought === '__synthesizing__')
   761→        if (!hasSynth) {
   762→          steps.push({
   763→            id: `synth_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
   764→            stepNumber: steps.length + 1,
   765→            thought: '__synthesizing__',
   766→            status: 'thinking',
   767→            startedAt: Date.now(),
   768→            kind: 'super_thought',
   769→          })
   770→        }
   771→        return { ...m, steps }
   772→      }),
   773→    }))
   774→  } else if (event === 'token') {
   775→    set((s) => ({
   776→      status: 'streaming',
   777→      messages: s.messages.map((m) => {
   778→        if (m.id !== assistantId) return m
   779→        // Drop the synthesizing marker step if it exists, since real tokens are now arriving
   780→        const steps = (m.steps ?? []).filter(
   781→          (st) => !(st.kind === 'super_thought' && st.thought === '__synthesizing__')
   782→        )
   783→        return { ...m, content: m.content + (data.content ?? ''), steps }
   784→      }),
   785→    }))
   786→  } else if (event === 'memory_update') {
   787→    // Optimistically add/update memory in the right panel
   788→    set((s) => {
   789→      const exists = s.memories.find((m) => m.key === data.key)
   790→      const item: MemoryItem = {
   791→        id: exists?.id ?? `tmp_${Date.now()}`,
   792→        key: data.key,
   793→        value: data.value,
   794→        category: data.category ?? 'general',
   795→        updatedAt: new Date().toISOString(),
   796→      }
   797→      const memories = exists
   798→        ? s.memories.map((m) => (m.key === data.key ? item : m))
   799→        : [item, ...s.memories]
   800→      return { memories }
   801→    })
   802→  } else if (event === 'error') {
   803→    set((s) => ({
   804→      messages: s.messages.map((m) =>
   805→        m.id === assistantId
   806→          ? {
   807→              ...m,
   808→              content:
   809→                m.content +
   810→                (m.content ? '\n\n' : '') +
   811→                `⚠️ **Error:** ${data.message ?? 'unknown error'}`,
   812→              isStreaming: false,
   813→            }
   814→          : m
   815→      ),
   816→      status: 'idle',
   817→      currentTool: null,
   818→      activeSubagents: [],
   819→    }))
   820→  }
   821→  // 'done' event: nothing special; outer loop will set isStreaming=false
   822→}
