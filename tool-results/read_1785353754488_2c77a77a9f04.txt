   177→let msgIdCounter = 0

   183→async function safeJson(res: Response): Promise<any> {
   184→  const text = await res.text()
   185→  try {
   186→    return JSON.parse(text)
   187→  } catch {
   188→    return { error: text || 'parse error' }
   189→  }
   190→}

   222→        console.warn('[loadConversations] API returned', res.status)

   238→        try { localStorage.setItem('agent007_conversations', JSON.stringify(convs)) } catch {}
   239→      }
   240→    } catch (e) {
   241→      console.error('loadConversations', e)
   242→      // Fallback: localStorage ONLY if API is unreachable (offline mode)
   243→      if (typeof window !== 'undefined') {
   244→        try {
   245→          const saved = localStorage.getItem('agent007_conversations')
   246→          if (saved) set({ conversations: JSON.parse(saved) })

   249→    }

   295→  loadMessages: async (conversationId) => {
   296→    try {
   297→      const res = await fetch(`/api/conversations/${conversationId}`)
   298→      if (!res.ok) {
   299→        console.warn('[loadMessages] API returned', res.status)
   300→        set({ messages: [] })
   301→        return
   302→      }
   303→      const data = await safeJson(res)
   304→      const conv = data.conversation
   305→      if (!conv || !conv.Message || conv.Message.length === 0) {
   306→        // UPGRADE #70 — DB is the ONLY source of truth. No localStorage fallback for messages.
   307→        // Every device sees the SAME messages from Postgres.
   308→        set({ messages: [] })
   309→        return
   310→      }
   311→      // Reconstruct messages from DB rows: collapse tool/thought rows under the *next* assistant message
   312→      const rows: any[] = conv.Message ?? []
   313→      const messages: ChatMessage[] = []
   314→      let pendingSteps: ToolStep[] = []
   315→      // Track open dispatches so subsequent subagent rows can be linked.
   316→      // Map dispatchId → { subagentId, subagentName, subagentColor, subagentIcon, task, stepId }
   317→      const dispatchMap = new Map<string, any>()
   318→      let stepCounter = 0
   319→      for (const r of rows) {
   320→        if (r.role === 'user') {
   321→          const atts = r.attachments ? safeParseAttachments(r.attachments) : undefined
   322→          messages.push({
   323→            id: r.id,
   324→            role: 'user',
   325→            content: r.content,
   326→            attachments: atts,
   327→            createdAt: new Date(r.createdAt).getTime(),
   328→          })
   329→        } else if (r.role === 'thought') {
   330→          // Check if this is a sub-agent thought (prefixed with [subagent:id])
   331→          const subMatch = r.content?.match(/^\[subagent:([^\]]+)\]\s*([\s\S]*)$/)
   332→          if (subMatch) {
   333→            const subId = subMatch[1]
   334→            // Find the most recent dispatch for this sub-agent
   335→            let parentDispatch: any = null
   336→            for (const [did, info] of dispatchMap.entries()) {
   337→              if (info.subagentId === subId) parentDispatch = { did, ...info }
   338→            }
   339→            stepCounter++
   340→            pendingSteps.push({
   341→              id: r.id,
   342→              stepNumber: stepCounter,
   343→              thought: subMatch[2],
   344→              status: 'done',
   345→              startedAt: new Date(r.createdAt).getTime(),
   346→              finishedAt: new Date(r.createdAt).getTime(),
   347→              kind: 'subagent_thought',
   348→              dispatchId: parentDispatch?.did,
   349→              subagentId: subId,
   350→              subagentName: parentDispatch?.subagentName,
   351→              subagentColor: parentDispatch?.subagentColor,
   352→              subagentIcon: parentDispatch?.subagentIcon,
   353→            })
   354→          } else {
   355→            stepCounter++
   356→            pendingSteps.push({
   357→              id: r.id,
   358→              stepNumber: stepCounter,
   359→              thought: r.content,
   360→              status: 'done',
   361→              startedAt: new Date(r.createdAt).getTime(),
   362→              finishedAt: new Date(r.createdAt).getTime(),
   363→              kind: 'super_thought',
   364→            })
   365→          }
   366→        } else if (r.role === 'tool') {
   367→          // Sub-agent dispatch?
   368→          if (r.toolName === 'subagent_dispatch') {
   369→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   370→            stepCounter++
   371→            const stepId = `dispatch_${meta.dispatchId ?? r.id}`
   372→            const stepInfo: any = {
   373→              subagentId: meta.agentId,
   374→              subagentName: meta.agentName,
   375→              subagentColor: meta.color,
   376→              subagentIcon: meta.icon,
   377→              task: meta.task,
   378→            }
   379→            if (meta.dispatchId) dispatchMap.set(meta.dispatchId, stepInfo)
   380→            pendingSteps.push({
   381→              id: stepId,
   382→              stepNumber: stepCounter,
   383→              subagentId: meta.agentId,
   384→              subagentName: meta.agentName,
   385→              subagentColor: meta.color,
   386→              subagentIcon: meta.icon,
   387→              subagentTask: meta.task,
   388→              dispatchId: meta.dispatchId,
   389→              status: 'done',
   390→              startedAt: new Date(r.createdAt).getTime(),
   391→              finishedAt: new Date(r.createdAt).getTime(),
   392→              kind: 'subagent_dispatch',
   393→              // If a subagent_complete row exists for the same dispatchId, the
   394→              // subagentAnswer will be filled below when we encounter that row.
   395→            })
   396→          } else if (r.toolName === 'subagent_complete') {
   397→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   398→            // Find the original dispatch step and set the answer
   399→            const dispatchStep = pendingSteps.find(
   400→              (st) => st.dispatchId === meta.dispatchId && st.kind === 'subagent_dispatch'
   401→            )
   402→            if (dispatchStep) {
   403→              dispatchStep.subagentAnswer = r.toolResult ?? ''
   404→              dispatchStep.status = 'done'
   405→              dispatchStep.finishedAt = new Date(r.createdAt).getTime()
   406→            }
   407→          } else if (r.toolName === 'subagent_tool') {
   408→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   409→            const parentDispatch = meta.dispatchId ? dispatchMap.get(meta.dispatchId) : null
   410→            stepCounter++
   411→            pendingSteps.push({
   412→              id: meta.stepId ?? r.id,
   413→              stepNumber: stepCounter,
   414→              toolName: meta.tool,
   415→              toolArgs: meta.args,
   416→              toolResult: r.toolResult ?? '',
   417→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   418→              toolOk: true,
   419→              status: 'done',
   420→              startedAt: new Date(r.createdAt).getTime(),
   421→              finishedAt: new Date(r.createdAt).getTime(),
   422→              kind: 'subagent_tool',
   423→              dispatchId: meta.dispatchId,
   424→              subagentId: parentDispatch?.subagentId ?? meta.agentId,
   425→              subagentName: parentDispatch?.subagentName,
   426→              subagentColor: parentDispatch?.subagentColor,
   427→              subagentIcon: parentDispatch?.subagentIcon,
   428→            })
   429→          } else if (r.toolName === 'manage_action') {
   430→            // Reconstruct manage_action step from persisted row
   431→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   432→            stepCounter++
   433→            pendingSteps.push({
   434→              id: r.id,
   435→              stepNumber: stepCounter,
   436→              status: 'done',
   437→              startedAt: new Date(r.createdAt).getTime(),
   438→              finishedAt: new Date(r.createdAt).getTime(),
   439→              kind: 'manage_action',
   440→              manageAction: meta.action,
   441→              manageAttrs: meta.attrs,
   442→              manageResult: { ok: !/failed|error/i.test(r.toolResult ?? ''), message: r.toolResult ?? '' },
   443→            })
   444→          } else {
   445→            // Regular super-agent tool
   446→            stepCounter++
   447→            pendingSteps.push({
   448→              id: r.id,
   449→              stepNumber: stepCounter,
   450→              toolName: r.toolName,
   451→              toolArgs: r.toolArgs ? safeParseJson(r.toolArgs) : undefined,
   452→              toolResult: r.toolResult ?? '',
   453→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   454→              toolOk: true,
   455→              status: 'done',
   456→              startedAt: new Date(r.createdAt).getTime(),
   457→              finishedAt: new Date(r.createdAt).getTime(),
   458→              kind: 'super_tool',
   459→            })
   460→          }
   461→        } else if (r.role === 'assistant') {
   462→          messages.push({
   463→            id: r.id,
   464→            role: 'assistant',
   465→            content: r.content,
   466→            steps: pendingSteps,
   467→            createdAt: new Date(r.createdAt).getTime(),
   468→          })
   469→          pendingSteps = []
   470→          stepCounter = 0
   471→          dispatchMap.clear()
   472→        }
   473→      }
   474→      set({ messages })
   475→    } catch (e) {
   476→      console.error('loadMessages', e)
   477→      set({ messages: [] })
   478→    }
   479→  },

   514→  sendMessage: async (text) => {
   515→    const trimmed = text.trim()
   516→    if (!trimmed && get().attachments.length === 0) return
   517→    const state = get()
   518→    if (state.status !== 'idle') return
   519→
   520→    let conversationId = state.currentConversationId
   521→    if (!conversationId) {
   522→      conversationId = await get().createConversation()
   523→    }
   524→
   525→    const userMsg: ChatMessage = {
   526→      id: makeId('u'),
   527→      role: 'user',
   528→      content: trimmed,
   529→      attachments: state.attachments.length ? [...state.attachments] : undefined,
   530→      createdAt: Date.now(),
   531→    }
   532→    const assistantId = makeId('a')
   533→    const assistantMsg: ChatMessage = {
   534→      id: assistantId,
   535→      role: 'assistant',
   536→      content: '',
   537→      steps: [],
   538→      isStreaming: true,
   539→      createdAt: Date.now(),
   540→    }
   541→    set((s) => {
   542→      const newMessages = [...s.messages, userMsg, assistantMsg]
   543→      // UPGRADE #125 — Rec 2C: Skip localStorage write during streaming (debounced to done/error)
   544→      // Previously wrote to localStorage on EVERY token, causing synchronous blocking
   545→      if (typeof window !== 'undefined') {
   546→        try {
   547→          const convId = s.currentConversationId
   548→          // Only update conversation title in localStorage (lightweight)
   549→          const saved = localStorage.getItem('agent007_conversations')
   550→          if (saved) {
   551→            const convs = JSON.parse(saved)
   552→            const idx = convs.findIndex((c: any) => c.id === convId)
   553→            if (idx >= 0) {
   554→              convs[idx].title = trimmed.slice(0, 50)
   555→              convs[idx].updatedAt = new Date().toISOString()
   556→              localStorage.setItem('agent007_conversations', JSON.stringify(convs))
   557→            }
   558→          }
   559→        } catch {}
   560→      }
   561→      return {
   562→        messages: newMessages,
   563→        status: 'thinking',
   564→        currentTool: null,
   565→        attachments: [],
   566→        abortFlag: { current: false },
   567→        activeSubagents: [],
   568→        subagentActivity: {},
   569→      }
   570→    })
   571→
   572→    const abortFlag = get().abortFlag
   573→    try {
   574→      // UPGRADE #161: Increased to 180s (server maxDuration is now 300s on Vercel Pro).
   575→      // Before: 90s timeout (server was 60s). After: 180s gives the agent
   576→      // enough time to complete complex missions with subagent dispatches.
   577→      const res = await fetch('/api/agent', {
   578→        method: 'POST',
   579→        headers: { 'Content-Type': 'application/json' },

   623→            // needs to flush in case the 'done' event and the stream close

   837→function parseSse(raw: string): { event: string; data: any } | null {
   838→  const lines = raw.split('\n')
   839→  let event = 'message'
   840→  let data = ''
   841→  for (const ln of lines) {
   842→    if (ln.startsWith('event:')) event = ln.slice(6).trim()
   843→    else if (ln.startsWith('data:')) data += ln.slice(5).trim()
   844→  }
   845→  try {
   846→    return { event, data: data ? JSON.parse(data) : {} }
   847→  } catch {
   848→    return { event, data: { raw: data } }
   849→  }
   850→}

  1125→  } else if (event === 'token') {
  1126→    // UPGRADE #125 — Rec 2A: Batch token updates (throttle to 100ms)
  1127→    // Instead of updating state on EVERY token (which causes re-renders 100+ times/sec),
  1128→    // accumulate tokens in a buffer and flush to state every 100ms.
  1129→    _pendingTokens += (data.content ?? '')
  1130→    if (!_tokenFlushTimer) {
  1131→      _tokenFlushTimer = setTimeout(() => {
  1132→        const tokensToAdd = _pendingTokens
  1133→        _pendingTokens = ''
  1134→        _tokenFlushTimer = null
  1135→        if (tokensToAdd) {
  1136→          set((s) => ({
  1137→            status: 'streaming',
  1138→            messages: s.messages.map((m) => {
  1139→              if (m.id !== assistantId) return m
  1140→              const steps = (m.steps ?? []).filter(
  1141→                (st) => !(st.kind === 'super_thought' && st.thought === '__synthesizing__')
  1142→              )
  1143→              return { ...m, content: m.content + tokensToAdd, steps }
  1144→            }),
  1145→          }))
  1146→        }
  1147→      }, 100)
  1148→    }
  1149→  } else if (event === 'memory_update') {
  1150→    // Optimistically add/update memory in the right panel
  1151→    set((s) => {
  1152→      const exists = s.memories.find((m) => m.key === data.key)
  1153→      const item: MemoryItem = {
  1154→        id: exists?.id ?? `tmp_${Date.now()}`,
  1155→        key: data.key,
  1156→        value: data.value,
  1157→        category: data.category ?? 'general',
  1158→        updatedAt: new Date().toISOString(),
  1159→      }
  1160→      const memories = exists
  1161→        ? s.memories.map((m) => (m.key === data.key ? item : m))
  1162→        : [item, ...s.memories]
  1163→      return { memories }
  1164→    })
  1165→  } else if (event === 'manage_action') {
  1166→    // Two-phase event: status='running' pushes a new step; status='done'|'error'
  1167→    // updates that step with the result.
  1168→    const stepId = data.stepId
  1169→    if (data.status === 'running') {
  1170→      set((s) => ({
  1171→        status: 'tool_running',
  1172→        currentTool: `manage:${data.action}`,
  1173→        messages: s.messages.map((m) => {
  1174→          if (m.id !== assistantId) return m
  1175→          const steps = [...(m.steps ?? [])]
  1176→          steps.push({
  1177→            id: stepId,
  1178→            stepNumber: data.stepNumber ?? steps.length + 1,
  1179→            thought: data.thought,
  1180→            status: 'running',
  1181→            startedAt: Date.now(),
  1182→            kind: 'manage_action',
  1183→            manageAction: data.action,
  1184→            manageAttrs: data.attrs,
  1185→          })
  1186→          return { ...m, steps }
  1187→        }),
  1188→      }))
  1189→    } else {
  1190→      // done or error
  1191→      set((s) => ({
  1192→        status: 'thinking',
  1193→        currentTool: null,
  1194→        messages: s.messages.map((m) => {
  1195→          if (m.id !== assistantId) return m
  1196→          const steps = (m.steps ?? []).map((st): ToolStep =>
  1197→            st.id === stepId
  1198→              ? {
  1199→                  ...st,
  1200→                  status: data.status === 'error' ? 'error' : 'done',
  1201→                  manageResult: data.result,
  1202→                  finishedAt: Date.now(),
  1203→                }
  1204→              : st
  1205→          )
  1206→          return { ...m, steps }
  1207→        }),
  1208→      }))
  1209→    }
  1210→  } else if (event === 'subagents_updated') {
  1211→    // The orchestrator created/edited/deleted/toggled a sub-agent. Bump the
  1212→    // version so any mounted Sub-Agents panel re-fetches the list. Also
  1213→    // refresh memories + conversations (cheap) so the rest of the UI is fresh.
  1214→    set((s) => ({ subagentsVersion: s.subagentsVersion + 1 }))
  1215→  } else if (event === 'heartbeat') {
  1216→    // UPGRADE #63 — Update heartbeat state so the dashboard shows real-time progress
  1217→    set((s) => ({ heartbeat: data }))
  1218→  } else if (event === 'error') {
  1219→    // UPGRADE #128: Flush any pending tokens BEFORE showing the error
  1220→    // (UPGRADE #125 token batching could swallow content if error arrives
  1221→    // before the 100ms flush timer fires)
  1222→    const pendingContent = _pendingTokens
  1223→    _pendingTokens = ''
  1224→    if (_tokenFlushTimer) {
  1225→      clearTimeout(_tokenFlushTimer)
  1226→      _tokenFlushTimer = null
  1227→    }
  1228→
  1229→    const msg: string = (data?.message ?? '').toString()
  1230→    // UPGRADE #131: Fixed regex — was matching "rate limit" in ALL error messages
  1231→    // because the error text from agent.ts says "Rate limit reached on all active providers"
  1232→    // even when the actual cause is a DB failure or network timeout.
  1233→    // NOW: only treat as rate limit if the message explicitly says "rate-limiting" (with hyphen)
  1234→    // or contains "429" or "too many requests" — NOT the generic "rate limit" phrase.
  1235→    // UPGRADE #131: Stop detecting rate limits from message text entirely.
  1236→    // The error message from friendlyLlmError() was containing trigger words
  1237→    // that the regex was matching — creating a circular reference where the
  1238→    // fix caused the bug. NOW: only show the rate-limit banner if the server
  1239→    // explicitly sends { rateLimited: true } in the error data.
  1240→    const isRateLimit = data?.rateLimited === true
  1241→    const isDBError = /database|prisma|Can't reach|connection|ECONNREFUSED|ETIMEDOUT/i.test(msg)
  1242→    const isNetworkError = /fetch failed|network|timeout|ECONNRESET|socket hang up|aborted/i.test(msg)
  1243→
  1244→    // UPGRADE #149 — Use the friendly message from friendlyLlmError() directly.
  1245→    // Before: the chat-store OVERWROTE the detailed server message with a generic
  1246→    // "A provider is at capacity" banner. This hid the failure breakdown that
  1247→    // the new callLlmWithRetry produces (which provider 429'd, which had a
  1248→    // network error, etc.). After: if the server sent a message, show it
  1249→    // verbatim — it already contains the actionable diagnostic info.
  1250→    let userMessage = data.message ?? 'unknown error'
  1251→    if (isDBError && !data.message) {
  1252→      userMessage = '⚠️ The database is temporarily unreachable (Vercel cold start). Please wait 10 seconds and click Retry below.'
  1253→    } else if (isNetworkError && !data.message) {
  1254→      userMessage = '⚠️ Network error — a provider was temporarily unreachable. Please click Retry below.'
  1255→    } else if (isRateLimit && !data.message) {
  1256→      // Only use the generic message if the server didn't send one
  1257→      userMessage = '⏳ A provider is at capacity. Please wait 30 seconds and click Retry below.'
  1258→    } else if (!data.message) {
  1259→      // Generic error — show the actual error text
  1260→      userMessage = `⚠️ ${msg.slice(0, 300)}\n\nThis may be a temporary issue. Click Retry below to try again.`
  1261→    }
  1262→    // If data.message exists, we keep it as-is (it has the full breakdown)
  1263→
  1264→    set((s) => ({
  1265→      messages: s.messages.map((m) =>
  1266→        m.id === assistantId
  1267→          ? {
  1268→              ...m,
  1269→              content:
  1270→                (m.content || '') +
  1271→                (pendingContent ? pendingContent : '') +
  1272→                (m.content || pendingContent ? '\n\n' : '') +
  1273→                `⚠️ **Error:** ${userMessage}`,
  1274→              isStreaming: false,
  1275→            }
  1276→          : m
  1277→      ),
  1278→      status: 'idle',
  1279→      currentTool: null,
  1280→      heartbeat: null,
  1281→      activeSubagents: [],
  1282→      rateLimitedUntil: isRateLimit ? Date.now() + 30_000 : s.rateLimitedUntil,
  1283→    }))
  1284→  }
  1285→  // UPGRADE #154: 'done' event — flush any pending tokens IMMEDIATELY.
  1286→  // Before: the 'done' handler did nothing, relying on the outer loop to
  1287→  // flush tokens. But if the 'done' event and stream close arrived in the
  1288→  // same chunk, the 100ms token flush timer might not fire before the UI
  1289→  // stopped rendering — causing incomplete responses.
  1290→  // After: flush ALL pending tokens synchronously when 'done' arrives.
  1291→  else if (event === 'done') {
  1292→    const pendingContent = _pendingTokens
  1293→    _pendingTokens = ''
  1294→    if (_tokenFlushTimer) {
  1295→      clearTimeout(_tokenFlushTimer)
  1296→      _tokenFlushTimer = null
  1297→    }
  1298→    if (pendingContent) {
  1299→      set((s) => ({
  1300→        messages: s.messages.map((m) => {
  1301→          if (m.id !== assistantId) return m
  1302→          const steps = (m.steps ?? []).filter(
  1303→            (st) => !(st.kind === 'super_thought' && st.thought === '__synthesizing__')
  1304→          )
  1305→          return { ...m, content: m.content + pendingContent, steps }
  1306→        }),
  1307→      }))
  1308→    }
  1309→  }
  1310→}
