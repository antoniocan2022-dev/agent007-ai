   138→export const useChatStore = create<ChatState>((set, get) => ({
   139→  conversations: [],
   140→  currentConversationId: null,
   141→  messages: [],

   144→  status: 'idle',

   146→  memories: [],
   147→  activeSubagents: [],
   148→  subagentActivity: {},
   149→  leftOpen: true,
   150→  rightOpen: true,
   151→  abortFlag: { current: false },
   152→  activeTab: 'chat',
   153→  changePasswordOpen: false,
   154→
   155→  loadConversations: async () => {
   156→    try {
   157→      const res = await fetch('/api/conversations')
   158→      const data = await safeJson(res)
   159→      set({ conversations: data.conversations ?? [] })
   160→    } catch (e) {
   161→      console.error('loadConversations', e)
   162→    }
   163→  },

   166→    const res = await fetch('/api/conversations', {
   167→      method: 'POST',
   168→      headers: { 'Content-Type': 'application/json' },
   169→      body: JSON.stringify({ title: 'New Conversation' }),
   170→    })
   171→    const data = await safeJson(res)
   172→    const conv: ConversationMeta = data.conversation
   173→    set((s) => ({
   174→      conversations: [conv, ...s.conversations],
   175→      currentConversationId: conv.id,
   176→      messages: [],
   177→    }))
   178→    return conv.id
   179→  },
   180→
   181→  selectConversation: async (id) => {
   182→    set({ currentConversationId: id, messages: [] })
   183→    await get().loadMessages(id)
   184→  },
   185→
   186→  deleteConversation: async (id) => {
   187→    await fetch(`/api/conversations/${id}`, { method: 'DELETE' })
   188→    set((s) => ({
   189→      conversations: s.conversations.filter((c) => c.id !== id),
   190→      currentConversationId: s.currentConversationId === id ? null : s.currentConversationId,
   191→      messages: s.currentConversationId === id ? [] : s.messages,
   192→    }))
   193→  },
   194→
   195→  loadMessages: async (conversationId) => {
   196→    try {
   197→      const res = await fetch(`/api/conversations/${conversationId}`)
   198→      const data = await safeJson(res)
   199→      const conv = data.conversation
   200→      if (!conv) {
   201→        set({ messages: [] })
   202→        return
   203→      }
   204→      // Reconstruct messages from DB rows: collapse tool/thought rows under the *next* assistant message
   205→      const rows: any[] = conv.messages ?? []
   206→      const messages: ChatMessage[] = []
   207→      let pendingSteps: ToolStep[] = []
   208→      // Track open dispatches so subsequent subagent rows can be linked.
   209→      // Map dispatchId → { subagentId, subagentName, subagentColor, subagentIcon, task, stepId }
   210→      const dispatchMap = new Map<string, any>()
   211→      let stepCounter = 0
   212→      for (const r of rows) {
   213→        if (r.role === 'user') {
   214→          const atts = r.attachments ? safeParseAttachments(r.attachments) : undefined
   215→          messages.push({
   216→            id: r.id,
   217→            role: 'user',
   218→            content: r.content,
   219→            attachments: atts,
   220→            createdAt: new Date(r.createdAt).getTime(),
   221→          })
   222→        } else if (r.role === 'thought') {
   223→          // Check if this is a sub-agent thought (prefixed with [subagent:id])
   224→          const subMatch = r.content?.match(/^\[subagent:([^\]]+)\]\s*(.*)$/s)
   225→          if (subMatch) {
   226→            const subId = subMatch[1]
   227→            // Find the most recent dispatch for this sub-agent
   228→            let parentDispatch: any = null
   229→            for (const [did, info] of dispatchMap.entries()) {
   230→              if (info.subagentId === subId) parentDispatch = { did, ...info }
   231→            }
   232→            stepCounter++
   233→            pendingSteps.push({
   234→              id: r.id,
   235→              stepNumber: stepCounter,
   236→              thought: subMatch[2],
   237→              status: 'done',
   238→              startedAt: new Date(r.createdAt).getTime(),
   239→              finishedAt: new Date(r.createdAt).getTime(),
   240→              kind: 'subagent_thought',
   241→              dispatchId: parentDispatch?.did,
   242→              subagentId: subId,
   243→              subagentName: parentDispatch?.subagentName,
   244→              subagentColor: parentDispatch?.subagentColor,
   245→              subagentIcon: parentDispatch?.subagentIcon,
   246→            })
   247→          } else {
   248→            stepCounter++
   249→            pendingSteps.push({
   250→              id: r.id,
   251→              stepNumber: stepCounter,
   252→              thought: r.content,
   253→              status: 'done',
   254→              startedAt: new Date(r.createdAt).getTime(),
   255→              finishedAt: new Date(r.createdAt).getTime(),
   256→              kind: 'super_thought',
   257→            })
   258→          }
   259→        } else if (r.role === 'tool') {
   260→          // Sub-agent dispatch?
   261→          if (r.toolName === 'subagent_dispatch') {
   262→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   263→            stepCounter++
   264→            const stepId = `dispatch_${meta.dispatchId ?? r.id}`
   265→            const stepInfo: any = {
   266→              subagentId: meta.agentId,
   267→              subagentName: meta.agentName,
   268→              subagentColor: meta.color,
   269→              subagentIcon: meta.icon,
   270→              task: meta.task,
   271→            }
   272→            if (meta.dispatchId) dispatchMap.set(meta.dispatchId, stepInfo)
   273→            pendingSteps.push({
   274→              id: stepId,
   275→              stepNumber: stepCounter,
   276→              subagentId: meta.agentId,
   277→              subagentName: meta.agentName,
   278→              subagentColor: meta.color,
   279→              subagentIcon: meta.icon,
   280→              subagentTask: meta.task,
   281→              dispatchId: meta.dispatchId,
   282→              status: 'done',
   283→              startedAt: new Date(r.createdAt).getTime(),
   284→              finishedAt: new Date(r.createdAt).getTime(),
   285→              kind: 'subagent_dispatch',
   286→              // If a subagent_complete row exists for the same dispatchId, the
   287→              // subagentAnswer will be filled below when we encounter that row.
   288→            })
   289→          } else if (r.toolName === 'subagent_complete') {
   290→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   291→            // Find the original dispatch step and set the answer
   292→            const dispatchStep = pendingSteps.find(
   293→              (st) => st.dispatchId === meta.dispatchId && st.kind === 'subagent_dispatch'
   294→            )
   295→            if (dispatchStep) {
   296→              dispatchStep.subagentAnswer = r.toolResult ?? ''
   297→              dispatchStep.status = 'done'
   298→              dispatchStep.finishedAt = new Date(r.createdAt).getTime()
   299→            }
   300→          } else if (r.toolName === 'subagent_tool') {
   301→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   302→            const parentDispatch = meta.dispatchId ? dispatchMap.get(meta.dispatchId) : null
   303→            stepCounter++
   304→            pendingSteps.push({
   305→              id: meta.stepId ?? r.id,
   306→              stepNumber: stepCounter,
   307→              toolName: meta.tool,
   308→              toolArgs: meta.args,
   309→              toolResult: r.toolResult ?? '',
   310→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   311→              toolOk: true,
   312→              status: 'done',
   313→              startedAt: new Date(r.createdAt).getTime(),
   314→              finishedAt: new Date(r.createdAt).getTime(),
   315→              kind: 'subagent_tool',
   316→              dispatchId: meta.dispatchId,
   317→              subagentId: parentDispatch?.subagentId ?? meta.agentId,
   318→              subagentName: parentDispatch?.subagentName,
   319→              subagentColor: parentDispatch?.subagentColor,
   320→              subagentIcon: parentDispatch?.subagentIcon,
   321→            })
   322→          } else {
   323→            // Regular super-agent tool
   324→            stepCounter++
   325→            pendingSteps.push({
   326→              id: r.id,
   327→              stepNumber: stepCounter,
   328→              toolName: r.toolName,
   329→              toolArgs: r.toolArgs ? safeParseJson(r.toolArgs) : undefined,
   330→              toolResult: r.toolResult ?? '',
   331→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   332→              toolOk: true,
   333→              status: 'done',
   334→              startedAt: new Date(r.createdAt).getTime(),
   335→              finishedAt: new Date(r.createdAt).getTime(),
   336→              kind: 'super_tool',
   337→            })
   338→          }
   339→        } else if (r.role === 'assistant') {
   340→          messages.push({
   341→            id: r.id,
   342→            role: 'assistant',
   343→            content: r.content,
   344→            steps: pendingSteps,
   345→            createdAt: new Date(r.createdAt).getTime(),
   346→          })
   347→          pendingSteps = []
   348→          stepCounter = 0
   349→          dispatchMap.clear()
   350→        }
   351→      }
   352→      set({ messages })
   353→    } catch (e) {
   354→      console.error('loadMessages', e)
   355→      set({ messages: [] })
   356→    }
   357→  },

   360→

   363→  addAttachment: (a) => set((s) => ({ attachments: [...s.attachments, a] })),
   364→  removeAttachment: (filename) =>
   365→    set((s) => ({ attachments: s.attachments.filter((a) => a.filename !== filename) })),
   366→  clearAttachments: () => set({ attachments: [] }),
   367→
   368→  loadMemories: async () => {

   370→      const res = await fetch('/api/memory')
   371→      const data = await safeJson(res)

   373→    } catch (e) {
   374→      console.error('loadMemories', e)
   375→    }
   376→  },
   377→
   378→  resetSubagentActivity: () => set({ activeSubagents: [], subagentActivity: {} }),
   379→
   380→  sendMessage: async (text) => {
   381→    const trimmed = text.trim()
   382→    if (!trimmed && get().attachments.length === 0) return
   383→    const state = get()
   384→    if (state.status !== 'idle') return
   385→
   386→    let conversationId = state.currentConversationId
   387→    if (!conversationId) {
   388→      conversationId = await get().createConversation()
   389→    }
   390→
   391→    const userMsg: ChatMessage = {
   392→      id: makeId('u'),
   393→      role: 'user',
   394→      content: trimmed,
   395→      attachments: state.attachments.length ? [...state.attachments] : undefined,
   396→      createdAt: Date.now(),
   397→    }
   398→    const assistantId = makeId('a')
   399→    const assistantMsg: ChatMessage = {
   400→      id: assistantId,
   401→      role: 'assistant',
   402→      content: '',
   403→      steps: [],
   404→      isStreaming: true,
   405→      createdAt: Date.now(),
   406→    }
   407→    set((s) => ({
   408→      messages: [...s.messages, userMsg, assistantMsg],
   409→      status: 'thinking',
   410→      currentTool: null,
   411→      attachments: [],
   412→      abortFlag: { current: false },
   413→      activeSubagents: [],
   414→      subagentActivity: {},
   415→    }))
   416→
   417→    const abortFlag = get().abortFlag
   418→    try {
   419→      const res = await fetch('/api/agent', {
   420→        method: 'POST',
   421→        headers: { 'Content-Type': 'application/json' },
   422→        body: JSON.stringify({
   423→          message: trimmed,
   424→          conversationId,
   425→          attachments: userMsg.attachments ?? [],
   426→          language: state.language,
   427→        }),
   428→      })
   429→      if (!res.ok || !res.body) {
   430→        const errText = await res.text()
   431→        throw new Error(errText || `HTTP ${res.status}`)
   432→      }
   433→
   434→      const reader = res.body.getReader()
   435→      const decoder = new TextDecoder()
   436→      let buffer = ''
   437→      // Track which step is "in progress" so we can mark it done
   438→      let currentStepId: string | null = null
   439→
   440→      while (true) {
   441→        if (abortFlag.current) break
   442→        const { done, value } = await reader.read()
   443→        if (done) break
   444→        buffer += decoder.decode(value, { stream: true })
   445→
   446→        // SSE messages separated by \n\n
   447→        let idx: number
   448→        while ((idx = buffer.indexOf('\n\n')) >= 0) {
   449→          const rawEvent = buffer.slice(0, idx)
   450→          buffer = buffer.slice(idx + 2)
   451→          const evt = parseSse(rawEvent)
   452→          if (!evt) continue
   453→          applyEvent(evt.event, evt.data, assistantId, set, get, () => {
   454→            currentStepId = evt.data?.stepId ?? currentStepId
   455→          })
   456→          if (evt.event === 'done' || evt.event === 'error') {
   457→            // finish
   458→          }
   459→        }
   460→      }
   461→
   462→      set((s) => ({
   463→        messages: s.messages.map((m) =>
   464→          m.id === assistantId ? { ...m, isStreaming: false } : m
   465→        ),
   466→        status: 'idle',
   467→        currentTool: null,
   468→        activeSubagents: [],
   469→      }))
   470→      // Refresh conversation list (title may have changed) + memories
   471→      get().loadConversations()
   472→      get().loadMemories()
   473→    } catch (e: any) {
   474→      console.error('sendMessage error', e)
   475→      set((s) => ({
   476→        messages: s.messages.map((m) =>
   477→          m.id === assistantId
   478→            ? {
   479→                ...m,
   480→                isStreaming: false,
   481→                content:
   482→                  m.content +
   483→                  (m.content ? '\n\n' : '') +
   484→                  `⚠️ **Error:** ${e?.message ?? String(e)}`,
   485→              }
   486→            : m
   487→        ),
   488→        status: 'idle',
   489→        currentTool: null,
   490→        activeSubagents: [],
   491→      }))
   492→    }
   493→  },
   494→
   495→  stopStreaming: () => {
   496→    get().abortFlag.current = true
   497→    set({ status: 'idle', currentTool: null })
   498→  },
   499→
   500→  toggleLeft: () => set((s) => ({ leftOpen: !s.leftOpen })),
   501→  toggleRight: () => set((s) => ({ rightOpen: !s.rightOpen })),
   502→  setLeft: (v) => set({ leftOpen: v }),
   503→  setRight: (v) => set({ rightOpen: v }),
   504→  setActiveTab: (tab) => set({ activeTab: tab }),
   505→  setChangePasswordOpen: (v) => set({ changePasswordOpen: v }),
   506→}))

   508→function safeParseJson(s: string): any {
   509→  try {
   510→    return JSON.parse(s)
   511→  } catch {
   512→    return s
   513→  }
   514→}
   515→function safeParseAttachments(s: string): AttachmentMeta[] | undefined {
   516→  try {
   517→    const a = JSON.parse(s)
   518→    if (Array.isArray(a)) return a
   519→  } catch {
   520→    /* ignore */
   521→  }
   522→  return undefined
   523→}
   524→
   525→function parseSse(raw: string): { event: string; data: any } | null {
   526→  const lines = raw.split('\n')
   527→  let event = 'message'
   528→  let data = ''
   529→  for (const ln of lines) {
   530→    if (ln.startsWith('event:')) event = ln.slice(6).trim()
   531→    else if (ln.startsWith('data:')) data += ln.slice(5).trim()
   532→  }
   533→  try {
   534→    return { event, data: data ? JSON.parse(data) : {} }
   535→  } catch {
   536→    return { event, data: { raw: data } }
   537→  }
   538→}

   540→function applyEvent(
   541→  event: string,
   542→  data: any,
   543→  assistantId: string,
   544→  set: (fn: (s: ChatState) => Partial<ChatState>) => void,
   545→  get: () => ChatState,
   546→  _track: () => void
   547→) {
   548→  if (event === 'thought') {
   549→    set((s) => ({
   550→      status: 'thinking',
   551→      messages: s.messages.map((m) => {
   552→        if (m.id !== assistantId) return m
   553→        const steps = [...(m.steps ?? [])]
   554→        // create a "thinking-only" step so users see the reasoning
   555→        const stepId = `thought_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`
   556→        steps.push({
   557→          id: stepId,
   558→          stepNumber: steps.length + 1,
   559→          thought: data.content,
   560→          status: 'done',
   561→          startedAt: Date.now(),
   562→          finishedAt: Date.now(),
   563→          kind: 'super_thought',
   564→        })
   565→        return { ...m, steps }
   566→      }),
   567→    }))
   568→  } else if (event === 'tool_call') {
   569→    const stepId = data.stepId
   570→    set((s) => ({
   571→      status: 'tool_running',
   572→      currentTool: data.name,
   573→      messages: s.messages.map((m) => {
   574→        if (m.id !== assistantId) return m
   575→        const steps = [...(m.steps ?? [])]
   576→        steps.push({
   577→          id: stepId,
   578→          stepNumber: data.stepNumber ?? steps.length + 1,
   579→          thought: data.thought,
   580→          toolName: data.name,
   581→          toolArgs: data.args,
   582→          status: 'running',
   583→          startedAt: Date.now(),
   584→          kind: 'super_tool',
   585→        })
   586→        return { ...m, steps }
   587→      }),
   588→    }))
   589→  } else if (event === 'tool_result') {
   590→    const stepId = data.stepId
   591→    set((s) => ({
   592→      status: 'thinking',
   593→      currentTool: null,
   594→      messages: s.messages.map((m) => {
   595→        if (m.id !== assistantId) return m
   596→        const steps = (m.steps ?? []).map((st) =>
   597→          st.id === stepId
   598→            ? {
   599→                ...st,
   600→                status: data.ok === false ? 'error' : 'done',
   601→                toolResult: data.result,
   602→                toolPreview: data.preview,
   603→                toolOk: data.ok,
   604→                artifacts: data.artifacts,
   605→                finishedAt: Date.now(),
   606→              }
   607→            : st
   608→        )
   609→        return { ...m, steps }
   610→      }),
   611→    }))
   612→  } else if (event === 'subagent_dispatch') {
   613→    const dispatchId = data.dispatchId
   614→    const agentId = data.agentId
   615→    set((s) => {
   616→      const activeSubagents = s.activeSubagents.includes(dispatchId)
   617→        ? s.activeSubagents
   618→        : [...s.activeSubagents, dispatchId]
   619→      const subagentActivity = { ...s.subagentActivity, [agentId]: 'working' as const }
   620→      return {
   621→        status: 'thinking',
   622→        activeSubagents,
   623→        subagentActivity,
   624→        messages: s.messages.map((m) => {
   625→          if (m.id !== assistantId) return m
   626→          const steps = [...(m.steps ?? [])]
   627→          const stepId = `dispatch_${dispatchId}`
   628→          steps.push({
   629→            id: stepId,
   630→            stepNumber: data.stepNumber ?? steps.length + 1,
   631→            subagentId: agentId,
   632→            subagentName: data.agentName,
   633→            subagentColor: data.color,
   634→            subagentIcon: data.icon,
   635→            subagentTask: data.task,
   636→            dispatchId,
   637→            status: 'running',
   638→            startedAt: Date.now(),
   639→            kind: 'subagent_dispatch',
   640→          })
   641→          return { ...m, steps }
   642→        }),
   643→      }
   644→    })
   645→  } else if (event === 'subagent_thought') {
   646→    const dispatchId = data.dispatchId
   647→    set((s) => ({
   648→      messages: s.messages.map((m) => {
   649→        if (m.id !== assistantId) return m
   650→        const steps = [...(m.steps ?? [])]
   651→        // Find the dispatch step for this dispatchId to get sub-agent info
   652→        const dispatchStep = steps.find(
   653→          (st) => st.dispatchId === dispatchId && st.kind === 'subagent_dispatch'
   654→        )
   655→        const stepId = `subthought_${dispatchId}_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`
   656→        steps.push({
   657→          id: stepId,
   658→          stepNumber: steps.length + 1,
   659→          thought: data.content,
   660→          status: 'done',
   661→          startedAt: Date.now(),
   662→          finishedAt: Date.now(),
   663→          kind: 'subagent_thought',
   664→          dispatchId,
   665→          subagentId: dispatchStep?.subagentId,
   666→          subagentName: dispatchStep?.subagentName,
   667→          subagentColor: dispatchStep?.subagentColor,
   668→          subagentIcon: dispatchStep?.subagentIcon,
   669→        })
   670→        return { ...m, steps }
   671→      }),
   672→    }))
   673→  } else if (event === 'subagent_tool_call') {
   674→    const dispatchId = data.dispatchId
   675→    const stepId = data.stepId
   676→    set((s) => ({
   677→      status: 'tool_running',
   678→      currentTool: data.name,
   679→      messages: s.messages.map((m) => {
   680→        if (m.id !== assistantId) return m
   681→        const steps = [...(m.steps ?? [])]
   682→        const dispatchStep = steps.find(
   683→          (st) => st.dispatchId === dispatchId && st.kind === 'subagent_dispatch'
   684→        )
   685→        steps.push({
   686→          id: stepId,
   687→          stepNumber: data.stepNumber ?? steps.length + 1,
   688→          thought: data.thought,
   689→          toolName: data.name,
   690→          toolArgs: data.args,
   691→          status: 'running',
   692→          startedAt: Date.now(),
   693→          kind: 'subagent_tool',
   694→          dispatchId,
   695→          subagentId: dispatchStep?.subagentId,
   696→          subagentName: dispatchStep?.subagentName,
   697→          subagentColor: dispatchStep?.subagentColor,
   698→          subagentIcon: dispatchStep?.subagentIcon,
   699→        })
   700→        return { ...m, steps }
   701→      }),
   702→    }))
   703→  } else if (event === 'subagent_tool_result') {
   704→    const stepId = data.stepId
   705→    set((s) => ({
   706→      status: 'thinking',
   707→      currentTool: null,
   708→      messages: s.messages.map((m) => {
   709→        if (m.id !== assistantId) return m
   710→        const steps = (m.steps ?? []).map((st) =>
   711→          st.id === stepId
   712→            ? {
   713→                ...st,
   714→                status: data.ok === false ? 'error' : 'done',
   715→                toolResult: data.result,
   716→                toolPreview: data.preview,
   717→                toolOk: data.ok,
   718→                artifacts: data.artifacts,
   719→                finishedAt: Date.now(),
   720→              }
   721→            : st
   722→        )
   723→        return { ...m, steps }
   724→      }),
   725→    }))
   726→  } else if (event === 'subagent_complete') {
   727→    const dispatchId = data.dispatchId
   728→    set((s) => {
   729→      const activeSubagents = s.activeSubagents.filter((id) => id !== dispatchId)
   730→      // Mark the sub-agent activity as done (find which agent this dispatchId was for)
   731→      const messages = s.messages.map((m) => {
   732→        if (m.id !== assistantId) return m
   733→        const steps = (m.steps ?? []).map((st) => {
   734→          if (st.dispatchId === dispatchId && st.kind === 'subagent_dispatch') {
   735→            const subagentActivity = { ...s.subagentActivity }
   736→            if (st.subagentId) subagentActivity[st.subagentId] = 'done'
   737→            // Update via the outer set below (this closure can't return both)
   738→            return {
   739→              ...st,
   740→              status: 'done' as const,
   741→              subagentAnswer: data.answer,
   742→              finishedAt: Date.now(),
   743→            }
   744→          }
   745→          return st
   746→        })
   747→        return { ...m, steps }
   748→      })
   749→      // Determine subagent id for activity update
   750→      let agentId: string | undefined
   751→      for (const m of s.messages) {
   752→        if (m.id !== assistantId) continue
   753→        const ds = (m.steps ?? []).find(
   754→          (st) => st.dispatchId === dispatchId && st.kind === 'subagent_dispatch'
   755→        )
   756→        if (ds?.subagentId) {
   757→          agentId = ds.subagentId
   758→          break
   759→        }
   760→      }
   761→      const subagentActivity = { ...s.subagentActivity }
   762→      if (agentId) subagentActivity[agentId] = 'done'
   763→      return { activeSubagents, subagentActivity, messages }
   764→    })
   765→  } else if (event === 'synthesis') {
   766→    set((s) => ({
   767→      status: 'streaming',
   768→      messages: s.messages.map((m) => {
   769→        if (m.id !== assistantId) return m
   770→        const steps = [...(m.steps ?? [])]
   771→        // Push a "synthesizing" marker step (only if not already present)
   772→        const hasSynth = steps.some((st) => st.kind === ('super_thought' as any) && st.thought === '__synthesizing__')
   773→        if (!hasSynth) {
   774→          steps.push({
   775→            id: `synth_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
   776→            stepNumber: steps.length + 1,
   777→            thought: '__synthesizing__',
   778→            status: 'thinking',
   779→            startedAt: Date.now(),
   780→            kind: 'super_thought',
   781→          })
   782→        }
   783→        return { ...m, steps }
   784→      }),
   785→    }))
   786→  } else if (event === 'token') {
   787→    set((s) => ({
   788→      status: 'streaming',
   789→      messages: s.messages.map((m) => {
   790→        if (m.id !== assistantId) return m
   791→        // Drop the synthesizing marker step if it exists, since real tokens are now arriving
   792→        const steps = (m.steps ?? []).filter(
   793→          (st) => !(st.kind === 'super_thought' && st.thought === '__synthesizing__')
   794→        )
   795→        return { ...m, content: m.content + (data.content ?? ''), steps }
   796→      }),
   797→    }))
   798→  } else if (event === 'memory_update') {
   799→    // Optimistically add/update memory in the right panel
   800→    set((s) => {
   801→      const exists = s.memories.find((m) => m.key === data.key)
   802→      const item: MemoryItem = {
   803→        id: exists?.id ?? `tmp_${Date.now()}`,
   804→        key: data.key,
   805→        value: data.value,
   806→        category: data.category ?? 'general',
   807→        updatedAt: new Date().toISOString(),
   808→      }
   809→      const memories = exists
   810→        ? s.memories.map((m) => (m.key === data.key ? item : m))
   811→        : [item, ...s.memories]
   812→      return { memories }
   813→    })
   814→  } else if (event === 'error') {
   815→    set((s) => ({
   816→      messages: s.messages.map((m) =>
   817→        m.id === assistantId
   818→          ? {
   819→              ...m,
   820→              content:
   821→                m.content +
   822→                (m.content ? '\n\n' : '') +
   823→                `⚠️ **Error:** ${data.message ?? 'unknown error'}`,
   824→              isStreaming: false,
   825→            }
   826→          : m
   827→      ),
   828→      status: 'idle',
   829→      currentTool: null,
   830→      activeSubagents: [],
   831→    }))
   832→  }
   833→  // 'done' event: nothing special; outer loop will set isStreaming=false
   834→}
