   746→   746→  const apiKey = process.env.GROQ_API_KEY!
   747→   747→  // UPGRADE #159: Use auto-discovered model first, then fallbacks.
   748→   748→  // UPGRADE #176 fix #4: Removed 'llama-3.2-90b-vision-preview' — deprecated
   749→   749→  // by Groq, returns HTTP 400 ("Model does not exist or you do not have

   751→   751→  // this dead model before falling through to OpenAI.

   753→   753→    preferredModel,

   755→   755→    'llama-3.1-8b-instant',

   761→   761→  // the prompt itself is already 12-15KB. Groq enforces: prompt_tokens +
   762→   762→  // max_tokens ≤ model context window (32K for llama-3.3-70b). If the
   763→   763→  // prompt is 4K tokens, max_tokens=12000 is fine (16K total ≤ 32K). But
   764→   764→  // if the prompt is 8K tokens, max_tokens=12000 pushes total to 20K —
   765→   765→  // still OK, but the REQUEST BODY size may exceed Groq's gateway limit.
   766→   766→  // Fix: reduce max_tokens to 4096 (sufficient for most responses) and
   767→   767→  // skip Groq entirely if the prompt exceeds 28K chars (~7K tokens) to
   768→   768→  // avoid the 413 altogether.
   769→   769→  const promptSize = JSON.stringify(messages).length
   770→   770→  const maxTokens = 4096  // was 12000 — 4096 is plenty for agent responses
   771→   771→  // UPGRADE #183 fix A: Raised Groq limit from 28K → 100K chars.
   772→   772→  // Groq's llama-3.3-70b-versatile has a 32K TOKEN context window.
   773→   773→  // 100K chars ≈ 25K tokens (safe under 32K). The old 28K limit was
   774→   774→  // too conservative (only 7K tokens) and caused Groq to be skipped
   775→   775→  // on normal-length conversations. Antonio reported 176K char prompts

  1185→  1185→// Without this, the agent gets stuck in a loop: it writes <dispatch_subagent>
  1186→  1186→// as text, the parser doesn't recognize it, treats it as a final answer,
  1187→  1187→// and the agent never actually dispatches the subagent.
  1188→  1188→//
  1189→  1189→// Format: <dispatch_subagent id="scout">task description</dispatch_subagent>
  1190→  1190→// We convert this to: tool = { name: 'dispatch_subagent', args: { id, task } }
  1191→  1191→export const DISPATCH_SUBAGENT_RE = /<dispatch_subagent\s+id=["']([^"']+)["']\s*>([\s\S]*?)<\/dispatch_subagent>/i

  1193→  1193→export interface Parsed {
  1194→  1194→  thought?: string
  1195→  1195→  tool?: { name: string; args: any }
  1196→  1196→  // UPGRADE #169 C2: `dispatch` was missing — subagents.ts read parsed.dispatch
  1197→  1197→  // but Parsed never had a dispatch field. The 3-tier hierarchy (CEO → Leader →
  1198→  1198→  // Specialist) was therefore broken: Leaders could not delegate to Specialists
  1199→  1199→  // because the dispatch was never propagated. We now populate it from both
  1200→  1200→  // <dispatch_subagent> tags AND <tool name="dispatch_subagent"> tool calls.
  1201→  1201→  dispatch?: { agentId: string; task: string }
  1202→  1202→  textAfterTool: string
  1203→  1203→  textBeforeTool: string
  1204→  1204→  raw: string
  1205→  1205→}

  1207→  1207→export function parseAssistant(content: string): Parsed {
  1208→  1208→  const thoughtMatch = content.match(THOUGHT_RE)
  1209→  1209→  const thought = thoughtMatch?.[1]?.trim()
  1210→  1210→  const toolMatch = content.match(TOOL_RE)
  1211→  1211→
  1212→  1212→  // ── UPGRADE #63 — Also check for <dispatch_subagent> tags ──────────
  1213→  1213→  // The LLM often writes <dispatch_subagent id="scout">task</dispatch_subagent>
  1214→  1214→  // as text instead of using <tool name="dispatch_subagent">. We detect both
  1215→  1215→  // formats and convert dispatch_subagent tags to proper tool calls.
  1216→  1216→  const dispatchMatch = content.match(DISPATCH_SUBAGENT_RE)
  1217→  1217→
  1218→  1218→  // UPGRADE #169 C2: Also extract a `dispatch` field for subagents.ts to use
  1219→  1219→  // when delegating to specialists. We populate it from either format:
  1220→  1220→  //   - <dispatch_subagent id="quill">task</dispatch_subagent>
  1221→  1221→  //   - <tool name="dispatch_subagent">{"id":"quill","task":"..."}</tool>
  1222→  1222→  let dispatch: Parsed['dispatch']
  1223→  1223→
  1224→  1224→  // Prefer the proper <tool> format if present; otherwise fall back to <dispatch_subagent>
  1225→  1225→  let tool: Parsed['tool']
  1226→  1226→  let textBeforeTool = content
  1227→  1227→  let textAfterTool = ''
  1228→  1228→
  1229→  1229→  if (toolMatch) {
  1230→  1230→    const name = (toolMatch[1] ?? '').trim()
  1231→  1231→    if (!name) {
  1232→  1232→      return { thought, tool: undefined, dispatch: undefined, textBeforeTool: content.replace(THOUGHT_RE, '').trim(), textAfterTool: '', raw: content }
  1233→  1233→    }
  1234→  1234→    let args: any = {}
  1235→  1235→    const raw = (toolMatch[2] ?? '').trim()
  1236→  1236→    if (raw) {
  1237→  1237→      try {
  1238→  1238→        args = JSON.parse(raw)
  1239→  1239→      } catch {
  1240→  1240→        const m: Record<string, string> = {}
  1241→  1241→        const re = /"([^"]+)"\s*:\s*"([^"]*)"/g
  1242→  1242→        let mm: RegExpExecArray | null
  1243→  1243→        while ((mm = re.exec(raw))) m[mm[1]] = mm[2]
  1244→  1244→        args = m
  1245→  1245→      }
  1246→  1246→    }
  1247→  1247→    tool = { name, args }
  1248→  1248→    const idx = content.indexOf(toolMatch[0])
  1249→  1249→    textBeforeTool = content.slice(0, idx).replace(THOUGHT_RE, '').trim()
  1250→  1250→    textAfterTool = content.slice(idx + toolMatch[0].length).trim()
  1251→  1251→
  1252→  1252→    // UPGRADE #169 C2: Populate dispatch from <tool name="dispatch_subagent"> format
  1253→  1253→    if (name === 'dispatch_subagent') {
  1254→  1254→      const agentId = (args?.id ?? args?.agentId ?? '').toString().trim()
  1255→  1255→      const task = (args?.task ?? args?.goal ?? '').toString().trim()
  1256→  1256→      if (agentId) {
  1257→  1257→        dispatch = { agentId, task }
  1258→  1258→      }
  1259→  1259→    }
  1260→  1260→  } else if (dispatchMatch) {
  1261→  1261→    // ── UPGRADE #63 — Convert <dispatch_subagent> text to a real tool call ──
  1262→  1262→    const subagentId = (dispatchMatch[1] ?? '').trim()
  1263→  1263→    const task = (dispatchMatch[2] ?? '').trim()
  1264→  1264→    if (subagentId) {
  1265→  1265→      tool = { name: 'dispatch_subagent', args: { id: subagentId, task } }
  1266→  1266→      // UPGRADE #169 C2: Also populate the dispatch field so subagents.ts:1583
  1267→  1267→      // can detect the delegation request.
  1268→  1268→      dispatch = { agentId: subagentId, task }
  1269→  1269→      const idx = content.indexOf(dispatchMatch[0])
  1270→  1270→      textBeforeTool = content.slice(0, idx).replace(THOUGHT_RE, '').trim()
  1271→  1271→      textAfterTool = content.slice(idx + dispatchMatch[0].length).trim()
  1272→  1272→    }
  1273→  1273→  }
  1274→  1274→  return {
  1275→  1275→    thought,
  1276→  1276→    tool,
  1277→  1277→    dispatch,
  1278→  1278→    textBeforeTool,
  1279→  1279→    textAfterTool,
  1280→  1280→    raw: content,
  1281→  1281→  }
  1282→  1282→}

  1284→  1284→/** Rough token estimator (~4 chars/token, standard approximation). */
  1285→  1285→function estimateTokens(messages: Array<{ role: string; content: string }>): number {
  1286→  1286→  let chars = 0
  1287→  1287→  for (const m of messages) chars += (m.content ?? '').length
  1288→  1288→  return Math.ceil(chars / 4)
  1289→  1289→}

  1291→  1291→/** Build the LLM message history from the DB rows of the conversation. */
  1292→  1292→export async function buildHistoryMessages(
  1293→  1293→  conversationId: string,
  1294→  1294→  currentUserMessage: string,
  1295→  1295→  currentAttachments: AttachmentMeta[]
  1296→  1296→): Promise<Array<{ role: 'system' | 'user' | 'assistant'; content: string }>> {
  1297→  1297→  // UPGRADE #128: Wrap DB query in try/catch — if DB is unreachable,
  1298→  1298→  // continue with just the current message instead of crashing the entire agent
  1299→  1299→  let priorMessages: any[] = []
  1300→  1300→  try {
  1301→  1301→    priorMessages = await db.message.findMany({
  1302→  1302→      where: { conversationId },
  1303→  1303→      orderBy: { createdAt: 'asc' },
  1304→  1304→    })
  1305→  1305→  } catch (dbErr: any) {
  1306→  1306→    console.warn('[buildHistoryMessages] DB query failed, continuing without history:', dbErr?.message?.slice(0, 100))
  1307→  1307→    // Return just the current message — the agent will still respond,
  1308→  1308→    // just without conversation history context
  1309→  1309→  }
  1310→  1310→  const msgs: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = []
  1311→  1311→  for (const m of priorMessages) {
  1312→  1312→    if (m.role === 'user') {
  1313→  1313→      let content = m.content
  1314→  1314→      const atts = m.attachments ? (JSON.parse(m.attachments) as AttachmentMeta[]) : []
  1315→  1315→      const textFiles = atts.filter((a) => a.textContent)
  1316→  1316→      const images = atts.filter((a) => a.mimeType.startsWith('image/'))
  1317→  1317→      if (textFiles.length) {
  1318→  1318→        content +=
  1319→  1319→          '\n\n[ATTACHED TEXT FILES]\n' +
  1320→  1320→          textFiles.map((a) => `--- ${a.originalName} ---\n${a.textContent?.slice(0, 8000)}`).join('\n\n')
  1321→  1321→      }
  1322→  1322→      if (images.length) {
  1323→  1323→        content += `\n\n[ATTACHED IMAGES: ${images.map((a) => a.originalName).join(', ')}] Use the vision tool with image_index to analyze them.`
  1324→  1324→      }
  1325→  1325→      msgs.push({ role: 'user', content })
  1326→  1326→    } else if (m.role === 'assistant') {
  1327→  1327→      msgs.push({ role: 'assistant', content: m.content })
  1328→  1328→    } else if (m.role === 'tool') {
  1329→  1329→      msgs.push({
  1330→  1330→        role: 'user',
  1331→  1331→        content: `[TOOL_RESULT] ${m.toolName}: ${m.toolResult ?? ''}`,
  1332→  1332→      })
  1333→  1333→    } else if (m.role === 'thought') {
  1334→  1334→      // skip — thoughts are internal; we re-feed them via assistant content's <thought> tags
  1335→  1335→    }
  1336→  1336→  }
  1337→  1337→  // current message
  1338→  1338→  let userContent = currentUserMessage
  1339→  1339→  const textFiles = currentAttachments.filter((a) => a.textContent)
  1340→  1340→  const images = currentAttachments.filter((a) => a.mimeType.startsWith('image/'))
  1341→  1341→  if (textFiles.length) {
  1342→  1342→    userContent +=
  1343→  1343→      '\n\n[ATTACHED TEXT FILES]\n' +
  1344→  1344→      textFiles.map((a) => `--- ${a.originalName} ---\n${a.textContent?.slice(0, 8000)}`).join('\n\n')
  1345→  1345→  }
  1346→  1346→  if (images.length) {
  1347→  1347→    userContent += `\n\n[ATTACHED IMAGES: ${images.map((a) => a.originalName).join(', ')}] Use the vision tool with image_index to analyze them.`
  1348→  1348→  }
  1349→  1349→  msgs.push({ role: 'user', content: userContent })
  1350→  1350→
  1351→  1351→  // Auto-truncate if too long — keep the most recent ~30k tokens of history
  1352→  1352→  // and add a marker so the model knows earlier context was dropped.
  1353→  1353→  const MAX_TOKENS = 50_000
  1354→  1354→  const KEEP_TOKENS = 30_000
  1355→  1355→  if (estimateTokens(msgs) > MAX_TOKENS) {
  1356→  1356→    // Walk from the end of msgs, accumulating until we hit KEEP_TOKENS budget
  1357→  1357→    let keptTokens = 0
  1358→  1358→    let cutIndex = msgs.length
  1359→  1359→    for (let i = msgs.length - 1; i >= 0; i--) {
  1360→  1360→      const t = Math.ceil((msgs[i].content ?? '').length / 4)
  1361→  1361→      if (keptTokens + t > KEEP_TOKENS * 4) {
  1362→  1362→        cutIndex = i + 1
  1363→  1363→        break
  1364→  1364→      }
  1365→  1365→      keptTokens += t
  1366→  1366→      cutIndex = i
  1367→  1367→    }
  1368→  1368→    const trimmed = msgs.slice(cutIndex)
  1369→  1369→    return [
  1370→  1370→      {
  1371→  1371→        role: 'user',
  1372→  1372→        content:
  1373→  1373→          '[Earlier conversation history truncated to fit context window. Earlier tool results and assistant messages were dropped.]',
  1374→  1374→      },
  1375→  1375→      ...trimmed,
  1376→  1376→    ]
  1377→  1377→  }
  1378→  1378→
  1379→  1379→  return msgs
  1380→  1380→}
