    86→/* ----------------------------- Dashboard tab ----------------------------- */
    87→export function DashboardTab() {

    89→  const [loading, setLoading] = useState(true)
    90→  const [error, setError] = useState('')

    92→    monthlyGoal: 20000,
    93→    dailyGrowthTarget: 20,
    94→    currencySymbol: '$',
    95→    displayMode: 'detailed',
    96→  })
    97→  const [addModalOpen, setAddModalOpen] = useState(false)

   101→  const [refreshing, setRefreshing] = useState(false)
   102→  const [lastRefreshedAt, setLastRefreshedAt] = useState<Date | null>(null)
   103→
   104→  // Subscribe to refresh signals from Agent007
   105→  const refreshVersion = useChatStore((s) => s.refreshVersion)
   106→  const autoRefreshEnabled = useChatStore((s) => s.autoRefreshEnabled)
   107→  const setAutoRefreshEnabled = useChatStore((s) => s.setAutoRefreshEnabled)
   108→
   109→  const loadIncome = useCallback(async () => {
   110→    try {
   111→      const res = await fetch('/api/income')
   112→      const json = await res.json()
   113→      if (json.error) throw new Error(json.error)
   114→      setData(json)
   115→      setError('')
   116→    } catch (e: any) {
   117→      setError(e?.message ?? 'Failed to load income data')
   118→    } finally {
   119→      setLoading(false)
   120→    }
   121→  }, [])
   122→

   124→    try {

   128→    } catch {
   129→      /* ignore */
   130→    }
   131→  }, [])
   132→

   136→      const res = await fetch('/api/dashboard/widgets')

   156→  // After: 1 seed POST (non-blocking) + 3 parallel GETs = ~10s total instead of ~40s

   158→    let cancelled = false
   159→    ;(async () => {
   160→      // Fire the seed POST in the background (don't await — it's idempotent
   161→      // and the GETs below will return whatever data exists).
   162→      fetch('/api/income', {
   163→        method: 'POST',
   164→        headers: { 'Content-Type': 'application/json' },
   165→        body: JSON.stringify({ seedIfEmpty: true }),
   166→      }).catch(() => {/* ignore */})
   167→

   169→      // first cold-start DB init will satisfy all 3 simultaneously.
   170→      try {
   171→        await Promise.all([
   172→          loadIncome(),
   173→          loadSettings(),
   174→          loadCustomWidgets(),
   175→        ])
   176→      } catch {
   177→        /* ignore — individual loaders have their own error handling */
   178→      }
   179→      if (!cancelled) setLastRefreshedAt(new Date())
   180→    })()

   182→  }, [loadIncome, loadSettings, loadCustomWidgets])
   183→
   184→  // Re-fetch when Agent007 emits a refresh signal
   185→  useEffect(() => {
   186→    if (refreshVersion === 0) return // skip initial

   192→  //   2 monitor calls (each 30s timeout) — blocking the dashboard for 65+ seconds.
   193→  // After: scheduled tasks run via Vercel Cron (every 30 minutes, configured in
   194→  //   vercel.json). The dashboard no longer blocks. Users can still trigger
   195→  //   schedules manually via the Schedules tab "Run Now" button.
   196→
   197→  const today = data?.aggregates.today ?? 0
   198→  const yesterday = data?.aggregates.yesterday ?? 0
   199→  const growthPct =
   200→    yesterday > 0 ? ((today - yesterday) / yesterday) * 100 : today > 0 ? 100 : 0

   208→        <div className="max-w-6xl mx-auto">
   209→          {/* Header */}
   210→          <div className="flex items-center justify-between gap-3 mb-6 flex-wrap">
   211→            <div>
   212→              <h1 className="text-2xl sm:text-3xl font-bold tracking-tight">
   213→                <span className="neon-text-cyan">Income</span>{' '}
   214→                <span className="neon-text-purple">Tracker</span>
   215→              </h1>
   216→              <p className="text-xs text-[#7c89b5] mt-1 tracking-wide">
