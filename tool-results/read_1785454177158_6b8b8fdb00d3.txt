   182→let msgIdCounter = 0

   184→  msgIdCounter += 1

   188→async function safeJson(res: Response): Promise<any> {
   189→  const text = await res.text()
   190→  try {
   191→    return JSON.parse(text)
   192→  } catch {
   193→    return { error: text || 'parse error' }
   194→  }
   195→}

   225→  loadConversations: async () => {

   246→      }
   247→    } catch (e) {
   248→      console.error('loadConversations', e)
   249→      // Fallback: localStorage ONLY if API is unreachable (offline mode)
   250→      if (typeof window !== 'undefined') {
   251→        try {
   252→          const saved = localStorage.getItem('agent007_conversations')
   253→          if (saved) set({ conversations: JSON.parse(saved) })

   256→    }

   302→  loadMessages: async (conversationId) => {
   303→    try {
   304→      const res = await fetch(`/api/conversations/${conversationId}`)
   305→      if (!res.ok) {
   306→        console.warn('[loadMessages] API returned', res.status)
   307→        set({ messages: [] })
   308→        return
   309→      }
   310→      const data = await safeJson(res)
   311→      const conv = data.conversation
   312→      if (!conv || !conv.Message || conv.Message.length === 0) {
   313→        // UPGRADE #70 — DB is the ONLY source of truth. No localStorage fallback for messages.
   314→        // Every device sees the SAME messages from Postgres.
   315→        set({ messages: [] })
   316→        return
   317→      }
   318→      // Reconstruct messages from DB rows: collapse tool/thought rows under the *next* assistant message
   319→      const rows: any[] = conv.Message ?? []
   320→      const messages: ChatMessage[] = []
   321→      let pendingSteps: ToolStep[] = []
   322→      // Track open dispatches so subsequent subagent rows can be linked.
   323→      // Map dispatchId → { subagentId, subagentName, subagentColor, subagentIcon, task, stepId }
   324→      const dispatchMap = new Map<string, any>()
   325→      let stepCounter = 0
   326→      for (const r of rows) {
   327→        if (r.role === 'user') {
   328→          const atts = r.attachments ? safeParseAttachments(r.attachments) : undefined
   329→          messages.push({
   330→            id: r.id,
   331→            role: 'user',
   332→            content: r.content,
   333→            attachments: atts,
   334→            createdAt: new Date(r.createdAt).getTime(),
   335→          })
   336→        } else if (r.role === 'thought') {
   337→          // Check if this is a sub-agent thought (prefixed with [subagent:id])
   338→          const subMatch = r.content?.match(/^\[subagent:([^\]]+)\]\s*([\s\S]*)$/)
   339→          if (subMatch) {
   340→            const subId = subMatch[1]
   341→            // Find the most recent dispatch for this sub-agent
   342→            let parentDispatch: any = null
   343→            for (const [did, info] of dispatchMap.entries()) {
   344→              if (info.subagentId === subId) parentDispatch = { did, ...info }
   345→            }
   346→            stepCounter++
   347→            pendingSteps.push({
   348→              id: r.id,
   349→              stepNumber: stepCounter,
   350→              thought: subMatch[2],
   351→              status: 'done',
   352→              startedAt: new Date(r.createdAt).getTime(),
   353→              finishedAt: new Date(r.createdAt).getTime(),
   354→              kind: 'subagent_thought',
   355→              dispatchId: parentDispatch?.did,
   356→              subagentId: subId,
   357→              subagentName: parentDispatch?.subagentName,
   358→              subagentColor: parentDispatch?.subagentColor,
   359→              subagentIcon: parentDispatch?.subagentIcon,
   360→            })
   361→          } else {
   362→            stepCounter++
   363→            pendingSteps.push({
   364→              id: r.id,
   365→              stepNumber: stepCounter,
   366→              thought: r.content,
   367→              status: 'done',
   368→              startedAt: new Date(r.createdAt).getTime(),
   369→              finishedAt: new Date(r.createdAt).getTime(),
   370→              kind: 'super_thought',
   371→            })
   372→          }
   373→        } else if (r.role === 'tool') {
   374→          // Sub-agent dispatch?
   375→          if (r.toolName === 'subagent_dispatch') {
   376→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   377→            stepCounter++
   378→            const stepId = `dispatch_${meta.dispatchId ?? r.id}`
   379→            const stepInfo: any = {
   380→              subagentId: meta.agentId,
   381→              subagentName: meta.agentName,
   382→              subagentColor: meta.color,
   383→              subagentIcon: meta.icon,
   384→              task: meta.task,
   385→            }
   386→            if (meta.dispatchId) dispatchMap.set(meta.dispatchId, stepInfo)
   387→            pendingSteps.push({
   388→              id: stepId,
   389→              stepNumber: stepCounter,
   390→              subagentId: meta.agentId,
   391→              subagentName: meta.agentName,
   392→              subagentColor: meta.color,
   393→              subagentIcon: meta.icon,
   394→              subagentTask: meta.task,
   395→              dispatchId: meta.dispatchId,
   396→              status: 'done',
   397→              startedAt: new Date(r.createdAt).getTime(),
   398→              finishedAt: new Date(r.createdAt).getTime(),
   399→              kind: 'subagent_dispatch',
   400→              // If a subagent_complete row exists for the same dispatchId, the
   401→              // subagentAnswer will be filled below when we encounter that row.
   402→            })
   403→          } else if (r.toolName === 'subagent_complete') {
   404→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   405→            // Find the original dispatch step and set the answer
   406→            const dispatchStep = pendingSteps.find(
   407→              (st) => st.dispatchId === meta.dispatchId && st.kind === 'subagent_dispatch'
   408→            )
   409→            if (dispatchStep) {
   410→              dispatchStep.subagentAnswer = r.toolResult ?? ''
   411→              dispatchStep.status = 'done'
   412→              dispatchStep.finishedAt = new Date(r.createdAt).getTime()
   413→            }
   414→          } else if (r.toolName === 'subagent_tool') {
   415→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   416→            const parentDispatch = meta.dispatchId ? dispatchMap.get(meta.dispatchId) : null
   417→            stepCounter++
   418→            pendingSteps.push({
   419→              id: meta.stepId ?? r.id,
   420→              stepNumber: stepCounter,
   421→              toolName: meta.tool,
   422→              toolArgs: meta.args,
   423→              toolResult: r.toolResult ?? '',
   424→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   425→              toolOk: true,
   426→              status: 'done',
   427→              startedAt: new Date(r.createdAt).getTime(),
   428→              finishedAt: new Date(r.createdAt).getTime(),
   429→              kind: 'subagent_tool',
   430→              dispatchId: meta.dispatchId,
   431→              subagentId: parentDispatch?.subagentId ?? meta.agentId,
   432→              subagentName: parentDispatch?.subagentName,
   433→              subagentColor: parentDispatch?.subagentColor,
   434→              subagentIcon: parentDispatch?.subagentIcon,
   435→            })
   436→          } else if (r.toolName === 'manage_action') {
   437→            // Reconstruct manage_action step from persisted row
   438→            const meta = r.toolArgs ? safeParseJson(r.toolArgs) : {}
   439→            stepCounter++
   440→            pendingSteps.push({
   441→              id: r.id,
   442→              stepNumber: stepCounter,
   443→              status: 'done',
   444→              startedAt: new Date(r.createdAt).getTime(),
   445→              finishedAt: new Date(r.createdAt).getTime(),
   446→              kind: 'manage_action',
   447→              manageAction: meta.action,
   448→              manageAttrs: meta.attrs,
   449→              manageResult: { ok: !/failed|error/i.test(r.toolResult ?? ''), message: r.toolResult ?? '' },
   450→            })
   451→          } else {
   452→            // Regular super-agent tool
   453→            stepCounter++
   454→            pendingSteps.push({
   455→              id: r.id,
   456→              stepNumber: stepCounter,
   457→              toolName: r.toolName,
   458→              toolArgs: r.toolArgs ? safeParseJson(r.toolArgs) : undefined,
   459→              toolResult: r.toolResult ?? '',
   460→              toolPreview: (r.toolResult ?? '').slice(0, 160),
   461→              toolOk: true,
   462→              status: 'done',
   463→              startedAt: new Date(r.createdAt).getTime(),
   464→              finishedAt: new Date(r.createdAt).getTime(),
   465→              kind: 'super_tool',
   466→            })
   467→          }
   468→        } else if (r.role === 'assistant') {
   469→          messages.push({
   470→            id: r.id,
   471→            role: 'assistant',
   472→            content: r.content,
   473→            steps: pendingSteps,
   474→            createdAt: new Date(r.createdAt).getTime(),
   475→          })
   476→          pendingSteps = []
   477→          stepCounter = 0
   478→          dispatchMap.clear()
   479→        }
   480→      }
   481→      set({ messages })
   482→    } catch (e) {
   483→      console.error('loadMessages', e)
   484→      set({ messages: [] })
   485→    }
   486→  },

   521→  sendMessage: async (text) => {
   522→    const trimmed = text.trim()
   523→    if (!trimmed && get().attachments.length === 0) return
   524→    const state = get()
   525→    if (state.status !== 'idle') return
   526→
   527→    let conversationId = state.currentConversationId
   528→    if (!conversationId) {
   529→      conversationId = await get().createConversation()
   530→    }
   531→
   532→    const userMsg: ChatMessage = {
   533→      id: makeId('u'),
   534→      role: 'user',
   535→      content: trimmed,
   536→      attachments: state.attachments.length ? [...state.attachments] : undefined,
   537→      createdAt: Date.now(),
   538→    }
   539→    const assistantId = makeId('a')
   540→    const assistantMsg: ChatMessage = {
   541→      id: assistantId,
   542→      role: 'assistant',
   543→      content: '',
   544→      steps: [],
   545→      isStreaming: true,
   546→      createdAt: Date.now(),
   547→    }
   548→    set((s) => {
   549→      const newMessages = [...s.messages, userMsg, assistantMsg]
   550→      // UPGRADE #125 — Rec 2C: Skip localStorage write during streaming (debounced to done/error)
   551→      // Previously wrote to localStorage on EVERY token, causing synchronous blocking
   552→      if (typeof window !== 'undefined') {
   553→        try {
   554→          const convId = s.currentConversationId
   555→          // Only update conversation title in localStorage (lightweight)
   556→          const saved = localStorage.getItem('agent007_conversations')
   557→          if (saved) {
   558→            const convs = JSON.parse(saved)
   559→            const idx = convs.findIndex((c: any) => c.id === convId)
   560→            if (idx >= 0) {
   561→              convs[idx].title = trimmed.slice(0, 50)
   562→              convs[idx].updatedAt = new Date().toISOString()
   563→              localStorage.setItem('agent007_conversations', JSON.stringify(convs))
   564→            }
   565→          }
   566→        } catch {}
   567→      }
   568→      return {
   569→        messages: newMessages,
   570→        status: 'thinking',
   571→        currentTool: null,
   572→        attachments: [],
   573→        abortFlag: { current: false },
   574→        activeSubagents: [],
   575→        subagentActivity: {},
   576→      }

   851→function parseSse(raw: string): { event: string; data: any } | null {
   852→  const lines = raw.split('\n')
   853→  let event = 'message'
   854→  let data = ''
   855→  for (const ln of lines) {
   856→    if (ln.startsWith('event:')) event = ln.slice(6).trim()
   857→    else if (ln.startsWith('data:')) data += ln.slice(5).trim()
   858→  }
   859→  try {
   860→    return { event, data: data ? JSON.parse(data) : {} }
   861→  } catch {
   862→    return { event, data: { raw: data } }
   863→  }
   864→}

  1123→        const steps = [...(m.steps ?? [])]
  1124→        // Push a "synthesizing" marker step (only if not already present)
  1125→        const hasSynth = steps.some((st) => st.kind === ('super_thought' as any) && st.thought === '__synthesizing__')
  1126→        if (!hasSynth) {
  1127→          steps.push({
  1128→            id: `synth_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,

  1130→            thought: '__synthesizing__',

  1134→          })

  1138→    }))
  1139→  } else if (event === 'token') {
  1140→    // UPGRADE #125 — Rec 2A: Batch token updates (throttle to 100ms)
  1141→    // Instead of updating state on EVERY token (which causes re-renders 100+ times/sec),
  1142→    // accumulate tokens in a buffer and flush to state every 100ms.
  1143→    _pendingTokens += (data.content ?? '')

  1155→                (st) => !(st.kind === 'super_thought' && st.thought === '__synthesizing__')

  1157→              return { ...m, content: m.content + tokensToAdd, steps }

  1159→          }))
  1160→        }
  1161→      }, 100)
  1162→    }
  1163→  } else if (event === 'memory_update') {
  1164→    // Optimistically add/update memory in the right panel

  1167→      const item: MemoryItem = {
  1168→        id: exists?.id ?? `tmp_${Date.now()}`,
  1169→        key: data.key,
  1170→        value: data.value,
  1171→        category: data.category ?? 'general',
  1172→        updatedAt: new Date().toISOString(),
  1173→      }
  1174→      const memories = exists
  1175→        ? s.memories.map((m) => (m.key === data.key ? item : m))
  1176→        : [item, ...s.memories]
  1177→      return { memories }
  1178→    })
  1179→  } else if (event === 'manage_action') {
  1180→    // Two-phase event: status='running' pushes a new step; status='done'|'error'
  1181→    // updates that step with the result.
  1182→    const stepId = data.stepId
  1183→    if (data.status === 'running') {
  1184→      set((s) => ({
  1185→        status: 'tool_running',
  1186→        currentTool: `manage:${data.action}`,
  1187→        messages: s.messages.map((m) => {
  1188→          if (m.id !== assistantId) return m
  1189→          const steps = [...(m.steps ?? [])]
  1190→          steps.push({
  1191→            id: stepId,
  1192→            stepNumber: data.stepNumber ?? steps.length + 1,
  1193→            thought: data.thought,
  1194→            status: 'running',
  1195→            startedAt: Date.now(),

  1197→            manageAction: data.action,
  1198→            manageAttrs: data.attrs,
  1199→          })
  1200→          return { ...m, steps }
  1201→        }),
  1202→      }))
  1203→    } else {
  1204→      // done or error
  1205→      set((s) => ({
  1206→        status: 'thinking',
  1207→        currentTool: null,
  1208→        messages: s.messages.map((m) => {
  1209→          if (m.id !== assistantId) return m
  1210→          const steps = (m.steps ?? []).map((st): ToolStep =>
  1211→            st.id === stepId
  1212→              ? {
  1213→                  ...st,
  1214→                  status: data.status === 'error' ? 'error' : 'done',
  1215→                  manageResult: data.result,
  1216→                  finishedAt: Date.now(),
  1217→                }
  1218→              : st

  1223→    }
  1224→  } else if (event === 'subagents_updated') {

  1229→  } else if (event === 'heartbeat') {
  1230→    // UPGRADE #63 — Update heartbeat state so the dashboard shows real-time progress
  1231→    set((s) => ({ heartbeat: data }))
  1232→  } else if (event === 'error') {
  1233→    // UPGRADE #128: Flush any pending tokens BEFORE showing the error
  1234→    // (UPGRADE #125 token batching could swallow content if error arrives
  1235→    // before the 100ms flush timer fires)
  1236→    const pendingContent = _pendingTokens
  1237→    _pendingTokens = ''
  1238→    if (_tokenFlushTimer) {
  1239→      clearTimeout(_tokenFlushTimer)
  1240→      _tokenFlushTimer = null
  1241→    }
  1242→
  1243→    const msg: string = (data?.message ?? '').toString()
  1244→    // UPGRADE #131: Fixed regex — was matching "rate limit" in ALL error messages
  1245→    // because the error text from agent.ts says "Rate limit reached on all active providers"
  1246→    // even when the actual cause is a DB failure or network timeout.
  1247→    // NOW: only treat as rate limit if the message explicitly says "rate-limiting" (with hyphen)
  1248→    // or contains "429" or "too many requests" — NOT the generic "rate limit" phrase.

  1250→    // The error message from friendlyLlmError() was containing trigger words

  1255→    const isDBError = /database|prisma|Can't reach|connection|ECONNREFUSED|ETIMEDOUT/i.test(msg)

  1276→    // If data.message exists, we keep it as-is (it has the full breakdown)
  1277→
  1278→    set((s) => ({

  1280→        m.id === assistantId

  1285→                (pendingContent ? pendingContent : '') +

  1290→          : m
  1291→      ),
  1292→      status: 'idle',
  1293→      currentTool: null,
  1294→      heartbeat: null,
  1295→      activeSubagents: [],

  1297→    }))
  1298→  }
  1299→  // UPGRADE #154: 'done' event — flush any pending tokens IMMEDIATELY.
  1300→  // Before: the 'done' handler did nothing, relying on the outer loop to

  1304→  // After: flush ALL pending tokens synchronously when 'done' arrives.
  1305→  else if (event === 'done') {

  1307→    _pendingTokens = ''
