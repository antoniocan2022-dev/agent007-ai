   629→function BackupRestoreSection({ onToast }: { onToast: (msg: string) => void }) {
   630→  const [exporting, setExporting] = useState(false)
   631→  const [importing, setImporting] = useState(false)
   632→  const [importResult, setImportResult] = useState<any>(null)
   633→
   634→  const handleExport = async () => {
   635→    if (exporting) return
   636→    setExporting(true)
   637→    try {
   638→      const res = await fetch('/api/backup')
   639→      if (!res.ok) throw new Error(`Export failed (HTTP ${res.status})`)
   640→      const blob = await res.blob()
   641→      const url = URL.createObjectURL(blob)
   642→      const a = document.createElement('a')
   643→      a.href = url
   644→      a.download = `agent007-backup-${new Date().toISOString().slice(0, 10)}.json`
   645→      document.body.appendChild(a)
   646→      a.click()
   647→      document.body.removeChild(a)
   648→      URL.revokeObjectURL(url)
   649→      onToast('Backup downloaded successfully')
   650→    } catch (e: any) {
   651→      onToast(`Export failed: ${e?.message ?? 'unknown error'}`)
   652→    } finally {
   653→      setExporting(false)
   654→    }
   655→  }
   656→
   657→  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
   658→    const file = e.target.files?.[0]
   659→    if (!file || importing) return
   660→    setImporting(true)
   661→    setImportResult(null)
   662→    try {
   663→      const text = await file.text()
   664→      const backup = JSON.parse(text)
   665→      const res = await fetch('/api/backup', {
   666→        method: 'POST',
   667→        headers: { 'Content-Type': 'application/json' },
   668→        body: JSON.stringify({ backup }),
   669→      })
   670→      const result = await res.json()
   671→      if (!res.ok || !result.ok) throw new Error(result.error || `HTTP ${res.status}`)
   672→      setImportResult(result.stats)
   673→      onToast(
   674→        `Restored: ${result.stats.conversations} conversations, ${result.stats.messages} messages, ${result.stats.customSubagents} agents`
   675→      )
   676→    } catch (e: any) {
   677→      onToast(`Import failed: ${e?.message ?? 'invalid file'}`)
   678→    } finally {
   679→      setImporting(false)
   680→      if (e.target) e.target.value = ''
   681→    }
   682→  }
   683→
   684→  return (
   685→    <section className="glass rounded-xl p-4 border-cyan-400/15">
   686→      <div className="flex items-center gap-2 mb-3">
   687→        <Database className="w-4 h-4 text-cyan-300" />
   688→        <h2 className="text-sm font-semibold text-[#e0e7ff] tracking-wide">BACKUP & RESTORE</h2>
   689→      </div>
   690→      <p className="text-[11px] text-[#7c89b5] mb-3 leading-relaxed">
   691→        Export all dashboard data (conversations, memories, income entries, schedules, custom
   692→        sub-agents, settings) as a JSON file. Restore on any Agent007 instance.
   693→      </p>
   694→      <div className="flex flex-col sm:flex-row gap-2">
   695→        <button
   696→          onClick={handleExport}
   697→          disabled={exporting}
   698→          className="flex-1 neon-btn-cyan rounded-lg py-2 text-xs font-bold tracking-wider flex items-center justify-center gap-2 disabled:opacity-60"
   699→          style={{ touchAction: 'manipulation' }}
   700→        >
   701→          {exporting ? (
   702→            <Loader2 className="w-3.5 h-3.5 animate-spin" />
   703→          ) : (
   704→            <Download className="w-3.5 h-3.5" />
   705→          )}
   706→          {exporting ? 'EXPORTING…' : 'EXPORT BACKUP'}
   707→        </button>
   708→        <label
   709→          className="flex-1 glass border-cyan-400/30 hover:border-cyan-400/60 rounded-lg py-2 text-xs font-bold tracking-wider flex items-center justify-center gap-2 cursor-pointer transition text-cyan-200"
   710→          style={{ touchAction: 'manipulation' }}
   711→        >
   712→          {importing ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Upload className="w-3.5 h-3.5" />}
   713→          {importing ? 'RESTORING…' : 'RESTORE BACKUP'}
   714→          <input
   715→            type="file"
   716→            accept="application/json,.json"
   717→            onChange={handleImport}
   718→            disabled={importing}
   719→            className="hidden"
   720→          />
   721→        </label>
   722→      </div>
   723→      {importResult && (
   724→        <div className="mt-3 text-[10px] text-[#9bb5d4] glass rounded-lg p-2 border-emerald-400/20">
   725→          <div className="flex items-center gap-1 mb-1 text-emerald-200 font-semibold">
   726→            <CheckCircle2 className="w-3 h-3" />
   727→            Last restore result:
   728→          </div>
   729→          <div className="grid grid-cols-2 gap-x-3 gap-y-0.5">
   730→            <span>Conversations: {importResult.conversations}</span>
   731→            <span>Messages: {importResult.messages}</span>
   732→            <span>Memories: {importResult.memories}</span>
   733→            <span>Income: {importResult.incomeEntries}</span>
   734→            <span>Schedules: {importResult.schedules}</span>
   735→            <span>Custom agents: {importResult.customSubagents}</span>
   736→          </div>
   737→          {importResult.errors?.length > 0 && (
   738→            <div className="mt-1 text-amber-200">{importResult.errors.length} warnings (skipped some rows)</div>
   739→          )}
   740→        </div>
   741→      )}
   742→    </section>

   753→  const [searchQuery, setSearchQuery] = useState('')

   759→  const loadDocs = useCallback(async () => {
   760→    try {
   761→      const res = await fetch('/api/kb')

   764→    } catch {
   765→      setDocs([])
   766→    } finally {
   767→      setLoading(false)
   768→    }
   769→  }, [])
   770→
   771→  useEffect(() => {
   772→    loadDocs()
   773→  }, [loadDocs])
   774→
   775→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   776→    const file = e.target.files?.[0]
   777→    if (!file || uploading) return
   778→    setUploading(true)
   779→    try {
   780→      const fd = new FormData()
   781→      fd.append('file', file)
   782→      const res = await fetch('/api/kb', { method: 'POST', body: fd })
   783→      const data = await res.json()
   784→      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`)
   785→      onToast(`Uploaded "${data.doc.filename}" — ${data.doc.chunkCount} chunks indexed`)

   788→      onToast(`Upload failed: ${e?.message ?? 'error'}`)

   790→      setUploading(false)

   798→      await fetch(`/api/kb?id=${docId}`, { method: 'DELETE' })
   799→      onToast(`Deleted "${filename}"`)

   801→    } catch (e: any) {
   802→      onToast(`Delete failed: ${e?.message ?? 'error'}`)

   811→      const res = await fetch('/api/kb/search', {

   813→        headers: { 'Content-Type': 'application/json' },

   818→      setSearchResults(data.results || [])
   819→    } catch (e: any) {
   820→      onToast(`Search failed: ${e?.message ?? 'error'}`)
   821→    } finally {
   822→      setSearching(false)
   823→    }
   824→  }
   825→
   826→  return (
   827→    <section className="glass rounded-xl p-4 border-cyan-400/15">
   828→      <div className="flex items-center gap-2 mb-3">
   829→        <BookMarked className="w-4 h-4 text-cyan-300" />
   830→        <h2 className="text-sm font-semibold text-[#e0e7ff] tracking-wide">KNOWLEDGE BASE (RAG)</h2>
   831→      </div>
   832→      <p className="text-[11px] text-[#7c89b5] mb-3 leading-relaxed">
   833→        Upload documents (TXT, MD, CSV, JSON, PDF, code files). Agent007 and all sub-agents can
   834→        search your knowledge base via the <code className="text-cyan-200">kb_search</code> tool
   835→        to ground their answers in your own data.
   836→      </p>
   837→
   838→      {/* Upload button */}
   839→      <label
   840→        className="flex items-center justify-center gap-2 w-full glass border-cyan-400/30 hover:border-cyan-400/60 rounded-lg py-2 text-xs font-bold tracking-wider cursor-pointer transition text-cyan-200 mb-3"
   841→        style={{ touchAction: 'manipulation' }}
   842→      >
   843→        {uploading ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Upload className="w-3.5 h-3.5" />}
   844→        {uploading ? 'INDEXING…' : 'UPLOAD DOCUMENT'}
   845→        <input
   846→          ref={fileRef}
   847→          type="file"
   848→          onChange={handleUpload}
   849→          disabled={uploading}
   850→          className="hidden"
   851→          accept=".txt,.md,.csv,.json,.js,.ts,.tsx,.jsx,.py,.go,.rs,.java,.c,.cpp,.h,.sh,.sql,.yaml,.yml,.xml,.html,.css,.pdf"

   861→            value={searchQuery}
