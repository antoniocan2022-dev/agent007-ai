   233→   233→export function SubAgentsPanel() {
   234→   234→  const subagentsVersion = useChatStore((s) => s.subagentsVersion)
   235→   235→  const bumpSubagents = useChatStore((s) => s.bumpSubagents)
   236→   236→
   237→   237→  const [rows, setRows] = useState<SubagentRow[]>([])
   238→   238→  const [loading, setLoading] = useState(true)
   239→   239→  const [error, setError] = useState('')
   240→   240→  const [editing, setEditing] = useState<SubagentRow | null>(null)
   241→   241→  const [creating, setCreating] = useState(false)
   242→   242→  const [templatesOpen, setTemplatesOpen] = useState(false)
   243→   243→  // When the user picks a quick template, we open the create modal with the
   244→   244→  // preset values pre-filled. We pass the preset via `templatePreset` state.
   245→   245→  const [templatePreset, setTemplatePreset] = useState<QuickTemplate['preset'] | null>(null)
   246→   246→
   247→   247→  const load = useCallback(async () => {
   248→   248→    setLoading(true)
   249→   249→    setError('')
   250→   250→    try {
   251→   251→      const res = await fetch('/api/subagents')
   252→   252→      const json = await res.json()
   253→   253→      if (!res.ok || json.error) throw new Error(json.error ?? `HTTP ${res.status}`)
   254→   254→      setRows(json.subagents ?? [])
   255→   255→    } catch (e: any) {
   256→   256→      setError(e?.message ?? 'Failed to load subagents')
   257→   257→    } finally {
   258→   258→      setLoading(false)
   259→   259→    }
   260→   260→  }, [])
   261→   261→
   262→   262→  useEffect(() => {
   263→   263→    load()
   264→   264→  }, [load, subagentsVersion])
   265→   265→
   266→   266→  const onToggle = async (r: SubagentRow) => {
   267→   267→    try {
   268→   268→      const res = await fetch(`/api/subagents/${encodeURIComponent(r.id)}`, {
   269→   269→        method: 'PUT',
   270→   270→        headers: { 'Content-Type': 'application/json' },
   271→   271→        body: JSON.stringify({ enabled: !r.enabled }),
   272→   272→      })
   273→   273→      const json = await res.json()
   274→   274→      if (!res.ok || json.error) throw new Error(json.error ?? `HTTP ${res.status}`)
   275→   275→      bumpSubagents()
   276→   276→      // optimistic refresh
   277→   277→      setRows((prev) =>
   278→   278→        prev.map((p) => (p.id === r.id ? { ...p, enabled: !r.enabled } : p))
   279→   279→      )
   280→   280→    } catch (e: any) {
   281→   281→      setError(e?.message ?? 'Failed to toggle subagent')
   282→   282→    }
   283→   283→  }
   284→   284→
   285→   285→  const onDelete = async (r: SubagentRow) => {
   286→   286→    if (
   287→   287→      !confirm(
   288→   288→        r.isBuiltin
   289→   289→          ? `Reset built-in agent "${r.name}" to defaults? Any overlay edits will be removed.`
   290→   290→          : `Delete custom sub-agent "${r.name}"? This cannot be undone.`
   291→   291→      )
   292→   292→    ) {
   293→   293→      return
   294→   294→    }
   295→   295→    try {
   296→   296→      const res = await fetch(`/api/subagents/${encodeURIComponent(r.id)}`, {
   297→   297→        method: 'DELETE',
   298→   298→      })
   299→   299→      const json = await res.json()
   300→   300→      if (!res.ok || json.error) throw new Error(json.error ?? `HTTP ${res.status}`)
   301→   301→      bumpSubagents()
   302→   302→      setRows((prev) => prev.filter((p) => p.id !== r.id))
   303→   303→    } catch (e: any) {
   304→   304→      setError(e?.message ?? 'Failed to delete subagent')
   305→   305→    }
   306→   306→  }
   307→   307→
   308→   308→  const onEditClick = async (r: SubagentRow) => {
   309→   309→    try {
   310→   310→      const res = await fetch(`/api/subagents/${encodeURIComponent(r.id)}`)
   311→   311→      const json = await res.json()
   312→   312→      if (!res.ok || json.error) throw new Error(json.error ?? `HTTP ${res.status}`)
   313→   313→      setEditing(json.subagent as SubagentRow)
   314→   314→    } catch (e: any) {
   315→   315→      setError(e?.message ?? 'Failed to load subagent details')
   316→   316→    }
   317→   317→  }
   318→   318→
   319→   319→  const onCloseEdit = (didMutate: boolean) => {
   320→   320→    setEditing(null)
   321→   321→    setCreating(false)
   322→   322→    setTemplatePreset(null)
   323→   323→    if (didMutate) {
   324→   324→      bumpSubagents()
   325→   325→      load()
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  return (
   330→   330→    <section className="glass rounded-xl p-4 sm:p-5">
   331→   331→      <div className="flex items-center justify-between gap-2 mb-3 flex-wrap">
   332→   332→        <div className="flex items-center gap-2">
   333→   333→          <Users className="w-4 h-4 text-cyan-300" />
   334→   334→          <h2 className="text-sm font-semibold text-[#e0e7ff] tracking-wide">SUB-AGENTS</h2>
   335→   335→          <span className="text-[9px] text-[#5b6a92] tracking-wider">
   336→   336→            {rows.length} total ({rows.filter((r) => r.isBuiltin).length} built-in +{' '}
   337→   337→            {rows.filter((r) => !r.isBuiltin).length} custom)
   338→   338→          </span>
   339→   339→        </div>
   340→   340→        <div className="flex items-center gap-2 flex-wrap">
   341→   341→          {/* Quick Templates dropdown (#9) */}
   342→   342→          <div className="relative">
   343→   343→            <button
   344→   344→              onClick={() => setTemplatesOpen((v) => !v)}
   345→   345→              className="h-8 px-3 rounded-lg glass border-cyan-400/30 hover:border-cyan-400/70 text-cyan-200 text-[10px] font-bold tracking-wider flex items-center gap-1.5 transition"
   346→   346→              style={{ touchAction: 'manipulation' }}
   347→   347→              aria-haspopup="menu"
   348→   348→              aria-expanded={templatesOpen}
   349→   349→              aria-label="Quick agent templates"
   350→   350→            >
   351→   351→              <Sparkles className="w-3.5 h-3.5" />
   352→   352→              QUICK TEMPLATES
   353→   353→              <ChevronDown
   354→   354→                className={`w-3 h-3 transition-transform ${templatesOpen ? 'rotate-180' : ''}`}
   355→   355→              />
   356→   356→            </button>
   357→   357→            <AnimatePresence>
   358→   358→              {templatesOpen && (
   359→   359→                <>
   360→   360→                  {/* click-away catcher */}
   361→   361→                  <div
   362→   362→                    className="fixed inset-0 z-30"
   363→   363→                    onClick={() => setTemplatesOpen(false)}
   364→   364→                    aria-hidden
   365→   365→                  />
   366→   366→                  <motion.div
   367→   367→                    initial={{ opacity: 0, y: -6, scale: 0.96 }}
   368→   368→                    animate={{ opacity: 1, y: 0, scale: 1 }}
   369→   369→                    exit={{ opacity: 0, y: -6, scale: 0.96 }}
   370→   370→                    transition={{ duration: 0.15 }}
   371→   371→                    className="absolute right-0 top-full mt-1 w-64 glass-strong rounded-lg p-1.5 z-40"

   415→   415→                    <div className="mt-1 pt-1.5 border-t border-cyan-400/10 px-2 text-[8px] text-[#5b6a92] tracking-wide">
   416→   416→                      Click to pre-fill the new-agent modal (no auto-submit)
   417→   417→                    </div>
   418→   418→                  </motion.div>
   419→   419→                </>
   420→   420→              )}
   421→   421→            </AnimatePresence>
   422→   422→          </div>
   423→   423→          <button
   424→   424→            onClick={() => {

   428→   428→            className="h-8 px-3 rounded-lg neon-btn-cyan text-[10px] font-bold tracking-wider flex items-center gap-1.5"

   433→   433→          </button>
   434→   434→        </div>
   435→   435→      </div>
   436→   436→
   437→   437→      {error && (
   438→   438→        <div className="flex items-start gap-2 mb-3 px-3 py-2 rounded-lg bg-pink-500/10 border border-pink-400/40 text-pink-200 text-[11px]">
   439→   439→          <AlertCircle className="w-3.5 h-3.5 mt-0.5 flex-shrink-0" />
   440→   440→          <span>{error}</span>
   441→   441→          <button

   443→   443→            className="ml-auto text-pink-300 hover:text-pink-100"
   444→   444→            style={{ touchAction: 'manipulation' }}

   451→   451→
   452→   452→      {loading ? (

   454→   454→          <Loader2 className="w-4 h-4 animate-spin mr-2" />
   455→   455→          Loading sub-agents…
   456→   456→        </div>
   457→   457→      ) : (
   458→   458→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 max-h-[600px] overflow-y-auto scroll-cyan pr-1">
