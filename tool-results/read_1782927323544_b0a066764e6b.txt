    46→/* ------------------------------------------------------------------ *
    47→ * 1-hour in-memory cache for web_search + page_reader (#11).
    48→ *
    49→ * Only successful results are cached. image_gen / vision / code_exec /
    50→ * memory_* / file_read / wikipedia_* / free_apis_directory are NOT cached
    51→ * (either they're non-deterministic, expensive-to-cache, or already cheap).
    52→ * ------------------------------------------------------------------ */
    53→interface CacheEntry {
    54→  result: ToolResult
    55→  at: number
    56→}
    57→const _toolCache = new Map<string, CacheEntry>()
    58→const CACHE_TTL_MS = 60 * 60 * 1000 // 1 hour
    59→
    60→function cacheKey(toolName: string, args: any): string {
    61→  return toolName + ':' + JSON.stringify(args ?? {})
    62→}
    63→
    64→function getCached(toolName: string, args: any): ToolResult | null {
    65→  const k = cacheKey(toolName, args)
    66→  const e = _toolCache.get(k)
    67→  if (!e) return null
    68→  if (Date.now() - e.at > CACHE_TTL_MS) {
    69→    _toolCache.delete(k)
    70→    return null
    71→  }
    72→  return e.result
    73→}
    74→
    75→function setCached(toolName: string, args: any, result: ToolResult): void {
    76→  // Only cache successful results
    77→  if (!result.ok) return
    78→  _toolCache.set(cacheKey(toolName, args), { result, at: Date.now() })
    79→}

    81→/* ----------------------------- Web search ----------------------------- */
    82→export async function toolWebSearch(
    83→  args: { query?: string; num?: number; recency_days?: number },
    84→  _ctx: ToolContext
    85→): Promise<ToolResult> {
    86→  const query = (args?.query ?? '').toString().trim()
    87→  if (!query) return badResult('Missing "query" argument for web_search')
    88→  // Cache lookup (1-hour TTL)
    89→  const cached = getCached('web_search', args)
    90→  if (cached) {
    91→    return {
    92→      ...cached,
    93→      preview: '[cached] ' + cached.preview,
    94→    }
    95→  }
    96→  try {
    97→    const zai = await getZai()
    98→    const results = await zai.functions.invoke('web_search', {
    99→      query,
   100→      num: Math.min(Math.max(args.num ?? 5, 1), 10),
   101→      recency_days: args.recency_days,
   102→    })
   103→    if (!Array.isArray(results) || results.length === 0) {
   104→      return okResult(
   105→        `No results found for "${query}"`,
   106→        `No web search results found for query: "${query}".`
   107→      )
   108→    }
   109→    const formatted = results
   110→      .map((r: any, i: number) => {
   111→        const snip = (r.snippet || '').toString().slice(0, 400)
   112→        return `${i + 1}. **${r.name || r.url}**\n   URL: ${r.url}\n   ${snip}${r.date ? `\n   Date: ${r.date}` : ''}`
   113→      })
   114→      .join('\n\n')
   115→    const preview = results
   116→      .slice(0, 3)
   117→      .map((r: any) => `• ${r.name || r.url}`)
   118→      .join('\n')
   119→    const out = okResult(preview, formatted)
   120→    setCached('web_search', args, out)
   121→    return out
   122→  } catch (e: any) {
   123→    return badResult(`web_search failed: ${e?.message ?? String(e)}`)
   124→  }
   125→}

   127→/* ----------------------------- Page reader ---------------------------- */
   128→export async function toolPageReader(
   129→  args: { url?: string },
   130→  _ctx: ToolContext
   131→): Promise<ToolResult> {
   132→  const url = (args?.url ?? '').toString().trim()
   133→  if (!url) return badResult('Missing "url" argument for page_reader')
   134→  // Cache lookup (1-hour TTL)
   135→  const cached = getCached('page_reader', args)
   136→  if (cached) {
   137→    return {
   138→      ...cached,
   139→      preview: '[cached] ' + cached.preview,
   140→    }
   141→  }
   142→
   143→  // URL validation — pre-check with a HEAD request to avoid wasting the
   144→  // page_reader call (and an LLM iteration) on 404s / 5xxs. This was a
   145→  // recurring issue during cybersecurity testing where the agent would
   146→  // try URLs like "owasp.org/Top10/2025/0x01_2025-Broken_Access_Control"
   147→  // that returned 404, wasting iterations.
   148→  try {
   149→    const urlObj = new URL(url)
   150→    // Only validate http/https URLs
   151→    if (urlObj.protocol === 'http:' || urlObj.protocol === 'https:') {
   152→      const controller = new AbortController()
   153→      const timeout = setTimeout(() => controller.abort(), 5000)
   154→      try {
   155→        const headRes = await fetch(url, {
   156→          method: 'HEAD',
   157→          signal: controller.signal,
   158→          redirect: 'follow',
   159→          headers: { 'User-Agent': 'Agent007-AI/2.0' },
   160→        })
   161→        clearTimeout(timeout)
   162→        // 405 = server doesn't support HEAD (common) — allow these through
   163→        // 403 = often blocks bots but page_reader may still work — allow through
   164→        if (headRes.status === 404 || headRes.status >= 500) {
   165→          const skip = `URL returned HTTP ${headRes.status} — skipping page_reader to save iterations. Try a different URL or use web_search to find the correct one.`
   166→          return badResult(skip)
   167→        }
   168→      } catch (headErr: any) {
   169→        // HEAD failed (timeout, CORS, DNS) — don't block, fall through to page_reader
   170→        clearTimeout(timeout)
   171→      }
   172→    }
   173→  } catch {
   174→    // Invalid URL — fall through to page_reader which will report the error
   175→  }
   176→
   177→  try {
   178→    const zai = await getZai()
   179→    const res: any = await zai.functions.invoke('page_reader', { url })
   180→    const html = res?.data?.html ?? ''
   181→    const title = res?.data?.title ?? url
   182→    const text = stripHtml(html).slice(0, 6000)
   183→    const out = okResult(
   184→      `Read page: ${title}\n${text.slice(0, 400)}...`,
   185→      `Page: ${title}\nURL: ${url}\n\n${text}`
   186→    )
   187→    setCached('page_reader', args, out)
   188→    return out
   189→  } catch (e: any) {
   190→    return badResult(`page_reader failed: ${e?.message ?? String(e)}`)
   191→  }
   192→}
   193→
   194→function stripHtml(html: string): string {
   195→  return html
   196→    .replace(/<script[\s\S]*?<\/script>/gi, '')
   197→    .replace(/<style[\s\S]*?<\/style>/gi, '')
   198→    .replace(/<[^>]+>/g, ' ')
   199→    .replace(/&nbsp;/g, ' ')
   200→    .replace(/&amp;/g, '&')
   201→    .replace(/&lt;/g, '<')
   202→    .replace(/&gt;/g, '>')
   203→    .replace(/\s+/g, ' ')
   204→    .trim()
   205→}

   208→const VALID_SIZES = [
   209→  '1024x1024',
   210→  '768x1344',
   211→  '864x1152',
   212→  '1344x768',
   213→  '1152x864',
   214→  '1440x720',

   218→export async function toolImageGen(
   219→  args: { prompt?: string; size?: string },
   220→  _ctx: ToolContext
   221→): Promise<ToolResult> {
   222→  const prompt = (args?.prompt ?? '').toString().trim()
   223→  if (!prompt) return badResult('Missing "prompt" argument for image_gen')
   224→  const size = (VALID_SIZES as readonly string[]).includes(args.size ?? '')
   225→    ? (args.size as (typeof VALID_SIZES)[number])
   226→    : '1024x1024'
   227→  try {
   228→    const zai = await getZai()
   229→    const resp = await zai.images.generations.create({ prompt, size })
   230→    const b64 = resp?.data?.[0]?.base64
   231→    if (!b64) return badResult('image_gen returned no data')
   232→    const dataUrl = `data:image/png;base64,${b64}`
   233→    return {
   234→      ok: true,
   235→      preview: `Generated image (${size}): "${prompt.slice(0, 80)}"`,
   236→      result: `Image generated successfully. The image is embedded in this message as a PNG data URL (omitted here for brevity). Prompt: "${prompt}". Size: ${size}.`,
   237→      artifacts: [{ type: 'image', data: dataUrl, label: prompt.slice(0, 80) }],
   238→    }
   239→  } catch (e: any) {
   240→    return badResult(`image_gen failed: ${e?.message ?? String(e)}`)
   241→  }

   245→export async function toolVision(
   246→  args: { prompt?: string; image_index?: number; analyze_all?: boolean },
   247→  ctx: ToolContext
   248→): Promise<ToolResult> {
   249→  const prompt = (args?.prompt ?? 'Describe this in detail. Analyze all visual elements, text, data, colors, composition, and context. Provide a comprehensive analysis.').toString().trim()
   250→  const idx = Number(args?.image_index ?? 0)
   251→  const analyzeAll = args?.analyze_all === true || args?.image_index === -1
   252→
   253→  // Support BOTH images AND videos
   254→  const media = ctx.attachments.filter(
   255→    (a) => (a.mimeType.startsWith('image/') || a.mimeType.startsWith('video/')) && a.dataUrl
   256→  )
   257→  if (media.length === 0) {
   258→    // Check for video without dataUrl (saved to disk)
   259→    const videos = ctx.attachments.filter((a) => a.mimeType.startsWith('video/'))
   260→    if (videos.length > 0) {
   261→      const vid = videos[Math.min(idx, videos.length - 1)]
   262→      return okResult(
   263→        `Video analysis requested for ${vid.originalName}`,
   264→        `Video Analysis: ${vid.originalName}\n\nThe video was uploaded but the vision API requires a URL. Use python_exec with OpenCV or ffmpeg to extract key frames:\n\n\`\`\`python\nimport cv2\ncap = cv2.VideoCapture("${(vid as any).videoPath || vid.filename}")\nframe_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))\nfps = cap.get(cv2.CAP_PROP_FPS)\nduration = frame_count / fps if fps > 0 else 0\nprint(f"Video: {frame_count} frames, {fps:.1f} FPS, {duration:.1f}s")\nfor i, pct in enumerate([0.2, 0.4, 0.6, 0.8, 0.95]):\n    cap.set(cv2.CAP_PROP_POS_FRAMES, int(frame_count * pct))\n    ret, frame = cap.read()\n    if ret:\n        cv2.imwrite(f"/tmp/frame_{i}.jpg", frame)\n        print(f"Frame {i} saved at {pct*100:.0f}%")\ncap.release()\n\`\`\`\n\nThen re-upload the extracted frames as images and use the vision tool on each.`
   265→      )
   266→    }
   267→    return badResult(
   268→      'No attached image or video available for analysis. Ask the user to attach an image or video first.'
   269→    )
   270→  }
   271→
   272→  // === MULTI-IMAGE / MULTI-MEDIA MODE ===
   273→  if (analyzeAll && media.length > 1) {
   274→    try {
   275→      const zai = await getZai()
   276→      const content: any[] = [
   277→        { type: 'text', text: `${prompt}\n\nAnalyze ALL ${media.length} attached media items together. Compare, contrast, and synthesize insights across them.` },
   278→      ]
   279→      for (const m of media.slice(0, 10)) {
   280→        const isVideo = m.mimeType.startsWith('video/')
   281→        if (isVideo) {
   282→          content.push({ type: 'video_url', video_url: { url: m.dataUrl } })
   283→        } else {
   284→          content.push({ type: 'image_url', image_url: { url: m.dataUrl } })
   285→        }
   286→      }
   287→      const visionResp: any = await zai.chat.completions.createVision({
   288→        model: 'glm-4.5v',
   289→        messages: [{ role: 'user', content }],
   290→      })
   291→      const text = visionResp?.choices?.[0]?.message?.content ?? visionResp?.choices?.[0]?.message?.reasoning_content ?? ''
   292→      return okResult(`Multi-media analysis (${media.length} items): ${(text as string).slice(0, 200)}`, `Multi-Media Analysis of ${media.length} attached items: ${media.map((m) => m.originalName).join(', ')}\n\nPrompt: ${prompt}\n\nAnalysis:\n${text}`)
   293→    } catch (e: any) {
   294→      return badResult(`vision (multi) failed: ${e?.message ?? String(e)}`)
   295→    }
   296→  }
   297→
   298→  // === SINGLE MEDIA MODE ===
   299→  const item = media[Math.min(Math.max(idx, 0), media.length - 1)]
   300→  const isVideo = item.mimeType.startsWith('video/')
   301→  try {
   302→    const zai = await getZai()
   303→    const content: any[] = [{ type: 'text', text: prompt }]
   304→    if (isVideo) {
   305→      content.push({ type: 'video_url', video_url: { url: item.dataUrl } })
   306→    } else {
   307→      content.push({ type: 'image_url', image_url: { url: item.dataUrl } })
   308→    }
   309→    // Also include any attached file URLs (PDFs, documents) for multimodal analysis
   310→    const docs = ctx.attachments.filter(
   311→      (a) => !a.mimeType.startsWith('image/') && !a.mimeType.startsWith('video/') && a.dataUrl
   312→    )
   313→    for (const doc of docs.slice(0, 3)) {
   314→      content.push({ type: 'file_url', file_url: { url: doc.dataUrl } })
   315→    }
   316→    const visionResp: any = await zai.chat.completions.createVision({
   317→      model: 'glm-4.5v',
   318→      messages: [{ role: 'user', content }],
   319→    })
   320→    const text = visionResp?.choices?.[0]?.message?.content ?? visionResp?.choices?.[0]?.message?.reasoning_content ?? ''
   321→    const mediaType = isVideo ? 'Video' : 'Image'
   322→    return okResult(`${mediaType} analysis: ${(text as string).slice(0, 200)}`, `${mediaType} Analysis of ${item.originalName}:\n\nPrompt: ${prompt}\n\nAnalysis:\n${text}`)
   323→  } catch (e: any) {
   324→    return badResult(`vision failed: ${e?.message ?? String(e)}`)
   325→  }

   328→/* ----------------------------- Audio Transcription ----------------------------- */
   329→/**
   330→ * transcribe — converts audio to text using z-ai-web-dev-sdk ASR.
   331→ * Supports: WAV, MP3, OGG, FLAC, AAC, M4A, and any audio format.
   332→ */
   333→export async function toolTranscribe(
   334→  args: { audio_index?: number; prompt?: string },
   335→  ctx: ToolContext
   336→): Promise<ToolResult> {
   337→  const idx = Number(args.audio_index ?? 0)
   338→  const audioFiles = ctx.attachments.filter((a) => a.mimeType.startsWith('audio/'))
   339→  if (audioFiles.length === 0) {
   340→    return badResult('No attached audio file available. Ask the user to attach an audio file (WAV, MP3, OGG, etc.).')
   341→  }
   342→  const audio = audioFiles[Math.min(idx, audioFiles.length - 1)]
   343→  try {
   344→    let audioBase64 = (audio as any).audioBase64
   345→    if (!audioBase64) {
   346→      const { promises: fs } = await import('node:fs')
   347→      const path = await import('node:path')
   348→      const UPLOAD_DIR = '/home/z/my-project/download/uploads'
   349→      let full = path.join(UPLOAD_DIR, audio.filename)
   350→      try { await fs.access(full) } catch {
   351→        const files = await fs.readdir(UPLOAD_DIR)
   352→        const match = files.find((f) => f.endsWith('-' + audio.filename) || f.endsWith(audio.filename))
   353→        if (match) full = path.join(UPLOAD_DIR, match)
   354→      }
   355→      const buf = await fs.readFile(full)
   356→      audioBase64 = buf.toString('base64')
   357→    }
   358→    const zai = await getZai()
   359→    const response: any = await zai.audio.asr.create({ file_base64: audioBase64 })
   360→    const transcript = response?.text || ''
   361→    let result = `Audio Transcription of ${audio.originalName}:\n\nTranscript:\n${transcript}`
   362→    if (args.prompt) {
   363→      result += `\n\nAnalysis Request: ${args.prompt}\n\nThe transcript above can be used to answer your question. Key points extracted from the audio.`
   364→    }
   365→    return okResult(`Transcribed ${audio.originalName} (${transcript.length} chars)`, result)
   366→  } catch (e: any) {
   367→    return badResult(`transcribe failed: ${e?.message ?? String(e)}`)
   368→  }
   369→}

   491→        text = f.read()
   492→    print(f"Text length: {len(text)} chars")
   493→    print("---CONTENT---")
   494→    print(text[:10000])
   495→except:
   496→    with open("${filePath}", "rb") as f:
   497→        raw = f.read()
   498→    print(f"Binary file: {len(raw)} bytes")
   499→    print("---CONTENT---")
   500→    print(raw[:5000].decode("utf-8", errors="replace"))
   501→`
   502→  }
   503→
   504→  try {
   505→    const res = await fetch('http://localhost:3000/api/code/python', {
   506→      method: 'POST',
   507→      headers: { 'Content-Type': 'application/json' },
   508→      body: JSON.stringify({ code: pythonCode }),
   509→    })
   510→    const data = await res.json()
   511→    const output = data.stdout || ''
   512→    const errors = data.stderr || ''
   513→    if (data.exitCode !== 0 || errors) {
   514→      return okResult(`Document analysis (fallback): ${doc.originalName}`, `Document Analysis: ${doc.originalName}\nFormat: ${doc.mimeType}\n\nPython extraction error: ${errors.slice(0, 200)}\n\nFallback content:\n${doc.textContent || 'No text available'}\n\nPrompt: ${prompt}`)
   515→    }
   516→    let analysis = `Document Analysis: ${doc.originalName}\nFormat: ${doc.mimeType} (.${ext})\nSize: ${(doc.size / 1024).toFixed(1)}KB\n\n`
   517→    analysis += `Extraction Output:\n${output.slice(0, 15000)}\n\n`
   518→    analysis += `Analysis Request: ${prompt}\nAnalysis Type: ${analysisType}\n`
   519→    return okResult(`Document analyzed via Python extraction (${output.length} chars)`, analysis)
   520→  } catch (e: any) {
   521→    return badResult(`document_analyze failed: ${e?.message ?? String(e)}`)
   522→  }
   523→}

   525→/* ----------------------------- Code execution ------------------------- */
   526→export async function toolCodeExec(
   527→  args: { code?: string },
   528→  _ctx: ToolContext
   529→): Promise<ToolResult> {
   530→  const raw = (args?.code ?? '').toString()
   531→  if (!raw.trim()) return badResult('Missing "code" argument for code_exec')
   532→
   533→  const logs: string[] = []
   534→  const sandboxConsole = {
   535→    log: (...a: any[]) => logs.push(a.map(stringifyForLog).join(' ')),
   536→    error: (...a: any[]) => logs.push('[error] ' + a.map(stringifyForLog).join(' ')),
   537→    warn: (...a: any[]) => logs.push('[warn] ' + a.map(stringifyForLog).join(' ')),
   538→    info: (...a: any[]) => logs.push(a.map(stringifyForLog).join(' ')),
   539→  }
   540→
   541→  // Decide expression vs statement form
   542→  let codeToRun = raw.trim()
   543→  let isExpression = false
   544→  try {
   545→    // Try parsing as expression first
   546→    new Function(`return (${codeToRun})`)
   547→    isExpression = true
   548→  } catch {
   549→    isExpression = false
   550→  }
   551→
   552→  // Wrap in an IIFE so `return` is legal at the top of the function body.
   553→  // vm.Script does NOT allow top-level return statements, so we MUST wrap.
   554→  const wrapped = isExpression
   555→    ? `"use strict";\n(() => { return (${codeToRun}); })()`
   556→    : `"use strict";\n(() => {\n${codeToRun}\n})()`
   557→
   558→  const sandbox = {
   559→    Math,
   560→    JSON,
   561→    console: sandboxConsole,
   562→    Date,
   563→    Array,
   564→    Object,
   565→    String,
   566→    Number,
   567→    Boolean,
   568→    Math_utils: Math,
   569→    parseInt,
   570→    parseFloat,
   571→    isNaN,
   572→    isFinite,
   573→    undefined,
   574→  }
   575→
   576→  try {
   577→    const script = new vm.Script(wrapped, { filename: 'sandbox.js' })
   578→    const context = vm.createContext(sandbox)
   579→    let value: any
   580→    try {
   581→      value = script.runInContext(context, { timeout: 3000, microtaskMode: 'afterEvaluate' })
   582→    } catch (err: any) {
   583→      const msg = err?.message ?? String(err)
   584→      return badResult(`Runtime error: ${msg}\nConsole output:\n${logs.join('\n')}`)
   585→    }
   586→    let valueStr: string
   587→    try {
   588→      valueStr =
   589→        value === undefined ? 'undefined' : typeof value === 'string' ? value : JSON.stringify(value, null, 2)
   590→      if (valueStr === undefined) valueStr = 'undefined'
   591→    } catch {
   592→      valueStr = String(value)
   593→    }
   594→    const resultText = `Console output:\n${logs.join('\n')}${logs.length ? '\n' : ''}\nReturn value:\n${valueStr}`
   595→    const preview = `console: ${logs.length} line(s) • returns: ${valueStr.slice(0, 80)}`
   596→    return okResult(preview, resultText)
   597→  } catch (e: any) {
   598→    return badResult(`code_exec failed: ${e?.message ?? String(e)}`)
   599→  }
   600→}

   602→function stringifyForLog(v: any): string {
   603→  if (typeof v === 'string') return v
   604→  try {
   605→    return JSON.stringify(v)
   606→  } catch {
   607→    return String(v)
   608→  }

   615→): Promise<ToolResult> {

   671→  _ctx: ToolContext

   719→/* ----------------------------- Wikipedia search ----------------------- */
   720→/**
   721→ * Search Wikipedia's free API. No API key required.
   722→ *
   723→ * NOTE: We deliberately omit `origin=*` from the URL. That parameter is only
   724→ * needed for cross-origin browser requests (it triggers Wikimedia's CORS
   725→ * bypass which, combined with Node.js's undici TLS fingerprint, gets blocked
   726→ * with HTTP 403 by Wikimedia's bot detection in some sandbox environments).
   727→ * Server-side requests work fine without it.
   728→ *
   729→ * The User-Agent follows Wikimedia's policy (https://meta.wikimedia.org/wiki/User-Agent_policy):
   730→ * a descriptive UA with a contact URL. This also helps avoid 403 blocks.
   731→ *
   732→ * Docs: https://www.mediawiki.org/wiki/API:Search
   733→ */
   734→export async function toolWikipediaSearch(
   735→  args: { query?: string; limit?: number },
   736→  _ctx: ToolContext
   737→): Promise<ToolResult> {
   738→  const query = (args?.query ?? '').toString().trim()
   739→  if (!query) return badResult('Missing "query" argument for wikipedia_search')
   740→  const limit = Math.min(Math.max(args.limit ?? 5, 1), 20)
   741→  try {
   742→    const url =
   743→      'https://en.wikipedia.org/w/api.php?action=query&list=search' +
   744→      `&srsearch=${encodeURIComponent(query)}` +
   745→      `&srlimit=${limit}` +
   746→      '&format=json'
   747→    const resp = await fetch(url, {
   748→      headers: { 'User-Agent': 'Agent007-AI/1.0 (https://github.com/agent007; research bot)' },
   749→    })
   750→    if (!resp.ok) return badResult(`wikipedia_search HTTP ${resp.status}`)
   751→    const data: any = await resp.json()
   752→    const items: any[] = data?.query?.search ?? []
   753→    if (items.length === 0) {
   754→      return okResult(
   755→        `No Wikipedia articles matched "${query}".`,
   756→        `No Wikipedia articles found for query: "${query}".`
   757→      )
   758→    }
   759→    const formatted = items
   760→      .map((it, i) => {
   761→        const title = it.title ?? ''
   762→        const snip = (it.snippet ?? '').toString().replace(/<[^>]+>/g, '').slice(0, 400)
   763→        const url = `https://en.wikipedia.org/?curid=${it.pageid}`
   764→        return `${i + 1}. **${title}**\n   URL: ${url}\n   ${snip}`
   765→      })
   766→      .join('\n\n')
   767→    const preview = items
   768→      .slice(0, 3)
   769→      .map((it) => `• ${it.title}`)
   770→      .join('\n')
   771→    return okResult(preview, formatted)
   772→  } catch (e: any) {
   773→    return badResult(`wikipedia_search failed: ${e?.message ?? String(e)}`)
   774→  }
   775→}

   777→/* ----------------------------- Wikipedia read ------------------------- */
   778→/**
   779→ * Read a full Wikipedia article. Uses the parse API to get wikitext,
   780→ * then strips wikitext markup to readable plain text. Truncated to 8000 chars.
   781→ *
   782→ * Same `origin=*` omission + descriptive UA as `toolWikipediaSearch` —
   783→ * see that function's docstring for rationale.
   784→ *
   785→ * Docs: https://www.mediawiki.org/wiki/API:Parsing_wikitext
   786→ */
   787→export async function toolWikipediaRead(
   788→  args: { title?: string },
   789→  _ctx: ToolContext
   790→): Promise<ToolResult> {
   791→  const title = (args?.title ?? '').toString().trim()
   792→  if (!title) return badResult('Missing "title" argument for wikipedia_read')
   793→  try {
   794→    const url =
   795→      'https://en.wikipedia.org/w/api.php?action=parse' +
   796→      `&page=${encodeURIComponent(title)}` +
   797→      '&prop=wikitext&format=json'
   798→    const resp = await fetch(url, {
   799→      headers: { 'User-Agent': 'Agent007-AI/1.0 (https://github.com/agent007; research bot)' },
   800→    })
   801→    if (!resp.ok) return badResult(`wikipedia_read HTTP ${resp.status}`)
   802→    const data: any = await resp.json()
   803→    if (data?.error) {
   804→      return badResult(`wikipedia_read: ${data.error.info ?? data.error.code}`)
   805→    }
   806→    const wikitext: string = data?.parse?.wikitext?.['*'] ?? ''
   807→    if (!wikitext) {
   808→      return okResult(
   809→        `No content for "${title}".`,
   810→        `Wikipedia article "${title}" returned empty content.`
   811→      )
   812→    }
   813→    const plain = stripWikitext(wikitext).slice(0, 8000)
   814→    const articleUrl = `https://en.wikipedia.org/wiki/${encodeURIComponent(title.replace(/ /g, '_'))}`
   815→    return okResult(
   816→      `Read Wikipedia: ${title}\n${plain.slice(0, 400)}...`,
   817→      `Title: ${title}\nURL: ${articleUrl}\n\n${plain}`
   818→    )
   819→  } catch (e: any) {
   820→    return badResult(`wikipedia_read failed: ${e?.message ?? String(e)}`)
   821→  }
   822→}

   824→/**
   825→ * Strip MediaWiki wikitext markup to readable plain text.
   826→ * Best-effort — handles the most common markup patterns.
   827→ */
   828→function stripWikitext(wt: string): string {
   829→  return wt
   830→    // HTML comments
   831→    .replace(/<!--[\s\S]*?-->/g, '')
   832→    // Refs and citations
   833→    .replace(/<ref[^>]*\/>/gi, '')
   834→    .replace(/<ref[^>]*>[\s\S]*?<\/ref>/gi, '')
   835→    // Other HTML tags (keep their inner text)
   836→    .replace(/<[^>]+>/g, '')
   837→    // Headings: == Foo == → Foo
   838→    .replace(/^={2,}\s*(.+?)\s*={2,}$/gm, '$1')
   839→    // Bold/italic
   840→    .replace(/'''([^']+)'''/g, '$1')
   841→    .replace(/''([^']+)''/g, '$1')
   842→    // Internal wiki links: [[Foo|Bar]] → Bar, [[Foo]] → Foo
   843→    .replace(/\[\[[^\]]*\|([^\]]+)\]\]/g, '$1')
   844→    .replace(/\[\[([^\]]+)\]\]/g, '$1')
   845→    // External links: [http://x label] → label
   846→    .replace(/\[[^\s]+\s+([^\]]+)\]/g, '$1')
   847→    .replace(/\[([^\]]+)\]/g, '$1')
   848→    // Templates: {{...}}
   849→    .replace(/\{\{[^}]*\}\}/g, '')
   850→    // Tables: {| ... |}
   851→    .replace(/\{\|[\s\S]*?\|\}/g, '')
   852→    // Lists: "* item" → "item"
   853→    .replace(/^[\s]*[\*#:;]+\s*/gm, '')
   854→    // Multiple blank lines
   855→    .replace(/\n{3,}/g, '\n\n')
   856→    .trim()
   857→}

   859→/* ----------------------------- Free APIs directory -------------------- */
   860→/**
   861→ * Query the public-apis.org directory for free public APIs matching a keyword.
   862→ * No API key required. Returns list of API name + description + auth + HTTPS + link.
   863→ * Docs: https://github.com/davemachado/public-api
   864→ */
   865→export async function toolFreeApisDirectory(
   866→  args: { query?: string },
   867→  _ctx: ToolContext
   868→): Promise<ToolResult> {
   869→  const query = (args?.query ?? '').toString().trim()
   870→  if (!query) return badResult('Missing "query" argument for free_apis_directory')
   871→  try {
   872→    // public-apis.org returns a single object { count, entries: [...] }.
   873→    // We can filter by title (case-insensitive contains) on the client side
   874→    // to support multi-word queries like "real estate" or "open data".
   875→    const url = 'https://api.publicapis.org/entries'
   876→    const resp = await fetch(url, {
   877→      headers: { 'User-Agent': 'Agent007-AI/1.0 (research)' },
   878→    })
   879→    if (!resp.ok) return badResult(`free_apis_directory HTTP ${resp.status}`)
   880→    const data: any = await resp.json()
   881→    const all: any[] = Array.isArray(data?.entries) ? data.entries : []
   882→    if (all.length === 0) {
   883→      return okResult(
   884→        `No free APIs found.`,
   885→        `Free APIs directory returned no entries.`
   886→      )
   887→    }
   888→    const q = query.toLowerCase()
   889→    const matched = all.filter((e) => {
   890→      const title = (e.API ?? e.name ?? '').toString().toLowerCase()
   891→      const desc = (e.Description ?? '').toString().toLowerCase()
   892→      const cat = (e.Category ?? '').toString().toLowerCase()
   893→      return title.includes(q) || desc.includes(q) || cat.includes(q)
   894→    }).slice(0, 15)
   895→
   896→    if (matched.length === 0) {
   897→      return okResult(
   898→        `No free APIs matched "${query}".`,
   899→        `No free public APIs matched query "${query}". Try a broader keyword (e.g. "weather", "crypto", "data").`
   900→      )
   901→    }
   902→    const formatted = matched
   903→      .map((e, i) => {
   904→        const name = e.API ?? e.name ?? 'Unknown'
   905→        const desc = e.Description ?? ''
   906→        const auth = e.Auth ?? 'none'
   907→        const https = e.HTTPS ? 'Yes' : 'No'
   908→        const link = e.Link ?? ''
   909→        const cors = e.Cors ?? 'unknown'
   910→        return `${i + 1}. **${name}** (${e.Category ?? 'misc'})\n   Description: ${desc}\n   Auth: ${auth} • HTTPS: ${https} • CORS: ${cors}\n   Link: ${link}`
   911→      })
   912→      .join('\n\n')
   913→    const preview = matched
   914→      .slice(0, 3)
   915→      .map((e) => `• ${e.API ?? e.name}`)
   916→      .join('\n')
   917→    return okResult(preview, formatted)
   918→  } catch (e: any) {
   919→    return badResult(`free_apis_directory failed: ${e?.message ?? String(e)}`)
   920→  }
   921→}

   966→/* ----------------------------- Knowledge base search ----------------- */
   967→/**

   975→export async function toolKbSearch(
   976→  args: { query?: string; limit?: number },

   978→): Promise<ToolResult> {
   979→  const query = (args?.query ?? '').toString().trim()
   980→  if (!query) return badResult('Missing "query" argument for kb_search')
   981→  try {
   982→    // Dynamic import to avoid circular dependency + keep this lazy
   983→    const { searchKnowledgeBase, formatKbContext } = await import('@/lib/knowledge-base')
   984→    const { getSessionUserId } = await import('@/lib/session-user')
   985→    const userId = await getSessionUserId()
   986→    if (!userId) {

  1055→/* ----------------------------- File Write / Patch --------------------- */
  1056→/**
  1057→ * file_write — write or patch a source file in the project.
  1058→ * Restricted to the project directory. Creates a .bak backup before overwriting.
  1059→ *
  1060→ * Two modes:
  1061→ *   1. { path, content } — full file write (overwrites)
  1062→ *   2. { path, old_string, new_string } — surgical patch (replaces first occurrence)
  1063→ *
  1064→ * This lets the Developer agent ACTUALLY APPLY fixes, not just propose them.
  1065→ * Safety: blocks .env, node_modules, .git, .next, package.json, bun.lock.
  1066→ */
  1067→export async function toolFileWrite(
  1068→  args: { path?: string; content?: string; old_string?: string; new_string?: string },
  1069→  _ctx: ToolContext
  1070→): Promise<ToolResult> {
  1071→  const relPath = (args?.path ?? '').toString().trim()
  1072→  if (!relPath) return badResult('Missing "path" argument for file_write')
  1073→
  1074→  const PROJECT_ROOT = '/home/z/my-project'
  1075→  const resolved = path.resolve(PROJECT_ROOT, relPath)
  1076→  if (!resolved.startsWith(PROJECT_ROOT + '/') && resolved !== PROJECT_ROOT) {
  1077→    return badResult(`file_write: path "${relPath}" escapes project directory`)
  1078→  }
  1079→
  1080→  // Block sensitive files
  1081→  const blocked = ['.env', 'node_modules', '.git', '.next', 'package.json', 'bun.lock', 'prisma/schema.prisma']
  1082→  for (const b of blocked) {
  1083→    if (resolved.includes('/' + b + '/') || resolved.endsWith('/' + b)) {
  1084→      return badResult(`file_write: access to "${b}" is blocked for safety`)
  1085→    }
  1086→  }
  1087→
  1088→  try {
  1089→    // Read existing content (if file exists)
  1090→    let existing = ''
  1091→    let fileExists = true
  1092→    try {
  1093→      existing = await fs.readFile(resolved, 'utf8')
  1094→    } catch {
  1095→      fileExists = false
  1096→    }
  1097→
  1098→    // Create backup if file exists
  1099→    if (fileExists) {
  1100→      const backupPath = resolved + '.bak'
  1101→      await fs.writeFile(backupPath, existing)
  1102→    }
  1103→
  1104→    let newContent: string
  1105→    let mode: string
  1106→
  1107→    if (args.old_string !== undefined && args.new_string !== undefined) {
  1108→      // Patch mode — replace first occurrence of old_string with new_string
  1109→      mode = 'patch'
  1110→      const oldStr = args.old_string
  1111→      const newStr = args.new_string

  1113→        return badResult(`file_write (patch): old_string not found in ${relPath}. No changes made.`)
  1114→      }

  1118→      // Full write mode
  1119→      mode = 'full-write'
  1120→      newContent = args.content
  1121→    } else {
  1122→      return badResult('file_write: either {content} for full write OR {old_string, new_string} for patch mode')
  1123→    }
  1124→
  1125→    // Ensure parent directory exists
  1126→    await fs.mkdir(path.dirname(resolved), { recursive: true })
  1127→
  1128→    // Write the new content
  1129→    await fs.writeFile(resolved, newContent, 'utf8')
  1130→
  1131→    const oldLines = existing ? existing.split('\n').length : 0
  1132→    const newLines = newContent.split('\n').length
  1133→
  1134→    return okResult(
  1135→      `${mode === 'patch' ? 'Patched' : 'Wrote'} ${relPath} (${newLines} lines, was ${oldLines})`,
  1136→      `File: ${relPath}\nMode: ${mode}\nBackup: ${fileExists ? resolved + '.bak' : '(new file)'}\nOld lines: ${oldLines}\nNew lines: ${newLines}\nSize: ${newContent.length} bytes\n\nThe fix has been APPLIED to disk. Use source_read to verify.`
  1137→    )
  1138→  } catch (e: any) {
  1139→    return badResult(`file_write failed: ${e?.message ?? String(e)}`)
  1140→  }

  1493→export async function toolSendCommunication(args: { channel?: string; message?: string; to?: string; subject?: string }, _ctx: ToolContext): Promise<ToolResult> {
  1494→  const channel = (args.channel ?? 'whatsapp').toString()
  1495→  const message = (args.message ?? '').toString().trim()
  1496→  if (!message) return badResult('Missing "message" argument for send_communication')
  1497→  try {
  1498→    const res = await fetch('http://localhost:3000/api/commands/send', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ to: args.to || undefined, channel, message, subject: args.subject }) })
  1499→    const data = await res.json()
  1500→    if (data.ok) return okResult(`Sent via ${channel}: ${message.slice(0, 80)}`, `✅ Sent via ${channel.toUpperCase()}.\n\nMessage: ${message}\n\nResult: ${data.message}`)
  1501→    return badResult(`Failed to send via ${channel}: ${data.error || data.message}. Configure in Settings → WhatsApp Connect.`)
  1502→  } catch (e: any) { return badResult(`send_communication failed: ${e?.message ?? String(e)}`) }
  1503→}
  1504→TOOL_REGISTRY.send_communication = { fn: toolSendCommunication, icon: 'send', label: 'Send Communication (SMS/WhatsApp/Email)' }

  1506→export async function toolCheckInboundCommands(args: { status?: string; limit?: number }, _ctx: ToolContext): Promise<ToolResult> {
  1507→  const status = (args.status ?? 'pending').toString()
  1508→  const limit = Math.min(50, Math.max(1, args.limit ?? 20))
  1509→  try {
  1510→    const params = new URLSearchParams({ limit: String(limit) })
  1511→    if (status !== 'all') params.set('status', status)
  1512→    const res = await fetch(`http://localhost:3000/api/commands/inbound?${params}`)
  1513→    const data = await res.json()
  1514→    const commands = data.commands ?? []
  1515→    if (commands.length === 0) return okResult(`No ${status} commands`, `No ${status} inbound commands. Check again later.`)
  1516→    const formatted = commands.map((c: any, i: number) => `[${i + 1}] ID: ${c.id}\n    Source: ${(c.source || '').toUpperCase()}\n    From: ${c.fromNumber || c.fromEmail || 'unknown'}\n    Time: ${c.receivedAt ? new Date(c.receivedAt).toLocaleString() : ''}\n    Status: ${c.status}\n    Command: ${c.command}`).join('\n\n')
  1517→    return okResult(`${commands.length} ${status} command(s)`, `Inbound Commands (${status}):\nFound ${commands.length} command(s).\n\n${formatted}\n\nUse execute_inbound_command with the command ID to execute + reply.`)
  1518→  } catch (e: any) { return badResult(`check_inbound_commands failed: ${e?.message ?? String(e)}`) }
  1519→}
  1520→TOOL_REGISTRY.check_inbound_commands = { fn: toolCheckInboundCommands, icon: 'inbox', label: 'Check Inbound Commands (SMS/WhatsApp/Email)' }

  1522→export async function toolExecuteInboundCommand(args: { command_id?: string; reply_message?: string; mark_completed?: boolean }, _ctx: ToolContext): Promise<ToolResult> {
  1523→  const commandId = (args.command_id ?? '').toString().trim()
  1524→  if (!commandId) return badResult('Missing "command_id" argument for execute_inbound_command')
  1525→  const replyMessage = (args.reply_message ?? '').toString().trim()
  1526→  try {
  1527→    const res = await fetch('http://localhost:3000/api/commands/execute', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ commandId, replyMessage, markCompleted: args.mark_completed !== false }) })
  1528→    const data = await res.json()
  1529→    if (data.ok) return okResult(`Command ${commandId} executed`, `✅ Inbound command executed.\n\nCommand: ${data.command}\nSource: ${data.source}\nFrom: ${data.fromNumber || data.fromEmail}\nReply sent: ${replyMessage || '(auto)'}\nResult: ${data.result}`)
  1530→    return badResult(`Failed: ${data.error || data.message}`)
  1531→  } catch (e: any) { return badResult(`execute_inbound_command failed: ${e?.message ?? String(e)}`) }
  1532→}
