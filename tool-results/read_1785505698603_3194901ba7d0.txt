    94→export async function toolDecisionMatrix(args: any, _ctx: ToolContext): Promise<ToolResult> {
    95→  // UPGRADE #166: REAL implementation — uses weighted scoring with actual criteria.
    96→  const options = args?.options
    97→  const criteria = args?.criteria
    98→
    99→  if (!options || !Array.isArray(options) || options.length === 0) {
   100→    return badResult('Missing "options" array. Example: {"options":["A","B","C"],"criteria":{"speed":8,"cost":3,"quality":9}}')
   101→  }
   102→  if (!criteria || typeof criteria !== 'object') {
   103→    return badResult('Missing "criteria" object. Example: {"criteria":{"speed":8,"cost":3}}')
   104→  }
   105→
   106→  // If scores are provided for each option, use them. Otherwise, ask the LLM.
   107→  // UPGRADE #166: NEVER use Math.random(). If no scores, return "needs more data."
   108→  const criteriaEntries = Object.entries(criteria)
   109→  const totalWeight = criteriaEntries.reduce((sum, [_, w]) => sum + Number(w), 0)
   110→
   111→  const results = options.map((opt: any) => {
   112→    const optStr = typeof opt === 'string' ? opt : JSON.stringify(opt)
   113→    const scores = (args?.scores?.[optStr] || args?.scores?.[opt?.id || opt?.name]) as Record<string, number> | undefined
   114→
   115→    if (scores) {
   116→      // REAL scores provided — calculate weighted sum
   117→      const weightedSum = criteriaEntries.reduce((sum, [key, weight]) => {
   118→        return sum + (Number(scores[key] ?? 0) * Number(weight))
   119→      }, 0)
   120→      const normalizedScore = totalWeight > 0 ? (weightedSum / totalWeight).toFixed(1) : '0'
   121→      return { option: optStr, score: Number(normalizedScore), scores, hasData: true }
   122→    } else {
   123→      // No scores — don't fake it. Report honestly.
   124→      return { option: optStr, score: null, hasData: false }
   125→    }
   126→  })
   127→
   128→  const scored = results.filter((r: any) => r.hasData)
   129→  const unscored = results.filter((r: any) => !r.hasData)
   130→
   131→  if (scored.length === 0) {
   132→    return okResult(
   133→      `Decision matrix: ${options.length} options, 0 scored — needs data`,
   134→      `DECISION MATRIX\n${'='.repeat(60)}\n\nOPTIONS: ${options.map((o: any) => `"${typeof o === 'string' ? o : o.name || o.id}"`).join(', ')}\nCRITERIA: ${criteriaEntries.map(([k, w]) => `${k} (weight: ${w})`).join(', ')}\n\nSTATUS: No scores provided.\n\nTo use this tool:\n  1. Provide scores for each option: {"scores":{"Option A":{"speed":8,"cost":3}}}\n  2. The tool calculates weighted scores\n  3. The highest-scoring option is recommended\n\nAlternatively, ask the LLM directly: "Which option is best and why?"`
   135→    )
   136→  }
   137→
   138→  // Sort by score (highest first)
   139→  scored.sort((a: any, b: any) => b.score - a.score)
   140→  const winner = scored[0]
   141→
   142→  return okResult(
   143→    `Decision: "${winner.option}" wins with ${winner.score}/100`,
   144→    `DECISION MATRIX\n${'='.repeat(60)}\n\nCRITERIA (weights):\n${criteriaEntries.map(([k, w]) => `  ${k}: ${w}`).join('\n')}\n\nRESULTS:\n${scored.map((r: any) => `  ${r.option}: ${r.score}/100 ${r.option === winner.option ? '← WINNER' : ''}`).join('\n')}${unscored.length > 0 ? `\n\nUNSCORED (no data):\n${unscored.map((r: any) => `  ${r.option}`).join('\n')}` : ''}\n\nRECOMMENDATION: Choose "${winner.option}" (highest weighted score).`
   145→  )
   146→}

   342→export async function toolAutonomousDecisionMaker(args: any, _ctx: ToolContext): Promise<ToolResult> {
   343→  const decision = (args?.decision ?? 'what to focus on this week').toString()
   344→  const options = args?.options as string[] | undefined

   346→  try {

   349→    const all = await getAllPersistentMemory().catch(() => [])

   351→    const successCount = learnings.filter(m => m.score >= 60).length
   352→    const failCount = learnings.filter(m => m.score < 40).length
   353→
   354→    // Recall relevant learnings for this decision
   355→    const { recallPersistentMemory } = await import('./persistent-memory')
   356→    const relevant = await recallPersistentMemory(decision.slice(0, 100), 3).catch(() => [])
   357→
   358→    let llmDecision = ''
   359→    try {
   360→      const { callLlmWithRetry } = await import('./agent')
   361→      const optionsList = options && Array.isArray(options) && options.length > 0
   362→        ? `\nOPTIONS:\n${options.map((o, i) => `  ${String.fromCharCode(65 + i)}. ${o}`).join('\n')}`
   363→        : ''

   365→        ? `\nRELEVANT LEARNINGS (from past runs):\n${relevant.map(m => `  - [score: ${m.score}/100] ${m.value.slice(0, 200)}`).join('\n')}`
   366→        : ''
   367→      const systemPrompt = `You are a strategic decision maker. Given a decision and context, analyze and recommend the best option. Be concise (max 300 words). Use real data, not placeholder numbers.`
   368→      const userPrompt = `DECISION: ${decision}${optionsList}${learningContext}\n\nBased on ${successCount} successful past outcomes and ${failCount} failures, recommend the best path forward. Output:\nRECOMMENDATION: <option name>\nRATIONALE: <2-3 sentences>\nRISKS: <1-2 sentences>`
   369→
   370→      const result = await callLlmWithRetry([
   371→        { role: 'system', content: systemPrompt },
   372→        { role: 'user', content: userPrompt },
   373→      ])
   374→      llmDecision = result?.choices?.[0]?.message?.content ?? ''
   375→    } catch (e: any) {
   376→      llmDecision = `LLM unavailable: ${e?.message?.slice(0, 100) || 'unknown error'} — falling back to learning-based recommendation only.`
   377→    }
   378→
   379→    return okResult(
   380→      `Decision: analyzed "${decision.slice(0, 50)}" with ${relevant.length} relevant learnings`,
   381→      `AUTONOMOUS DECISION MAKER — LLM-DRIVEN\n${'='.repeat(60)}\n\n` +
   382→      `DECISION: "${decision}"\n\n` +
   383→      `REAL CONTEXT:\n` +
   384→      `  • Self-learning history: ${learnings.length} entries\n` +
   385→      `  • Successful past outcomes: ${successCount}\n` +
   386→      `  • Failed past outcomes: ${failCount}\n` +
   387→      `  • Relevant learnings for this decision: ${relevant.length}\n\n` +
   388→      `LLM ANALYSIS:\n${llmDecision || '(LLM did not return content)'}\n\n` +
   389→      `EXECUTION: This decision was made using ACTUAL data, not hardcoded metrics.`
   390→    )
   391→  } catch (e: any) {
   392→    return badResult(`Decision analysis failed: ${e?.message?.slice(0, 100)}`)
   393→  }

   449→// 5e. REAL Tool Usage Analyzer — counts ACTUAL tools in TOOL_REGISTRY
   450→export async function toolToolUsageAnalyzer(args: any, _ctx: ToolContext): Promise<ToolResult> {
   451→  // Import TOOL_REGISTRY to count actual tools
   452→  const { TOOL_REGISTRY } = await import('./tools')
   453→  const allTools = Object.keys(TOOL_REGISTRY)
   454→  const total = allTools.length
   455→
   456→  // Group by prefix (e.g., "memory_" → memory, "web_" → web)
   457→  const categories: Record<string, string[]> = {}
   458→  for (const name of allTools) {
   459→    const cat = name.includes('_') ? name.slice(0, name.indexOf('_')) : 'core'
   460→    if (!categories[cat]) categories[cat] = []
   461→    categories[cat].push(name)
   462→  }
   463→
   464→  const sorted = Object.entries(categories)
   465→    .map(([cat, tools]) => ({ cat, count: tools.length, sample: tools.slice(0, 3) }))
   466→    .sort((a, b) => b.count - a.count)
   467→
   468→  const top10 = sorted.slice(0, 10)
   469→
   470→  return okResult(
   471→    `Tool usage analysis: ${total} real tools in TOOL_REGISTRY across ${sorted.length} categories`,
   472→    `TOOL USAGE ANALYZER — REAL TOOL_REGISTRY COUNTS\n${'='.repeat(60)}\n\n` +
   473→    `TOTAL TOOLS: ${total}\n` +
   474→    `TOTAL CATEGORIES: ${sorted.length}\n\n` +
   475→    `TOP 10 CATEGORIES BY SIZE:\n${top10.map(c =>
   476→      `  ${c.cat}: ${c.count} tools (e.g., ${c.sample.join(', ')})`
   477→    ).join('\n')}\n\n` +
   478→    `HOW TO USE THIS:\n` +
   479→    `  • If a category has 0 useful tools for your task, dispatch a specialist with that capability\n` +
   480→    `  • If a category has 50+ tools, use smart_tool_router to find the right one\n` +
   481→    `  • Most tools support the JSON args format: <tool name="X">{"key":"value"}</tool>\n\n` +
   482→    `NOTE: This tool no longer reports fake "$890/mo projected" or "+78% conversion"\n` +
   483→    `metrics. The old version returned Math.random data. This version reports\n` +
   484→    `REAL tool counts from the actual TOOL_REGISTRY.`
   485→  )
   486→}

   490→export async function toolToolBoundaryAudit(args: any, _ctx: ToolContext): Promise<ToolResult> {
   491→  // UPGRADE #166: REAL tool — checks which tools a subagent ACTUALLY used
   492→  // vs what it was ALLOWED to use. Reports violations.
   493→  const agentId = (args?.agentId ?? '').toString().trim()
   494→  const toolsUsed = args?.toolsUsed as string[] | undefined
   495→
   496→  if (!agentId) {
   497→    return badResult('Missing "agentId" — which subagent to audit? Example: {"agentId":"scout","toolsUsed":["web_search","page_reader"]}')
   498→  }
   499→
   500→  // Get the subagent's allowed tools
   501→  try {
   502→    const { getAllSubagents } = await import('./subagents')
   503→    const allSubs = await getAllSubagents({ includeDisabled: false })
   504→    const sub = allSubs.find(s => s.id === agentId || s.name.toLowerCase() === agentId.toLowerCase())
   505→
   506→    if (!sub) {
   507→      return badResult(`Unknown subagent: "${agentId}"`)
   508→    }
   509→
   510→    const allowedTools = sub.allowedTools ?? []
   511→    const usedTools = toolsUsed ?? []
   512→
   513→    // Check for violations (used a tool NOT in allowedTools)
   514→    const violations = usedTools.filter(t => !allowedTools.includes(t))
   515→    const allowedButUnused = allowedTools.filter(t => !usedTools.includes(t))
   516→
   517→    return okResult(
   518→      violations.length === 0
   519→        ? `Boundary audit: ${agentId} OK — all ${usedTools.length} tools were allowed`
   520→        : `Boundary audit: ${agentId} — ${violations.length} VIOLATION(S)`,
   521→      `TOOL BOUNDARY AUDIT\n${'='.repeat(60)}\n\nAgent: ${sub.name} (${sub.id})\n\nALLOWED TOOLS (${allowedTools.length}):\n${allowedTools.map(t => `  ✅ ${t}`).join('\n')}\n\nTOOLS USED (${usedTools.length}):\n${usedTools.map(t => `  ${allowedTools.includes(t) ? '✅' : '❌'} ${t}`).join('\n')}\n\n${violations.length > 0 ? `⚠ VIOLATIONS (${violations.length}):\n${violations.map(v => `  ❌ ${v} — NOT in allowedTools`).join('\n')}` : '✅ No violations — all tools were within the allowed set.'}\n\n${allowedButUnused.length > 0 ? `\nALLOWED BUT UNUSED (${allowedButUnused.length}):\n${allowedButUnused.map(t => `  💤 ${t}`).join('\n')}` : ''}`
   522→    )
   523→  } catch (e: any) {
   524→    return badResult(`Audit failed: ${e?.message?.slice(0, 100)}`)
   525→  }
   526→}
