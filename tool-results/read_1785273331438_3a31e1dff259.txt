   509→   509→ * Best-effort extract an artifact (URL, transaction ID, etc.) from the team's output.
   510→   510→ * Looks for common patterns.
   511→   511→ */
   512→   512→function extractArtifact(output: string, type: PipelineStage['artifactType']): string | null {
   513→   513→  if (type === 'none') return null
   514→   514→  if (type === 'url') {
   515→   515→    // Find first http(s) URL that looks like a deliverable (not example.com)
   516→   516→    const urlMatch = output.match(/https?:\/\/(?!example\.com|localhost|example\.org)[^\s<>"')\]]+/i)
   517→   517→    return urlMatch ? urlMatch[0] : null
   518→   518→  }
   519→   519→  if (type === 'transaction_id') {
   520→   520→    // Stripe-style IDs (ch_, pi_, txn_, etc.)
   521→   521→    const txMatch = output.match(/\b(ch|pi|txn|sub|in)_[A-Za-z0-9]{10,}\b/)
   522→   522→    return txMatch ? txMatch[0] : null
   523→   523→  }
   524→   524→  if (type === 'message_id') {
   525→   525→    // Telegram / Discord-style message IDs
   526→   526→    const msgMatch = output.match(/\bmsg[_-]?(\d{6,})\b/i) || output.match(/\b(\d{10,})\b/)
   527→   527→    return msgMatch ? msgMatch[0] : null
   528→   528→  }
   529→   529→  if (type === 'file_path') {
   530→   530→    // Path-like strings
   531→   531→    const pathMatch = output.match(/(?:^|\s)(\/[\w./-]+\.\w+)/) || output.match(/([\w-]+\.(?:js|ts|tsx|jsx|py|md|json|html|css))/)
   532→   532→    return pathMatch ? pathMatch[1] : null
   533→   533→  }
   534→   534→  if (type === 'data') {
   535→   535→    // For data artifacts, return the first 500 chars as the "artifact"
   536→   536→    return output.slice(0, 500) || null
   537→   537→  }
   538→   538→  return null
   539→   539→}
