   258→   258→const MAX_ROUNDS_PER_STAGE = 3

   260→   260→/**
   261→   261→ * Run a single team leader with the Super Agent verification loop.
   262→   262→ * Returns the final approved output + verification metadata.
   263→   263→ */
   264→   264→async function runTeamWithVerificationLoop(opts: {
   265→   265→  missionId: string
   266→   266→  stage: PipelineStage
   267→   267→  objective: string
   268→   268→  missionContext: string
   269→   269→  previousTeamOutput?: string
   270→   270→}): Promise<{
   271→   271→  output: string
   272→   272→  rounds: number
   273→   273→  finalScore: number
   274→   274→  finalVerification: VerificationResult
   275→   275→  artifactValue: string | null
   276→   276→  artifactVerified: boolean
   277→   277→}> {
   278→   278→  const { missionId, stage, objective, missionContext, previousTeamOutput } = opts
   279→   279→  let currentPrompt = stage.promptTemplate(objective)
   280→   280→  let lastVerification: VerificationResult | null = null
   281→   281→  let teamOutput = ''
   282→   282→
   283→   283→  for (let round = 1; round <= MAX_ROUNDS_PER_STAGE; round++) {
   284→   284→    // ── Notify: stage start (round 1 only)
   285→   285→    if (round === 1) {
   286→   286→      await notifyTelegram(`🚀 Stage ${stage.stage}/${stage.team} STARTED\nMission: ${missionId}\nObjective: ${objective.slice(0, 100)}`)
   287→   287→      await logApprovalEvent({
   288→   288→        missionId,
   289→   289→        stageId: `stage_${stage.stage}`,
   290→   290→        round,
   291→   291→        agentRole: 'team_leader',
   292→   292→        agentId: stage.leader,
   293→   293→        action: 'started',
   294→   294→        feedback: `Stage ${stage.stage} (${stage.name}) started by ${stage.leader}`,
   295→   295→      })
   296→   296→    } else {
   297→   297→      // Retry round — note in audit log
   298→   298→      await logApprovalEvent({
   299→   299→        missionId,
   300→   300→        stageId: `stage_${stage.stage}`,
   301→   301→        round,
   302→   302→        agentRole: 'team_leader',
   303→   303→        agentId: stage.leader,
   304→   304→        action: 'retry_submitted',
   305→   305→        feedback: `Round ${round}: team re-submitting after Super Agent feedback`,
   306→   306→      })
   307→   307→    }
   308→   308→
   309→   309→    // ── Run the team leader
   310→   310→    // Special case: CEO stage uses callLlmWithRetry directly (no 'ceo' subagent exists).
   311→   311→    // The CEO is the apex LLM that aggregates everything into a final executive report.
   312→   312→    if (stage.team === 'ceo') {
   313→   313→      try {
   314→   314→        const { callLlmWithRetry } = await import('./agent')
   315→   315→        const ceoResponse = await callLlmWithRetry([
   316→   316→          {
   317→   317→            role: 'system',
   318→   318→            content: `You are the CEO of Agent007 — the apex executive reporting to the human owner (Antonio).
   319→   319→
   320→   320→A mission has just completed all of its team stages. Your job: AGGREGATE everything the teams produced and present a CLEAR, EXECUTIVE summary to Antonio.
   321→   321→
   322→   322→Antonio is busy. He needs to understand:
   323→   323→1. Did the mission succeed? (success / partial / failed)
   324→   324→2. What was actually delivered? (URLs, files, transaction IDs — concrete artifacts, not promises)
   325→   325→3. What is the revenue impact? (if applicable)
   326→   326→4. What risks or notes should he be aware of?
   327→   327→5. What are the next 3 recommended actions?
   328→   328→
   329→   329→FORMAT (strict):
   330→   330→🎯 MISSION: [one-line description]
   331→   331→📊 OUTCOME: [success/partial/failed]
   332→   332→💰 REVENUE IMPACT: [if applicable, otherwise "N/A"]
   333→   333→✅ KEY DELIVERABLES:
   334→   334→   - [bullet list with URLs/IDs]
   335→   335→⚠️ RISKS/NOTES:
   336→   336→   - [if any, otherwise "None"]
   337→   337→📈 NEXT STEPS:
   338→   338→   1. [action]
   339→   339→   2. [action]
   340→   340→   3. [action]
   341→   341→
   342→   342→RULES:
   343→   343→- Keep the report under 300 words.
   344→   344→- Be honest — if delivery failed, say so. Antonio trusts the CEO because the CEO never lies.
   345→   345→- Quote real artifact values (URLs, IDs), not vague descriptions.
   346→   346→- If revenue is $0, say "$0 so far — see next steps for monetization plan".`
   347→   347→          },
   348→   348→          { role: 'user', content: currentPrompt },
   349→   349→        ], { thinking: false })
   350→   350→
   351→   351→        teamOutput = typeof ceoResponse === 'string'
   352→   352→          ? ceoResponse
   353→   353→          : (ceoResponse?.content ?? ceoResponse?.message?.content ?? '(CEO produced no output)')
   354→   354→      } catch (ceoErr: any) {
   355→   355→        teamOutput = `🎯 MISSION: ${objective.slice(0, 100)}\n📊 OUTCOME: partial (CEO LLM unavailable)\n💰 REVENUE IMPACT: See artifacts below\n✅ KEY DELIVERABLES:\n   - (See stage results in audit trail)\n⚠️ RISKS/NOTES:\n   - CEO LLM was unavailable; this is a fallback report.\n📈 NEXT STEPS:\n   1. Review the audit trail manually.\n   2. Approve mission via dashboard.\n   3. Schedule the next mission.`
   356→   356→      }
   357→   357→    } else {
   358→   358→      const result = await runSubagent({
   359→   359→        subagentId: stage.leader,
   360→   360→        task: currentPrompt,
   361→   361→        attachments: [],
   362→   362→        language: 'en',
   363→   363→        emit: async () => {},  // silent — pipeline runs in background
   364→   364→        parentConversationId: `mission_${missionId}`,
   365→   365→        dispatchId: `pipeline_${missionId}_stage${stage.stage}_round${round}`,
   366→   366→      })
   367→   367→      teamOutput = result.answer
   368→   368→    }
   369→   369→
   370→   370→    // ── Special case: CEO stage doesn't get verified (it IS the verifier of the whole mission)
   371→   371→    if (stage.team === 'ceo') {
   372→   372→      await logApprovalEvent({
   373→   373→        missionId,
   374→   374→        stageId: `stage_${stage.stage}`,
   375→   375→        round,
   376→   376→        agentRole: 'ceo',
   377→   377→        agentId: 'ceo',
   378→   378→        action: 'submitted',
   379→   379→        feedback: teamOutput.slice(0, 500),
   380→   380→        score: 100,
   381→   381→      })
   382→   382→      return {
   383→   383→        output: teamOutput,
   384→   384→        rounds: round,
   385→   385→        finalScore: 100,
   386→   386→        finalVerification: {
   387→   387→          approved: true,
   388→   388→          verdict: 'APPROVED',
   389→   389→          score: 100,
   390→   390→          strengths: ['CEO final report — apex of mission'],
   391→   391→          weaknesses: [],
   392→   392→          corrections: [],
   393→   393→          summary: 'CEO stage auto-approved',
   394→   394→        },
   395→   395→        artifactValue: null,
   396→   396→        artifactVerified: true,
   397→   397→      }
   398→   398→    }
   399→   399→
   400→   400→    // ── Log submission to audit trail
   401→   401→    await logApprovalEvent({
   402→   402→      missionId,
   403→   403→      stageId: `stage_${stage.stage}`,
   404→   404→      round,
   405→   405→      agentRole: 'team_leader',
   406→   406→      agentId: stage.leader,
   407→   407→      action: 'submitted',
   408→   408→      feedback: teamOutput.slice(0, 500),
   409→   409→    })
   410→   410→
   411→   411→    // ── Super Agent verifies
   412→   412→    lastVerification = await superAgentVerify({
   413→   413→      teamOutput,
   414→   414→      missionContext,
   415→   415→      stageRequirements: stage.requirements,
   416→   416→      previousTeamOutput,
   417→   417→      round,
   418→   418→      maxRounds: MAX_ROUNDS_PER_STAGE,
   419→   419→    })
   420→   420→
   421→   421→    // ── Log verification result to audit trail
   422→   422→    await logApprovalEvent({
   423→   423→      missionId,
   424→   424→      stageId: `stage_${stage.stage}`,
   425→   425→      round,
   426→   426→      agentRole: 'super_agent',
   427→   427→      agentId: 'super_agent',
   428→   428→      action: lastVerification.approved ? 'approved' : 'rejected',
   429→   429→      score: lastVerification.score,
   430→   430→      feedback: formatVerificationResult(lastVerification).slice(0, 1000),
   431→   431→    })
   432→   432→
   433→   433→    // ── Approved → done
   434→   434→    if (lastVerification.approved) {
   435→   435→      await notifyTelegram(`✅ Stage ${stage.stage} (${stage.team}) APPROVED\nScore: ${lastVerification.score}/100, ${round} round(s)`)
   436→   436→      break
   437→   437→    }
   438→   438→
   439→   439→    // ── Notify rejection
   440→   440→    await notifyTelegram(`⚠️ Stage ${stage.stage} (${stage.team}) ${lastVerification.verdict}\nScore: ${lastVerification.score}/100, round ${round}/${MAX_ROUNDS_PER_STAGE}\nCorrections: ${lastVerification.corrections.length}`)
   441→   441→
   442→   442→    // ── Last round exhausted — break out, mission will escalate
   443→   443→    if (round === MAX_ROUNDS_PER_STAGE) {
   444→   444→      await logApprovalEvent({
   445→   445→        missionId,
   446→   446→        stageId: `stage_${stage.stage}`,
   447→   447→        round,
   448→   448→        agentRole: 'super_agent',
   449→   449→        agentId: 'super_agent',
   450→   450→        action: 'escalated',
   451→   451→        feedback: `Stage failed after ${MAX_ROUNDS_PER_STAGE} rounds — escalating to CEO`,
   452→   452→      })
   453→   453→      await notifyTelegram(`❌ Stage ${stage.stage} ESCALATED\nFailed after ${MAX_ROUNDS_PER_STAGE} rounds. CEO will note in final report.`)
   454→   454→      break
   455→   455→    }
   456→   456→
   457→   457→    // ── Build retry prompt for next round
   458→   458→    currentPrompt = buildRetryPrompt(
   459→   459→      stage.promptTemplate(objective),
   460→   460→      lastVerification,
   461→   461→      round + 1,
   462→   462→      MAX_ROUNDS_PER_STAGE
   463→   463→    )
   464→   464→  }
   465→   465→
   466→   466→  // ── Extract artifact from team output (best-effort)
   467→   467→  const artifactValue = extractArtifact(teamOutput, stage.artifactType)
   468→   468→  const artifactVerified = artifactValue !== null && lastVerification?.approved === true
   469→   469→
   470→   470→  return {
   471→   471→    output: teamOutput,
   472→   472→    rounds: lastVerification ? (lastVerification.approved ? 1 : MAX_ROUNDS_PER_STAGE) : 1,
   473→   473→    finalScore: lastVerification?.score ?? 0,
   474→   474→    finalVerification: lastVerification ?? {
   475→   475→      approved: false,
   476→   476→      verdict: 'REJECTED',
   477→   477→      score: 0,
   478→   478→      strengths: [],
   479→   479→      weaknesses: ['No verification performed'],
   480→   480→      corrections: [],
   481→   481→      summary: 'Verification skipped',
   482→   482→    },
   483→   483→    artifactValue,
   484→   484→    artifactVerified,
   485→   485→  }
   486→   486→}

   488→   488→/**
   489→   489→ * Best-effort extract an artifact (URL, transaction ID, etc.) from the team's output.
   490→   490→ * Looks for common patterns.
   491→   491→ */
   492→   492→function extractArtifact(output: string, type: PipelineStage['artifactType']): string | null {
   493→   493→  if (type === 'none') return null
   494→   494→  if (type === 'url') {
   495→   495→    // Find first http(s) URL that looks like a deliverable (not example.com)
   496→   496→    const urlMatch = output.match(/https?:\/\/(?!example\.com|localhost|example\.org)[^\s<>"')\]]+/i)
   497→   497→    return urlMatch ? urlMatch[0] : null
   498→   498→  }
   499→   499→  if (type === 'transaction_id') {
   500→   500→    // Stripe-style IDs (ch_, pi_, txn_, etc.)
   501→   501→    const txMatch = output.match(/\b(ch|pi|txn|sub|in)_[A-Za-z0-9]{10,}\b/)
   502→   502→    return txMatch ? txMatch[0] : null
   503→   503→  }
   504→   504→  if (type === 'message_id') {
   505→   505→    // Telegram / Discord-style message IDs
   506→   506→    const msgMatch = output.match(/\bmsg[_-]?(\d{6,})\b/i) || output.match(/\b(\d{10,})\b/)
   507→   507→    return msgMatch ? msgMatch[0] : null
   508→   508→  }
   509→   509→  if (type === 'file_path') {
   510→   510→    // Path-like strings
   511→   511→    const pathMatch = output.match(/(?:^|\s)(\/[\w./-]+\.\w+)/) || output.match(/([\w-]+\.(?:js|ts|tsx|jsx|py|md|json|html|css))/)
   512→   512→    return pathMatch ? pathMatch[1] : null
   513→   513→  }
   514→   514→  if (type === 'data') {
   515→   515→    // For data artifacts, return the first 500 chars as the "artifact"
   516→   516→    return output.slice(0, 500) || null
   517→   517→  }
   518→   518→  return null
   519→   519→}

   521→   521→/**
   522→   522→ * Check if owner has approved a high-stakes mission.
   523→   523→ * Rec 7 — polls the approval-audit-log for an 'owner_approved' event.
   524→   524→ */
   525→   525→async function waitForOwnerApproval(missionId: string, timeoutMs: number = 24 * 60 * 60 * 1000): Promise<boolean> {
   526→   526→  const startTime = Date.now()
   527→   527→  const pollInterval = 60 * 1000  // check every minute
   528→   528→
   529→   529→  await notifyTelegram(`⏸️ MISSION REQUIRES OWNER APPROVAL\nMission: ${missionId}\n\nThis is a high-stakes mission. Reply with /approve_${missionId.slice(0, 8)} to proceed, or /reject_${missionId.slice(0, 8)} to cancel.\n\nAuto-cancel in 24 hours if no response.`)
   530→   530→
   531→   531→  while (Date.now() - startTime < timeoutMs) {
   532→   532→    try {
   533→   533→      const { hasOwnerApproval } = await import('./approval-audit-log')
   534→   534→      const approved = await hasOwnerApproval(missionId)
   535→   535→      if (approved) return true
   536→   536→
   537→   537→      // Check for explicit rejection
   538→   538→      const { hasOwnerRejection } = await import('./approval-audit-log')
   539→   539→      const rejected = await hasOwnerRejection(missionId)
   540→   540→      if (rejected) return false
   541→   541→    } catch {}
   542→   542→
   543→   543→    await new Promise((r) => setTimeout(r, pollInterval))
   544→   544→  }
   545→   545→
   546→   546→  await notifyTelegram(`⏰ MISSION AUTO-CANCELLED\nMission: ${missionId}\nReason: owner did not respond within 24 hours.`)
   547→   547→  return false
   548→   548→}

   550→   550→/**
   551→   551→ * Run the full mission pipeline end-to-end.
   552→   552→ *
   553→   553→ * This is the MAIN ENTRY POINT. Called by:
   554→   554→ *   - The orchestrator when it detects a "mission" task
   555→   555→ *   - The /api/mission-active/run endpoint (manual trigger)
   556→   556→ *   - The scheduled tick (for autonomous missions)
   557→   557→ */
   558→   558→export async function runMissionPipeline(opts: {
   559→   559→  missionId: string
   560→   560→  pipelineType: string
   561→   561→  objective: string
   562→   562→  missionTitle?: string
   563→   563→  skipOwnerApproval?: boolean  // for testing only
   564→   564→}): Promise<PipelineRunResult> {
   565→   565→  const { missionId, pipelineType, objective } = opts
   566→   566→
   567→   567→  const pipeline = MISSION_PIPELINES[pipelineType] ?? MISSION_PIPELINES.generic
   568→   568→  const stages: MissionStageSummary[] = []
   569→   569→
   570→   570→  // UPGRADE #144 — Initialize heartbeat (real-time monitoring)
   571→   571→  const initialHeartbeat: MissionHeartbeat = {
   572→   572→    missionId,
   573→   573→    missionTitle: opts.missionTitle ?? missionId,
   574→   574→    pipelineType: pipeline.type,
   575→   575→    status: 'working',
   576→   576→    currentStage: {
   577→   577→      stageId: `stage_1`,
   578→   578→      stageNumber: 1,
   579→   579→      totalStages: pipeline.stages.length,
   580→   580→      name: pipeline.stages[0].name,
   581→   581→      team: pipeline.stages[0].team,
   582→   582→      leader: pipeline.stages[0].leader,
   583→   583→      startedAt: new Date().toISOString(),
   584→   584→      elapsedMs: 0,
   585→   585→      round: 1,
   586→   586→      maxRounds: 3,
   587→   587→    },
   588→   588→    completedStages: [],
   589→   589→    estimatedRemainingMs: null,
   590→   590→    estimatedCompletionAt: null,
   591→   591→    lastActivityAt: new Date().toISOString(),
   592→   592→    lastError: null,
   593→   593→    ceoWatchdog: { verdict: 'healthy', message: 'Mission started', checkedAt: new Date().toISOString() },
   594→   594→    updatedAt: new Date().toISOString(),
   595→   595→  }
   596→   596→  await saveHeartbeat(initialHeartbeat).catch(() => {})
   597→   597→
   598→   598→  // ── Rec 7: Owner approval gate for high-stakes missions
   599→   599→  if (pipeline.requiresOwnerApproval && !opts.skipOwnerApproval) {
   600→   600→    await notifyTelegram(`🎯 MISSION STARTED: ${opts.missionTitle ?? missionId}\nPipeline: ${pipeline.name}\nStages: ${pipeline.stages.length}\n⚠️ This mission requires your approval before final execution.`)
   601→   601→  } else {
   602→   602→    await notifyTelegram(`🎯 MISSION STARTED: ${opts.missionTitle ?? missionId}\nPipeline: ${pipeline.name}\nStages: ${pipeline.stages.length}`)
   603→   603→  }
   604→   604→
   605→   605→  // Build mission context (grows as stages complete)
   606→   606→  let missionContext = `MISSION OBJECTIVE: ${objective}\n\n`
   607→   607→  let previousTeamOutput: string | undefined
   608→   608→
   609→   609→  // ── Run each stage sequentially
   610→   610→  for (const stage of pipeline.stages) {
   611→   611→    try {
   612→   612→      const result = await runTeamWithVerificationLoop({
   613→   613→        missionId,
   614→   614→        stage,
   615→   615→        objective,
   616→   616→        missionContext,
   617→   617→        previousTeamOutput,
   618→   618→      })
   619→   619→
   620→   620→      const stageSummary: MissionStageSummary = {
   621→   621→        stage: stage.stage,
   622→   622→        team: stage.team,
   623→   623→        leader: stage.leader,
   624→   624→        artifactValue: result.artifactValue,
   625→   625→        artifactVerified: result.artifactVerified,
   626→   626→        finalScore: result.finalScore,
   627→   627→        rounds: result.rounds,
   628→   628→        approvedAt: result.finalVerification.approved ? new Date().toISOString() : null,
   629→   629→      }
   630→   630→      stages.push(stageSummary)
   631→   631→
   632→   632→      // UPGRADE #144 — Update heartbeat after each stage
   633→   633→      try {
   634→   634→        const hb = await buildHeartbeatFromAuditLog({
   635→   635→          missionId,
   636→   636→          missionTitle: opts.missionTitle ?? missionId,
   637→   637→          pipelineType: pipeline.type,
   638→   638→          totalStages: pipeline.stages.length,
   639→   639→        })
   640→   640→        // If this wasn't the last stage, set current stage to next one
   641→   641→        if (stage.stage < pipeline.stages.length) {
   642→   642→          const nextStage = pipeline.stages[stage.stage]  // 0-indexed; stage.stage is 1-indexed
   643→   643→          if (nextStage && !hb.currentStage) {
   644→   644→            hb.currentStage = {
   645→   645→              stageId: `stage_${nextStage.stage}`,
   646→   646→              stageNumber: nextStage.stage,
   647→   647→              totalStages: pipeline.stages.length,
   648→   648→              name: nextStage.name,
   649→   649→              team: nextStage.team,
   650→   650→              leader: nextStage.leader,
   651→   651→              startedAt: new Date().toISOString(),
   652→   652→              elapsedMs: 0,
   653→   653→              round: 1,
   654→   654→              maxRounds: 3,
   655→   655→            }
   656→   656→            hb.status = 'working'
   657→   657→          }
   658→   658→        } else {
   659→   659→          // Last stage done
   660→   660→          hb.status = 'completed'
   661→   661→          hb.currentStage = null
   662→   662→        }
   663→   663→        hb.updatedAt = new Date().toISOString()
   664→   664→        await saveHeartbeat(hb)
   665→   665→      } catch {}
   666→   666→
   667→   667→      // Update mission context with this stage's output
   668→   668→      missionContext += `STAGE ${stage.stage} (${stage.team}/${stage.leader}) OUTPUT:\n${result.output.slice(0, 3000)}\n\nFINAL SCORE: ${result.finalScore}/100\nVERDICT: ${result.finalVerification.verdict}\n\n---\n\n`
   669→   669→      previousTeamOutput = result.output
   670→   670→
   671→   671→      // If stage is the CEO (final), capture the report
   672→   672→      if (stage.team === 'ceo') {
   673→   673→        // CEO stage output IS the report
   674→   674→        const ceoReport: import('./ceo-presenter').CeoReport = {
   675→   675→          missionId,
   676→   676→          missionTitle: opts.missionTitle ?? missionId,
   677→   677→          objective,
   678→   678→          outcome: stages.every((s) => s.artifactVerified || s.team === 'ceo') ? 'success' : 'partial',
   679→   679→          revenueImpact: 'See full report',
   680→   680→          keyDeliverables: stages.filter((s) => s.artifactValue).map((s) => `Stage ${s.stage} (${s.team}): ${s.artifactValue}`),
   681→   681→          risksNotes: [],
   682→   682→          nextSteps: ['Review CEO report', 'Approve mission via dashboard', 'Schedule follow-up'],
   683→   683→          fullReport: result.output,
   684→   684→          generatedAt: new Date().toISOString(),
   685→   685→        }
   686→   686→
   687→   687→        // Send CEO report via Telegram (the CEO's output is already a formatted report)
   688→   688→        await notifyTelegram(`🎯 MISSION COMPLETE — CEO REPORT\n\n${result.output.slice(0, 3500)}\n\n— Agent007 CEO`)
   689→   689→
   690→   690→        // Persist to DB
   691→   691→        try {
   692→   692→          const { ceoPersistReport } = await import('./ceo-presenter')
   693→   693→          await ceoPersistReport(ceoReport)
   694→   694→        } catch {}
   695→   695→
   696→   696→        return {
   697→   697→          missionId,
   698→   698→          pipelineType: pipeline.type,
   699→   699→          success: true,
   700→   700→          stages,
   701→   701→          ceoReport,
   702→   702→        }
   703→   703→      }
   704→   704→    } catch (stageErr: any) {
   705→   705→      // Stage crashed — log + abort
   706→   706→      await logApprovalEvent({
   707→   707→        missionId,
   708→   708→        stageId: `stage_${stage.stage}`,
   709→   709→        round: 1,
   710→   710→        agentRole: 'system',
   711→   711→        agentId: 'system',
   712→   712→        action: 'escalated',
   713→   713→        feedback: `Stage crashed: ${stageErr?.message?.slice(0, 200)}`,
   714→   714→      })
   715→   715→      await notifyTelegram(`❌ MISSION FAILED at stage ${stage.stage} (${stage.team})\nError: ${stageErr?.message?.slice(0, 200)}`)
   716→   716→
   717→   717→      // UPGRADE #144 — Update heartbeat to failed
   718→   718→      try {
   719→   719→        const hb = await buildHeartbeatFromAuditLog({
   720→   720→          missionId,
   721→   721→          missionTitle: opts.missionTitle ?? missionId,
   722→   722→          pipelineType: pipeline.type,
   723→   723→          totalStages: pipeline.stages.length,
   724→   724→        })
   725→   725→        hb.status = 'failed'
   726→   726→        hb.lastError = `Stage ${stage.stage} (${stage.team}) crashed: ${stageErr?.message?.slice(0, 200) ?? 'unknown'}`
   727→   727→        hb.updatedAt = new Date().toISOString()
   728→   728→        await saveHeartbeat(hb)
   729→   729→      } catch {}
   730→   730→
   731→   731→      return {
   732→   732→        missionId,
   733→   733→        pipelineType: pipeline.type,
   734→   734→        success: false,
   735→   735→        stages,
   736→   736→        error: `Stage ${stage.stage} (${stage.team}) crashed: ${stageErr?.message ?? 'unknown'}`,
   737→   737→      }
   738→   738→    }
   739→   739→  }
   740→   740→
   741→   741→  // ── Rec 7: Owner approval gate (only for high-stakes missions that completed successfully)
   742→   742→  if (pipeline.requiresOwnerApproval && !opts.skipOwnerApproval) {
   743→   743→    const approved = await waitForOwnerApproval(missionId)
   744→   744→    if (!approved) {
   745→   745→      return {
   746→   746→        missionId,
   747→   747→        pipelineType: pipeline.type,
   748→   748→        success: false,
   749→   749→        stages,
   750→   750→        error: 'Owner did not approve the mission',
   751→   751→        requiresOwnerApproval: true,
   752→   752→        pausedForOwnerApproval: true,
   753→   753→      }
   754→   754→    }
   755→   755→  }
   756→   756→
   757→   757→  return {
   758→   758→    missionId,
   759→   759→    pipelineType: pipeline.type,
   760→   760→    success: true,
   761→   761→    stages,
   762→   762→  }
   763→   763→}
