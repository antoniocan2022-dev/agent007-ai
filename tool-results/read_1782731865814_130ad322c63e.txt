   462→ *     enabled) on top of the built-in defaults.

   477→function parseAllowedTools(raw: string): string[] {
   478→  try {
   479→    const parsed = JSON.parse(raw)
   480→    if (Array.isArray(parsed)) {
   481→      return parsed.filter((s) => typeof s === 'string' && s.length > 0)
   482→    }
   483→  } catch {
   484→    /* ignore */
   485→  }
   486→  // Fallback: comma-separated string
   487→  return raw
   488→    .split(',')
   489→    .map((s) => s.trim())
   490→    .filter(Boolean)
   491→}

   493→/**
   494→ * Returns ALL subagents (12 built-in + custom), with overlay edits applied.
   495→ * Disabled agents are filtered out unless `includeDisabled=true`.
   496→ *
   497→ * IMPORTANT: This is async because it reads from the DB. Callers that need
   498→ * a synchronous lookup (e.g. runSubagent) should call this first and pass
   499→ * the resulting list down.
   500→ */
   501→export async function getAllSubagents(opts?: { includeDisabled?: boolean }): Promise<Subagent[]> {
   502→  const includeDisabled = opts?.includeDisabled ?? false
   503→  let customRows: any[] = []
   504→  try {
   505→    const userId = await getOperatorUserId()
   506→    if (userId) {
   507→      customRows = await db.customSubagent.findMany({ where: { userId } })
   508→    }
   509→  } catch (e) {
   510→    console.error('[subagents] getAllSubagents DB load failed:', e)
   511→  }
   512→
   513→  const overlayMap = new Map<string, any>()
   514→  const customList: Subagent[] = []
   515→  for (const row of customRows) {
   516→    if (row.isBuiltinOverlay) {
   517→      overlayMap.set(row.id, row)
   518→    } else {
   519→      customList.push({
   520→        id: row.id,
   521→        name: row.name,
   522→        role: row.role,
   523→        specialty: row.specialty,
   524→        color: row.color,
   525→        icon: row.icon,
   526→        allowedTools: parseAllowedTools(row.allowedTools),
   527→        systemPrompt: row.systemPrompt,
   528→        isBuiltin: false,
   529→        enabled: row.enabled ?? true,
   530→      })
   531→    }
   532→  }
   533→
   534→  // Apply overlays on top of built-ins
   535→  const mergedBuiltins: Subagent[] = SUBAGENTS.map((b) => {
   536→    const ov = overlayMap.get(b.id)
   537→    if (!ov) return { ...b, enabled: b.enabled ?? true }
   538→    return {
   539→      ...b,
   540→      name: ov.name ?? b.name,
   541→      role: ov.role ?? b.role,
   542→      specialty: ov.specialty ?? b.specialty,
   543→      color: ov.color ?? b.color,
   544→      icon: ov.icon ?? b.icon,
   545→      allowedTools: ov.allowedTools ? parseAllowedTools(ov.allowedTools) : b.allowedTools,
   546→      systemPrompt: ov.systemPrompt ?? b.systemPrompt,
   547→      enabled: ov.enabled ?? b.enabled ?? true,
   548→    }
   549→  })
   550→
   551→  const all = [...mergedBuiltins, ...customList]
   552→  return includeDisabled ? all : all.filter((s) => s.enabled !== false)
   553→}

   555→/**
   556→ * Sync lookup of a subagent by id within a provided list. Falls back to the
   557→ * built-in registry (with overlays NOT applied) if no list is provided.
   558→ */
   559→export function findSubagentIn(list: Subagent[], id: string): Subagent | undefined {
   560→  return list.find((s) => s.id === id)
   561→}

   564→ * prompt + restricted tool set, then returns its final answer.
   565→ * ------------------------------------------------------------------ */
   566→
   567→export interface SubagentEventEmit {

   569→}

   571→export interface RunSubagentOptions {
   572→  subagentId: string
   573→  task: string
   574→  attachments: AttachmentMeta[]
   575→  language: 'en' | 'zh'
   576→  emit: SubagentEventEmit
   577→  parentConversationId: string
   578→  dispatchId: string
   579→}
   580→
   581→export interface RunSubagentResult {
   582→  answer: string
   583→  steps: Array<{
   584→    id: string
   585→    thought?: string
   586→    toolName?: string
   587→    toolArgs?: any
   588→    toolResult?: ToolResult
   589→    startedAt: number
   590→    finishedAt?: number
   591→  }>
   592→}
   593→
   594→const SUBAGENT_MAX_ITERATIONS = 6
   595→
   596→export async function runSubagent(opts: RunSubagentOptions): Promise<RunSubagentResult> {
   597→  // Look up the sub-agent definition from the merged list (built-in + DB-loaded
   598→  // custom + built-in overlays). This lets the Super Agent dispatch to custom
   599→  // agents and respect overlay edits at runtime.
   600→  const allSubs = await getAllSubagents({ includeDisabled: true })
   601→  const sub = allSubs.find((s) => s.id === opts.subagentId)
   602→  if (!sub) {
   603→    const err = `Unknown sub-agent id: "${opts.subagentId}"`
   604→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
   605→    return { answer: `⚠️ ${err}`, steps: [] }
   606→  }
   607→  if (sub.enabled === false) {
   608→    const err = `Sub-agent "${sub.name}" is currently disabled. Re-enable it in Settings → Sub-Agents.`
   609→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
   610→    return { answer: `⚠️ ${err}`, steps: [] }
   611→  }
   612→
   613→  const zai = await getZai()
   614→  const allowed = new Set(sub.allowedTools)
   615→  const ctx: ToolContext = { attachments: opts.attachments, language: opts.language }
   616→
   617→  const languageInstruction =
   618→    opts.language === 'zh'
   619→      ? 'LANGUAGE: Reply in 中文 for your FINAL answer.'
   620→      : 'LANGUAGE: Reply in English for your FINAL answer unless the task is in another language.'
   621→
   622→  const systemPrompt = `${sub.systemPrompt}
   623→
   624→${languageInstruction}
   625→
   626→CURRENT UTC TIME: ${new Date().toUTCString()}
   627→
   628→You are operating autonomously inside Agent007's multi-agent network. The Super Agent has given you a specific task. Execute it end-to-end using only your allowed tools. Then return a clear, structured final answer.`
   629→
   630→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
   631→    { role: 'system', content: systemPrompt },
   632→    { role: 'user', content: opts.task },
   633→  ]
   634→
   635→  const steps: RunSubagentResult['steps'] = []
   636→  let finalAnswer = ''
   637→  let iter = 0
   638→
   639→  while (iter < SUBAGENT_MAX_ITERATIONS) {
   640→    iter++
   641→    let completion: any
   642→    try {
   643→      completion = await zai.chat.completions.create({
   644→        messages: conversationMessages,
   645→        thinking: { type: 'enabled' },
   646→      })
   647→    } catch (e: any) {
   648→      finalAnswer = friendlyLlmError(e)
   649→      break
   650→    }
   651→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
   652→    if (!content.trim()) {
   653→      finalAnswer = `(${sub.name} produced no output.)`
   654→      break
   655→    }
   656→    const parsed = parseAssistant(content)
   657→
   658→    if (parsed.thought) {
   659→      await opts.emit('subagent_thought', { dispatchId: opts.dispatchId, content: parsed.thought })
   660→    }
   661→
   662→    if (!parsed.tool) {
   663→      // final answer
   664→      finalAnswer = content.replace(THOUGHT_RE, '').trim() || content.trim()
   665→      break
   666→    }
   667→
   668→    const toolName = parsed.tool.name
   669→    const toolArgs = parsed.tool.args
   670→    const stepId = `sub_${opts.dispatchId}_${iter}_${Math.random().toString(36).slice(2, 8)}`
   671→
   672→    // Enforce the sub-agent's allowed tool list
   673→    if (!allowed.has(toolName)) {
   674→      const errResult: ToolResult = {
   675→        ok: false,
   676→        preview: `Tool "${toolName}" not allowed for ${sub.name}`,
   677→        result: `BLOCKED: ${sub.name} is not permitted to call "${toolName}". Allowed tools: ${sub.allowedTools.join(', ')}. Use one of those instead.`,
   678→      }
   679→      const step = {
   680→        id: stepId,
   681→        thought: parsed.thought,
   682→        toolName,
   683→        toolArgs,
   684→        toolResult: errResult,
   685→        startedAt: Date.now(),
   686→        finishedAt: Date.now(),
   687→      }
   688→      steps.push(step)
   689→      await opts.emit('subagent_tool_call', {
   690→        dispatchId: opts.dispatchId,
   691→        stepId,
   692→        name: toolName,
   693→        args: toolArgs,
   694→        thought: parsed.thought,
   695→        stepNumber: iter,
   696→      })
   697→      await opts.emit('subagent_tool_result', {
   698→        dispatchId: opts.dispatchId,
   699→        stepId,
   700→        result: errResult.result,
   701→        preview: errResult.preview,
   702→        ok: false,
   703→        artifacts: undefined,
   704→      })
   705→      conversationMessages.push({ role: 'assistant', content })
   706→      conversationMessages.push({
   707→        role: 'user',
   708→        content: `[TOOL_RESULT] ${toolName}: ${errResult.result}`,
   709→      })
   710→      // Persist the blocked attempt for reload reconstruction
   711→      try {
   712→        await db.message.create({
   713→          data: {
   714→            conversationId: opts.parentConversationId,
   715→            role: 'tool',
   716→            content: `[subagent:${sub.id}:blocked] ${toolName} ${JSON.stringify(toolArgs)}`,
   717→            toolName: 'subagent_tool',
   718→            toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, tool: toolName, args: toolArgs }),
   719→            toolResult: errResult.result,
   720→          },
   721→        })
   722→      } catch {
   723→        /* ignore */
   724→      }
   725→      continue
   726→    }
   727→
   728→    const step = {
   729→      id: stepId,
   730→      thought: parsed.thought,
   731→      toolName,
   732→      toolArgs,
   733→      startedAt: Date.now(),
   734→    }
   735→    steps.push(step)
   736→
   737→    await opts.emit('subagent_tool_call', {
   738→      dispatchId: opts.dispatchId,
   739→      stepId,
   740→      name: toolName,
   741→      args: toolArgs,
   742→      thought: parsed.thought,
   743→      stepNumber: iter,
   744→    })
   745→
   746→    const toolResult = await dispatchTool(toolName, toolArgs, ctx)
   747→    step.toolResult = toolResult
   748→    step.finishedAt = Date.now()
   749→
   750→    await opts.emit('subagent_tool_result', {
   751→      dispatchId: opts.dispatchId,
   752→      stepId,
   753→      result: toolResult.result,
   754→      preview: toolResult.preview,
   755→      ok: toolResult.ok,
   756→      artifacts: toolResult.artifacts,
   757→    })
   758→
   759→    // Persist subagent tool activity for reload reconstruction
   760→    try {
   761→      if (step.thought) {
   762→        await db.message.create({
   763→          data: {
   764→            conversationId: opts.parentConversationId,
   765→            role: 'thought',
   766→            content: `[subagent:${sub.id}] ${step.thought}`,
   767→          },
   768→        })
   769→      }
   770→      await db.message.create({
   771→        data: {
   772→          conversationId: opts.parentConversationId,
   773→          role: 'tool',
   774→          content: `[subagent:${sub.id}:tool] ${toolName} ${JSON.stringify(toolArgs)}`,
   775→          toolName: 'subagent_tool',
   776→          toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, stepId, tool: toolName, args: toolArgs }),
   777→          toolResult: toolResult.result,
   778→        },
   779→      })
   780→    } catch {
   781→      /* ignore persistence errors */
   782→    }
   783→
   784→    conversationMessages.push({ role: 'assistant', content })
   785→    conversationMessages.push({
   786→      role: 'user',
   787→      content: `[TOOL_RESULT] ${toolName}: ${toolResult.result}`,
   788→    })
   789→  }
   790→
   791→  if (!finalAnswer) {
   792→    finalAnswer = `${sub.name} reached its tool-call limit. Here is what it found so far:\n\n` +
   793→      steps
   794→        .filter((s) => s.toolResult?.result)
   795→        .map((s, i) => `${i + 1}. **${s.toolName}** → ${s.toolResult!.result.slice(0, 500)}`)
   796→        .join('\n\n')
   797→  }
   798→
   799→  await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: finalAnswer })
   800→
   801→  // Persist the sub-agent's final answer for reload reconstruction
   802→  try {
   803→    await db.message.create({
   804→      data: {
   805→        conversationId: opts.parentConversationId,
   806→        role: 'tool',
   807→        content: `[subagent:${sub.id}:complete] ${finalAnswer.slice(0, 500)}`,
   808→        toolName: 'subagent_complete',
   809→        toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, task: opts.task }),
   810→        toolResult: finalAnswer,
   811→      },
   812→    })
   813→  } catch {
   814→    /* ignore */
   815→  }
   816→
   817→  return { answer: finalAnswer, steps }
   818→}
