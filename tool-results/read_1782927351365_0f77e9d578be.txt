   484→    const parsed = JSON.parse(raw)
   485→    if (Array.isArray(parsed)) {
   486→      return parsed.filter((s) => typeof s === 'string' && s.length > 0)
   487→    }
   488→  } catch {
   489→    /* ignore */
   490→  }
   491→  // Fallback: comma-separated string
   492→  return raw
   493→    .split(',')
   494→    .map((s) => s.trim())
   495→    .filter(Boolean)
   496→}

   498→/**
   499→ * Returns ALL subagents (12 built-in + custom), with overlay edits applied.
   500→ * Disabled agents are filtered out unless `includeDisabled=true`.
   501→ *
   502→ * IMPORTANT: This is async because it reads from the DB. Callers that need
   503→ * a synchronous lookup (e.g. runSubagent) should call this first and pass
   504→ * the resulting list down.
   505→ */
   506→export async function getAllSubagents(opts?: { includeDisabled?: boolean }): Promise<Subagent[]> {
   507→  const includeDisabled = opts?.includeDisabled ?? false
   508→  let customRows: any[] = []
   509→  try {
   510→    const userId = await getOperatorUserId()
   511→    if (userId) {
   512→      customRows = await db.customSubagent.findMany({ where: { userId } })
   513→    }
   514→  } catch (e) {
   515→    console.error('[subagents] getAllSubagents DB load failed:', e)
   516→  }
   517→
   518→  const overlayMap = new Map<string, any>()
   519→  const customList: Subagent[] = []
   520→  for (const row of customRows) {
   521→    if (row.isBuiltinOverlay) {
   522→      overlayMap.set(row.id, row)
   523→    } else {
   524→      customList.push({
   525→        id: row.id,
   526→        name: row.name,
   527→        role: row.role,
   528→        specialty: row.specialty,
   529→        color: row.color,
   530→        icon: row.icon,
   531→        allowedTools: parseAllowedTools(row.allowedTools),
   532→        systemPrompt: row.systemPrompt,
   533→        isBuiltin: false,
   534→        enabled: row.enabled ?? true,
   535→      })
   536→    }
   537→  }
   538→
   539→  // Apply overlays on top of built-ins
   540→  const mergedBuiltins: Subagent[] = SUBAGENTS.map((b) => {
   541→    const ov = overlayMap.get(b.id)
   542→    if (!ov) return { ...b, enabled: b.enabled ?? true }
   543→    return {
   544→      ...b,
   545→      name: ov.name ?? b.name,
   546→      role: ov.role ?? b.role,
   547→      specialty: ov.specialty ?? b.specialty,
   548→      color: ov.color ?? b.color,
   549→      icon: ov.icon ?? b.icon,
   550→      allowedTools: ov.allowedTools ? parseAllowedTools(ov.allowedTools) : b.allowedTools,
   551→      systemPrompt: ov.systemPrompt ?? b.systemPrompt,
   552→      enabled: ov.enabled ?? b.enabled ?? true,
   553→    }
   554→  })
   555→
   556→  const all = [...mergedBuiltins, ...customList]
   557→  return includeDisabled ? all : all.filter((s) => s.enabled !== false)
   558→}

   560→/**
   561→ * Sync lookup of a subagent by id within a provided list. Falls back to the
   562→ * built-in registry (with overlays NOT applied) if no list is provided.
   563→ */
   564→export function findSubagentIn(list: Subagent[], id: string): Subagent | undefined {
   565→  return list.find((s) => s.id === id)

   590→    thought?: string

   593→    toolResult?: ToolResult
   594→    startedAt: number
   595→    finishedAt?: number
   596→  }>

   599→const SUBAGENT_MAX_ITERATIONS = 15
   600→
   601→/* Per-agent request throttle (#10). Ensures each individual sub-agent waits
   602→ * at least MIN_AGENT_INTERVAL_MS between its own LLM calls, on top of the
   603→ * app-wide throttle in agent.ts. */
   604→const _agentLastCallAt: Record<string, number> = {}
   605→const MIN_AGENT_INTERVAL_MS = 1500
   606→async function throttleAgentCall(agentId: string): Promise<void> {
   607→  const now = Date.now()
   608→  const last = _agentLastCallAt[agentId] || 0
   609→  const wait = Math.max(0, last + MIN_AGENT_INTERVAL_MS - now)
   610→  if (wait > 0) await new Promise((r) => setTimeout(r, wait))
   611→  _agentLastCallAt[agentId] = Date.now()
   612→}
   613→
   614→export async function runSubagent(opts: RunSubagentOptions): Promise<RunSubagentResult> {
   615→  // Look up the sub-agent definition from the merged list (built-in + DB-loaded
   616→  // custom + built-in overlays). This lets the Super Agent dispatch to custom
   617→  // agents and respect overlay edits at runtime.
   618→  const allSubs = await getAllSubagents({ includeDisabled: true })
   619→  // Match by id (case-sensitive) OR by name (case-insensitive) — this lets the
   620→  // Super Agent dispatch to custom agents like "Cybersecurity A" using their
   621→  // human-readable name even though their id is a cuid like "cmqzahs7d...".
   622→  const sub = allSubs.find(
   623→    (s) => s.id === opts.subagentId || s.name.toLowerCase() === opts.subagentId.toLowerCase()
   624→  )
   625→  if (!sub) {
   626→    const available = allSubs.map((s) => `${s.name} (id: ${s.id})`).join(', ')
   627→    const err = `Unknown sub-agent: "${opts.subagentId}". Available: ${available}`
   628→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
   629→    return { answer: `⚠️ ${err}`, steps: [] }
   630→  }
   631→  if (sub.enabled === false) {
   632→    const err = `Sub-agent "${sub.name}" is currently disabled. Re-enable it in Settings → Sub-Agents.`
   633→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
   634→    return { answer: `⚠️ ${err}`, steps: [] }
   635→  }
   636→
   637→  const allowed = new Set(sub.allowedTools)
   638→  const ctx: ToolContext = { attachments: opts.attachments, language: opts.language }
   639→
   640→  const languageInstruction =
   641→    opts.language === 'zh'
   642→      ? 'LANGUAGE: Reply in 中文 for your FINAL answer.'
   643→      : 'LANGUAGE: Reply in English for your FINAL answer unless the task is in another language.'
   644→
   645→  const systemPrompt = `${sub.systemPrompt}
   646→
   647→${languageInstruction}
   648→
   649→CURRENT UTC TIME: ${new Date().toUTCString()}
   650→
   651→You are operating autonomously inside Agent007's multi-agent network. The Super Agent has given you a specific task. Execute it end-to-end using only your allowed tools. Then return a clear, structured final answer.`
   652→
   653→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
   654→    { role: 'system', content: systemPrompt },
   655→    { role: 'user', content: opts.task },
   656→  ]
   657→
   658→  const steps: RunSubagentResult['steps'] = []
   659→  let finalAnswer = ''
   660→  let iter = 0
   661→
   662→  while (iter < SUBAGENT_MAX_ITERATIONS) {
   663→    iter++
   664→    // Per-agent throttle — keep each sub-agent's calls >=1.5s apart
   665→    await throttleAgentCall(sub.id)
   666→    let completion: any
   667→    try {
   668→      completion = await callLlmWithRetry(conversationMessages)
   669→    } catch (e: any) {
   670→      finalAnswer = friendlyLlmError(e)
   671→      break
   672→    }
   673→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
   674→    if (!content.trim()) {
   675→      finalAnswer = `(${sub.name} produced no output.)`
   676→      break
   677→    }
   678→    const parsed = parseAssistant(content)
   679→
   680→    if (parsed.thought) {
   681→      await opts.emit('subagent_thought', { dispatchId: opts.dispatchId, content: parsed.thought })
   682→    }
   683→
   684→    if (!parsed.tool) {
   685→      // FIX 2: Stuck detection for sub-agents.
   686→      // If the sub-agent produced ONLY a thought with no substantial answer,
   687→      // auto-recover by prompting it to continue (instead of treating the
   688→      // thought as the final answer).
   689→      const textAfterThought = content.replace(THOUGHT_RE, '').trim()
   690→      const isThoughtOnly = !!parsed.thought && textAfterThought.length < 20
   691→      const stuckPatterns = /(wait|waiting|haven't|will wait|let me wait|as i wait)/i
   692→      const isStuck = isThoughtOnly && stuckPatterns.test(parsed.thought)
   693→
   694→      if ((isStuck || isThoughtOnly) && iter < SUBAGENT_MAX_ITERATIONS - 1) {
   695→        await opts.emit('subagent_thought', {
   696→          dispatchId: opts.dispatchId,
   697→          content: `[AUTO-RECOVERY] Thought-only response. Prompting to continue...`,
   698→        })
   699→        conversationMessages.push({ role: 'assistant', content })
   700→        conversationMessages.push({
   701→          role: 'user',
   702→          content: '[SYSTEM] You produced only a thought. Please either call a tool or give your final answer now.',
   703→        })
   704→        continue
   705→      }
   706→
   707→      // final answer
   708→      finalAnswer = textAfterThought || content.trim()
   709→      break
   710→    }
   711→
   712→    const toolName = parsed.tool.name
   713→    const toolArgs = parsed.tool.args
   714→    const stepId = `sub_${opts.dispatchId}_${iter}_${Math.random().toString(36).slice(2, 8)}`
   715→
   716→    // Enforce the sub-agent's allowed tool list
   717→    if (!allowed.has(toolName)) {
   718→      const errResult: ToolResult = {
   719→        ok: false,
   720→        preview: `Tool "${toolName}" not allowed for ${sub.name}`,
   721→        result: `BLOCKED: ${sub.name} is not permitted to call "${toolName}". Allowed tools: ${sub.allowedTools.join(', ')}. Use one of those instead.`,
   722→      }
   723→      const step = {
   724→        id: stepId,
   725→        thought: parsed.thought,
   726→        toolName,
   727→        toolArgs,
   728→        toolResult: errResult,
   729→        startedAt: Date.now(),
   730→        finishedAt: Date.now(),
   731→      }
   732→      steps.push(step)
   733→      await opts.emit('subagent_tool_call', {
   734→        dispatchId: opts.dispatchId,
   735→        stepId,
   736→        name: toolName,
   737→        args: toolArgs,
   738→        thought: parsed.thought,
   739→        stepNumber: iter,
   740→      })
   741→      await opts.emit('subagent_tool_result', {
   742→        dispatchId: opts.dispatchId,
   743→        stepId,
   744→        result: errResult.result,
   745→        preview: errResult.preview,
   746→        ok: false,
   747→        artifacts: undefined,
   748→      })
   749→      conversationMessages.push({ role: 'assistant', content })
   750→      conversationMessages.push({
   751→        role: 'user',
   752→        content: `[TOOL_RESULT] ${toolName}: ${errResult.result}`,
   753→      })
   754→      // Persist the blocked attempt for reload reconstruction
   755→      try {
   756→        await db.message.create({
   757→          data: {
   758→            conversationId: opts.parentConversationId,
   759→            role: 'tool',
   760→            content: `[subagent:${sub.id}:blocked] ${toolName} ${JSON.stringify(toolArgs)}`,
   761→            toolName: 'subagent_tool',
   762→            toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, tool: toolName, args: toolArgs }),
   763→            toolResult: errResult.result,
   764→          },
   765→        })
   766→      } catch {
   767→        /* ignore */
   768→      }
   769→      continue
   770→    }
   771→
   772→    const step = {
   773→      id: stepId,
   774→      thought: parsed.thought,
   775→      toolName,
   776→      toolArgs,
   777→      startedAt: Date.now(),
   778→    }
   779→    steps.push(step)
   780→
   781→    await opts.emit('subagent_tool_call', {
   782→      dispatchId: opts.dispatchId,
   783→      stepId,
   784→      name: toolName,
   785→      args: toolArgs,
   786→      thought: parsed.thought,
   787→      stepNumber: iter,
   788→    })
   789→
   790→    const toolResult = await dispatchTool(toolName, toolArgs, ctx)
   791→    step.toolResult = toolResult
   792→    step.finishedAt = Date.now()
   793→
   794→    await opts.emit('subagent_tool_result', {
   795→      dispatchId: opts.dispatchId,
   796→      stepId,
   797→      result: toolResult.result,
   798→      preview: toolResult.preview,
   799→      ok: toolResult.ok,
   800→      artifacts: toolResult.artifacts,
   801→    })
   802→
   803→    // Persist subagent tool activity for reload reconstruction
   804→    try {
   805→      if (step.thought) {
   806→        await db.message.create({
   807→          data: {
   808→            conversationId: opts.parentConversationId,
   809→            role: 'thought',
   810→            content: `[subagent:${sub.id}] ${step.thought}`,
   811→          },
   812→        })
   813→      }
   814→      await db.message.create({
   815→        data: {
   816→          conversationId: opts.parentConversationId,
   817→          role: 'tool',
   818→          content: `[subagent:${sub.id}:tool] ${toolName} ${JSON.stringify(toolArgs)}`,
   819→          toolName: 'subagent_tool',
   820→          toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, stepId, tool: toolName, args: toolArgs }),
   821→          toolResult: toolResult.result,
   822→        },
   823→      })
   824→    } catch {
   825→      /* ignore persistence errors */
   826→    }
   827→
   828→    conversationMessages.push({ role: 'assistant', content })
   829→    conversationMessages.push({
   830→      role: 'user',
   831→      content: `[TOOL_RESULT] ${toolName}: ${toolResult.result}`,
   832→    })
   833→  }
   834→
   835→  if (!finalAnswer) {
   836→    finalAnswer = `${sub.name} reached its tool-call limit. Here is what it found so far:\n\n` +
   837→      steps
   838→        .filter((s) => s.toolResult?.result)
   839→        .map((s, i) => `${i + 1}. **${s.toolName}** → ${s.toolResult!.result.slice(0, 500)}`)
   840→        .join('\n\n')
   841→  }
   842→
   843→  await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: finalAnswer })
   844→
   845→  // Persist the sub-agent's final answer for reload reconstruction
   846→  try {
   847→    await db.message.create({
   848→      data: {
   849→        conversationId: opts.parentConversationId,
   850→        role: 'tool',
   851→        content: `[subagent:${sub.id}:complete] ${finalAnswer.slice(0, 500)}`,
   852→        toolName: 'subagent_complete',
   853→        toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, task: opts.task }),
   854→        toolResult: finalAnswer,
   855→      },
   856→    })
   857→  } catch {
   858→    /* ignore */
   859→  }
   860→
   861→  return { answer: finalAnswer, steps }
   862→}
