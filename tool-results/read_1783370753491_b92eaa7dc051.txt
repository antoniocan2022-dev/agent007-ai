   149→/* ------------------------------------------------------------------ *
   150→ * Rate-limit resilience helpers (#1, #2 of AGENT007-IMPROVEMENTS-1)
   151→ *
   152→ * RATE_LIMIT_INFO — singleton updated on each 429. The /api/health/llm
   153→ * endpoint reads this to drive the green/amber/gray status indicator in
   154→ * the chat header.
   155→ *
   156→ * throttleLlm() — enforces a ~0.5s minimum spacing between LLM calls
   157→ * app-wide (in-process). Keeps us under the provider's RPM limit.
   158→ *
   159→ * callLlmWithRetry() — wraps zai.chat.completions.create with:
   160→ *   - throttleLlm() before each call
   161→ *   - 429 detection + exponential backoff (1s, 2s, 4s, 8s — 3 retries)
   162→ *   - fallback LLM provider stub (OpenAI-compatible) if all retries fail
   163→ * ------------------------------------------------------------------ */
   164→export const RATE_LIMIT_INFO: {
   165→  last429At: number | null
   166→  retryingNow: boolean
   167→} = {
   168→  last429At: null,
   169→  retryingNow: false,
   170→}
   171→
   172→const RATE_LIMIT_COOLDOWN_MS = 60_000
   173→
   174→let _lastLlmCallAt = 0
   175→const MIN_LLM_INTERVAL_MS = 500 // Reduced from 2000 — owner requested no rate limiting
   176→
   177→async function throttleLlm(): Promise<void> {
   178→  const now = Date.now()
   179→  const wait = Math.max(0, _lastLlmCallAt + MIN_LLM_INTERVAL_MS - now)
   180→  if (wait > 0) await new Promise((r) => setTimeout(r, wait))
   181→  _lastLlmCallAt = Date.now()
   182→}

   184→function isRateLimitError(e: any): boolean {
   185→  const status: number | undefined = e?.status ?? e?.response?.status
   186→  if (status === 429) return true
   187→  const lower = (e?.message ?? String(e)).toLowerCase()
   188→  return (
   189→    lower.includes('429') ||
   190→    lower.includes('too many requests') ||
   191→    lower.includes('rate limit')
   192→  )
   193→}
   194→
   195→const BACKOFF_DELAYS_MS = [500, 1000, 2000, 4000, 8000, 16000] // 6 retries, starts faster
   196→
   197→/**
   198→ * Call zai.chat.completions.create with thinking enabled, applying:
   199→ *   - app-wide ~0.5s throttle
   200→ *   - 6 retries with exponential backoff on 429s (0.5s → 1s → 2s → 4s → 8s → 16s)
   201→ *   - fallback LLM provider if every retry fails
   202→ *
   203→ * Throws the original (last) error if everything fails — callers should
   204→ * catch and call friendlyLlmError() to produce a user-visible message.
   205→ */
   206→export async function callLlmWithRetry(
   207→  messages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }>,
   208→  opts?: { thinking?: boolean }
   209→): Promise<any> {
   210→  let lastErr: any = null
   211→
   212→  // FAST PATH: If OPENAI_API_KEY is set in env, skip z-ai entirely
   213→  // (z-ai doesn't work on Vercel and wastes 5-10s trying to connect)
   214→  if (process.env.OPENAI_API_KEY) {
   215→    try {
   216→      return await callFallbackLlm(messages)
   217→    } catch (fallbackErr) {
   218→      throw fallbackErr
   219→    }
   220→  }
   221→
   222→  // Try primary provider (z-ai)
   223→  try {
   224→    const zai = await getZai()
   225→    const thinking = opts?.thinking === false ? undefined : { type: 'enabled' as const }
   226→
   227→    for (let attempt = 0; attempt <= BACKOFF_DELAYS_MS.length; attempt++) {
   228→      if (attempt > 0) {
   229→        RATE_LIMIT_INFO.retryingNow = true
   230→        const delay = BACKOFF_DELAYS_MS[attempt - 1]
   231→        await new Promise((r) => setTimeout(r, delay))
   232→      }
   233→      await throttleLlm()
   234→      try {
   235→        const completion = await zai.chat.completions.create({
   236→          messages,
   237→          ...(thinking ? { thinking } : {}),
   238→        })
   239→        RATE_LIMIT_INFO.retryingNow = false
   240→        return completion
   241→      } catch (e: any) {
   242→        lastErr = e
   243→        if (isRateLimitError(e)) {
   244→          RATE_LIMIT_INFO.last429At = Date.now()
   245→          continue // retry on rate limit
   246→        }
   247→        // Non-rate-limit error (config not found, auth, etc.) — break and try fallback
   248→        break
   249→      }
   250→    }
   251→  } catch (e: any) {
   252→    lastErr = e
   253→  }
   254→
   255→  // Primary failed — try fallback LLM (OpenAI)
   256→  RATE_LIMIT_INFO.retryingNow = false
   257→  try {
   258→    return await callFallbackLlm(messages)
   259→  } catch (fallbackErr) {
   260→    // Fallback also failed — throw the fallback error (more informative)
   261→    throw fallbackErr
   262→  }
   263→}

   268→  last429At: number | null
   269→  cooldownMs: number

   279→  }

   282→export const THOUGHT_RE = /<thought>([\s\S]*?)<\/thought>/i
   283→// Match both <tool name="x">{json}</tool> AND <tool name="x"/> (self-closing)
   284→// The LLM sometimes generates self-closing tags for tools with no args.
   285→export const TOOL_RE = /<tool\s+name=["']([^"']+)["']\s*(?:\/>|>([\s\S]*?)<\/tool>)/i
   286→
   287→export interface Parsed {
   288→  thought?: string
   289→  tool?: { name: string; args: any }
   290→  textAfterTool: string
   291→  textBeforeTool: string
   292→  raw: string
   293→}
   294→
   295→export function parseAssistant(content: string): Parsed {
   296→  const thoughtMatch = content.match(THOUGHT_RE)
   297→  const thought = thoughtMatch?.[1]?.trim()
   298→  const toolMatch = content.match(TOOL_RE)
   299→  let tool: Parsed['tool']
   300→  let textBeforeTool = content
   301→  let textAfterTool = ''
   302→  if (toolMatch) {
   303→    const name = (toolMatch[1] ?? '').trim()
   304→    if (!name) {
   305→      // No tool name captured — shouldn't happen but guard against it
   306→      return { thought, tool: undefined, textBeforeTool: content.replace(THOUGHT_RE, '').trim(), textAfterTool: '', raw: content }
   307→    }
   308→    let args: any = {}
   309→    // Regex: /<tool\s+name=["']([^"']+)["']\s*(?:\/>|>([\s\S]*?)<\/tool>)/i
   310→    // Group 1 = tool name
   311→    // Group 2 = content between tags (for closed tags) OR undefined (for self-closing)
   312→    const raw = (toolMatch[2] ?? '').trim()
   313→    if (raw) {
   314→      try {
   315→        args = JSON.parse(raw)
   316→      } catch {
   317→        // try to salvage key="value" pairs
   318→        const m: Record<string, string> = {}
   319→        const re = /"([^"]+)"\s*:\s*"([^"]*)"/g
   320→        let mm: RegExpExecArray | null
   321→        while ((mm = re.exec(raw))) m[mm[1]] = mm[2]
   322→        args = m
   323→      }
   324→    }
   325→    tool = { name, args }
   326→    const idx = content.indexOf(toolMatch[0])
   327→    textBeforeTool = content.slice(0, idx).replace(THOUGHT_RE, '').trim()
   328→    textAfterTool = content.slice(idx + toolMatch[0].length).trim()
   329→  }
   330→  return {
   331→    thought,
   332→    tool,
   333→    textBeforeTool,
   334→    textAfterTool,
   335→    raw: content,
   336→  }
   337→}

   339→/** Rough token estimator (~4 chars/token, standard approximation). */
   340→function estimateTokens(messages: Array<{ role: string; content: string }>): number {
   341→  let chars = 0
   342→  for (const m of messages) chars += (m.content ?? '').length
   343→  return Math.ceil(chars / 4)

   356→  const msgs: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = []
   357→  for (const m of priorMessages) {
   358→    if (m.role === 'user') {
   359→      let content = m.content
   360→      const atts = m.attachments ? (JSON.parse(m.attachments) as AttachmentMeta[]) : []
   361→      const textFiles = atts.filter((a) => a.textContent)
   362→      const images = atts.filter((a) => a.mimeType.startsWith('image/'))
   363→      if (textFiles.length) {
   364→        content +=
   365→          '\n\n[ATTACHED TEXT FILES]\n' +
   366→          textFiles.map((a) => `--- ${a.originalName} ---\n${a.textContent?.slice(0, 8000)}`).join('\n\n')
   367→      }
   368→      if (images.length) {
   369→        content += `\n\n[ATTACHED IMAGES: ${images.map((a) => a.originalName).join(', ')}] Use the vision tool with image_index to analyze them.`
   370→      }
   371→      msgs.push({ role: 'user', content })
   372→    } else if (m.role === 'assistant') {
   373→      msgs.push({ role: 'assistant', content: m.content })
   374→    } else if (m.role === 'tool') {
   375→      msgs.push({
   376→        role: 'user',
   377→        content: `[TOOL_RESULT] ${m.toolName}: ${m.toolResult ?? ''}`,
   378→      })
   379→    } else if (m.role === 'thought') {
   380→      // skip — thoughts are internal; we re-feed them via assistant content's <thought> tags
   381→    }
   382→  }
   383→  // current message
   384→  let userContent = currentUserMessage
   385→  const textFiles = currentAttachments.filter((a) => a.textContent)
   386→  const images = currentAttachments.filter((a) => a.mimeType.startsWith('image/'))
   387→  if (textFiles.length) {
   388→    userContent +=
   389→      '\n\n[ATTACHED TEXT FILES]\n' +
   390→      textFiles.map((a) => `--- ${a.originalName} ---\n${a.textContent?.slice(0, 8000)}`).join('\n\n')
   391→  }
   392→  if (images.length) {
   393→    userContent += `\n\n[ATTACHED IMAGES: ${images.map((a) => a.originalName).join(', ')}] Use the vision tool with image_index to analyze them.`
   394→  }
   395→  msgs.push({ role: 'user', content: userContent })
   396→
   397→  // Auto-truncate if too long — keep the most recent ~30k tokens of history
   398→  // and add a marker so the model knows earlier context was dropped.
   399→  const MAX_TOKENS = 50_000
   400→  const KEEP_TOKENS = 30_000
   401→  if (estimateTokens(msgs) > MAX_TOKENS) {
   402→    // Walk from the end of msgs, accumulating until we hit KEEP_TOKENS budget
   403→    let keptTokens = 0
   404→    let cutIndex = msgs.length
   405→    for (let i = msgs.length - 1; i >= 0; i--) {
   406→      const t = Math.ceil((msgs[i].content ?? '').length / 4)
   407→      if (keptTokens + t > KEEP_TOKENS * 4) {
   408→        cutIndex = i + 1
   409→        break
   410→      }
   411→      keptTokens += t
   412→      cutIndex = i
   413→    }
   414→    const trimmed = msgs.slice(cutIndex)
   415→    return [
   416→      {
   417→        role: 'user',
   418→        content:
   419→          '[Earlier conversation history truncated to fit context window. Earlier tool results and assistant messages were dropped.]',
   420→      },
   421→      ...trimmed,
   422→    ]
   423→  }
   424→
   425→  return msgs
   426→}

   428→export async function runAgent(opts: AgentRunOptions): Promise<AgentRunResult> {
   429→  const { conversationId, userMessage, attachments, language, emit } = opts
   430→
   431→  // 1) Recall relevant memories for context
   432→  const recalled = await recallMemories(userMessage.slice(0, 200), 8)
   433→  const memoryBlock = formatMemoryForPrompt(recalled)
   434→
   435→  const languageInstruction =
   436→    language === 'zh'
   437→      ? 'LANGUAGE INSTRUCTION: The user has toggled the agent to Chinese. Reply in 中文 (Chinese) for your FINAL answer regardless of input language.'
   438→      : 'LANGUAGE INSTRUCTION: The user has toggled the agent to English. Reply in English for your FINAL answer unless the user wrote in another language.'
   439→
   440→  const systemPrompt = `${SYSTEM_PROMPT}
   441→
   442→${languageInstruction}
   443→
   444→RECALLED MEMORIES (use as context, do not blindly trust if outdated):
   445→${memoryBlock}
   446→
   447→CURRENT UTC TIME: ${new Date().toUTCString()}`
   448→
   449→  const history = await buildHistoryMessages(conversationId, userMessage, attachments)
   450→  const ctx: ToolContext = { attachments, language }
   451→
   452→  const steps: AgentRunResult['steps'] = []
   453→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
   454→    { role: 'system', content: systemPrompt },
   455→    ...history,
   456→  ]
   457→
   458→  let finalAnswer = ''
   459→  let iter = 0
   460→
   461→  while (iter < MAX_ITERATIONS) {
   462→    iter++
   463→    let completion: any
   464→    try {
   465→      completion = await callLlmWithRetry(conversationMessages)
   466→    } catch (e: any) {
   467→      const friendly = friendlyLlmError(e)
   468→      await emit('error', { message: friendly })
   469→      finalAnswer = friendly
   470→      break
   471→    }
   472→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
   473→    if (!content || !content.trim()) {
   474→      finalAnswer = '(The agent produced no output. Please try rephrasing.)'
   475→      break
   476→    }
   477→
   478→    const parsed = parseAssistant(content)
   479→
   480→    // Emit thought if present
   481→    if (parsed.thought) {
   482→      await emit('thought', { content: parsed.thought })
   483→    }
   484→
   485→    // If no tool, this is the final answer
   486→    if (!parsed.tool) {
   487→      finalAnswer = content.replace(THOUGHT_RE, '').trim() || content.trim()
   488→      // Stream tokens (chunked) — SDK doesn't natively stream tokens here, so we send in ~80-char chunks for typing effect
   489→      const chunks = chunkText(finalAnswer, 80)
   490→      for (const c of chunks) {
   491→        await emit('token', { content: c })
   492→      }
   493→      break
   494→    }
   495→
   496→    // Tool call
   497→    const step: any = {
   498→      id: `step_${iter}_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   499→      thought: parsed.thought,
   500→      toolName: parsed.tool.name,
   501→      toolArgs: parsed.tool.args,
   502→      startedAt: Date.now(),
   503→    }
   504→    steps.push(step)
   505→    await emit('tool_call', {
   506→      stepId: step.id,
   507→      name: step.toolName,
   508→      args: step.toolArgs,
   509→      thought: step.thought,
   510→      stepNumber: iter,
   511→    })
   512→
   513→    // Execute
   514→    const toolResult = await dispatchTool(step.toolName!, step.toolArgs, ctx)
   515→    step.toolResult = toolResult
   516→    step.finishedAt = Date.now()
   517→    await emit('tool_result', {
   518→      stepId: step.id,
   519→      result: toolResult.result,
   520→      preview: toolResult.preview,
   521→      ok: toolResult.ok,
   522→      artifacts: toolResult.artifacts,
   523→    })
   524→
   525→    // If memory was stored, also emit memory_update for the right panel
   526→    if (step.toolName === 'memory_store' && toolResult.ok) {
   527→      await emit('memory_update', {
   528→        key: step.toolArgs?.key,
   529→        value: step.toolArgs?.value,
   530→        category: step.toolArgs?.category ?? 'general',
   531→      })
   532→    }
   533→
   534→    // Feed back to model. We append the assistant's raw tool-call message + a tool result.
   535→    conversationMessages.push({ role: 'assistant', content })
   536→    conversationMessages.push({
   537→      role: 'user',
   538→      content: `[TOOL_RESULT] ${step.toolName}: ${toolResult.result}`,
   539→    })
   540→
   541→    // Persist intermediate tool/thought rows so reloads show full trace
   542→    try {
   543→      if (step.thought) {
   544→        await db.message.create({
   545→          data: {
   546→            conversationId,
   547→            role: 'thought',
   548→            content: step.thought,
   549→          },
   550→        })
   551→      }
   552→      await db.message.create({
   553→        data: {
   554→          conversationId,
   555→          role: 'tool',
   556→          content: `[tool call] ${step.toolName} ${JSON.stringify(step.toolArgs)}`,
   557→          toolName: step.toolName,
   558→          toolArgs: JSON.stringify(step.toolArgs),
   559→          toolResult: toolResult.result,
   560→        },
   561→      })
   562→    } catch {
   563→      // ignore persistence errors mid-loop
   564→    }
   565→  }
   566→
   567→  if (!finalAnswer) {
   568→    finalAnswer =
   569→      "I've reached my tool-call limit for this turn. Here's what I have so far — let me know if you'd like me to continue."
   570→    await emit('token', { content: finalAnswer })
   571→  }
   572→
   573→  // Persist the final assistant message
   574→  const assistantRow = await db.message.create({
   575→    data: {
   576→      conversationId,
   577→      role: 'assistant',
   578→      content: finalAnswer,
   579→    },
   580→  })
   581→
   582→  // Update conversation title if it's still the default
   583→  const conv = await db.conversation.findUnique({ where: { id: conversationId } })
   584→  if (conv && (conv.title === 'New Conversation' || !conv.title)) {
   585→    const title = userMessage.slice(0, 50).trim() || 'New Conversation'
   586→    await db.conversation.update({ where: { id: conversationId }, data: { title } })
   587→  }
   588→
   589→  return {
   590→    finalAnswer,
   591→    steps,
   592→    persistedAssistantMessageId: assistantRow.id,
   593→  }
   594→}

   596→export function chunkText(text: string, size: number): string[] {
   597→  if (!text) return []
   598→  const out: string[] = []
   599→  for (let i = 0; i < text.length; i += size) out.push(text.slice(i, i + size))
   600→  return out
   601→}

   603→/**
   604→ * Convert a raw LLM API exception into a friendly, user-visible message.
   605→ * Detects rate-limit (429) errors specifically and produces a clear, actionable
   606→ * message instead of dumping the raw API JSON.
   607→ */
   608→export function friendlyLlmError(e: any): string {
   609→  const raw: string = e?.message ?? String(e)
   610→  const status: number | undefined = e?.status ?? e?.response?.status
   611→  const lower = raw.toLowerCase()
   612→
   613→  // Detect which provider failed
   614→  const isOpenai = lower.includes('openai') || lower.includes('fallback') || lower.includes('gpt-4o')
   615→  const isZai = lower.includes('z-ai') || lower.includes('zai') || lower.includes('glm')
   616→  const providerName = isOpenai ? 'OpenAI' : isZai ? 'Z.ai (GLM)' : 'AI provider'
   617→
   618→  if (status === 429 || lower.includes('429') || lower.includes('too many requests') || lower.includes('rate limit')) {
   619→    return `⏳ Agent007's ${providerName} is rate-limiting requests. Please wait 60 seconds and try again.`
   620→  }
   621→  if (status === 401 || status === 403 || lower.includes('unauthorized') || lower.includes('forbidden')) {
   622→    // Check for region block specifically
   623→    if (lower.includes('unsupported_country_region_territory') || lower.includes('region') && lower.includes('not supported')) {
   624→      return `🌍 Agent007's ${providerName} is blocked in this server region.
   625→
   626→The API key is VALID, but ${providerName} refuses to serve requests from this geographic location.
   627→
   628→WHICH PROVIDER FAILED: ${providerName}
   629→HTTP STATUS: ${status}
   630→
   631→TO FIX:
   632→1. Deploy to Vercel (US servers) — ${providerName} works there
   633→2. Use Z.ai SDK as primary (already working in dev)
   634→3. The key itself is fine — no need to change it
   635→
   636→Agent007 is still functional via Z.ai SDK (GLM-4-Plus).`
   637→    }
   638→    return `🔐 Agent007's ${providerName} rejected the request (auth/permission).
   639→
   640→This means the API key is invalid, expired, or doesn't have permission.
   641→
   642→WHICH PROVIDER FAILED: ${providerName}
   643→HTTP STATUS: ${status ?? 'unknown'}
   644→
   645→TO FIX:
   646→${isOpenai
   647→      ? '1. Check your OpenAI API key is valid at https://platform.openai.com/api-keys\n2. Ensure you have credits at https://platform.openai.com/account/billing\n3. Update the key in Settings → API Key Manager\n4. Or set OPENAI_API_KEY as a Vercel env var'
   648→      : '1. The Z.ai SDK may have a temporary auth issue\n2. Add an OPENAI_API_KEY as fallback in Settings → API Key Manager\n3. Or set OPENAI_API_KEY as a Vercel env var'}
   649→
   650→The operator has been notified. Please contact antonio.can2022@hotmail.com if this persists.`
   651→  }
   652→  if (status === 500 || status === 502 || status === 503 || lower.includes('server error') || lower.includes('service unavailable')) {
   653→    return `🛠️ Agent007's ${providerName} is having a server-side issue (HTTP ${status}). Please retry in a moment.`
   654→  }
   655→  if (lower.includes('timeout') || lower.includes('timed out')) {
   656→    return `⏱️ Agent007's ${providerName} took too long to respond. Please try again.`
   657→  }
   658→  return `⚠️ ${raw.slice(0, 200)}
   659→
   660→This may be a temporary issue. Try again, or if it persists, add an OPENAI_API_KEY in Settings → API Keys.`
   661→}
