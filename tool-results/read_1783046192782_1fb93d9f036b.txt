   518→/* ------------------------------------------------------------------ *

   522→  const [analytics, setAnalytics] = useState<any>(null)
   523→  const [loading, setLoading] = useState(true)
   524→
   525→  useEffect(() => {
   526→    let cancelled = false
   527→    fetch('/api/analytics/agents')
   528→      .then((r) => r.json())
   529→      .then((data) => {

   627→function BackupRestoreSection({ onToast }: { onToast: (msg: string) => void }) {
   628→  const [exporting, setExporting] = useState(false)
   629→  const [importing, setImporting] = useState(false)
   630→  const [importResult, setImportResult] = useState<any>(null)
   631→
   632→  const handleExport = async () => {
   633→    if (exporting) return
   634→    setExporting(true)
   635→    try {
   636→      const res = await fetch('/api/backup')
   637→      if (!res.ok) throw new Error(`Export failed (HTTP ${res.status})`)
   638→      const blob = await res.blob()
   639→      const url = URL.createObjectURL(blob)
   640→      const a = document.createElement('a')
   641→      a.href = url
   642→      a.download = `agent007-backup-${new Date().toISOString().slice(0, 10)}.json`
   643→      document.body.appendChild(a)
   644→      a.click()
   645→      document.body.removeChild(a)
   646→      URL.revokeObjectURL(url)
   647→      onToast('Backup downloaded successfully')
   648→    } catch (e: any) {
   649→      onToast(`Export failed: ${e?.message ?? 'unknown error'}`)
   650→    } finally {
   651→      setExporting(false)
   652→    }
   653→  }
   654→
   655→  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
   656→    const file = e.target.files?.[0]
   657→    if (!file || importing) return
   658→    setImporting(true)
   659→    setImportResult(null)
   660→    try {
   661→      const text = await file.text()
   662→      const backup = JSON.parse(text)
   663→      const res = await fetch('/api/backup', {
   664→        method: 'POST',
   665→        headers: { 'Content-Type': 'application/json' },
   666→        body: JSON.stringify({ backup }),
   667→      })
   668→      const result = await res.json()
   669→      if (!res.ok || !result.ok) throw new Error(result.error || `HTTP ${res.status}`)
   670→      setImportResult(result.stats)
   671→      onToast(
   672→        `Restored: ${result.stats.conversations} conversations, ${result.stats.messages} messages, ${result.stats.customSubagents} agents`
   673→      )
   674→    } catch (e: any) {
   675→      onToast(`Import failed: ${e?.message ?? 'invalid file'}`)
   676→    } finally {
   677→      setImporting(false)
   678→      if (e.target) e.target.value = ''
   679→    }
   680→  }
   681→
   682→  return (
   683→    <section className="glass rounded-xl p-4 border-cyan-400/15">
   684→      <div className="flex items-center gap-2 mb-3">
   685→        <Database className="w-4 h-4 text-cyan-300" />
   686→        <h2 className="text-sm font-semibold text-[#e0e7ff] tracking-wide">BACKUP & RESTORE</h2>
   687→      </div>
   688→      <p className="text-[11px] text-[#7c89b5] mb-3 leading-relaxed">
   689→        Export all dashboard data (conversations, memories, income entries, schedules, custom
   690→        sub-agents, settings) as a JSON file. Restore on any Agent007 instance.
   691→      </p>
   692→      <div className="flex flex-col sm:flex-row gap-2">
   693→        <button
   694→          onClick={handleExport}
   695→          disabled={exporting}
   696→          className="flex-1 neon-btn-cyan rounded-lg py-2 text-xs font-bold tracking-wider flex items-center justify-center gap-2 disabled:opacity-60"
   697→          style={{ touchAction: 'manipulation' }}
   698→        >
   699→          {exporting ? (
   700→            <Loader2 className="w-3.5 h-3.5 animate-spin" />
   701→          ) : (
   702→            <Download className="w-3.5 h-3.5" />
   703→          )}
   704→          {exporting ? 'EXPORTING…' : 'EXPORT BACKUP'}
   705→        </button>
   706→        <label
   707→          className="flex-1 glass border-cyan-400/30 hover:border-cyan-400/60 rounded-lg py-2 text-xs font-bold tracking-wider flex items-center justify-center gap-2 cursor-pointer transition text-cyan-200"
   708→          style={{ touchAction: 'manipulation' }}
   709→        >
   710→          {importing ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Upload className="w-3.5 h-3.5" />}

   712→          <input

   715→            onChange={handleImport}
   716→            disabled={importing}
   717→            className="hidden"
   718→          />

   721→      {importResult && (
   722→        <div className="mt-3 text-[10px] text-[#9bb5d4] glass rounded-lg p-2 border-emerald-400/20">
   723→          <div className="flex items-center gap-1 mb-1 text-emerald-200 font-semibold">

  1305→function ApiKeyManagerSection({ onToast }: { onToast: (msg: string) => void }) {
  1306→  const [keys, setKeys] = useState<any[]>([])

  1312→

  1315→

  1321→
  1322→  return (
  1323→    <section className="glass rounded-xl p-4 border-cyan-400/15">
