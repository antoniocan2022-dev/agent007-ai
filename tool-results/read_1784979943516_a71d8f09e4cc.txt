     6→// Helper: fetch internal URL with better error handling for Vercel
     7→async function internalFetch(url: string, options?: any): Promise<any> {
     8→  try {
     9→    const res = await fetch(url, {
    10→      ...options,
    11→      redirect: 'follow',
    12→      signal: options?.signal ?? AbortSignal.timeout(15000),
    13→    })
    14→    if (!res.ok) {
    15→      const text = await res.text().catch(() => '')
    16→      return { ok: false, error: `HTTP ${res.status}: ${text.slice(0, 200)}`, _httpError: true }
    17→    }
    18→    const contentType = res.headers.get('content-type') || ''
    19→    if (!contentType.includes('application/json')) {
    20→      const text = await res.text().catch(() => '')
    21→      return { ok: false, error: `Non-JSON response (${contentType}): ${text.slice(0, 100)}`, _parseError: true }
    22→    }
    23→    return await res.json().catch(() => ({}))
    24→  } catch (e: any) {
    25→    return { ok: false, error: e?.message ?? String(e), _fetchError: true }
    26→  }
    27→}

    56→/* Regex to find <dispatch agent="..." task="..."/> OR <dispatch agent="..." task="..."> tags.
    57→ * Accepts both self-closing (/> ) and non-self-closing (> ) forms because the LLM
    58→ * sometimes emits one or the other. Uses non-greedy [\s\S]*? for the task value. */
    59→const DISPATCH_RE = /<dispatch\s+agent=["']([^"']+)["']\s+task=["']([\s\S]*?)["']\s*\/?>/i

    61→// UPGRADE #63 + #86 — Detect <dispatch_subagent ...> in ALL formats the LLM emits:
    62→//   (a) <dispatch_subagent id="quill">task text</dispatch_subagent>   (paired)
    63→//   (b) <dispatch_subagent id="quill" task="..."/>                      (self-closing w/ task attr)
    64→//   (c) <dispatch_subagent id="quill"/>                                  (self-closing, task in body)
    65→//   (d) <dispatch_subagent id="quill" task="...">task</dispatch_subagent> (mixed)
    66→// Without this, dispatches leak as raw XML text to the user (the "weird incomprehensible answer" bug).
    67→const DISPATCH_SUBAGENT_RE = /<dispatch_subagent\s+id=["']([^"']+)["']\s*(?:task=["']([\s\S]*?)["']\s*)?(?:\/>|>([\s\S]*?)<\/dispatch_subagent>)/i

    70→// These are NOT real tools — the correct way is <tool name="parallel_executor">. If the LLM emits
    71→// pseudo-XML, we strip it from the final answer so the user never sees raw tags.
    72→const PSEUDO_XML_RE = /<\/?(?:parallel_executor|reasoning_trace|reasoning|execution|plan|action|reflect|reflection|analyze)(?:\s+[^>]*)?(?:\/>|>[\s\S]*?<\/(?:parallel_executor|reasoning_trace|reasoning|execution|plan|action|reflect|reflection|analyze)>)/gi
    73→const REASONING_TRACE_BLOCK_RE = /(?:^|\n)\s*(?:REASONING\s*TRACE|REASONING|INTERNAL\s*MONOLOGUE|CHAIN\s*OF\s*THOUGHT|THINKING)\s*:?\s*\n[\s\S]*?(?=\n\s*(?:[A-Z][A-Z _]{4,}|<|YOU:|$))/gi

    75→// UPGRADE #95 — AUTO-CONVERTER: Convert <parallel_executor>{json}</parallel_executor>
    76→// to <tool name="parallel_executor">{json}</tool> BEFORE parsing.
    77→// This fixes the root cause: when the LLM uses the WRONG format, we auto-correct it
    78→// so the tools ACTUALLY RUN and the user sees real results (not raw XML).
    79→// Regex matches: <parallel_executor>{anything}</parallel_executor> OR <parallel_executor>{anything}/>
    80→const PARALLEL_EXECUTOR_CONVERTER_RE = /<parallel_executor(?:\s+[^>]*)?>\s*(\{[\s\S]*?\})\s*<\/parallel_executor>|<parallel_executor(?:\s+[^>]*)?>\s*(\{[\s\S]*?\})\s*\/>/gi
    81→
    82→/**
    83→ * UPGRADE #95 — Auto-convert pseudo-XML tool calls to proper <tool> format.
    84→ * Called BEFORE parseOrchestrator() so the tools actually execute.
    85→ *
    86→ * Handles:
    87→ *   <parallel_executor>{"tools":[...]}</parallel_executor>
    88→ *     → <tool name="parallel_executor">{"tools":[...]}</tool>
    89→ *
    90→ *   <parallel_executor>{"tools":[...]}/>
    91→ *     → <tool name="parallel_executor">{"tools":[...]}</tool>
    92→ *
    93→ * Also handles other common pseudo-XML tool patterns the LLM invents:
    94→ *   <search>{"query":"..."}</search>
    95→ *     → <tool name="search">{"query":"..."}</tool>
    96→ *   <analyze>{"data":"..."}</analyze>
    97→ *     → <tool name="analyze">{"data":"..."}</tool>
    98→ */
    99→function autoConvertPseudoToolCalls(content: string): string {
   100→  // 1. Convert <parallel_executor>{json}</parallel_executor> → <tool name="parallel_executor">{json}</tool>
   101→  let converted = content.replace(PARALLEL_EXECUTOR_CONVERTER_RE, (match, json1, json2) => {
   102→    const json = json1 || json2
   103→    return `<tool name="parallel_executor">${json}</tool>`
   104→  })
   105→
   106→  // 2. Convert other common pseudo-XML tool patterns:
   107→  //    <search>{json}</search> → <tool name="search">{json}</tool>
   108→  //    <analyze>{json}</analyze> → <tool name="analyze">{json}</analyze>
   109→  //    <fetch>{json}</fetch> → <tool name="fetch">{json}</fetch>
   110→  //    etc.
   111→  const otherPseudoTools = ['search', 'fetch', 'analyze', 'summarize', 'translate', 'generate', 'process', 'execute']
   112→  for (const toolName of otherPseudoTools) {
   113→    const re = new RegExp(`<${toolName}(?:\\s+[^>]*)?>\\s*(\\{[\\s\\S]*?\\})\\s*</${toolName}>`, 'gi')
   114→    converted = converted.replace(re, `<tool name="${toolName}">$1</tool>`)
   115→  }
   116→
   117→  return converted
   118→}

   124→interface OrchestratorParsed {
   125→  thought?: string
   126→  tool?: { name: string; args: any }
   127→  dispatch?: { agentId: string; task: string }
   128→  manage?: { action: string; attrs: Record<string, string>; raw: string }
   129→  textAfter: string
   130→  raw: string
   131→}

   133→function parseOrchestrator(content: string): OrchestratorParsed {
   134→  const thoughtMatch = content.match(THOUGHT_RE)
   135→  const thought = thoughtMatch?.[1]?.trim()
   136→
   137→  const dispatchMatch = content.match(DISPATCH_RE)
   138→  // UPGRADE #63 — Also check <dispatch_subagent id="...">task</dispatch_subagent> format
   139→  const dispatchSubagentMatch = content.match(DISPATCH_SUBAGENT_RE)
   140→  const toolMatch = content.match(TOOL_RE)
   141→  const manageMatch = content.match(MANAGE_RE)
   142→
   143→  // Priority: dispatch > dispatch_subagent > manage > tool
   144→  if (dispatchMatch) {
   145→    const agentId = dispatchMatch[1].trim().toLowerCase()
   146→    const task = dispatchMatch[2].trim()
   147→    return {
   148→      thought,
   149→      dispatch: { agentId, task },
   150→      textAfter: content.slice(content.indexOf(dispatchMatch[0]) + dispatchMatch[0].length).replace(THOUGHT_RE, '').trim(),
   151→      raw: content,
   152→    }
   153→  }
   154→  // UPGRADE #63 + #86 — Handle <dispatch_subagent id="..."> in ALL formats
   155→  // Capture groups: [1]=id, [2]=task attr (if present), [3]=body content (if paired tag)
   156→  if (dispatchSubagentMatch) {
   157→    const agentId = dispatchSubagentMatch[1].trim().toLowerCase()
   158→    // Prefer task attribute (group 2); fall back to body content (group 3)
   159→    const task = (dispatchSubagentMatch[2] ?? dispatchSubagentMatch[3] ?? '').trim() || 'execute task'
   160→    return {
   161→      thought,
   162→      dispatch: { agentId, task },
   163→      textAfter: content.slice(content.indexOf(dispatchSubagentMatch[0]) + dispatchSubagentMatch[0].length).replace(THOUGHT_RE, '').trim(),
   164→      raw: content,
   165→    }
   166→  }
   167→  if (manageMatch && manageMatch.length > 0) {
   168→    const tag = manageMatch[0]
   169→    const attrs = parseManageAttrs(tag)
   170→    const action = (attrs.action ?? '').toString().trim().toLowerCase()
   171→    return {
   172→      thought,
   173→      manage: { action, attrs, raw: tag },
   174→      textAfter: content.replace(tag, '').replace(THOUGHT_RE, '').trim(),
   175→      raw: content,
   176→    }
   177→  }
   178→  if (toolMatch) {
   179→    const name = (toolMatch[1] ?? '').trim()
   180→    if (!name) return { thought, textAfter: content.replace(THOUGHT_RE, '').trim(), raw: content }
   181→    let args: any = {}
   182→    const raw = (toolMatch[2] ?? '').trim()
   183→    if (raw) {
   184→      try {
   185→        args = JSON.parse(raw)
   186→      } catch {
   187→        const m: Record<string, string> = {}
   188→        const re = /"([^"]+)"\s*:\s*"([^"]*)"/g
   189→        let mm: RegExpExecArray | null
   190→        while ((mm = re.exec(raw))) m[mm[1]] = mm[2]
   191→        args = m
   192→      }
   193→    }
   194→    return { thought, tool: { name, args }, textAfter: '', raw: content }
   195→  }
   196→  return { thought, textAfter: content.replace(THOUGHT_RE, '').trim(), raw: content }
   197→}
   198→
   199→/** Parse attributes from a <manage .../> tag. Handles key="value" pairs with
   200→ * either single or double quotes. Also supports attribute values containing
   201→ * spaces because the regex is greedy on the quoted portion. */
   202→function parseManageAttrs(tag: string): Record<string, string> {
   203→  const attrs: Record<string, string> = {}
   204→  // Match: attrName="value" or attrName='value'
   205→  // We allow newlines inside the value via [\s\S]*? (non-greedy).
   206→  const re = /([a-zA-Z_][a-zA-Z0-9_]*)\s*=\s*(?:"([^"]*)"|'([^']*)')/g
   207→  let m: RegExpExecArray | null
   208→  while ((m = re.exec(tag))) {
   209→    const key = m[1]
   210→    const val = m[2] ?? m[3] ?? ''
   211→    attrs[key] = val
   212→  }
   213→  return attrs
   214→}

   474→/* ------------------------------------------------------------------ *
   475→ * Fast-path detection — when the user's message is a CLEAR, unambiguous
   476→ * create_agent request (short, single-intent, matches a strict regex),
   477→ * we skip the LLM round-trip entirely and execute the manage action
   478→ * directly. This makes "create a sub-agent named X" instant and immune

   485→interface FastPathCreateAgent {
   486→  action: 'create_agent'
   487→  attrs: Record<string, string>

   490→/* Match: "create a new sub-agent named 'Cybersecurity A'"
   491→ *        "add an agent called TESTFAST via fast path"
   492→ *        "build a subagent named Foo"
   493→ * Captures the agent name in group 1 (quotes optional). */
   494→const FAST_CREATE_RE =
   495→  /(?:create|add|build)\s+(?:a\s+)?(?:new\s+)?(?:sub-?agent|agent)\s+(?:named|called)\s+["']?([A-Za-z0-9 _\-]+?)["']?(?:[\s.,]|$)/i
   496→
   497→/* Match: role="..." / role: '...' / role is "..." / role named "..." */
   498→function extractAttr(message: string, key: string): string | null {
   499→  // key="value" or key='value' or key=value-with-no-spaces
   500→  const kvRe = new RegExp(
   501→    `${key}\\s*(?:=|:)\\s*["']?([^"'\\n,]+?)["']?(?:[\\s,]|$)`,
   502→    'i'
   503→  )
   504→  const m1 = message.match(kvRe)
   505→  if (m1) return m1[1].trim()
   506→  // "role is X" / "role named X" / "specialized in X"
   507→  const phraseRe = new RegExp(
   508→    `${key}\\s+(?:is|named|specialized\\s+in|specialising\\s+in|specialty)\\s+["']?([A-Za-z0-9 _\\-/]+?)["']?(?:[\\s.,]|$)`,
   509→    'i'
   510→  )
   511→  const m2 = message.match(phraseRe)
   512→  if (m2) return m2[1].trim()
   513→  return null
   514→}

   516→function detectFastPathManage(userMessage: string): FastPathCreateAgent | null {
   517→  if (!userMessage || userMessage.length > 500) return null
   518→  // Require an explicit "fast path" hint OR a very clear single-intent command.
   519→  // The "fast path" hint lets users opt in; we also accept very-short clear
   520→  // commands without the hint.
   521→  const hasFastHint = /\bfast[\s-]?path\b/i.test(userMessage)
   522→  const m = userMessage.match(FAST_CREATE_RE)
   523→  if (!m) return null
   524→  const name = m[1].trim()
   525→  if (!name || name.length < 2 || name.length > 80) return null
   526→
   527→  // If the message contains words suggesting other actions (dispatch, delete,
   528→  // log, schedule, set goal, etc.), DON'T fast-path — let the LLM handle it.
   529→  const otherActions =
   530→    /\b(?:dispatch|delete|remove|toggle|disable|enable|log\s+\$|log\s+income|set\s+(?:my\s+)?(?:income|growth|daily)|create\s+schedule|update\s+settings)\b/i.test(
   531→      userMessage
   532→    )
   533→  if (otherActions) return null
   534→
   535→  // Without the fast-path hint, require very short messages to avoid
   536→  // over-eager matching on long descriptive requests.
   537→  if (!hasFastHint && userMessage.length > 200) return null
   538→
   539→  const attrs: Record<string, string> = { name }
   540→  const role = extractAttr(userMessage, 'role')
   541→  if (role) attrs.role = role
   542→  const specialty = extractAttr(userMessage, 'specialty')
   543→  if (specialty) attrs.specialty = specialty
   544→  const color = extractAttr(userMessage, 'color')
   545→  if (color) attrs.color = color
   546→  const icon = extractAttr(userMessage, 'icon')
   547→  if (icon) attrs.icon = icon
   548→  const systemPrompt = extractAttr(userMessage, 'system_prompt')
   549→  if (systemPrompt) attrs.system_prompt = systemPrompt
   550→  const allowedTools = extractAttr(userMessage, 'allowed_tools')
   551→  if (allowedTools) attrs.allowed_tools = allowedTools
   552→
   553→  // If user provided a "specialized in X" phrase but no role, derive a role
   554→  if (!attrs.role) {
   555→    const specMatch = userMessage.match(
   556→      /specialized\s+in\s+([A-Za-z0-9 _\-/]+?)(?:[\s.,]|$)/i
   557→    )
   558→    if (specMatch) {
   559→      attrs.specialty = specMatch[1].trim()
   560→      attrs.role = specMatch[1].trim() + ' Specialist'
   561→    }
   562→  }
   563→
   564→  // Fast-path defaults so the action can succeed without forcing the user to
   565→  // specify everything. The user can always edit afterwards via the panel.
   566→  if (!attrs.role) {
   567→    attrs.role = `${name} Specialist`
   568→  }
   569→  if (!attrs.specialty) {
   570→    attrs.specialty = `Custom specialist created via fast-path`
   571→  }
   572→  if (!attrs.allowed_tools) {
   573→    // Sensible default: read-only research tools
   574→    attrs.allowed_tools = 'web_search,page_reader,wikipedia_search,wikipedia_read,free_apis_directory,memory_store,memory_recall'
   575→  }
   576→  if (!attrs.system_prompt) {
   577→    attrs.system_prompt = `You are ${name.toUpperCase()}, a custom specialist sub-agent of Agent007 AI.\n\nYour role: ${attrs.role}.\nYour specialty: ${attrs.specialty}.\n\nALLOWED TOOLS:\n- web_search — Google-style search for current info\n- page_reader — read full web pages\n- memory_store / memory_recall — persist + recall context\n- wikipedia_search / wikipedia_read — encyclopedic background\n- free_apis_directory — find public data APIs\n\nOUTPUT FORMAT:\n- <thought>brief reasoning</thought> before each action\n- <tool name="...">{json}</tool> to call a tool\n- Plain markdown final answer\n\nRULES:\n- Be concise and structured.\n- Cite sources for any factual claim.\n- Max 6 tool calls per turn.`
   578→  }
   579→
   580→  return { action: 'create_agent', attrs }
   581→}

   729→/**
   730→ * UPGRADE #117 — Query Complexity Router
   731→ *
   732→ * Classifies the user's message to decide whether the agent should:
   733→ *   - 'direct': Answer directly with a smart, deep response (90% of messages)
   734→ *   - 'dispatch': Genuinely needs subagent work (10% of messages)
   735→ *
   736→ * This prevents the agent from dispatching to a subagent for simple questions
   737→ * like "What's the best affiliate strategy?" — which should get an immediate
   738→ * 500-1500 word smart answer, not a slow dispatch-synthesize loop.
   739→ *
   740→ * Classification logic:
   741→ *   - Greetings, questions, explanations, advice, comparisons, brainstorming → 'direct'
   742→ *   - Multi-step research, content creation, code building, deployment → 'dispatch'
   743→ *   - Default (anything that doesn't match) → 'direct' (smart default)
   744→ */
   745→function classifyQuery(message: string): 'direct' | 'dispatch' {
   746→  const lower = message.toLowerCase().trim()
   747→  if (!lower) return 'direct'
   748→
   749→  // ── DISPATCH patterns — genuinely needs subagent work ────────────────
   750→  // These are ACTION requests that require tool execution
   751→  const dispatchPatterns = [
   752→    // Multi-step research requiring multiple tool calls
   753→    /^(research|investigate|find out|look up|search for)\s+(the|all|every|top\s+\d+)/i,
   754→    /\bsearch the web\b/i,
   755→    /\b(verify|check|validate)\s+(if|whether|that|the)\b/i, // verification tasks
   756→    // Content creation tasks
   757→    /^(write|create|build|design|publish|draft|generate)\s+(a|an|the|some)?\s*(blog|article|post|email|newsletter|script|landing\s+page|website|funnel|graphic|image|video|tweet|thread)/i,
   758→    /^(write|create)\s+(me\s+)?a\s+/i,
   759→    // Code/build tasks
   760→    /^(build|deploy|fix|repair|install|set\s+up|implement|code|develop|refactor)\s+/i,
   761→    /\b(deploy|push\s+to\s+production|ship\s+it)\b/i,
   762→    // System operations
   763→    /^(run|execute|start|stop|restart)\s+(the\s+)?(mission|tick|scan|audit|test|pipeline|workflow)/i,
   764→    /\bself.?heal\b/i,
   765→    /\b(check|audit|scan)\s+(tools?|system|infrastructure|security)\b/i,
   766→    // Explicit dispatch commands
   767→    /\b(dispatch|send\s+to|ask\s+the\s+(scout|aurora|echo|forge|pulse|developer|quantum|cybersecurity))\b/i,
   768→  ]
   769→
   770→  // ── DIRECT patterns — answer with smart response ────────────────────
   771→  // These are questions, analysis requests, advice, explanations
   772→  const directPatterns = [
   773→    // Greetings + social
   774→    /^(hi|hello|hey|good\s+(morning|afternoon|evening)|sup|yo)\b/i,
   775→    /^(thanks|thank\s+you|cool|nice|great|awesome|perfect)\b/i,

   777→    /\?$/,
   778→    // Question words
   779→    /^(what|why|how|when|where|who|which|whose|whom)\b/i,
   780→    /^(can|could|would|will|should|do|does|did|is|are|am|was|were|have|has|had)\s+(you|i|we|the)\b/i,

   782→    /^(explain|describe|tell\s+me\s+about|what\s+is|what\s+are|define|elaborate)\b/i,
   783→    // Advice/recommendations
   784→    /^(should\s+i|is\s+it\s+worth|do\s+you\s+recommend|what\s+do\s+you\s+(think|suggest|recommend|advise))\b/i,
   785→    /^(advice|recommend|suggest)\b/i,
   786→    // Comparisons
   787→    /^(compare|difference\s+between|vs\.?|versus)\b/i,
   788→    // Brainstorming
   789→    /^(brainstorm|ideas?\s+for|give\s+me\s+\d+\s+ideas|list\s+\d+\s+)/i,
   790→    // Analysis
   791→    /^(analyze|analysis|assess|evaluate|review)\b/i,
   792→    // Strategy
   793→    /(strategy|strategic|plan|approach|roadmap|game\s+plan)\b/i,
   794→    // Opinions
   795→    /^(what\s+do\s+you\s+(think|feel|believe)|your\s+opinion|your\s+thoughts)\b/i,
   796→    // Continue/status (short messages)
   797→    /^(continue|ok|okay|proceed|go\s+ahead|keep\s+going|status|update|what's\s+new|anything\s+new)\s*\.?\s*$/i,
   798→    // Follow-up questions
   799→    /^(and|but|or|so|then|also|additionally|moreover|furthermore)\b/i,
   800→    // "Tell me more" / "go deeper"
   801→    /(tell\s+me\s+more|go\s+deeper|elaborate|expand\s+on|dive\s+deeper)/i,
   802→    // "What about" / "How about"
   803→    /^(what\s+about|how\s+about)\b/i,
   804→  ]
   805→
   806→  // Check dispatch patterns first (more specific)
   807→  for (const pattern of dispatchPatterns) {
   808→    if (pattern.test(lower)) return 'dispatch'
   809→  }
   810→
   811→  // Check direct patterns
   812→  for (const pattern of directPatterns) {
   813→    if (pattern.test(lower)) return 'direct'
   814→  }
   815→
   816→  // Default: direct (smart response)
   817→  // Most conversational messages should get a smart direct answer
   818→  return 'direct'
   819→}

   821→export async function runOrchestrator(opts: OrchestratorRunOptions): Promise<OrchestratorRunResult> {
   822→  const { conversationId, userMessage, attachments, language, emit } = opts

   831→        where: { userId, status: { in: ['pending', 'executing'] } },

   849→            where: { id: p.id },
   850→            data: {
   851→              status: result.ok ? 'done' : 'failed',

   866→            where: { id: p.id },

   870→      }
   871→    }
   872→  } catch (replayOuterErr) {

   876→
   877→  // 0b) Fast-path: detect a clear, unambiguous create_agent request and
   878→  //     execute it directly without an LLM round-trip. Saves time, tokens,

   883→      conversationId,

   885→      language,
   886→      emit,
   887→      fastPath,
   888→    })
   889→  }
   890→

   892→  const recalled = await recallMemories(userMessage.slice(0, 200), 8)

   896→    language === 'zh'
   897→      ? 'LANGUAGE INSTRUCTION: The user has toggled the agent to Chinese. Reply in 中文 (Chinese) for your FINAL answer regardless of input language.'
   898→      : 'LANGUAGE INSTRUCTION: The user has toggled the agent to English. Reply in English for your FINAL answer unless the user wrote in another language.'

   902→  // in BASE_SYSTEM_PROMPT. This fixes the bug where the Super Agent would
   903→  // tell the user "that agent doesn't exist" when they addressed a custom

   910→    .map((a) => {

   916→
   917→CURRENTLY AVAILABLE SUB-AGENTS (built-in + custom, fetched live from DB):
   918→${agentListDynamic}

   921→you MUST dispatch that exact agent via <dispatch agent="agent_id" task="..."/> using the
   922→dispatch id shown at the start of each line above. For built-in agents the dispatch id is
   923→their lowercase name (aurora, vertex, etc.). For custom agents the dispatch id is their
   924→display name (e.g., "Cybersecurity A", "TRADER"). Do NOT do the work yourself with direct

   927→

   929→
   930→${ORCHESTRATOR_PROMPT_ADDENDUM}
   931→
   932→${dynamicAgentSection}
   933→
   934→${languageInstruction}
   935→
   936→RECALLED MEMORIES (use as context, do not blindly trust if outdated):
   937→${memoryBlock}
   938→
   939→CURRENT UTC TIME: ${new Date().toUTCString()}`
   940→
   941→  const history = await buildHistoryMessages(conversationId, userMessage, attachments)
   942→  const ctx: ToolContext = { attachments, language }
   943→
   944→  // UPGRADE #68 — Continue command support
   945→  const continuePatterns = /^(continue|keep going|go ahead|go on|ok|okay|yes|proceed|finish|done\?|are you done\?|status|update|what's the status|keep working|don't stop|resume)\s*\.?\s*$/i
   946→  const isContinueCommand = continuePatterns.test(userMessage.trim())
   947→
   948→  const steps: OrchestratorRunResult['steps'] = []
   949→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }>
   950→  if (isContinueCommand) {
   951→    const lastAssistant = [...history].reverse().find((m) => m.role === 'assistant')
   952→    conversationMessages = lastAssistant
   953→      ? [
   954→          { role: 'system', content: systemPrompt },
   955→          ...history,
   956→          { role: 'user', content: `[CONTINUE COMMAND] Owner typed "${userMessage}". CONTINUE your previous work. Don't start over — pick up where you left off. Your last response was:\n\n${lastAssistant.content.slice(0, 500)}\n\nNow EXECUTE the next step. Use <tool name="...">, <dispatch agent="..." task="..."/> or <dispatch_subagent id="...">task</dispatch_subagent> tags. Do not repeat yourself — advance the task.` },
   957→        ]
   958→      : [{ role: 'system', content: systemPrompt }, ...history]
   959→  } else {
   960→    conversationMessages = [
   961→      { role: 'system', content: systemPrompt },
   962→      ...history,
   963→    ]
   964→  }
   965→
   966→  let finalAnswer = ''
   967→  let iter = 0
   968→  let dispatchCount = 0
   969→  let manageCount = 0
   970→  // Cache the merged subagent list once per orchestrator run (12 built-in + custom + overlays).
   971→  let mergedSubagents: Subagent[] | null = null
   972→  const getMerged = async (): Promise<Subagent[]> => {
   973→    if (!mergedSubagents) mergedSubagents = await getAllSubagents({ includeDisabled: true })
   974→    return mergedSubagents
   975→  }
   976→
   977→  // ════════════════════════════════════════════════════════════════════
   978→  // UPGRADE #117 — Query Complexity Router
   979→  // ════════════════════════════════════════════════════════════════════
   980→  // Classify the user's message to decide whether to:
   981→  //   - 'direct': Answer directly with a smart, deep response (90% of messages)
   982→  //   - 'dispatch': Genuinely needs subagent work (10% of messages)
   983→  //
   984→  // This prevents the agent from dispatching to a subagent for simple
   985→  // questions like "What's the best affiliate strategy?" — which should
   986→  // get an immediate 500-1500 word smart answer, not a slow dispatch loop.
   987→  const queryType = classifyQuery(userMessage)
   988→  if (queryType === 'direct') {
   989→    // Inject a system nudge that tells the agent to answer directly
   990→    conversationMessages.push({
   991→      role: 'user',
   992→      content: `[SYSTEM ROUTER] This is a direct question/analysis/advice request. Do NOT dispatch to a subagent. Answer DIRECTLY with a deep, intelligent response (500-1500 words for complex questions, concise for simple ones). Use ## headers, **bold**, bullet lists. Provide examples. Show your reasoning. End with next steps.`,
   993→    })
   994→  }
   995→
   996→  while (iter < MAX_ITERATIONS) {
   997→    iter++
   998→
   999→    // UPGRADE #68 — Heartbeat emission (so dashboard shows progress)
  1000→    const lastStep = steps.length > 0 ? steps[steps.length - 1] : null
  1001→    try {
  1002→      await emit('heartbeat', {
  1003→        iteration: iter,
  1004→        maxIterations: MAX_ITERATIONS,
  1005→        toolsCalled: steps.length,
  1006→        dispatchesCalled: dispatchCount,
  1007→        manageActionsCalled: manageCount,
  1008→        lastToolName: (lastStep as any)?.toolName ?? null,
  1009→        lastThought: (lastStep as any)?.thought ? String((lastStep as any).thought).slice(0, 200) : null,
  1010→        startedAt: steps.length > 0 ? (steps[0] as any).startedAt : Date.now(),
  1011→        elapsedMs: steps.length > 0 ? Date.now() - (steps[0] as any).startedAt : 0,
  1012→        message: `Working — step ${iter}/${MAX_ITERATIONS}, ${steps.length} tool${steps.length === 1 ? '' : 's'} called, agent is alive`,
  1013→      })
  1014→    } catch {}
  1015→
  1016→    let completion: any
  1017→    try {
  1018→      completion = await callLlmWithRetry(conversationMessages)
  1019→    } catch (e: any) {
  1020→      const friendly = friendlyLlmError(e)
  1021→      // UPGRADE #131: Send rateLimited flag explicitly — don't rely on text matching
  1022→      const rateLimited = isRateLimitError(e)
  1023→      await emit('error', { message: friendly, rateLimited })
  1024→      finalAnswer = friendly
  1025→      break
  1026→    }
  1027→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
  1028→    if (!content.trim()) {
  1029→      finalAnswer = '(The agent produced no output. Please try rephrasing.)'
  1030→      break
  1031→    }
  1032→
  1033→    // UPGRADE #119 — Extract reasoning from the LLM response
  1034→    const reasoning: string | null =
  1035→      completion?.choices?.[0]?.message?.reasoning ||
  1036→      completion?.choices?.[0]?.message?.reasoning_content ||
  1037→      completion?.choices?.[0]?.message?.thinking ||
  1038→      completion?._reasoning ||
  1039→      null
  1040→    if (reasoning) {
  1041→      try { await emit('reasoning', { content: reasoning }) } catch {}
  1042→    }
  1043→
  1044→    // UPGRADE #95 — Auto-convert pseudo-XML tool calls BEFORE parsing.
  1045→    // This fixes the root cause: when the LLM emits <parallel_executor>{json}</parallel_executor>
  1046→    // (wrong format), we convert it to <tool name="parallel_executor">{json}</tool> (correct format)
  1047→    // so the tools ACTUALLY RUN and the user sees real results (not raw XML).
  1048→    const convertedContent = autoConvertPseudoToolCalls(content)
  1049→
  1050→    const parsed = parseOrchestrator(convertedContent)
  1051→
  1052→    // Emit thought
  1053→    if (parsed.thought) {
  1054→      await emit('thought', { content: parsed.thought })
  1055→    }
  1056→
  1057→    // 0) Manage path — parse <manage .../> tags and execute server-side.
  1058→    if (parsed.manage) {
  1059→      const { action, attrs } = parsed.manage
  1060→      if (manageCount >= MAX_MANAGE_ACTIONS) {
  1061→        const capMsg = `Reached max manage actions (${MAX_MANAGE_ACTIONS}). Please synthesize the answer from what you have.`
  1062→        conversationMessages.push({ role: 'assistant', content })
  1063→        conversationMessages.push({ role: 'user', content: `[SYSTEM] ${capMsg}` })
  1064→        continue
  1065→      }
  1066→      manageCount++
  1067→
  1068→      const stepId = makeId('manage')
  1069→      await emit('manage_action', {
  1070→        stepId,
  1071→        action,
  1072→        attrs,
  1073→        thought: parsed.thought,
  1074→        stepNumber: iter,
  1075→        status: 'running',
  1076→      })
  1077→
  1078→      // Persist a PendingManageAction row before executing (so we can replay
  1079→      // if this run crashes mid-flight).
  1080→      let pendingId: string | null = null
  1081→      try {
  1082→        const userId = await getOperatorUserId()
  1083→        if (userId) {
  1084→          const row = await db.pendingManageAction.create({
  1085→            data: {
  1086→              userId,
  1087→              action,
  1088→              attrs: JSON.stringify(attrs),
  1089→              status: 'executing',
  1090→            },
  1091→          })
  1092→          pendingId = row.id
  1093→        }
  1094→      } catch (e) {
  1095→        console.error('[orchestrator] failed to persist pending action:', e)
  1096→      }
  1097→
  1098→      const result = await executeManageAction(action, attrs)
  1099→      // Refresh the merged subagent list so subsequent dispatches see the new state.
  1100→      mergedSubagents = null
  1101→
  1102→      // Update the pending row with the result
  1103→      if (pendingId) {
  1104→        try {
  1105→          await db.pendingManageAction.update({
  1106→            where: { id: pendingId },
  1107→            data: {
  1108→              status: result.ok ? 'done' : 'failed',
  1109→              result: result.message,
  1110→            },
  1111→          })
  1112→        } catch {
  1113→          /* ignore */
  1114→        }
  1115→      }
  1116→
  1117→      await emit('manage_action', {
  1118→        stepId,
  1119→        action,
  1120→        attrs,
  1121→        result,
  1122→        stepNumber: iter,
  1123→        status: result.ok ? 'done' : 'error',
  1124→      })
  1125→
  1126→      // Persist for reload reconstruction
  1127→      try {
  1128→        await db.message.create({
  1129→          data: {
  1130→            conversationId,
  1131→            role: 'tool',
  1132→            content: `[manage:${action}] ${JSON.stringify(attrs).slice(0, 200)}`,
  1133→            toolName: 'manage_action',
  1134→            toolArgs: JSON.stringify({ action, attrs }),
  1135→            toolResult: result.message,
  1136→          },
  1137→        })
  1138→      } catch { /* ignore */ }
  1139→
  1140→      // If a subagent was created/edited/deleted/toggled, tell the client to refresh its UI.
  1141→      if (
  1142→        result.ok &&
  1143→        ['create_agent', 'edit_agent', 'delete_agent', 'toggle_agent'].includes(action)
  1144→      ) {
  1145→        await emit('subagents_updated', { action, attrs, result })
  1146→      }
  1147→
  1148→      // Feed back the result so the orchestrator can confirm to the user.
  1149→      conversationMessages.push({ role: 'assistant', content })
  1150→      conversationMessages.push({
  1151→        role: 'user',
  1152→        content: `[MANAGE_RESULT] ${action}: ${result.ok ? 'success' : 'failed'} — ${result.message}`,
  1153→      })
  1154→      continue
  1155→    }
  1156→

  1158→    if (parsed.dispatch) {
  1159→      // UPGRADE #68 + #86 — Multi-dispatch: detect ALL dispatch tags + execute in parallel
  1160→      // Handles: <dispatch agent="x" task="..."/> AND <dispatch_subagent id="x">task</dispatch_subagent>
  1161→      //          AND <dispatch_subagent id="x" task="..."/> AND <dispatch_subagent id="x"/>
  1162→      const allDispatchRe = /<dispatch(?:_subagent)?\s+(?:id|agent)=["']([^"']+)["']\s*(?:task=["']([\s\S]*?)["']\s*)?(?:\/>|>([\s\S]*?)<\/dispatch_subagent>)/gi
  1163→      const allDispatches: Array<{ agentId: string; task: string }> = []
  1164→      let dm: RegExpExecArray | null

  1166→        const agentId = dm[1].trim().toLowerCase()
  1167→        // task may come from attr (group 2) or body (group 3)
  1168→        const task = (dm[2] ?? dm[3] ?? '').trim() || 'execute task'
  1169→        allDispatches.push({ agentId, task })
  1170→      }
  1171→
  1172→      if (allDispatches.length > 1) {
  1173→        // Multi-dispatch: execute ALL in parallel via Promise.allSettled
  1174→        try { await emit('thought', { content: `[MULTI-DISPATCH] Detected ${allDispatches.length} dispatches. Executing in PARALLEL for max speed.` }) } catch {}
  1175→        const list = await getMerged()
  1176→        const { runSubagent } = await import('./subagents')
  1177→        const startTime = Date.now()
  1178→        const results = await Promise.allSettled(
  1179→          allDispatches.map(async (d) => {
  1180→            const sub = list.find((s) => s.id === d.agentId || s.name.toLowerCase() === d.agentId.toLowerCase())
  1181→            if (!sub || sub.enabled === false) return { id: d.agentId, name: d.agentId, error: 'not found or disabled' }
  1182→            dispatchCount++
  1183→            const dispatchId = `par_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`
  1184→            try { await emit('subagent_dispatch', { dispatchId, agentId: sub.id, agentName: sub.name, color: sub.color, icon: sub.icon, task: d.task, stepNumber: iter }) } catch {}
  1185→            const result = await runSubagent({
  1186→              subagentId: sub.id, task: d.task, dispatchId,
  1187→              attachments, language, emit,
  1188→              parentConversationId: conversationId,
  1189→            })
  1190→            return { id: sub.id, name: sub.name, answer: result.answer }
  1191→          })
  1192→        )
  1193→        const elapsedMs = Date.now() - startTime
  1194→        const succeeded = results.filter((r) => r.status === 'fulfilled').length
  1195→        // Feed ALL results back
  1196→        for (let i = 0; i < results.length; i++) {
  1197→          const r = results[i]
  1198→          const d = allDispatches[i]
  1199→          if (r.status === 'fulfilled' && r.value) {
  1200→            try { await emit('subagent_complete', { dispatchId: `par_${i}`, answer: (r.value.answer ?? '').slice(0, 2000) }) } catch {}
  1201→            conversationMessages.push({ role: 'assistant', content: `<dispatch_subagent id="${d.agentId}">${d.task}</dispatch_subagent>` })
  1202→            conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${r.value.name}: ${(r.value.answer ?? '').slice(0, 20000)}` })
  1203→          } else {
  1204→            conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${d.agentId}: ERROR — ${r.status === 'rejected' ? r.reason?.message : 'unknown'}` })
  1205→          }
  1206→        }
  1207→        try { await emit('heartbeat', { iteration: iter, maxIterations: MAX_ITERATIONS, toolsCalled: steps.length, dispatchesCalled: dispatchCount, lastToolName: 'multi_dispatch', lastThought: `${succeeded}/${results.length} subagents completed in ${elapsedMs}ms (parallel)`, startedAt: Date.now(), elapsedMs, message: `Multi-dispatch: ${succeeded}/${results.length} in ${elapsedMs}ms` }) } catch {}
  1208→        continue
  1209→      }
  1210→
  1211→      // Single dispatch (existing logic)
  1212→      const { agentId, task } = parsed.dispatch
  1213→      const list = await getMerged()
  1214→      // Match by id (case-sensitive) OR by name (case-insensitive) — this lets
  1215→      // the Super Agent dispatch to custom agents like "Cybersecurity A" using
  1216→      // their human-readable name (their db id is an unpredictable cuid).
  1217→      const sub = list.find(
  1218→        (s) => s.id === agentId || s.name.toLowerCase() === agentId.toLowerCase()
  1219→      )
  1220→      if (!sub) {
  1221→        // Unknown agent — feed back an error to the orchestrator.
  1222→        // Show BOTH name and id so the LLM can pick the right dispatch id next time.
  1223→        const available = list.map((s) => `${s.name} (dispatch_id: ${s.isBuiltin ? s.id : s.name})`).join(', ')
  1224→        const errMsg = `Unknown sub-agent: "${agentId}". Available: ${available}`
  1225→        await emit('error', { message: errMsg })
  1226→        conversationMessages.push({ role: 'assistant', content })
  1227→        conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${agentId}: ERROR — ${errMsg}` })
  1228→        // Persist the failed dispatch attempt
  1229→        try {
  1230→          await db.message.create({
  1231→            data: {
  1232→              conversationId,
  1233→              role: 'tool',
  1234→              content: `[dispatch:unknown] ${agentId} task="${task}"`,
  1235→              toolName: 'subagent_dispatch',
  1236→              toolArgs: JSON.stringify({ agentId, task, error: errMsg }),
  1237→              toolResult: errMsg,
  1238→            },
  1239→          })
  1240→        } catch { /* ignore */ }
  1241→        continue
  1242→      }
  1243→      if (sub.enabled === false) {
  1244→        const errMsg = `Sub-agent "${sub.name}" is currently disabled. Re-enable it via the Sub-Agents panel or the toggle_agent manage tag.`
  1245→        await emit('error', { message: errMsg })
  1246→        conversationMessages.push({ role: 'assistant', content })
  1247→        conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${agentId}: ERROR — ${errMsg}` })
  1248→        continue
  1249→      }
  1250→      if (dispatchCount >= MAX_DISPATCHES) {
  1251→        const capMsg = `Reached max sub-agent dispatches (${MAX_DISPATCHES}). Please synthesize the answer from what you have.`
  1252→        conversationMessages.push({ role: 'assistant', content })
  1253→        conversationMessages.push({ role: 'user', content: `[SYSTEM] ${capMsg}` })
  1254→        continue
  1255→      }

  1766→/**
  1767→ * Scan a sub-agent's answer for dollar amounts that look like earnings, and
  1768→ * log them as IncomeEntry rows with source = agentId. Fire-and-forget.
  1769→ *
  1770→ * We're deliberately conservative — only log amounts that appear near income
  1771→ * keywords (earned, income, revenue, MRR, /day, /mo, /week, /month, profit,
  1772→ * ROI, yield). This avoids logging "$0 cost" or "$1,000 capital" as income.
  1773→ */
  1774→function autoLogIncomeFromAnswer(agentId: string, answer: string): void {
  1775→  if (!answer || typeof answer !== 'string') return
  1776→  // Strip code blocks to avoid logging amounts from code samples
  1777→  const cleaned = answer.replace(/```[\s\S]*?```/g, ' ')
  1778→  // Find all $X or $X.Y or $X,YYY mentions
  1779→  const re = /\$([\d,]+(?:\.\d{1,2})?)\s*(?:\/(?:day|d|mo|month|m|week|wk|w|year|yr|y))?/gi
  1780→  const incomeKeywords = /(earned|income|revenue|mrr|arr|profit|yield|roi|royalt|paying|paid|generat)/i
  1781→  const periodKeywords = /\/(day|d|mo|month|m|week|wk|w|year|yr|y)\b/i
  1782→  let m: RegExpExecArray | null
  1783→  const candidates: Array<{ amount: number; line: string }> = []
  1784→  while ((m = re.exec(cleaned))) {
  1785→    const amountStr = m[1].replace(/,/g, '')
  1786→    const amount = parseFloat(amountStr)
  1787→    if (!isFinite(amount) || amount <= 0 || amount > 1_000_000) continue
  1788→    // Look at a window of text around this match for income keywords
  1789→    const start = Math.max(0, m.index - 80)
  1790→    const end = Math.min(cleaned.length, m.index + m[0].length + 80)
  1791→    const window = cleaned.slice(start, end)
  1792→    // If the amount has a period suffix (/day /mo etc.) OR nearby income keyword → log it
  1793→    if (periodKeywords.test(m[0]) || incomeKeywords.test(window)) {
  1794→      candidates.push({ amount, line: m[0] })
  1795→    }
  1796→  }
  1797→  if (!candidates.length) return
  1798→  // Cap to 3 per sub-agent answer to avoid spamming the table
  1799→  const toLog = candidates.slice(0, 3)
  1800→  // Fire-and-forget DB inserts
  1801→  ;(async () => {
  1802→    try {
  1803→      const { db } = await import('@/lib/db')
  1804→      const now = new Date()
  1805→      for (const c of toLog) {
  1806→        await db.incomeEntry.create({
  1807→          data: {
  1808→            amount: c.amount,
  1809→            source: agentId.charAt(0).toUpperCase() + agentId.slice(1),
  1810→            notes: `Auto-logged from ${agentId} sub-agent answer: "${c.line}"`,
  1811→            date: now,
  1812→          },
  1813→        })
  1814→      }
  1815→    } catch (e) {
  1816→      console.error('[orchestrator] autoLogIncomeFromAnswer failed:', e)
  1817→    }
  1818→  })()
  1819→}
