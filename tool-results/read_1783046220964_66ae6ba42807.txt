    23→/* Regex to find <dispatch agent="..." task="..."/> OR <dispatch agent="..." task="..."> tags.
    24→ * Accepts both self-closing (/> ) and non-self-closing (> ) forms because the LLM
    25→ * sometimes emits one or the other. Uses non-greedy [\s\S]*? for the task value. */
    26→const DISPATCH_RE = /<dispatch\s+agent=["']([^"']+)["']\s+task=["']([\s\S]*?)["']\s*\/?>/i

    28→/* Regex to find <manage action="..." attr="..." ... /> self-closing tags.

    30→const MANAGE_RE = /<manage\s+[^>]*?\/>/gi

    32→interface OrchestratorParsed {
    33→  thought?: string
    34→  tool?: { name: string; args: any }
    35→  dispatch?: { agentId: string; task: string }
    36→  manage?: { action: string; attrs: Record<string, string>; raw: string }
    37→  textAfter: string
    38→  raw: string
    39→}

    41→function parseOrchestrator(content: string): OrchestratorParsed {
    42→  const thoughtMatch = content.match(THOUGHT_RE)
    43→  const thought = thoughtMatch?.[1]?.trim()
    44→
    45→  const dispatchMatch = content.match(DISPATCH_RE)
    46→  const toolMatch = content.match(TOOL_RE)
    47→  const manageMatch = content.match(MANAGE_RE)
    48→
    49→  // Priority: dispatch > manage > tool (manage and dispatch are both "agent actions")
    50→  if (dispatchMatch) {
    51→    const agentId = dispatchMatch[1].trim().toLowerCase()
    52→    const task = dispatchMatch[2].trim()
    53→    return {
    54→      thought,
    55→      dispatch: { agentId, task },
    56→      textAfter: content.slice(content.indexOf(dispatchMatch[0]) + dispatchMatch[0].length).replace(THOUGHT_RE, '').trim(),
    57→      raw: content,
    58→    }
    59→  }
    60→  if (manageMatch && manageMatch.length > 0) {
    61→    const tag = manageMatch[0]
    62→    const attrs = parseManageAttrs(tag)
    63→    const action = (attrs.action ?? '').toString().trim().toLowerCase()
    64→    return {
    65→      thought,
    66→      manage: { action, attrs, raw: tag },
    67→      textAfter: content.replace(tag, '').replace(THOUGHT_RE, '').trim(),
    68→      raw: content,
    69→    }
    70→  }
    71→  if (toolMatch) {
    72→    const name = toolMatch[1].trim()
    73→    let args: any = {}
    74→    const raw = toolMatch[2].trim()
    75→    try {
    76→      args = JSON.parse(raw)
    77→    } catch {
    78→      const m: Record<string, string> = {}
    79→      const re = /"([^"]+)"\s*:\s*"([^"]*)"/g
    80→      let mm: RegExpExecArray | null
    81→      while ((mm = re.exec(raw))) m[mm[1]] = mm[2]
    82→      args = m
    83→    }
    84→    return { thought, tool: { name, args }, textAfter: '', raw: content }
    85→  }
    86→  return { thought, textAfter: content.replace(THOUGHT_RE, '').trim(), raw: content }
    87→}

    89→/** Parse attributes from a <manage .../> tag. Handles key="value" pairs with
    90→ * either single or double quotes. Also supports attribute values containing
    91→ * spaces because the regex is greedy on the quoted portion. */
    92→function parseManageAttrs(tag: string): Record<string, string> {
    93→  const attrs: Record<string, string> = {}
    94→  // Match: attrName="value" or attrName='value'
    95→  // We allow newlines inside the value via [\s\S]*? (non-greedy).
    96→  const re = /([a-zA-Z_][a-zA-Z0-9_]*)\s*=\s*(?:"([^"]*)"|'([^']*)')/g
    97→  let m: RegExpExecArray | null
    98→  while ((m = re.exec(tag))) {
    99→    const key = m[1]
   100→    const val = m[2] ?? m[3] ?? ''
   101→    attrs[key] = val
   102→  }
   103→  return attrs
   104→}

   197→/* ------------------------------------------------------------------ *
   198→ * Fast-path detection — when the user's message is a CLEAR, unambiguous
   199→ * create_agent request (short, single-intent, matches a strict regex),
   200→ * we skip the LLM round-trip entirely and execute the manage action
   201→ * directly. This makes "create a sub-agent named X" instant and immune
   202→ * to rate-limiting.
   203→ *
   204→ * Returns null when the pattern is not clear enough → fall through to
   205→ * normal LLM orchestration.
   206→ * ------------------------------------------------------------------ */
   207→
   208→interface FastPathCreateAgent {
   209→  action: 'create_agent'
   210→  attrs: Record<string, string>
   211→}
   212→
   213→/* Match: "create a new sub-agent named 'Cybersecurity A'"
   214→ *        "add an agent called TESTFAST via fast path"
   215→ *        "build a subagent named Foo"
   216→ * Captures the agent name in group 1 (quotes optional). */
   217→const FAST_CREATE_RE =
   218→  /(?:create|add|build)\s+(?:a\s+)?(?:new\s+)?(?:sub-?agent|agent)\s+(?:named|called)\s+["']?([A-Za-z0-9 _\-]+?)["']?(?:[\s.,]|$)/i
   219→
   220→/* Match: role="..." / role: '...' / role is "..." / role named "..." */
   221→function extractAttr(message: string, key: string): string | null {
   222→  // key="value" or key='value' or key=value-with-no-spaces
   223→  const kvRe = new RegExp(
   224→    `${key}\\s*(?:=|:)\\s*["']?([^"'\\n,]+?)["']?(?:[\\s,]|$)`,
   225→    'i'
   226→  )
   227→  const m1 = message.match(kvRe)
   228→  if (m1) return m1[1].trim()
   229→  // "role is X" / "role named X" / "specialized in X"
   230→  const phraseRe = new RegExp(
   231→    `${key}\\s+(?:is|named|specialized\\s+in|specialising\\s+in|specialty)\\s+["']?([A-Za-z0-9 _\\-/]+?)["']?(?:[\\s.,]|$)`,
   232→    'i'
   233→  )
   234→  const m2 = message.match(phraseRe)
   235→  if (m2) return m2[1].trim()
   236→  return null
   237→}

   239→function detectFastPathManage(userMessage: string): FastPathCreateAgent | null {
   240→  if (!userMessage || userMessage.length > 500) return null
   241→  // Require an explicit "fast path" hint OR a very clear single-intent command.
   242→  // The "fast path" hint lets users opt in; we also accept very-short clear
   243→  // commands without the hint.
   244→  const hasFastHint = /\bfast[\s-]?path\b/i.test(userMessage)
   245→  const m = userMessage.match(FAST_CREATE_RE)
   246→  if (!m) return null
   247→  const name = m[1].trim()
   248→  if (!name || name.length < 2 || name.length > 80) return null
   249→
   250→  // If the message contains words suggesting other actions (dispatch, delete,
   251→  // log, schedule, set goal, etc.), DON'T fast-path — let the LLM handle it.
   252→  const otherActions =
   253→    /\b(?:dispatch|delete|remove|toggle|disable|enable|log\s+\$|log\s+income|set\s+(?:my\s+)?(?:income|growth|daily)|create\s+schedule|update\s+settings)\b/i.test(
   254→      userMessage
   255→    )
   256→  if (otherActions) return null
   257→
   258→  // Without the fast-path hint, require very short messages to avoid
   259→  // over-eager matching on long descriptive requests.
   260→  if (!hasFastHint && userMessage.length > 200) return null
   261→
   262→  const attrs: Record<string, string> = { name }
   263→  const role = extractAttr(userMessage, 'role')
   264→  if (role) attrs.role = role
   265→  const specialty = extractAttr(userMessage, 'specialty')
   266→  if (specialty) attrs.specialty = specialty
   267→  const color = extractAttr(userMessage, 'color')
   268→  if (color) attrs.color = color
   269→  const icon = extractAttr(userMessage, 'icon')
   270→  if (icon) attrs.icon = icon
   271→  const systemPrompt = extractAttr(userMessage, 'system_prompt')
   272→  if (systemPrompt) attrs.system_prompt = systemPrompt
   273→  const allowedTools = extractAttr(userMessage, 'allowed_tools')
   274→  if (allowedTools) attrs.allowed_tools = allowedTools
   275→
   276→  // If user provided a "specialized in X" phrase but no role, derive a role
   277→  if (!attrs.role) {
   278→    const specMatch = userMessage.match(
   279→      /specialized\s+in\s+([A-Za-z0-9 _\-/]+?)(?:[\s.,]|$)/i
   280→    )
   281→    if (specMatch) {
   282→      attrs.specialty = specMatch[1].trim()
   283→      attrs.role = specMatch[1].trim() + ' Specialist'
   284→    }
   285→  }
   286→
   287→  // Fast-path defaults so the action can succeed without forcing the user to
   288→  // specify everything. The user can always edit afterwards via the panel.
   289→  if (!attrs.role) {
   290→    attrs.role = `${name} Specialist`
   291→  }
   292→  if (!attrs.specialty) {
   293→    attrs.specialty = `Custom specialist created via fast-path`
   294→  }
   295→  if (!attrs.allowed_tools) {
   296→    // Sensible default: read-only research tools
   297→    attrs.allowed_tools = 'web_search,page_reader,wikipedia_search,wikipedia_read,free_apis_directory,memory_store,memory_recall'
   298→  }
   299→  if (!attrs.system_prompt) {
   300→    attrs.system_prompt = `You are ${name.toUpperCase()}, a custom specialist sub-agent of Agent007 AI.\n\nYour role: ${attrs.role}.\nYour specialty: ${attrs.specialty}.\n\nALLOWED TOOLS:\n- web_search — Google-style search for current info\n- page_reader — read full web pages\n- memory_store / memory_recall — persist + recall context\n- wikipedia_search / wikipedia_read — encyclopedic background\n- free_apis_directory — find public data APIs\n\nOUTPUT FORMAT:\n- <thought>brief reasoning</thought> before each action\n- <tool name="...">{json}</tool> to call a tool\n- Plain markdown final answer\n\nRULES:\n- Be concise and structured.\n- Cite sources for any factual claim.\n- Max 6 tool calls per turn.`
   301→  }
   302→
   303→  return { action: 'create_agent', attrs }
   304→}

   307→async function runFastPathManage(opts: {
   308→  conversationId: string
   309→  userMessage: string
   310→  language: 'en' | 'zh'
   311→  emit: OrchestratorEventEmit
   312→  fastPath: FastPathCreateAgent
   313→}): Promise<OrchestratorRunResult> {
   314→  const { conversationId, userMessage, emit, fastPath } = opts
   315→  const { action, attrs } = fastPath
   316→
   317→  // Persist the user message + an empty assistant row up-front so the timeline
   318→  // and DB stay consistent.
   319→  const assistantRow = await db.message.create({
   320→    data: { conversationId, role: 'assistant', content: '' },
   321→  })
   322→
   323→  await emit('thought', {
   324→    content:
   325→      '⚡ Fast-path: detected clear create_agent request, executing without LLM round-trip',
   326→  })
   327→
   328→  const stepId = makeId('manage')
   329→  await emit('manage_action', {
   330→    stepId,
   331→    action,
   332→    attrs,
   333→    thought: 'Fast-path create_agent (no LLM round-trip)',
   334→    stepNumber: 1,
   335→    status: 'running',
   336→    fastPath: true,
   337→  })
   338→
   339→  // Persist a PendingManageAction row before executing
   340→  let pendingId: string | null = null
   341→  try {
   342→    const userId = await getOperatorUserId()
   343→    if (userId) {
   344→      const row = await db.pendingManageAction.create({
   345→        data: {
   346→          userId,
   347→          action,
   348→          attrs: JSON.stringify(attrs),
   349→          status: 'executing',
   350→        },
   351→      })
   352→      pendingId = row.id
   353→    }
   354→  } catch (e) {
   355→    console.error('[orchestrator:fast-path] failed to persist pending action:', e)
   356→  }
   357→
   358→  const result = await executeManageAction(action, attrs)
   359→
   360→  // Update the pending row with the result
   361→  if (pendingId) {
   362→    try {
   363→      await db.pendingManageAction.update({
   364→        where: { id: pendingId },
   365→        data: {
   366→          status: result.ok ? 'done' : 'failed',
   367→          result: result.message,
   368→        },
   369→      })
   370→    } catch {
   371→      /* ignore */
   372→    }
   373→  }
   374→
   375→  await emit('manage_action', {
   376→    stepId,
   377→    action,
   378→    attrs,
   379→    result,
   380→    stepNumber: 1,
   381→    status: result.ok ? 'done' : 'error',
   382→    fastPath: true,
   383→  })
   384→
   385→  // Persist a tool/thought trace for reload reconstruction
   386→  try {
   387→    await db.message.create({
   388→      data: {
   389→        conversationId,
   390→        role: 'tool',
   391→        content: `[manage:${action}] ${JSON.stringify(attrs).slice(0, 200)}`,
   392→        toolName: 'manage_action',
   393→        toolArgs: JSON.stringify({ action, attrs, fastPath: true }),
   394→        toolResult: result.message,
   395→      },
   396→    })
   397→  } catch {
   398→    /* ignore */
   399→  }
   400→
   401→  if (
   402→    result.ok &&
   403→    ['create_agent', 'edit_agent', 'delete_agent', 'toggle_agent'].includes(action)
   404→  ) {
   405→    await emit('subagents_updated', { action, attrs, result, fastPath: true })
   406→  }
   407→
   408→  // Build a confirmation message and stream it as tokens
   409→  const agentName = attrs.name ?? 'agent'
   410→  const roleLine = attrs.role ? ` (${attrs.role})` : ''
   411→  let finalAnswer: string
   412→  if (result.ok) {
   413→    finalAnswer = `✅ Created sub-agent "${agentName}"${roleLine} via fast-path. Use the Sub-Agents panel to verify or edit it.`
   414→  } else {
   415→    finalAnswer = `⚠️ Fast-path create_agent for "${agentName}" failed: ${result.message}`
   416→  }
   417→
   418→  const chunks = chunkText(finalAnswer, 80)
   419→  for (const c of chunks) {
   420→    await emit('token', { content: c })
   421→  }
   422→
   423→  // Update the assistant row with the final answer
   424→  try {
   425→    await db.message.update({
   426→      where: { id: assistantRow.id },
   427→      data: { content: finalAnswer },
   428→    })
   429→  } catch {
   430→    /* ignore */
   431→  }
   432→
   433→  // Update conversation title
   434→  try {
   435→    const conv = await db.conversation.findUnique({ where: { id: conversationId } })
   436→    if (conv && (conv.title === 'New Conversation' || !conv.title)) {
   437→      const title = userMessage.slice(0, 50).trim() || 'New Conversation'
   438→      await db.conversation.update({ where: { id: conversationId }, data: { title } })
   439→    }
   440→  } catch {
   441→    /* ignore */
   442→  }
   443→
   444→  return {
   445→    finalAnswer,
   446→    steps: [],
   447→    persistedAssistantMessageId: assistantRow.id,
   448→  }
   449→}

   451→export async function runOrchestrator(opts: OrchestratorRunOptions): Promise<OrchestratorRunResult> {
   452→  const { conversationId, userMessage, attachments, language, emit } = opts
   453→
   454→  // 0a) Replay any pending manage actions left over from prior failed runs.
   455→  //     We surface them as `manage_action` events (status=done|error) so the
   456→  //     UI timeline shows what was (re)executed.
   457→  try {
   458→    const userId = await getOperatorUserId()
   459→    if (userId) {
   460→      const pending = await db.pendingManageAction.findMany({
   461→        where: { userId, status: { in: ['pending', 'executing'] } },
   462→        orderBy: { createdAt: 'asc' },
   463→        take: 10,
   464→      })
   465→      for (const p of pending) {
   466→        try {
   467→          const attrs = JSON.parse(p.attrs) as Record<string, string>
   468→          await emit('manage_action', {
   469→            stepId: `replay_${p.id}`,
   470→            action: p.action,
   471→            attrs,
   472→            thought: 'Replaying pending action from a prior interrupted run',
   473→            stepNumber: 0,
   474→            status: 'running',
   475→            replay: true,
   476→          })
   477→          const result = await executeManageAction(p.action, attrs)
   478→          await db.pendingManageAction.update({
   479→            where: { id: p.id },
   480→            data: {
   481→              status: result.ok ? 'done' : 'failed',
   482→              result: result.message,
   483→            },
   484→          })
   485→          await emit('manage_action', {
   486→            stepId: `replay_${p.id}`,
   487→            action: p.action,
   488→            attrs,
   489→            result,
   490→            stepNumber: 0,
   491→            status: result.ok ? 'done' : 'error',
   492→            replay: true,
   493→          })
   494→        } catch (replayErr: any) {
   495→          await db.pendingManageAction.update({
   496→            where: { id: p.id },
   497→            data: { status: 'failed', result: replayErr?.message ?? 'replay error' },
   498→          })
   499→        }
   500→      }
   501→    }
   502→  } catch (replayOuterErr) {
   503→    // Non-fatal — keep going with the new user message
   504→    console.error('[orchestrator] pending replay failed:', replayOuterErr)
   505→  }
   506→
   507→  // 0b) Fast-path: detect a clear, unambiguous create_agent request and
   508→  //     execute it directly without an LLM round-trip. Saves time, tokens,
   509→  //     and sidesteps any rate-limit hit entirely.
   510→  const fastPath = detectFastPathManage(userMessage)
   511→  if (fastPath) {
   512→    return runFastPathManage({
   513→      conversationId,
   514→      userMessage,
   515→      language,
   516→      emit,
   517→      fastPath,
   518→    })
   519→  }
   520→
   521→  // Recall memories for context
   522→  const recalled = await recallMemories(userMessage.slice(0, 200), 8)
   523→  const memoryBlock = formatMemoryForPrompt(recalled)
   524→
   525→  const languageInstruction =
   526→    language === 'zh'
   527→      ? 'LANGUAGE INSTRUCTION: The user has toggled the agent to Chinese. Reply in 中文 (Chinese) for your FINAL answer regardless of input language.'
   528→      : 'LANGUAGE INSTRUCTION: The user has toggled the agent to English. Reply in English for your FINAL answer unless the user wrote in another language.'
   529→
   530→  // Build a DYNAMIC sub-agent list so the LLM knows about custom agents
   531→  // (Cybersecurity A, TRADER, etc.) — not just the 12 built-ins hard-coded
   532→  // in BASE_SYSTEM_PROMPT. This fixes the bug where the Super Agent would
   533→  // tell the user "that agent doesn't exist" when they addressed a custom
   534→  // agent by name.
   535→  const allAgents = await getAllSubagents({ includeDisabled: false })
   536→  // IMPORTANT: For built-in agents, the dispatch id is their lowercase id (aurora, vertex, etc.).
   537→  // For custom agents, the dispatch id is their NAME (e.g., "Cybersecurity A") — because their
   538→  // db id is a cuid that the LLM can't predict. The runSubagent() lookup matches by id OR name.
   539→  const agentListDynamic = allAgents
   540→    .map((a) => {
   541→      const dispatchId = a.isBuiltin ? a.id : a.name
   542→      return `- ${dispatchId} (${a.name} — ${a.role})`
   543→    })
   544→    .join('\n')
   545→  const dynamicAgentSection = `
   546→
   547→CURRENTLY AVAILABLE SUB-AGENTS (built-in + custom, fetched live from DB):
   548→${agentListDynamic}
   549→
   550→IMPORTANT: When the user addresses a sub-agent by name (e.g. "Cybersecurity A, search..."),
   551→you MUST dispatch that exact agent via <dispatch agent="agent_id" task="..."/> using the
   552→dispatch id shown at the start of each line above. For built-in agents the dispatch id is
   553→their lowercase name (aurora, vertex, etc.). For custom agents the dispatch id is their
   554→display name (e.g., "Cybersecurity A", "TRADER"). Do NOT do the work yourself with direct
   555→web_search calls. Do NOT tell the user the agent doesn't exist if it appears in the list
   556→above. The list above is authoritative.`
   557→
   558→  const systemPrompt = `${BASE_SYSTEM_PROMPT}
   559→
   560→${ORCHESTRATOR_PROMPT_ADDENDUM}
   561→
   562→${dynamicAgentSection}
   563→
   564→${languageInstruction}
   565→
   566→RECALLED MEMORIES (use as context, do not blindly trust if outdated):
   567→${memoryBlock}
   568→
   569→CURRENT UTC TIME: ${new Date().toUTCString()}`
   570→
   571→  const history = await buildHistoryMessages(conversationId, userMessage, attachments)
   572→  const ctx: ToolContext = { attachments, language }
   573→
   574→  const steps: OrchestratorRunResult['steps'] = []
   575→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
   576→    { role: 'system', content: systemPrompt },
   577→    ...history,
   578→  ]
   579→
   580→  let finalAnswer = ''
   581→  let iter = 0
   582→  let dispatchCount = 0
   583→  let manageCount = 0
   584→  // Cache the merged subagent list once per orchestrator run (12 built-in + custom + overlays).
   585→  let mergedSubagents: Subagent[] | null = null
   586→  const getMerged = async (): Promise<Subagent[]> => {
   587→    if (!mergedSubagents) mergedSubagents = await getAllSubagents({ includeDisabled: true })
   588→    return mergedSubagents
   589→  }
   590→
   591→  while (iter < MAX_ITERATIONS) {
   592→    iter++
   593→    let completion: any
   594→    try {
   595→      completion = await callLlmWithRetry(conversationMessages)
   596→    } catch (e: any) {
   597→      const friendly = friendlyLlmError(e)
   598→      await emit('error', { message: friendly })
   599→      finalAnswer = friendly
   600→      break
   601→    }
   602→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
   603→    if (!content.trim()) {
   604→      finalAnswer = '(The agent produced no output. Please try rephrasing.)'
   605→      break
   606→    }
   607→
   608→    const parsed = parseOrchestrator(content)
   609→
   610→    // Emit thought
   611→    if (parsed.thought) {
   612→      await emit('thought', { content: parsed.thought })
   613→    }
   614→
   615→    // 0) Manage path — parse <manage .../> tags and execute server-side.
   616→    if (parsed.manage) {
   617→      const { action, attrs } = parsed.manage
   618→      if (manageCount >= MAX_MANAGE_ACTIONS) {
   619→        const capMsg = `Reached max manage actions (${MAX_MANAGE_ACTIONS}). Please synthesize the answer from what you have.`
   620→        conversationMessages.push({ role: 'assistant', content })
   621→        conversationMessages.push({ role: 'user', content: `[SYSTEM] ${capMsg}` })
   622→        continue
   623→      }
   624→      manageCount++
   625→
   626→      const stepId = makeId('manage')
   627→      await emit('manage_action', {
   628→        stepId,
   629→        action,
   630→        attrs,
   631→        thought: parsed.thought,
   632→        stepNumber: iter,
   633→        status: 'running',
   634→      })
   635→
   636→      // Persist a PendingManageAction row before executing (so we can replay
   637→      // if this run crashes mid-flight).
   638→      let pendingId: string | null = null
   639→      try {
   640→        const userId = await getOperatorUserId()
   641→        if (userId) {
   642→          const row = await db.pendingManageAction.create({
   643→            data: {
   644→              userId,
   645→              action,
   646→              attrs: JSON.stringify(attrs),
   647→              status: 'executing',
   648→            },
   649→          })
   650→          pendingId = row.id
   651→        }
   652→      } catch (e) {
   653→        console.error('[orchestrator] failed to persist pending action:', e)
   654→      }
   655→
   656→      const result = await executeManageAction(action, attrs)
   657→      // Refresh the merged subagent list so subsequent dispatches see the new state.
   658→      mergedSubagents = null
   659→
   660→      // Update the pending row with the result
   661→      if (pendingId) {
   662→        try {
   663→          await db.pendingManageAction.update({
   664→            where: { id: pendingId },
   665→            data: {
   666→              status: result.ok ? 'done' : 'failed',
   667→              result: result.message,
   668→            },
   669→          })
   670→        } catch {
   671→          /* ignore */
   672→        }
   673→      }
   674→
   675→      await emit('manage_action', {
   676→        stepId,
   677→        action,
   678→        attrs,
   679→        result,
   680→        stepNumber: iter,
   681→        status: result.ok ? 'done' : 'error',
   682→      })
   683→
   684→      // Persist for reload reconstruction
   685→      try {
   686→        await db.message.create({
   687→          data: {
   688→            conversationId,
   689→            role: 'tool',
   690→            content: `[manage:${action}] ${JSON.stringify(attrs).slice(0, 200)}`,
   691→            toolName: 'manage_action',
   692→            toolArgs: JSON.stringify({ action, attrs }),
   693→            toolResult: result.message,
   694→          },
   695→        })
   696→      } catch { /* ignore */ }
   697→
   698→      // If a subagent was created/edited/deleted/toggled, tell the client to refresh its UI.
   699→      if (
   700→        result.ok &&
   701→        ['create_agent', 'edit_agent', 'delete_agent', 'toggle_agent'].includes(action)
   702→      ) {
   703→        await emit('subagents_updated', { action, attrs, result })
   704→      }
   705→
   706→      // Feed back the result so the orchestrator can confirm to the user.
   707→      conversationMessages.push({ role: 'assistant', content })
   708→      conversationMessages.push({
   709→        role: 'user',
   710→        content: `[MANAGE_RESULT] ${action}: ${result.ok ? 'success' : 'failed'} — ${result.message}`,
   711→      })
   712→      continue
   713→    }
   714→
   715→    // 1) Dispatch path
   716→    if (parsed.dispatch) {
   717→      const { agentId, task } = parsed.dispatch
   718→      const list = await getMerged()
   719→      // Match by id (case-sensitive) OR by name (case-insensitive) — this lets
   720→      // the Super Agent dispatch to custom agents like "Cybersecurity A" using
   721→      // their human-readable name (their db id is an unpredictable cuid).
   722→      const sub = list.find(
   723→        (s) => s.id === agentId || s.name.toLowerCase() === agentId.toLowerCase()
   724→      )
   725→      if (!sub) {
   726→        // Unknown agent — feed back an error to the orchestrator.
   727→        // Show BOTH name and id so the LLM can pick the right dispatch id next time.
   728→        const available = list.map((s) => `${s.name} (dispatch_id: ${s.isBuiltin ? s.id : s.name})`).join(', ')
   729→        const errMsg = `Unknown sub-agent: "${agentId}". Available: ${available}`
   730→        await emit('error', { message: errMsg })
   731→        conversationMessages.push({ role: 'assistant', content })
   732→        conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${agentId}: ERROR — ${errMsg}` })
   733→        // Persist the failed dispatch attempt
   734→        try {
   735→          await db.message.create({
   736→            data: {
   737→              conversationId,
   738→              role: 'tool',
   739→              content: `[dispatch:unknown] ${agentId} task="${task}"`,
   740→              toolName: 'subagent_dispatch',
   741→              toolArgs: JSON.stringify({ agentId, task, error: errMsg }),
   742→              toolResult: errMsg,
   743→            },
   744→          })
   745→        } catch { /* ignore */ }
   746→        continue
   747→      }
   748→      if (sub.enabled === false) {
   749→        const errMsg = `Sub-agent "${sub.name}" is currently disabled. Re-enable it via the Sub-Agents panel or the toggle_agent manage tag.`
   750→        await emit('error', { message: errMsg })
   751→        conversationMessages.push({ role: 'assistant', content })
   752→        conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${agentId}: ERROR — ${errMsg}` })
   753→        continue
   754→      }
   755→      if (dispatchCount >= MAX_DISPATCHES) {
   756→        const capMsg = `Reached max sub-agent dispatches (${MAX_DISPATCHES}). Please synthesize the answer from what you have.`
   757→        conversationMessages.push({ role: 'assistant', content })
   758→        conversationMessages.push({ role: 'user', content: `[SYSTEM] ${capMsg}` })
   759→        continue
   760→      }
   761→      dispatchCount++
   762→
   763→      const dispatchId = makeId('disp')
   764→      const dispatchStepNumber = iter
   765→
   766→      // Emit dispatch event (UI shows the sub-agent header card)
   767→      await emit('subagent_dispatch', {
   768→        dispatchId,
   769→        agentId: sub.id,
   770→        agentName: sub.name,
   771→        color: sub.color,
   772→        icon: sub.icon,
   773→        task,
   774→        stepNumber: dispatchStepNumber,
   775→      })
   776→
   777→      // Persist the dispatch row for reload reconstruction
   778→      try {
   779→        await db.message.create({
   780→          data: {
   781→            conversationId,
   782→            role: 'tool',
   783→            content: `[dispatch] ${sub.id} task="${task.slice(0, 200)}"`,
   784→            toolName: 'subagent_dispatch',
   785→            toolArgs: JSON.stringify({ dispatchId, agentId: sub.id, agentName: sub.name, color: sub.color, icon: sub.icon, task }),
   786→            toolResult: null,
   787→          },
   788→        })
   789→      } catch { /* ignore */ }
   790→
   791→      // Run the sub-agent (it will emit its own subagent_thought/tool_call/tool_result events)
   792→      let subAnswer = ''
   793→      try {
   794→        const result = await runSubagent({
   795→          subagentId: sub.id,
   796→          task,
   797→          attachments,
   798→          language,
   799→          parentConversationId: conversationId,
   800→          dispatchId,
   801→          emit: async (ev, data) => {
   802→            await emit(ev, data)
   803→            if (ev === 'subagent_complete' && data?.answer) {
   804→              subAnswer = data.answer
   805→            }
   806→          },
   807→        })
   808→        subAnswer = result.answer
   809→      } catch (e: any) {
   810→        subAnswer = friendlyLlmError(e)
   811→        await emit('subagent_complete', { dispatchId, answer: subAnswer })
   812→      }
   813→
   814→      // Feed the sub-agent's result back to the orchestrator
   815→      conversationMessages.push({ role: 'assistant', content })
   816→      conversationMessages.push({
   817→        role: 'user',
   818→        content: `[SUBAGENT_RESULT] ${sub.id}: ${subAnswer}`,
   819→      })
   820→
   821→      // BEST-EFFORT auto-logging: if the sub-agent's answer mentions dollar
   822→      // amounts (e.g. "$12.50", "$1,200/mo", "$45/day"), log them as income
   823→      // entries with source = sub-agent id. Fire-and-forget — never blocks the
   824→      // orchestrator. We only consider amounts that look like earnings (positive
   825→      // dollar values, optionally followed by /day /mo /week /month).
   826→      try {
   827→        autoLogIncomeFromAnswer(sub.id, subAnswer)
   828→      } catch {
   829→        /* ignore */
   830→      }
   831→
   832→      // If a memory_store happened inside the sub-agent, the sub-agent already emitted it
   833→      // (we don't double-emit memory_update here)
   834→      continue
   835→    }
   836→
   837→    // 2) Direct tool path (same as the original agent loop)
   838→    if (parsed.tool) {
   839→      const step: any = {
   840→        id: makeId('step'),
   841→        thought: parsed.thought,
   842→        toolName: parsed.tool.name,
   843→        toolArgs: parsed.tool.args,
   844→        startedAt: Date.now(),
   845→      }
   846→      steps.push(step)
   847→      await emit('tool_call', {
   848→        stepId: step.id,
   849→        name: step.toolName,
   850→        args: step.toolArgs,
   851→        thought: step.thought,
   852→        stepNumber: iter,
   853→      })
   854→
   855→      const toolResult = await dispatchTool(step.toolName!, step.toolArgs, ctx)
   856→      step.toolResult = toolResult
   857→      step.finishedAt = Date.now()
   858→      await emit('tool_result', {
   859→        stepId: step.id,
   860→        result: toolResult.result,
   861→        preview: toolResult.preview,
   862→        ok: toolResult.ok,
   863→        artifacts: toolResult.artifacts,
   864→      })
   865→
   866→      if (step.toolName === 'memory_store' && toolResult.ok) {
   867→        await emit('memory_update', {
   868→          key: step.toolArgs?.key,
   869→          value: step.toolArgs?.value,
   870→          category: step.toolArgs?.category ?? 'general',
   871→        })
   872→      }
   873→
   874→      conversationMessages.push({ role: 'assistant', content })
   875→      conversationMessages.push({
   876→        role: 'user',
   877→        content: `[TOOL_RESULT] ${step.toolName}: ${toolResult.result}`,
   878→      })
   879→
   880→      // Persist intermediate tool/thought rows for reload reconstruction
   881→      try {
   882→        if (step.thought) {
   883→          await db.message.create({
   884→            data: { conversationId, role: 'thought', content: step.thought },
   885→          })
   886→        }
   887→        await db.message.create({
   888→          data: {
   889→            conversationId,
   890→            role: 'tool',
   891→            content: `[tool call] ${step.toolName} ${JSON.stringify(step.toolArgs)}`,
   892→            toolName: step.toolName,
   893→            toolArgs: JSON.stringify(step.toolArgs),
   894→            toolResult: toolResult.result,
   895→          },
   896→        })
   897→      } catch { /* ignore */ }
   898→      continue
   899→    }
   900→
   901→    // 3) Final answer path — but FIRST check for "stuck" condition (FIX 2)
   902→    // If the agent produced ONLY a thought (no tool/dispatch/manage) and the
   903→    // thought contains "wait"-like language, it's stuck waiting for input
   904→    // that will never come. Auto-recover by prompting it to continue.
   905→    const isThoughtOnly = !parsed.tool && !parsed.dispatch && !parsed.manage && !!parsed.thought
   906→    const stuckPatterns = /(wait|waiting|haven't provided|yet to|will wait|need to wait|i'll wait|let me wait|as i wait)/i
   907→    const isStuck = isThoughtOnly && parsed.thought && stuckPatterns.test(parsed.thought)
   908→
   909→    if (isStuck && iter < MAX_ITERATIONS - 1) {
   910→      // Auto-recovery: feed back a "continue" prompt + re-enter the loop
   911→      await emit('thought', { content: `[AUTO-RECOVERY] Detected stuck condition. Auto-continuing...` })
   912→      conversationMessages.push({ role: 'assistant', content })
   913→      conversationMessages.push({
   914→        role: 'user',
   915→        content: '[SYSTEM] You appear to be waiting. Do NOT wait — continue executing the task now. Dispatch the next sub-agent or use a tool or give your final answer.',
   916→      })
   917→      continue
   918→    }
   919→
   920→    // If it's thought-only but NOT stuck, check if the text after thought is meaningful
   921→    const textAfterThought = content.replace(THOUGHT_RE, '').trim()
   922→    if (isThoughtOnly && textAfterThought.length < 20 && iter < MAX_ITERATIONS - 1) {
   923→      // The agent produced only a thought with no substantial answer — likely stuck
   924→      await emit('thought', { content: `[AUTO-RECOVERY] Thought-only response with no answer. Prompting to continue...` })
   925→      conversationMessages.push({ role: 'assistant', content })
   926→      conversationMessages.push({
   927→        role: 'user',
   928→        content: '[SYSTEM] You produced only a thought with no action or answer. Please either: (1) dispatch a sub-agent, (2) call a tool, or (3) give your final answer now.',
   929→      })
   930→      continue
   931→    }
   932→
   933→    // 3a) Final answer path — emit synthesis signal then stream tokens
   934→    finalAnswer = content.replace(THOUGHT_RE, '').replace(DISPATCH_RE, '').trim() || content.trim()
   935→
   936→    // Emit a synthesis indicator so the UI shows "Synthesizing…" briefly
   937→    await emit('synthesis', { content: finalAnswer.slice(0, 80) })
   938→
   939→    const chunks = chunkText(finalAnswer, 80)
   940→    for (const c of chunks) {
   941→      await emit('token', { content: c })
   942→    }
   943→    break
   944→  }
   945→
   946→  if (!finalAnswer) {
   947→    // FIX 1: Auto-synthesize from what we have instead of just saying "reached limit".
   948→    // Gather all tool results collected so far, then produce a useful summary.
   949→    const collectedResults: string[] = []
   950→    for (const s of steps) {
   951→      if (s.toolName && s.toolResult?.result) {
   952→        const preview = s.toolResult.result.slice(0, 300)
   953→        collectedResults.push(`- ${s.toolName}: ${preview}`)
   954→      }
   955→    }
   956→    if (collectedResults.length > 0) {
   957→      finalAnswer = `I've processed ${collectedResults.length} step(s) this turn. Here's what I have so far:\n\n${collectedResults.join('\n\n')}\n\n---\n*To continue, type "continue" and I'll pick up where I left off.*`
   958→    } else {
   959→      finalAnswer =
   960→        "I've reached my iteration limit for this turn without completing the task. Please type 'continue' and I'll retry."
   961→    }
   962→    await emit('token', { content: finalAnswer })
   963→  }
   964→
   965→  // Persist the final assistant message
   966→  const assistantRow = await db.message.create({
   967→    data: { conversationId, role: 'assistant', content: finalAnswer },
   968→  })
   969→
   970→  // Update conversation title if it's still default
   971→  const conv = await db.conversation.findUnique({ where: { id: conversationId } })
   972→  if (conv && (conv.title === 'New Conversation' || !conv.title)) {
   973→    const title = userMessage.slice(0, 50).trim() || 'New Conversation'
   974→    await db.conversation.update({ where: { id: conversationId }, data: { title } })
   975→  }
   976→
   977→  // Notification hook: if mission_complete notifications are enabled, send
   978→  // (or log) an email to the operator with the conversation title + preview.
   979→  try {
   980→    const { getNotificationSettings, recentlyNotified } = await import('@/lib/settings')
   981→    const notif = await getNotificationSettings()
   982→    const looksLikeError = /^⚠️|error|failed|crashed/i.test(finalAnswer.slice(0, 50))
   983→    const eventType = looksLikeError ? 'mission_failed' : 'mission_complete'
   984→    if (notif.enabled && notif.events[eventType as keyof typeof notif.events]) {
   985→      if (!(await recentlyNotified(eventType, notif.minDelayMinutes))) {
   986→        const { sendEmail } = await import('@/lib/email')
   987→        const { getOperatorUserId } = await import('@/lib/settings')
   988→        const userId = await getOperatorUserId()
   989→        const convTitle = conv?.title ?? 'Mission'
   990→        const preview = finalAnswer.slice(0, 500)
   991→        sendEmail({
   992→          to: notif.email,
   993→          subject: looksLikeError
   994→            ? `Mission Failed: ${convTitle}`
   995→            : `Mission Complete: ${convTitle}`,
   996→          body: looksLikeError
   997→            ? `Agent007 encountered an issue while running a mission.\n\nConversation: ${convTitle}\n\nPreview:\n${preview}\n\nOpen the dashboard at / to investigate.`
   998→            : `Agent007 has completed a mission.\n\nConversation: ${convTitle}\n\nResult preview:\n${preview}\n\nOpen the dashboard at / to view the full report.`,
   999→          userId: userId ?? undefined,
  1000→          type: eventType,
  1001→        }).catch(() => {/* ignore */})
  1002→      }
  1003→    }
  1004→  } catch {
  1005→    /* ignore notification errors */
  1006→  }
  1007→
  1008→  return {

  1019→/**
  1020→ * Scan a sub-agent's answer for dollar amounts that look like earnings, and
  1021→ * log them as IncomeEntry rows with source = agentId. Fire-and-forget.
  1022→ *
  1023→ * We're deliberately conservative — only log amounts that appear near income
  1024→ * keywords (earned, income, revenue, MRR, /day, /mo, /week, /month, profit,
  1025→ * ROI, yield). This avoids logging "$0 cost" or "$1,000 capital" as income.
  1026→ */
  1027→function autoLogIncomeFromAnswer(agentId: string, answer: string): void {
  1028→  if (!answer || typeof answer !== 'string') return
  1029→  // Strip code blocks to avoid logging amounts from code samples
  1030→  const cleaned = answer.replace(/```[\s\S]*?```/g, ' ')
  1031→  // Find all $X or $X.Y or $X,YYY mentions
  1032→  const re = /\$([\d,]+(?:\.\d{1,2})?)\s*(?:\/(?:day|d|mo|month|m|week|wk|w|year|yr|y))?/gi
  1033→  const incomeKeywords = /(earned|income|revenue|mrr|arr|profit|yield|roi|royalt|paying|paid|generat)/i
  1034→  const periodKeywords = /\/(day|d|mo|month|m|week|wk|w|year|yr|y)\b/i
  1035→  let m: RegExpExecArray | null
  1036→  const candidates: Array<{ amount: number; line: string }> = []
  1037→  while ((m = re.exec(cleaned))) {
  1038→    const amountStr = m[1].replace(/,/g, '')
  1039→    const amount = parseFloat(amountStr)
  1040→    if (!isFinite(amount) || amount <= 0 || amount > 1_000_000) continue
  1041→    // Look at a window of text around this match for income keywords
  1042→    const start = Math.max(0, m.index - 80)
  1043→    const end = Math.min(cleaned.length, m.index + m[0].length + 80)
  1044→    const window = cleaned.slice(start, end)
  1045→    // If the amount has a period suffix (/day /mo etc.) OR nearby income keyword → log it
  1046→    if (periodKeywords.test(m[0]) || incomeKeywords.test(window)) {
  1047→      candidates.push({ amount, line: m[0] })
  1048→    }
  1049→  }
  1050→  if (!candidates.length) return
  1051→  // Cap to 3 per sub-agent answer to avoid spamming the table
  1052→  const toLog = candidates.slice(0, 3)
  1053→  // Fire-and-forget DB inserts
  1054→  ;(async () => {
  1055→    try {
  1056→      const { db } = await import('@/lib/db')
  1057→      const now = new Date()
  1058→      for (const c of toLog) {
  1059→        await db.incomeEntry.create({
  1060→          data: {
  1061→            amount: c.amount,
  1062→            source: agentId.charAt(0).toUpperCase() + agentId.slice(1),
  1063→            notes: `Auto-logged from ${agentId} sub-agent answer: "${c.line}"`,
  1064→            date: now,
  1065→          },
  1066→        })
  1067→      }
  1068→    } catch (e) {
  1069→      console.error('[orchestrator] autoLogIncomeFromAnswer failed:', e)
  1070→    }
  1071→  })()
  1072→}

  1473→          }
  1474→        }

  1482→              'update_settings: no recognized keys. Supported: monthly_goal, daily_growth_target, currency_symbol, display_mode.',
  1483→          }
  1484→        }
  1485→        return {
  1486→          ok: true,

  1488→        }
  1489→      }
  1490→
  1491→      default:

  1494→          message: `Unknown manage action: "${action}". Supported: create_agent, edit_agent, delete_agent, toggle_agent, set_income_goal, set_growth_target, log_income, create_schedule, delete_schedule, update_settings.`,
  1495→        }
  1496→    }
  1497→  } catch (e: any) {
  1498→    console.error('[orchestrator] executeManageAction failed:', e)
  1499→    return {
  1500→      ok: false,
  1501→      message: `Manage action "${action}" threw: ${e?.message ?? String(e)}`,
  1502→    }
  1503→  }
  1504→}

  1506→function validateHexColor(c?: string): string | null {
  1507→  if (!c || typeof c !== 'string') return null
  1508→  const trimmed = c.trim()
  1509→  if (/^#([0-9a-fA-F]{3}|[0-9a-fA-F]{6})$/.test(trimmed)) return trimmed
  1510→  return null
  1511→}
