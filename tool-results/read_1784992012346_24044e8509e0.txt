  1312→function parseAllowedTools(raw: string): string[] {
  1313→  try {

  1328→/**
  1329→ * Returns ALL subagents (12 built-in + custom), with overlay edits applied.
  1330→ * Disabled agents are filtered out unless `includeDisabled=true`.
  1331→ *
  1332→ * IMPORTANT: This is async because it reads from the DB. Callers that need
  1333→ * a synchronous lookup (e.g. runSubagent) should call this first and pass
  1334→ * the resulting list down.
  1335→ */
  1336→export async function getAllSubagents(opts?: { includeDisabled?: boolean }): Promise<Subagent[]> {
  1337→  const includeDisabled = opts?.includeDisabled ?? false
  1338→  let customRows: any[] = []
  1339→  try {
  1340→    // Ensure DB is ready before querying (fixes Vercel cold-start race condition
  1341→    // where getAllSubagents is called before ensureDbReady completes)
  1342→    const { ensureDbReady } = await import('./db')
  1343→    await ensureDbReady().catch(() => {})
  1344→    const userId = await getOperatorUserId()
  1345→    if (userId) {
  1346→      customRows = await db.customSubagent.findMany({ where: { userId } })
  1347→    }
  1348→  } catch (e) {
  1349→    console.error('[subagents] getAllSubagents DB load failed:', e)
  1350→    // Continue with empty customRows — built-in agents will still be returned
  1351→    // with FULL_ACCESS_TOOLS applied (see below)
  1352→  }
  1353→
  1354→  const overlayMap = new Map<string, any>()
  1355→  const customList: Subagent[] = []
  1356→  for (const row of customRows) {
  1357→    if (row.isBuiltinOverlay) {
  1358→      overlayMap.set(row.id, row)
  1359→    } else {
  1360→      customList.push({
  1361→        id: row.id,
  1362→        name: row.name,
  1363→        role: row.role,
  1364→        specialty: row.specialty,
  1365→        color: row.color,
  1366→        icon: row.icon,
  1367→        // FULL ACCESS — override any stored tools with ALL tools
  1368→        allowedTools: [...FULL_ACCESS_TOOLS,'multi_provider_compare'],
  1369→        systemPrompt: row.systemPrompt,
  1370→        isBuiltin: false,
  1371→        enabled: row.enabled ?? true,
  1372→      })
  1373→    }
  1374→  }
  1375→
  1376→  // Apply overlays on top of built-ins
  1377→  // UPGRADE #98 — TOOL RESTRICTION: Use each built-in subagent's specialized
  1378→  // allowedTools (defined in SUBAGENTS array) instead of overriding with FULL_ACCESS.
  1379→  // This creates true specialization: leaders have 15-20 tools, members have 7-12.
  1380→  const mergedBuiltins: Subagent[] = SUBAGENTS.map((b) => {
  1381→    const ov = overlayMap.get(b.id)
  1382→    return {
  1383→      ...b,
  1384→      name: ov?.name ?? b.name,
  1385→      role: ov?.role ?? b.role,
  1386→      specialty: ov?.specialty ?? b.specialty,
  1387→      color: ov?.color ?? b.color,
  1388→      icon: ov?.icon ?? b.icon,
  1389→      // UPGRADE #98: Use built-in allowedTools (specialized) or overlay's tools
  1390→      allowedTools: b.allowedTools?.length ? b.allowedTools : [...FULL_ACCESS_TOOLS],
  1391→      systemPrompt: ov?.systemPrompt ?? b.systemPrompt,
  1392→      enabled: ov?.enabled ?? b.enabled ?? true,
  1393→    }
  1394→  })
  1395→
  1396→  // DEDUPLICATION (upgrade #40 + #60): Filter out custom agents that have the
  1397→  // same id OR name (case-insensitive) as a built-in agent. This prevents the
  1398→  // 6 promoted agents (TRADER, Cybersecurity A/R, Developer, TESTFAST2,
  1399→  // FASTTEST3) from appearing twice — once from SUBAGENTS (builtin=true)
  1400→  // and once from the CustomSubagent DB table (builtin=false, created by
  1401→  // the old seeding code that ran before upgrade #38 promoted them).
  1402→  //
  1403→  // UPGRADE #60 FIX: The built-in agents were RENAMED in upgrade #57:
  1404→  //   - TESTFAST2 → QA Monitor (id: testfast2)
  1405→  //   - FASTTEST3 → External Monitor (id: fasttest3)
  1406→  // So the old DB entries (name="TESTFAST2", name="FASTTEST3") were slipping
  1407→  // through the name-based dedup. Now we also check against the OLD names.
  1408→  const OLD_NAMES_TO_FILTER = new Set([
  1409→    'testfast2',
  1410→    'fasttest3',
  1411→    'trader',
  1412→    'cybersecurity a',
  1413→    'cybersecurity r',
  1414→    'developer',
  1415→  ])
  1416→  const builtinIds = new Set(SUBAGENTS.map((b) => b.id.toLowerCase()))
  1417→  const builtinNames = new Set(SUBAGENTS.map((b) => b.name.toLowerCase()))
  1418→  const dedupedCustomList = customList.filter((c) => {
  1419→    const idMatch = builtinIds.has(c.id.toLowerCase())
  1420→    const nameMatch = builtinNames.has(c.name.toLowerCase())
  1421→    const oldNameMatch = OLD_NAMES_TO_FILTER.has(c.name.toLowerCase())
  1422→    return !idMatch && !nameMatch && !oldNameMatch
  1423→  })
  1424→
  1425→  const all = [...mergedBuiltins, ...dedupedCustomList]
  1426→  return includeDisabled ? all : all.filter((s) => s.enabled !== false)
  1427→}

  1470→/* Per-agent request throttle (#10). Ensures each individual sub-agent waits
  1471→ * at least MIN_AGENT_INTERVAL_MS between its own LLM calls, on top of the
  1472→ * app-wide throttle in agent.ts. */
  1473→const _agentLastCallAt: Record<string, number> = {}
  1474→const MIN_AGENT_INTERVAL_MS = 500 // Reduced from 1500 — owner requested no rate limiting
  1475→async function throttleAgentCall(agentId: string): Promise<void> {
  1476→  const now = Date.now()
  1477→  const last = _agentLastCallAt[agentId] || 0
  1478→  const wait = Math.max(0, last + MIN_AGENT_INTERVAL_MS - now)
  1479→  if (wait > 0) await new Promise((r) => setTimeout(r, wait))
  1480→  _agentLastCallAt[agentId] = Date.now()
  1481→}

  1483→export async function runSubagent(opts: RunSubagentOptions): Promise<RunSubagentResult> {
  1484→  // Look up the sub-agent definition from the merged list (built-in + DB-loaded
  1485→  // custom + built-in overlays). This lets the Super Agent dispatch to custom
  1486→  // agents and respect overlay edits at runtime.
  1487→  const allSubs = await getAllSubagents({ includeDisabled: true })
  1488→  // Match by id (case-sensitive) OR by name (case-insensitive) — this lets the
  1489→  // Super Agent dispatch to custom agents like "Cybersecurity A" using their
  1490→  // human-readable name even though their id is a cuid like "cmqzahs7d...".
  1491→  const sub = allSubs.find(
  1492→    (s) => s.id === opts.subagentId || s.name.toLowerCase() === opts.subagentId.toLowerCase()
  1493→  )
  1494→  if (!sub) {
  1495→    const available = allSubs.map((s) => `${s.name} (id: ${s.id})`).join(', ')
  1496→    const err = `Unknown sub-agent: "${opts.subagentId}". Available: ${available}`
  1497→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
  1498→    return { answer: `⚠️ ${err}`, steps: [] }
  1499→  }
  1500→  if (sub.enabled === false) {
  1501→    const err = `Sub-agent "${sub.name}" is currently disabled. Re-enable it in Settings → Sub-Agents.`
  1502→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
  1503→    return { answer: `⚠️ ${err}`, steps: [] }
  1504→  }
  1505→
  1506→  // UPGRADE #98 — TOOL RESTRICTION: Use the subagent's specialized allowedTools
  1507→  // instead of forcing FULL_ACCESS_TOOLS. This creates TRUE specialization:
  1508→  // each subagent only has access to tools relevant to their pod/role.
  1509→  // Leaders have 15-20 tools, members have 7-12 tools.
  1510→  // The Super Agent (orchestrator) still has all 667 tools.
  1511→  // If a subagent needs a tool they don't have, the leader provides it or
  1512→  // the Super Agent dispatches differently.
  1513→  const allowed = new Set(sub.allowedTools?.length ? sub.allowedTools : [...FULL_ACCESS_TOOLS])
  1514→  const ctx: ToolContext = { attachments: opts.attachments, language: opts.language }
  1515→
  1516→  const languageInstruction =
  1517→    opts.language === 'zh'
  1518→      ? 'LANGUAGE: Reply in 中文 for your FINAL answer.'
  1519→      : 'LANGUAGE: Reply in English for your FINAL answer unless the task is in another language.'
  1520→
  1521→  const systemPrompt = `${sub.systemPrompt}
  1522→
  1523→${languageInstruction}
  1524→
  1525→CURRENT UTC TIME: ${new Date().toUTCString()}
  1526→
  1527→You are operating autonomously inside Agent007's multi-agent network. The Super Agent has given you a specific task. Execute it end-to-end using only your allowed tools. Then return a clear, structured final answer.`
  1528→
  1529→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
  1530→    { role: 'system', content: systemPrompt },
  1531→    { role: 'user', content: opts.task },
  1532→  ]
  1533→
  1534→  const steps: RunSubagentResult['steps'] = []
  1535→  let finalAnswer = ''
  1536→  let iter = 0
  1537→
  1538→  while (iter < SUBAGENT_MAX_ITERATIONS) {
  1539→    iter++
  1540→    // Per-agent throttle — keep each sub-agent's calls >=1.5s apart
  1541→    await throttleAgentCall(sub.id)
  1542→    let completion: any
  1543→    try {
  1544→      completion = await callLlmWithRetry(conversationMessages)
  1545→    } catch (e: any) {
  1546→      finalAnswer = friendlyLlmError(e)
  1547→      break
  1548→    }
  1549→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
  1550→    if (!content.trim()) {
  1551→      finalAnswer = `(${sub.name} produced no output.)`
  1552→      break
  1553→    }
  1554→    const parsed = parseAssistant(content)
  1555→
  1556→    if (parsed.thought) {
  1557→      await opts.emit('subagent_thought', { dispatchId: opts.dispatchId, content: parsed.thought })
  1558→    }
  1559→
  1560→    if (!parsed.tool) {
  1561→      // FIX 2: Stuck detection for sub-agents.
  1562→      // If the sub-agent produced ONLY a thought with no substantial answer,
  1563→      // auto-recover by prompting it to continue (instead of treating the
  1564→      // thought as the final answer).
  1565→      const textAfterThought = content.replace(THOUGHT_RE, '').trim()
  1566→      const isThoughtOnly = !!parsed.thought && textAfterThought.length < 20
  1567→      const stuckPatterns = /(wait|waiting|haven't|will wait|let me wait|as i wait)/i
  1568→      const isStuck = isThoughtOnly && stuckPatterns.test(parsed.thought ?? "")
  1569→
  1570→      if ((isStuck || isThoughtOnly) && iter < SUBAGENT_MAX_ITERATIONS - 1) {
  1571→        await opts.emit('subagent_thought', {
  1572→          dispatchId: opts.dispatchId,
  1573→          content: `[AUTO-RECOVERY] Thought-only response. Prompting to continue...`,
  1574→        })
  1575→        conversationMessages.push({ role: 'assistant', content })
  1576→        conversationMessages.push({
  1577→          role: 'user',
  1578→          content: '[SYSTEM] You produced only a thought. Please either call a tool or give your final answer now.',
  1579→        })
  1580→        continue
  1581→      }
  1582→
  1583→      // final answer
  1584→      finalAnswer = textAfterThought || content.trim()
  1585→      break
  1586→    }
  1587→
  1588→    const toolName = parsed.tool.name
  1589→    const toolArgs = parsed.tool.args
  1590→    const stepId = `sub_${opts.dispatchId}_${iter}_${Math.random().toString(36).slice(2, 8)}`
  1591→
  1592→    // Enforce the sub-agent's allowed tool list
  1593→    // NOTE: With FULL_ACCESS_TOOLS, all 99+ tools are allowed, so this check
  1594→    // should NEVER block. If it does, it's a bug — the subagent should be
  1595→    // able to use any tool. The check remains as a safety net.
  1596→    let toolBlocked = false
  1597→    if (!allowed.has(toolName)) {
  1598→      // Auto-grant: if the tool exists in TOOL_REGISTRY, allow it anyway
  1599→      // (FULL_ACCESS means FULL ACCESS — no tool should be blocked)
  1600→      try {
  1601→        const { TOOL_REGISTRY } = await import('./tools')
  1602→        if (TOOL_REGISTRY[toolName]) {
  1603→          // Tool exists in registry — grant access + add to allowed set
  1604→          allowed.add(toolName)
  1605→          console.log(`[subagents] Auto-granted tool "${toolName}" to ${sub.name} (FULL_ACCESS)`)
  1606→        } else {
  1607→          // Tool doesn't exist at all — block it
  1608→          toolBlocked = true
  1609→          const errResult: ToolResult = {
  1610→            ok: false,
  1611→            preview: `Tool "${toolName}" not found in registry`,
  1612→            result: `BLOCKED: Tool "${toolName}" does not exist in the TOOL_REGISTRY. Available tools include: web_search, page_reader, image_gen, vision, code_exec, memory_store, memory_recall, file_read, file_write, source_read, and 455+ more.`,
  1613→          }
  1614→          const step: any = {
  1615→            id: stepId,
  1616→            thought: parsed.thought,
  1617→            toolName,
  1618→            toolArgs,
  1619→            toolResult: errResult,
  1620→            startedAt: Date.now(),
  1621→            finishedAt: Date.now(),
  1622→          }
  1623→          steps.push(step)
  1624→          await opts.emit('subagent_tool_call', {
  1625→            dispatchId: opts.dispatchId,
  1626→            stepId,
  1627→            name: toolName,
  1628→            args: toolArgs,
  1629→            thought: parsed.thought,
  1630→            stepNumber: iter,
  1631→          })
  1632→          await opts.emit('subagent_tool_result', {
  1633→            dispatchId: opts.dispatchId,
  1634→            stepId,
  1635→            result: errResult.result,
  1636→            preview: errResult.preview,
  1637→            ok: false,
  1638→            artifacts: undefined,
  1639→          })
  1640→          conversationMessages.push({ role: 'assistant', content })
  1641→          conversationMessages.push({
  1642→            role: 'user',
  1643→            content: `[TOOL_RESULT] ${toolName}: ${errResult.result}`,
  1644→          })
  1645→          continue
  1646→        }
  1647→      } catch (importErr) {
  1648→        // If import fails, auto-grant anyway (fail-open)
  1649→        allowed.add(toolName)
  1650→      }
  1651→    }
  1652→
  1653→    const step: any = {
  1654→      id: stepId,
  1655→      thought: parsed.thought,
  1656→      toolName,
  1657→      toolArgs,
  1658→      startedAt: Date.now(),
  1659→    }
  1660→    steps.push(step)
  1661→
  1662→    await opts.emit('subagent_tool_call', {
  1663→      dispatchId: opts.dispatchId,
  1664→      stepId,
  1665→      name: toolName,
  1666→      args: toolArgs,
  1667→      thought: parsed.thought,
  1668→      stepNumber: iter,
  1669→    })
  1670→
  1671→    const toolResult = await dispatchTool(toolName, toolArgs, ctx)
  1672→    step.toolResult = toolResult
  1673→    step.finishedAt = Date.now()
  1674→
  1675→    await opts.emit('subagent_tool_result', {
  1676→      dispatchId: opts.dispatchId,
  1677→      stepId,
  1678→      result: toolResult.result,
  1679→      preview: toolResult.preview,
  1680→      ok: toolResult.ok,
  1681→      artifacts: toolResult.artifacts,
  1682→    })
  1683→
  1684→    // Persist subagent tool activity for reload reconstruction
  1685→    try {
  1686→      if (step.thought) {
  1687→        await db.message.create({
  1688→          data: {
  1689→            conversationId: opts.parentConversationId,
  1690→            role: 'thought',
  1691→            content: `[subagent:${sub.id}] ${step.thought}`,
  1692→          },
  1693→        })
  1694→      }
  1695→      await db.message.create({
  1696→        data: {
  1697→          conversationId: opts.parentConversationId,
  1698→          role: 'tool',
  1699→          content: `[subagent:${sub.id}:tool] ${toolName} ${JSON.stringify(toolArgs)}`,
  1700→          toolName: 'subagent_tool',
  1701→          toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, stepId, tool: toolName, args: toolArgs }),
  1702→          toolResult: toolResult.result,
  1703→        },
  1704→      })
  1705→    } catch {
  1706→      /* ignore persistence errors */
  1707→    }
  1708→
  1709→    conversationMessages.push({ role: 'assistant', content })
  1710→    conversationMessages.push({
  1711→      role: 'user',
  1712→      content: `[TOOL_RESULT] ${toolName}: ${toolResult.result}`,
  1713→    })
  1714→  }
  1715→
  1716→  if (!finalAnswer) {
  1717→    finalAnswer = `${sub.name} reached its tool-call limit. Here is what it found so far:\n\n` +
  1718→      steps
  1719→        .filter((s) => s.toolResult?.result)
  1720→        .map((s, i) => `${i + 1}. **${s.toolName}** → ${s.toolResult!.result.slice(0, 500)}`)
  1721→        .join('\n\n')
  1722→  }
  1723→
  1724→  await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: finalAnswer })
  1725→
  1726→  // Persist the sub-agent's final answer for reload reconstruction
  1727→  try {
  1728→    await db.message.create({
  1729→      data: {
  1730→        conversationId: opts.parentConversationId,
  1731→        role: 'tool',
  1732→        content: `[subagent:${sub.id}:complete] ${finalAnswer.slice(0, 500)}`,
  1733→        toolName: 'subagent_complete',
  1734→        toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, task: opts.task }),
  1735→        toolResult: finalAnswer,
  1736→      },
  1737→    })
  1738→  } catch {
  1739→    /* ignore */
  1740→  }
  1741→
  1742→  return { answer: finalAnswer, steps }
  1743→}
