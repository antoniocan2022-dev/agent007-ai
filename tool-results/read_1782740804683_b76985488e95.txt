    23→/* Regex to find <dispatch agent="..." task="..."/> tags (self-closing).
    24→ * Uses non-greedy [\s\S]*? for the task value so apostrophes / quotes inside the
    25→ * task description don't break the match (the LLM frequently emits apostrophes
    26→ * like 'Quantum Labs' inside the task attribute value). */
    27→const DISPATCH_RE = /<dispatch\s+agent=["']([^"']+)["']\s+task=["']([\s\S]*?)["']\s*\/>/i
    28→
    29→/* Regex to find <manage action="..." attr="..." ... /> self-closing tags.
    30→ * Captures the full tag string; attribute parsing happens in parseManageTag. */
    31→const MANAGE_RE = /<manage\s+[^>]*?\/>/gi
    32→
    33→interface OrchestratorParsed {
    34→  thought?: string
    35→  tool?: { name: string; args: any }
    36→  dispatch?: { agentId: string; task: string }
    37→  manage?: { action: string; attrs: Record<string, string>; raw: string }
    38→  textAfter: string
    39→  raw: string
    40→}
    41→
    42→function parseOrchestrator(content: string): OrchestratorParsed {
    43→  const thoughtMatch = content.match(THOUGHT_RE)
    44→  const thought = thoughtMatch?.[1]?.trim()
    45→
    46→  const dispatchMatch = content.match(DISPATCH_RE)
    47→  const toolMatch = content.match(TOOL_RE)
    48→  const manageMatch = content.match(MANAGE_RE)
    49→
    50→  // Priority: dispatch > manage > tool (manage and dispatch are both "agent actions")
    51→  if (dispatchMatch) {
    52→    const agentId = dispatchMatch[1].trim().toLowerCase()
    53→    const task = dispatchMatch[2].trim()
    54→    return {
    55→      thought,
    56→      dispatch: { agentId, task },
    57→      textAfter: content.slice(content.indexOf(dispatchMatch[0]) + dispatchMatch[0].length).replace(THOUGHT_RE, '').trim(),
    58→      raw: content,
    59→    }
    60→  }
    61→  if (manageMatch && manageMatch.length > 0) {
    62→    const tag = manageMatch[0]
    63→    const attrs = parseManageAttrs(tag)
    64→    const action = (attrs.action ?? '').toString().trim().toLowerCase()
    65→    return {
    66→      thought,
    67→      manage: { action, attrs, raw: tag },
    68→      textAfter: content.replace(tag, '').replace(THOUGHT_RE, '').trim(),
    69→      raw: content,
    70→    }
    71→  }
    72→  if (toolMatch) {
    73→    const name = toolMatch[1].trim()
    74→    let args: any = {}
    75→    const raw = toolMatch[2].trim()
    76→    try {
    77→      args = JSON.parse(raw)
    78→    } catch {
    79→      const m: Record<string, string> = {}
    80→      const re = /"([^"]+)"\s*:\s*"([^"]*)"/g
    81→      let mm: RegExpExecArray | null
    82→      while ((mm = re.exec(raw))) m[mm[1]] = mm[2]
    83→      args = m
    84→    }
    85→    return { thought, tool: { name, args }, textAfter: '', raw: content }
    86→  }
    87→  return { thought, textAfter: content.replace(THOUGHT_RE, '').trim(), raw: content }
    88→}
    89→
    90→/** Parse attributes from a <manage .../> tag. Handles key="value" pairs with
    91→ * either single or double quotes. Also supports attribute values containing
    92→ * spaces because the regex is greedy on the quoted portion. */
    93→function parseManageAttrs(tag: string): Record<string, string> {
    94→  const attrs: Record<string, string> = {}
    95→  // Match: attrName="value" or attrName='value'
    96→  // We allow newlines inside the value via [\s\S]*? (non-greedy).
    97→  const re = /([a-zA-Z_][a-zA-Z0-9_]*)\s*=\s*(?:"([^"]*)"|'([^']*)')/g
    98→  let m: RegExpExecArray | null
    99→  while ((m = re.exec(tag))) {
   100→    const key = m[1]
   101→    const val = m[2] ?? m[3] ?? ''
   102→    attrs[key] = val
   103→  }
   104→  return attrs
   105→}

   195→export async function runOrchestrator(opts: OrchestratorRunOptions): Promise<OrchestratorRunResult> {
   196→  const { conversationId, userMessage, attachments, language, emit } = opts
   197→  const zai = await getZai()
   198→
   199→  // Recall memories for context
   200→  const recalled = await recallMemories(userMessage.slice(0, 200), 8)
   201→  const memoryBlock = formatMemoryForPrompt(recalled)
   202→
   203→  const languageInstruction =
   204→    language === 'zh'
   205→      ? 'LANGUAGE INSTRUCTION: The user has toggled the agent to Chinese. Reply in 中文 (Chinese) for your FINAL answer regardless of input language.'
   206→      : 'LANGUAGE INSTRUCTION: The user has toggled the agent to English. Reply in English for your FINAL answer unless the user wrote in another language.'
   207→
   208→  const systemPrompt = `${BASE_SYSTEM_PROMPT}
   209→
   210→${ORCHESTRATOR_PROMPT_ADDENDUM}
   211→
   212→${languageInstruction}
   213→
   214→RECALLED MEMORIES (use as context, do not blindly trust if outdated):
   215→${memoryBlock}
   216→
   217→CURRENT UTC TIME: ${new Date().toUTCString()}`
   218→
   219→  const history = await buildHistoryMessages(conversationId, userMessage, attachments)
   220→  const ctx: ToolContext = { attachments, language }
   221→
   222→  const steps: OrchestratorRunResult['steps'] = []
   223→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
   224→    { role: 'system', content: systemPrompt },
   225→    ...history,
   226→  ]
   227→
   228→  let finalAnswer = ''
   229→  let iter = 0
   230→  let dispatchCount = 0
   231→  let manageCount = 0
   232→  // Cache the merged subagent list once per orchestrator run (12 built-in + custom + overlays).
   233→  let mergedSubagents: Subagent[] | null = null
   234→  const getMerged = async (): Promise<Subagent[]> => {
   235→    if (!mergedSubagents) mergedSubagents = await getAllSubagents({ includeDisabled: true })
   236→    return mergedSubagents
   237→  }
   238→
   239→  while (iter < MAX_ITERATIONS) {
   240→    iter++
   241→    let completion: any
   242→    try {
   243→      completion = await zai.chat.completions.create({
   244→        messages: conversationMessages,
   245→        thinking: { type: 'enabled' },
   246→      })
   247→    } catch (e: any) {
   248→      const friendly = friendlyLlmError(e)
   249→      await emit('error', { message: friendly })
   250→      finalAnswer = friendly
   251→      break
   252→    }
   253→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
   254→    if (!content.trim()) {
   255→      finalAnswer = '(The agent produced no output. Please try rephrasing.)'
   256→      break
   257→    }
   258→
   259→    const parsed = parseOrchestrator(content)
   260→
   261→    // Emit thought
   262→    if (parsed.thought) {
   263→      await emit('thought', { content: parsed.thought })
   264→    }
   265→
   266→    // 0) Manage path — parse <manage .../> tags and execute server-side.
   267→    if (parsed.manage) {
   268→      const { action, attrs } = parsed.manage
   269→      if (manageCount >= MAX_MANAGE_ACTIONS) {
   270→        const capMsg = `Reached max manage actions (${MAX_MANAGE_ACTIONS}). Please synthesize the answer from what you have.`
   271→        conversationMessages.push({ role: 'assistant', content })
   272→        conversationMessages.push({ role: 'user', content: `[SYSTEM] ${capMsg}` })
   273→        continue
   274→      }
   275→      manageCount++
   276→
   277→      const stepId = makeId('manage')
   278→      await emit('manage_action', {
   279→        stepId,
   280→        action,
   281→        attrs,
   282→        thought: parsed.thought,
   283→        stepNumber: iter,
   284→        status: 'running',
   285→      })
   286→
   287→      const result = await executeManageAction(action, attrs)
   288→      // Refresh the merged subagent list so subsequent dispatches see the new state.
   289→      mergedSubagents = null
   290→
   291→      await emit('manage_action', {
   292→        stepId,
   293→        action,
   294→        attrs,
   295→        result,
   296→        stepNumber: iter,
   297→        status: result.ok ? 'done' : 'error',
   298→      })
   299→
   300→      // Persist for reload reconstruction
   301→      try {
   302→        await db.message.create({
   303→          data: {
   304→            conversationId,
   305→            role: 'tool',
   306→            content: `[manage:${action}] ${JSON.stringify(attrs).slice(0, 200)}`,
   307→            toolName: 'manage_action',
   308→            toolArgs: JSON.stringify({ action, attrs }),
   309→            toolResult: result.message,
   310→          },
   311→        })
   312→      } catch { /* ignore */ }
   313→
   314→      // If a subagent was created/edited/deleted/toggled, tell the client to refresh its UI.
   315→      if (
   316→        result.ok &&
   317→        ['create_agent', 'edit_agent', 'delete_agent', 'toggle_agent'].includes(action)
   318→      ) {
   319→        await emit('subagents_updated', { action, attrs, result })
   320→      }
   321→
   322→      // Feed back the result so the orchestrator can confirm to the user.
   323→      conversationMessages.push({ role: 'assistant', content })
   324→      conversationMessages.push({
   325→        role: 'user',
   326→        content: `[MANAGE_RESULT] ${action}: ${result.ok ? 'success' : 'failed'} — ${result.message}`,
   327→      })
   328→      continue
   329→    }
   330→
   331→    // 1) Dispatch path
   332→    if (parsed.dispatch) {
   333→      const { agentId, task } = parsed.dispatch
   334→      const list = await getMerged()
   335→      const sub = list.find((s) => s.id === agentId)
   336→      if (!sub) {
   337→        // Unknown agent — feed back an error to the orchestrator
   338→        const errMsg = `Unknown sub-agent: "${agentId}". Available: ${list.map((s) => s.id).join(', ')}`
   339→        await emit('error', { message: errMsg })
   340→        conversationMessages.push({ role: 'assistant', content })
   341→        conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${agentId}: ERROR — ${errMsg}` })
   342→        // Persist the failed dispatch attempt
   343→        try {
   344→          await db.message.create({
   345→            data: {
   346→              conversationId,
   347→              role: 'tool',
   348→              content: `[dispatch:unknown] ${agentId} task="${task}"`,
   349→              toolName: 'subagent_dispatch',
   350→              toolArgs: JSON.stringify({ agentId, task, error: errMsg }),
   351→              toolResult: errMsg,
   352→            },
   353→          })
   354→        } catch { /* ignore */ }
   355→        continue
   356→      }
   357→      if (sub.enabled === false) {
   358→        const errMsg = `Sub-agent "${sub.name}" is currently disabled. Re-enable it via the Sub-Agents panel or the toggle_agent manage tag.`
   359→        await emit('error', { message: errMsg })
   360→        conversationMessages.push({ role: 'assistant', content })
   361→        conversationMessages.push({ role: 'user', content: `[SUBAGENT_RESULT] ${agentId}: ERROR — ${errMsg}` })
   362→        continue
   363→      }
   364→      if (dispatchCount >= MAX_DISPATCHES) {
   365→        const capMsg = `Reached max sub-agent dispatches (${MAX_DISPATCHES}). Please synthesize the answer from what you have.`
   366→        conversationMessages.push({ role: 'assistant', content })
   367→        conversationMessages.push({ role: 'user', content: `[SYSTEM] ${capMsg}` })
   368→        continue
   369→      }
   370→      dispatchCount++
   371→
   372→      const dispatchId = makeId('disp')
   373→      const dispatchStepNumber = iter
   374→
   375→      // Emit dispatch event (UI shows the sub-agent header card)
   376→      await emit('subagent_dispatch', {
   377→        dispatchId,
   378→        agentId: sub.id,
   379→        agentName: sub.name,
   380→        color: sub.color,
   381→        icon: sub.icon,
   382→        task,
   383→        stepNumber: dispatchStepNumber,
   384→      })
   385→
   386→      // Persist the dispatch row for reload reconstruction
   387→      try {
   388→        await db.message.create({
   389→          data: {
   390→            conversationId,
   391→            role: 'tool',
   392→            content: `[dispatch] ${sub.id} task="${task.slice(0, 200)}"`,
   393→            toolName: 'subagent_dispatch',
   394→            toolArgs: JSON.stringify({ dispatchId, agentId: sub.id, agentName: sub.name, color: sub.color, icon: sub.icon, task }),
   395→            toolResult: null,
   396→          },
   397→        })
   398→      } catch { /* ignore */ }
   399→
   400→      // Run the sub-agent (it will emit its own subagent_thought/tool_call/tool_result events)
   401→      let subAnswer = ''
   402→      try {
   403→        const result = await runSubagent({
   404→          subagentId: sub.id,
   405→          task,
   406→          attachments,
   407→          language,
   408→          parentConversationId: conversationId,
   409→          dispatchId,
   410→          emit: async (ev, data) => {
   411→            await emit(ev, data)
   412→            if (ev === 'subagent_complete' && data?.answer) {
   413→              subAnswer = data.answer
   414→            }
   415→          },
   416→        })
   417→        subAnswer = result.answer
   418→      } catch (e: any) {
   419→        subAnswer = friendlyLlmError(e)
   420→        await emit('subagent_complete', { dispatchId, answer: subAnswer })
   421→      }
   422→
   423→      // Feed the sub-agent's result back to the orchestrator
   424→      conversationMessages.push({ role: 'assistant', content })
   425→      conversationMessages.push({
   426→        role: 'user',
   427→        content: `[SUBAGENT_RESULT] ${sub.id}: ${subAnswer}`,
   428→      })
   429→
   430→      // BEST-EFFORT auto-logging: if the sub-agent's answer mentions dollar
   431→      // amounts (e.g. "$12.50", "$1,200/mo", "$45/day"), log them as income
   432→      // entries with source = sub-agent id. Fire-and-forget — never blocks the
   433→      // orchestrator. We only consider amounts that look like earnings (positive
   434→      // dollar values, optionally followed by /day /mo /week /month).
   435→      try {
   436→        autoLogIncomeFromAnswer(sub.id, subAnswer)
   437→      } catch {
   438→        /* ignore */
   439→      }
   440→
   441→      // If a memory_store happened inside the sub-agent, the sub-agent already emitted it
   442→      // (we don't double-emit memory_update here)
   443→      continue
   444→    }
   445→
   446→    // 2) Direct tool path (same as the original agent loop)
   447→    if (parsed.tool) {
   448→      const step = {
   449→        id: makeId('step'),
   450→        thought: parsed.thought,
   451→        toolName: parsed.tool.name,
   452→        toolArgs: parsed.tool.args,
   453→        startedAt: Date.now(),
   454→      }
   455→      steps.push(step)
   456→      await emit('tool_call', {
   457→        stepId: step.id,
   458→        name: step.toolName,
   459→        args: step.toolArgs,
   460→        thought: step.thought,
   461→        stepNumber: iter,
   462→      })
   463→
   464→      const toolResult = await dispatchTool(step.toolName!, step.toolArgs, ctx)
   465→      step.toolResult = toolResult
   466→      step.finishedAt = Date.now()
   467→      await emit('tool_result', {
   468→        stepId: step.id,
   469→        result: toolResult.result,
   470→        preview: toolResult.preview,
   471→        ok: toolResult.ok,
   472→        artifacts: toolResult.artifacts,
   473→      })
   474→
   475→      if (step.toolName === 'memory_store' && toolResult.ok) {
   476→        await emit('memory_update', {
   477→          key: step.toolArgs?.key,
   478→          value: step.toolArgs?.value,
   479→          category: step.toolArgs?.category ?? 'general',
   480→        })
   481→      }
   482→
   483→      conversationMessages.push({ role: 'assistant', content })
   484→      conversationMessages.push({
   485→        role: 'user',
   486→        content: `[TOOL_RESULT] ${step.toolName}: ${toolResult.result}`,
   487→      })
   488→
   489→      // Persist intermediate tool/thought rows for reload reconstruction
   490→      try {
   491→        if (step.thought) {
   492→          await db.message.create({
   493→            data: { conversationId, role: 'thought', content: step.thought },
   494→          })
   495→        }
   496→        await db.message.create({
   497→          data: {
   498→            conversationId,
   499→            role: 'tool',
   500→            content: `[tool call] ${step.toolName} ${JSON.stringify(step.toolArgs)}`,
   501→            toolName: step.toolName,
   502→            toolArgs: JSON.stringify(step.toolArgs),
   503→            toolResult: toolResult.result,
   504→          },
   505→        })
   506→      } catch { /* ignore */ }
   507→      continue
   508→    }
   509→
   510→    // 3) Final answer path — emit synthesis signal then stream tokens
   511→    finalAnswer = content.replace(THOUGHT_RE, '').replace(DISPATCH_RE, '').trim() || content.trim()
   512→
   513→    // Emit a synthesis indicator so the UI shows "Synthesizing…" briefly
   514→    await emit('synthesis', { content: finalAnswer.slice(0, 80) })
   515→
   516→    const chunks = chunkText(finalAnswer, 80)
   517→    for (const c of chunks) {
   518→      await emit('token', { content: c })
   519→    }
   520→    break
   521→  }
   522→
   523→  if (!finalAnswer) {
   524→    finalAnswer =
   525→      "I've reached my iteration limit for this turn. Here's what I have so far — let me know if you'd like me to continue."
   526→    await emit('token', { content: finalAnswer })
   527→  }
   528→
   529→  // Persist the final assistant message
   530→  const assistantRow = await db.message.create({
   531→    data: { conversationId, role: 'assistant', content: finalAnswer },
   532→  })
   533→
   534→  // Update conversation title if it's still default
   535→  const conv = await db.conversation.findUnique({ where: { id: conversationId } })
   536→  if (conv && (conv.title === 'New Conversation' || !conv.title)) {
   537→    const title = userMessage.slice(0, 50).trim() || 'New Conversation'
   538→    await db.conversation.update({ where: { id: conversationId }, data: { title } })
   539→  }
   540→
   541→  // Notification hook: if mission_complete notifications are enabled, send
   542→  // (or log) an email to the operator with the conversation title + preview.
   543→  try {
   544→    const { getNotificationSettings, recentlyNotified } = await import('@/lib/settings')
   545→    const notif = await getNotificationSettings()
   546→    const looksLikeError = /^⚠️|error|failed|crashed/i.test(finalAnswer.slice(0, 50))
   547→    const eventType = looksLikeError ? 'mission_failed' : 'mission_complete'
   548→    if (notif.enabled && notif.events[eventType as keyof typeof notif.events]) {
   549→      if (!(await recentlyNotified(eventType, notif.minDelayMinutes))) {
   550→        const { sendEmail } = await import('@/lib/email')
   551→        const { getOperatorUserId } = await import('@/lib/settings')
   552→        const userId = await getOperatorUserId()
   553→        const convTitle = conv?.title ?? 'Mission'
   554→        const preview = finalAnswer.slice(0, 500)
   555→        sendEmail({
   556→          to: notif.email,
   557→          subject: looksLikeError
   558→            ? `Mission Failed: ${convTitle}`
   559→            : `Mission Complete: ${convTitle}`,
   560→          body: looksLikeError
   561→            ? `Agent007 encountered an issue while running a mission.\n\nConversation: ${convTitle}\n\nPreview:\n${preview}\n\nOpen the dashboard at / to investigate.`
   562→            : `Agent007 has completed a mission.\n\nConversation: ${convTitle}\n\nResult preview:\n${preview}\n\nOpen the dashboard at / to view the full report.`,
   563→          userId: userId ?? undefined,
   564→          type: eventType,
   565→        }).catch(() => {/* ignore */})
   566→      }
   567→    }
   568→  } catch {
   569→    /* ignore notification errors */
   570→  }
   571→
   572→  return {
   573→    finalAnswer,
   574→    steps,
   575→    persistedAssistantMessageId: assistantRow.id,
   576→  }
   577→}

   583→/**
   584→ * Scan a sub-agent's answer for dollar amounts that look like earnings, and
   585→ * log them as IncomeEntry rows with source = agentId. Fire-and-forget.
   586→ *
   587→ * We're deliberately conservative — only log amounts that appear near income
   588→ * keywords (earned, income, revenue, MRR, /day, /mo, /week, /month, profit,
   589→ * ROI, yield). This avoids logging "$0 cost" or "$1,000 capital" as income.
   590→ */
   591→function autoLogIncomeFromAnswer(agentId: string, answer: string): void {
   592→  if (!answer || typeof answer !== 'string') return
   593→  // Strip code blocks to avoid logging amounts from code samples
   594→  const cleaned = answer.replace(/```[\s\S]*?```/g, ' ')
   595→  // Find all $X or $X.Y or $X,YYY mentions
   596→  const re = /\$([\d,]+(?:\.\d{1,2})?)\s*(?:\/(?:day|d|mo|month|m|week|wk|w|year|yr|y))?/gi
   597→  const incomeKeywords = /(earned|income|revenue|mrr|arr|profit|yield|roi|royalt|paying|paid|generat)/i
   598→  const periodKeywords = /\/(day|d|mo|month|m|week|wk|w|year|yr|y)\b/i
   599→  let m: RegExpExecArray | null
   600→  const candidates: Array<{ amount: number; line: string }> = []
   601→  while ((m = re.exec(cleaned))) {
   602→    const amountStr = m[1].replace(/,/g, '')
   603→    const amount = parseFloat(amountStr)
   604→    if (!isFinite(amount) || amount <= 0 || amount > 1_000_000) continue
   605→    // Look at a window of text around this match for income keywords
   606→    const start = Math.max(0, m.index - 80)
   607→    const end = Math.min(cleaned.length, m.index + m[0].length + 80)
   608→    const window = cleaned.slice(start, end)
   609→    // If the amount has a period suffix (/day /mo etc.) OR nearby income keyword → log it
   610→    if (periodKeywords.test(m[0]) || incomeKeywords.test(window)) {
   611→      candidates.push({ amount, line: m[0] })
   612→    }
   613→  }
   614→  if (!candidates.length) return
   615→  // Cap to 3 per sub-agent answer to avoid spamming the table
   616→  const toLog = candidates.slice(0, 3)
   617→  // Fire-and-forget DB inserts
   618→  ;(async () => {
   619→    try {
   620→      const { db } = await import('@/lib/db')
   621→      const now = new Date()
   622→      for (const c of toLog) {
   623→        await db.incomeEntry.create({
   624→          data: {
   625→            amount: c.amount,
   626→            source: agentId.charAt(0).toUpperCase() + agentId.slice(1),
   627→            notes: `Auto-logged from ${agentId} sub-agent answer: "${c.line}"`,
   628→            date: now,
   629→          },
   630→        })
   631→      }
   632→    } catch (e) {
   633→      console.error('[orchestrator] autoLogIncomeFromAnswer failed:', e)
   634→    }
   635→  })()
   636→}
