   482→   482→function parseAllowedTools(raw: string): string[] {
   483→   483→  try {
   484→   484→    const parsed = JSON.parse(raw)
   485→   485→    if (Array.isArray(parsed)) {
   486→   486→      return parsed.filter((s) => typeof s === 'string' && s.length > 0)
   487→   487→    }
   488→   488→  } catch {
   489→   489→    /* ignore */
   490→   490→  }
   491→   491→  // Fallback: comma-separated string
   492→   492→  return raw
   493→   493→    .split(',')
   494→   494→    .map((s) => s.trim())
   495→   495→    .filter(Boolean)
   496→   496→}

   498→   498→/**
   499→   499→ * Returns ALL subagents (12 built-in + custom), with overlay edits applied.
   500→   500→ * Disabled agents are filtered out unless `includeDisabled=true`.
   501→   501→ *
   502→   502→ * IMPORTANT: This is async because it reads from the DB. Callers that need
   503→   503→ * a synchronous lookup (e.g. runSubagent) should call this first and pass
   504→   504→ * the resulting list down.
   505→   505→ */
   506→   506→export async function getAllSubagents(opts?: { includeDisabled?: boolean }): Promise<Subagent[]> {
   507→   507→  const includeDisabled = opts?.includeDisabled ?? false
   508→   508→  let customRows: any[] = []
   509→   509→  try {
   510→   510→    const userId = await getOperatorUserId()
   511→   511→    if (userId) {
   512→   512→      customRows = await db.customSubagent.findMany({ where: { userId } })
   513→   513→    }
   514→   514→  } catch (e) {
   515→   515→    console.error('[subagents] getAllSubagents DB load failed:', e)
   516→   516→  }
   517→   517→
   518→   518→  const overlayMap = new Map<string, any>()
   519→   519→  const customList: Subagent[] = []
   520→   520→  for (const row of customRows) {
   521→   521→    if (row.isBuiltinOverlay) {
   522→   522→      overlayMap.set(row.id, row)
   523→   523→    } else {
   524→   524→      customList.push({
   525→   525→        id: row.id,
   526→   526→        name: row.name,
   527→   527→        role: row.role,
   528→   528→        specialty: row.specialty,
   529→   529→        color: row.color,
   530→   530→        icon: row.icon,
   531→   531→        allowedTools: parseAllowedTools(row.allowedTools),
   532→   532→        systemPrompt: row.systemPrompt,
   533→   533→        isBuiltin: false,
   534→   534→        enabled: row.enabled ?? true,
   535→   535→      })
   536→   536→    }
   537→   537→  }
   538→   538→
   539→   539→  // Apply overlays on top of built-ins
   540→   540→  const mergedBuiltins: Subagent[] = SUBAGENTS.map((b) => {
   541→   541→    const ov = overlayMap.get(b.id)
   542→   542→    if (!ov) return { ...b, enabled: b.enabled ?? true }
   543→   543→    return {
   544→   544→      ...b,
   545→   545→      name: ov.name ?? b.name,
   546→   546→      role: ov.role ?? b.role,
   547→   547→      specialty: ov.specialty ?? b.specialty,
   548→   548→      color: ov.color ?? b.color,
   549→   549→      icon: ov.icon ?? b.icon,
   550→   550→      allowedTools: ov.allowedTools ? parseAllowedTools(ov.allowedTools) : b.allowedTools,
   551→   551→      systemPrompt: ov.systemPrompt ?? b.systemPrompt,
   552→   552→      enabled: ov.enabled ?? b.enabled ?? true,
   553→   553→    }
   554→   554→  })
   555→   555→
   556→   556→  const all = [...mergedBuiltins, ...customList]
   557→   557→  return includeDisabled ? all : all.filter((s) => s.enabled !== false)
   558→   558→}

   601→   601→/* Per-agent request throttle (#10). Ensures each individual sub-agent waits
   602→   602→ * at least MIN_AGENT_INTERVAL_MS between its own LLM calls, on top of the
   603→   603→ * app-wide throttle in agent.ts. */
   604→   604→const _agentLastCallAt: Record<string, number> = {}
   605→   605→const MIN_AGENT_INTERVAL_MS = 1500
   606→   606→async function throttleAgentCall(agentId: string): Promise<void> {
   607→   607→  const now = Date.now()
   608→   608→  const last = _agentLastCallAt[agentId] || 0
   609→   609→  const wait = Math.max(0, last + MIN_AGENT_INTERVAL_MS - now)
   610→   610→  if (wait > 0) await new Promise((r) => setTimeout(r, wait))
   611→   611→  _agentLastCallAt[agentId] = Date.now()
   612→   612→}

   614→   614→export async function runSubagent(opts: RunSubagentOptions): Promise<RunSubagentResult> {
   615→   615→  // Look up the sub-agent definition from the merged list (built-in + DB-loaded
   616→   616→  // custom + built-in overlays). This lets the Super Agent dispatch to custom
   617→   617→  // agents and respect overlay edits at runtime.
   618→   618→  const allSubs = await getAllSubagents({ includeDisabled: true })
   619→   619→  // Match by id (case-sensitive) OR by name (case-insensitive) — this lets the
   620→   620→  // Super Agent dispatch to custom agents like "Cybersecurity A" using their
   621→   621→  // human-readable name even though their id is a cuid like "cmqzahs7d...".
   622→   622→  const sub = allSubs.find(
   623→   623→    (s) => s.id === opts.subagentId || s.name.toLowerCase() === opts.subagentId.toLowerCase()
   624→   624→  )
   625→   625→  if (!sub) {
   626→   626→    const available = allSubs.map((s) => `${s.name} (id: ${s.id})`).join(', ')
   627→   627→    const err = `Unknown sub-agent: "${opts.subagentId}". Available: ${available}`
   628→   628→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
   629→   629→    return { answer: `⚠️ ${err}`, steps: [] }
   630→   630→  }
   631→   631→  if (sub.enabled === false) {
   632→   632→    const err = `Sub-agent "${sub.name}" is currently disabled. Re-enable it in Settings → Sub-Agents.`
   633→   633→    await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: `⚠️ ${err}` })
   634→   634→    return { answer: `⚠️ ${err}`, steps: [] }
   635→   635→  }
   636→   636→
   637→   637→  const allowed = new Set(sub.allowedTools)
   638→   638→  const ctx: ToolContext = { attachments: opts.attachments, language: opts.language }
   639→   639→
   640→   640→  const languageInstruction =
   641→   641→    opts.language === 'zh'
   642→   642→      ? 'LANGUAGE: Reply in 中文 for your FINAL answer.'
   643→   643→      : 'LANGUAGE: Reply in English for your FINAL answer unless the task is in another language.'
   644→   644→
   645→   645→  const systemPrompt = `${sub.systemPrompt}
   646→   646→
   647→   647→${languageInstruction}
   648→   648→
   649→   649→CURRENT UTC TIME: ${new Date().toUTCString()}
   650→   650→
   651→   651→You are operating autonomously inside Agent007's multi-agent network. The Super Agent has given you a specific task. Execute it end-to-end using only your allowed tools. Then return a clear, structured final answer.`
   652→   652→
   653→   653→  let conversationMessages: Array<{ role: 'system' | 'user' | 'assistant'; content: string }> = [
   654→   654→    { role: 'system', content: systemPrompt },
   655→   655→    { role: 'user', content: opts.task },
   656→   656→  ]
   657→   657→
   658→   658→  const steps: RunSubagentResult['steps'] = []
   659→   659→  let finalAnswer = ''
   660→   660→  let iter = 0
   661→   661→
   662→   662→  while (iter < SUBAGENT_MAX_ITERATIONS) {
   663→   663→    iter++
   664→   664→    // Per-agent throttle — keep each sub-agent's calls >=1.5s apart
   665→   665→    await throttleAgentCall(sub.id)
   666→   666→    let completion: any
   667→   667→    try {
   668→   668→      completion = await callLlmWithRetry(conversationMessages)
   669→   669→    } catch (e: any) {
   670→   670→      finalAnswer = friendlyLlmError(e)
   671→   671→      break
   672→   672→    }
   673→   673→    const content: string = completion?.choices?.[0]?.message?.content ?? ''
   674→   674→    if (!content.trim()) {
   675→   675→      finalAnswer = `(${sub.name} produced no output.)`
   676→   676→      break
   677→   677→    }
   678→   678→    const parsed = parseAssistant(content)
   679→   679→
   680→   680→    if (parsed.thought) {
   681→   681→      await opts.emit('subagent_thought', { dispatchId: opts.dispatchId, content: parsed.thought })
   682→   682→    }
   683→   683→
   684→   684→    if (!parsed.tool) {
   685→   685→      // FIX 2: Stuck detection for sub-agents.
   686→   686→      // If the sub-agent produced ONLY a thought with no substantial answer,
   687→   687→      // auto-recover by prompting it to continue (instead of treating the
   688→   688→      // thought as the final answer).
   689→   689→      const textAfterThought = content.replace(THOUGHT_RE, '').trim()
   690→   690→      const isThoughtOnly = !!parsed.thought && textAfterThought.length < 20
   691→   691→      const stuckPatterns = /(wait|waiting|haven't|will wait|let me wait|as i wait)/i
   692→   692→      const isStuck = isThoughtOnly && stuckPatterns.test(parsed.thought)
   693→   693→
   694→   694→      if ((isStuck || isThoughtOnly) && iter < SUBAGENT_MAX_ITERATIONS - 1) {
   695→   695→        await opts.emit('subagent_thought', {
   696→   696→          dispatchId: opts.dispatchId,
   697→   697→          content: `[AUTO-RECOVERY] Thought-only response. Prompting to continue...`,
   698→   698→        })
   699→   699→        conversationMessages.push({ role: 'assistant', content })
   700→   700→        conversationMessages.push({
   701→   701→          role: 'user',
   702→   702→          content: '[SYSTEM] You produced only a thought. Please either call a tool or give your final answer now.',
   703→   703→        })
   704→   704→        continue
   705→   705→      }
   706→   706→
   707→   707→      // final answer
   708→   708→      finalAnswer = textAfterThought || content.trim()
   709→   709→      break
   710→   710→    }
   711→   711→
   712→   712→    const toolName = parsed.tool.name
   713→   713→    const toolArgs = parsed.tool.args
   714→   714→    const stepId = `sub_${opts.dispatchId}_${iter}_${Math.random().toString(36).slice(2, 8)}`
   715→   715→
   716→   716→    // Enforce the sub-agent's allowed tool list
   717→   717→    if (!allowed.has(toolName)) {
   718→   718→      const errResult: ToolResult = {
   719→   719→        ok: false,
   720→   720→        preview: `Tool "${toolName}" not allowed for ${sub.name}`,
   721→   721→        result: `BLOCKED: ${sub.name} is not permitted to call "${toolName}". Allowed tools: ${sub.allowedTools.join(', ')}. Use one of those instead.`,
   722→   722→      }
   723→   723→      const step = {
   724→   724→        id: stepId,
   725→   725→        thought: parsed.thought,
   726→   726→        toolName,
   727→   727→        toolArgs,
   728→   728→        toolResult: errResult,
   729→   729→        startedAt: Date.now(),
   730→   730→        finishedAt: Date.now(),
   731→   731→      }
   732→   732→      steps.push(step)
   733→   733→      await opts.emit('subagent_tool_call', {
   734→   734→        dispatchId: opts.dispatchId,
   735→   735→        stepId,
   736→   736→        name: toolName,
   737→   737→        args: toolArgs,
   738→   738→        thought: parsed.thought,
   739→   739→        stepNumber: iter,
   740→   740→      })
   741→   741→      await opts.emit('subagent_tool_result', {
   742→   742→        dispatchId: opts.dispatchId,
   743→   743→        stepId,
   744→   744→        result: errResult.result,
   745→   745→        preview: errResult.preview,
   746→   746→        ok: false,
   747→   747→        artifacts: undefined,
   748→   748→      })
   749→   749→      conversationMessages.push({ role: 'assistant', content })
   750→   750→      conversationMessages.push({
   751→   751→        role: 'user',
   752→   752→        content: `[TOOL_RESULT] ${toolName}: ${errResult.result}`,
   753→   753→      })
   754→   754→      // Persist the blocked attempt for reload reconstruction
   755→   755→      try {
   756→   756→        await db.message.create({
   757→   757→          data: {
   758→   758→            conversationId: opts.parentConversationId,
   759→   759→            role: 'tool',
   760→   760→            content: `[subagent:${sub.id}:blocked] ${toolName} ${JSON.stringify(toolArgs)}`,
   761→   761→            toolName: 'subagent_tool',
   762→   762→            toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, tool: toolName, args: toolArgs }),
   763→   763→            toolResult: errResult.result,
   764→   764→          },
   765→   765→        })
   766→   766→      } catch {
   767→   767→        /* ignore */
   768→   768→      }
   769→   769→      continue
   770→   770→    }
   771→   771→
   772→   772→    const step = {
   773→   773→      id: stepId,
   774→   774→      thought: parsed.thought,
   775→   775→      toolName,
   776→   776→      toolArgs,
   777→   777→      startedAt: Date.now(),
   778→   778→    }
   779→   779→    steps.push(step)
   780→   780→
   781→   781→    await opts.emit('subagent_tool_call', {
   782→   782→      dispatchId: opts.dispatchId,
   783→   783→      stepId,
   784→   784→      name: toolName,
   785→   785→      args: toolArgs,
   786→   786→      thought: parsed.thought,
   787→   787→      stepNumber: iter,
   788→   788→    })
   789→   789→
   790→   790→    const toolResult = await dispatchTool(toolName, toolArgs, ctx)
   791→   791→    step.toolResult = toolResult
   792→   792→    step.finishedAt = Date.now()
   793→   793→
   794→   794→    await opts.emit('subagent_tool_result', {
   795→   795→      dispatchId: opts.dispatchId,
   796→   796→      stepId,
   797→   797→      result: toolResult.result,
   798→   798→      preview: toolResult.preview,
   799→   799→      ok: toolResult.ok,
   800→   800→      artifacts: toolResult.artifacts,
   801→   801→    })
   802→   802→
   803→   803→    // Persist subagent tool activity for reload reconstruction
   804→   804→    try {
   805→   805→      if (step.thought) {
   806→   806→        await db.message.create({
   807→   807→          data: {
   808→   808→            conversationId: opts.parentConversationId,
   809→   809→            role: 'thought',
   810→   810→            content: `[subagent:${sub.id}] ${step.thought}`,
   811→   811→          },
   812→   812→        })
   813→   813→      }
   814→   814→      await db.message.create({
   815→   815→        data: {
   816→   816→          conversationId: opts.parentConversationId,
   817→   817→          role: 'tool',
   818→   818→          content: `[subagent:${sub.id}:tool] ${toolName} ${JSON.stringify(toolArgs)}`,
   819→   819→          toolName: 'subagent_tool',
   820→   820→          toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, stepId, tool: toolName, args: toolArgs }),
   821→   821→          toolResult: toolResult.result,
   822→   822→        },
   823→   823→      })
   824→   824→    } catch {
   825→   825→      /* ignore persistence errors */
   826→   826→    }
   827→   827→
   828→   828→    conversationMessages.push({ role: 'assistant', content })
   829→   829→    conversationMessages.push({
   830→   830→      role: 'user',
   831→   831→      content: `[TOOL_RESULT] ${toolName}: ${toolResult.result}`,
   832→   832→    })
   833→   833→  }
   834→   834→
   835→   835→  if (!finalAnswer) {
   836→   836→    finalAnswer = `${sub.name} reached its tool-call limit. Here is what it found so far:\n\n` +
   837→   837→      steps
   838→   838→        .filter((s) => s.toolResult?.result)
   839→   839→        .map((s, i) => `${i + 1}. **${s.toolName}** → ${s.toolResult!.result.slice(0, 500)}`)
   840→   840→        .join('\n\n')
   841→   841→  }
   842→   842→
   843→   843→  await opts.emit('subagent_complete', { dispatchId: opts.dispatchId, answer: finalAnswer })
   844→   844→
   845→   845→  // Persist the sub-agent's final answer for reload reconstruction
   846→   846→  try {
   847→   847→    await db.message.create({
   848→   848→      data: {
   849→   849→        conversationId: opts.parentConversationId,
   850→   850→        role: 'tool',
   851→   851→        content: `[subagent:${sub.id}:complete] ${finalAnswer.slice(0, 500)}`,
   852→   852→        toolName: 'subagent_complete',
   853→   853→        toolArgs: JSON.stringify({ agentId: sub.id, dispatchId: opts.dispatchId, task: opts.task }),
   854→   854→        toolResult: finalAnswer,
   855→   855→      },
   856→   856→    })
   857→   857→  } catch {
   858→   858→    /* ignore */
   859→   859→  }
   860→   860→
   861→   861→  return { answer: finalAnswer, steps }
