   147→              <p className="text-xs text-[#7c89b5] mt-1 tracking-wide">

   439→/* ----------------------------- KPI card ----------------------------- */
   440→function KpiCard({
   441→  label,
   442→  value,
   443→  sub,
   444→  icon,
   445→  accent,
   446→  big,
   447→}: {
   448→  label: string
   449→  value: string
   450→  sub?: string
   451→  icon: React.ReactNode
   452→  accent: 'cyan' | 'purple' | 'pink'
   453→  big?: boolean
   454→}) {
   455→  const colorMap = {
   456→    cyan: 'text-cyan-300 border-cyan-400/30',
   457→    purple: 'text-purple-300 border-purple-400/30',
   458→    pink: 'text-pink-300 border-pink-400/30',
   459→  }
   460→  return (
   461→    <div className="glass rounded-xl p-3 sm:p-4">
   462→      <div className="flex items-center justify-between mb-2">
   463→        <span className="text-[9px] sm:text-[10px] tracking-[0.18em] text-[#7c89b5] font-semibold">
   464→          {label}
   465→        </span>
   466→        <div className={`w-7 h-7 rounded-md flex items-center justify-center bg-[#0a1020] border ${colorMap[accent]}`}>
   467→          {icon}
   468→        </div>
   469→      </div>
   470→      <div className={`${big ? 'text-2xl sm:text-3xl' : 'text-xl sm:text-2xl'} font-bold neon-text-cyan`}>
   471→        {value}
   472→      </div>
   473→      {sub && <div className="text-[10px] text-[#5b6a92] mt-1 tracking-wide">{sub}</div>}
   474→    </div>
   475→  )
   476→}
   477→
   478→function GrowthCard({
   479→  pct,
   480→  target,
   481→  currencySymbol: _currencySymbol,
   482→}: {
   483→  pct: number
   484→  target: number
   485→  currencySymbol: string
   486→}) {
   487→  const positive = pct >= 0
   488→  const metTarget = pct >= target
   489→  return (
   490→    <div className="glass rounded-xl p-3 sm:p-4">
   491→      <div className="flex items-center justify-between mb-2">
   492→        <span className="text-[9px] sm:text-[10px] tracking-[0.18em] text-[#7c89b5] font-semibold">
   493→          GROWTH %
   494→        </span>
   495→        <div
   496→          className={`w-7 h-7 rounded-md flex items-center justify-center bg-[#0a1020] border ${
   497→            positive ? 'text-emerald-300 border-emerald-400/30' : 'text-pink-300 border-pink-400/30'
   498→          }`}
   499→        >
   500→          {positive ? <TrendingUp className="w-4 h-4" /> : <TrendingDown className="w-4 h-4" />}
   501→        </div>
   502→      </div>
   503→      <div
   504→        className={`text-2xl sm:text-3xl font-bold ${
   505→          positive ? 'text-emerald-300' : 'text-pink-300'
   506→        }`}
   507→        style={{
   508→          textShadow: positive
   509→            ? '0 0 8px rgba(110,231,183,0.6), 0 0 18px rgba(110,231,183,0.3)'
   510→            : '0 0 8px rgba(236,72,153,0.6), 0 0 18px rgba(236,72,153,0.3)',
   511→        }}
   512→      >
   513→        {positive ? '+' : ''}
   514→        {pct.toFixed(1)}%
   515→      </div>
   516→      <div className="text-[10px] text-[#5b6a92] mt-1 tracking-wide">
   517→        vs yesterday • target +{target}%
   518→        {metTarget && <span className="text-emerald-300 ml-1">✓</span>}
   519→      </div>
   520→    </div>
   521→  )
   522→}

   524→/* ----------------------------- Income row ----------------------------- */

   529→}: {
   530→  entry: IncomeEntry
   531→  currencySymbol: string

   535→  const isSample = entry.source.toLowerCase() === 'sample'
   536→  return (
   537→    <div className="flex items-center gap-3 p-2 rounded-lg glass border-cyan-400/15 hover:border-cyan-400/40 transition group">
   538→      <div
   539→        className={`w-9 h-9 rounded-md flex items-center justify-center font-bold text-xs flex-shrink-0 ${
   540→          isSample
   541→            ? 'bg-[#1a1430] text-purple-200 border border-purple-400/30'
